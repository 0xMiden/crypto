1//! The `LiftedAir` super-trait for AIR definitions in the lifted STARK system.
2//!
3//! # Panic safety of `eval()`
4//!
5//! [`LiftedAir::eval`] is generic over `AB: LiftedAirBuilder`, so it cannot branch
6//! on the concrete builder type. All builders expose data through the same trait
7//! methods — [`main()`](crate::AirBuilder::main),
8//! [`permutation()`](crate::PermutationAirBuilder::permutation),
9//! [`public_values()`](crate::AirBuilder::public_values),
10//! [`permutation_randomness()`](crate::PermutationAirBuilder::permutation_randomness),
11//! [`permutation_values()`](crate::PermutationAirBuilder::permutation_values), and
12//! [`periodic_values()`](crate::PeriodicAirBuilder::periodic_values) — which return
13//! matrices or slices.
14//!
15//! If the symbolic evaluation in [`LiftedAir::log_quotient_degree`] succeeds (i.e.
16//! does not panic), it proves that the AIR's `eval()` only accesses indices within
17//! the declared dimensions. Any concrete builder constructed with matching dimensions
18//! is therefore safe from out-of-bounds panics.
19//!
20//! Use [`LiftedAir::is_valid_builder`] to verify that a concrete builder's
21//! dimensions match the AIR before calling `eval()`.
22

37/// Super-trait for AIR definitions used by the lifted STARK prover/verifier.
38///
39/// Inherits from upstream traits for width and public values.
40/// Adds Miden-specific auxiliary trace support and periodic column data.
41/// Every `LiftedAir` must provide an auxiliary trace (even if it is a minimal
42/// 1-column dummy).
43///
44/// # Type Parameters
45/// - `F`: Base field
46/// - `EF`: Extension field (for aux trace challenges and aux values)
47pub trait LiftedAir<F: Field, EF>: Sync + BaseAir<F> {
48    /// Return the periodic table data: a list of columns, each a `Vec<F>` of evaluations.
49    ///
50    /// Each inner `Vec<F>` represents one periodic column. Its length is the period of
51    /// that column, and the entries are the evaluations over a subgroup of that order.
52    ///
53    /// Default: no periodic columns.
54    fn periodic_columns(&self) -> Vec<Vec<F>> {
55        Vec::new()
56    }
57
58    /// Return a matrix with all periodic columns extended to a common height.
59    ///
60    /// Columns with smaller periods are repeated cyclically to fill the extended domain.
61    /// Returns `None` if there are no periodic columns.
62    fn periodic_columns_matrix(&self) -> Option<RowMajorMatrix<F>> {
63        let cols = self.periodic_columns();
64        if cols.is_empty() {
65            return None;
66        }
67
68        let max_period = cols.iter().map(Vec::len).max()?;
69        let num_cols = cols.len();
70
71        let mut values = Vec::with_capacity(max_period * num_cols);
72        for row in 0..max_period {
73            for col in &cols {
74                let period = col.len();
75                values.push(col[row % period]);
76            }
77        }
78
79        Some(RowMajorMatrix::new(values, num_cols))
80    }
81
82    /// Number of extension-field challenges required for the auxiliary trace.
83    fn num_randomness(&self) -> usize;
84
85    /// Number of extension-field columns in the auxiliary trace.
86    fn aux_width(&self) -> usize;
87
88    /// Number of extension-field aux values committed to the Fiat-Shamir transcript.
89    ///
90    /// These are the values returned by
91    /// [`AuxBuilder::build_aux_trace`](crate::AuxBuilder::build_aux_trace) alongside the aux
92    /// trace matrix. Their count may differ from [`aux_width`](Self::aux_width) (the number of
93    /// aux trace columns).
94    ///
95    /// These values are exposed to AIR constraints as *permutation values* via
96    /// [`PermutationAirBuilder::permutation_values`](crate::PermutationAirBuilder::permutation_values).
97    fn num_aux_values(&self) -> usize;
98
99    /// Number of variable-length public inputs this AIR expects.
100    ///
101    /// Each input is a slice of base-field elements that
102    /// [`reduced_aux_values`](Self::reduced_aux_values) reduces to a single value.
103    /// The prover validates that witnesses provide exactly this many slices.
104    ///
105    /// Implementors of [`reduced_aux_values`](Self::reduced_aux_values) should verify
106    /// that `var_len_public_inputs` contains exactly this many slices, returning
107    /// [`ReductionError`] otherwise.
108    fn num_var_len_public_inputs(&self) -> usize;
109
110    /// Reduce this AIR's aux values to a [`ReducedAuxValues`] contribution.
111    ///
112    /// Called by the verifier (with concrete field values, not symbolic expressions)
113    /// to compute each AIR's contribution to the global cross-AIR bus identity check.
114    /// The verifier accumulates contributions across all AIRs and checks that the
115    /// combined result is identity (prod=1, sum=0).
116    ///
117    /// # Arguments
118    /// - `aux_values`: prover-supplied aux values (from the proof)
119    /// - `challenges`: extension-field challenges (same as used for aux trace building)
120    /// - `public_values`: this AIR's public values (base field)
121    /// - `var_len_public_inputs`: reducible inputs for the cross-AIR identity check
122    ///
123    /// # Errors
124    ///
125    /// The verifier validates instance dimensions (public values length,
126    /// var-len public inputs count) before calling this method, so
127    /// implementations can assume correct input counts. However, the
128    /// *length of each individual var-len slice* is not validated upfront —
129    /// implementations that index into these slices must check lengths
130    /// themselves or use the `Result` return type to report errors.
131    ///
132    /// Default: returns identity (correct for AIRs without buses).
133    fn reduced_aux_values(
134        &self,
135        _aux_values: &[EF],
136        _challenges: &[EF],
137        _public_values: &[F],
138        _var_len_public_inputs: VarLenPublicInputs<'_, F>,
139    ) -> Result<ReducedAuxValues<EF>, ReductionError>
140    where
141        EF: ExtensionField<F>,
142    {
143        Ok(ReducedAuxValues::identity())
144    }
145
146    /// Return the [`AirLayout`] describing this AIR's dimensions.
147    ///
148    /// This is the single source of truth for building symbolic or layout builders.
149    /// `preprocessed_width` is always 0 because lifted AIRs forbid preprocessed traces.
150    fn air_layout(&self) -> AirLayout {
151        AirLayout {
152            preprocessed_width: 0,
153            main_width: self.width(),
154            num_public_values: self.num_public_values(),
155            permutation_width: self.aux_width(),
156            num_permutation_challenges: self.num_randomness(),
157            num_permutation_values: self.num_aux_values(),
158            num_periodic_columns: self.periodic_columns().len(),
159        }
160    }
161
162    /// Validate that this AIR satisfies the [`LiftedAir`] contract.
163    ///
164    /// The lifted STARK protocol relies on several structural properties of the AIR
165    /// that can be checked statically (i.e. without a witness). This method verifies
166    /// the subset that is machine-checkable; the full list of trust assumptions is
167    /// documented in the module docs of `miden-lifted-stark`. Both the prover and
168    /// verifier call this before proceeding, so a malformed AIR is caught early.
169    ///
170    /// # Checked properties
171    ///
172    /// - **No preprocessed trace** — the lifted STARK protocol does not support preprocessed
173    ///   (fixed) columns; their presence is an error.
174    /// - **Positive auxiliary width** — every lifted AIR must declare at least one auxiliary column
175    ///   (`aux_width() > 0`).
176    /// - **Well-formed periodic columns** — each periodic column must be non-empty and have a
177    ///   power-of-two length.
178    fn validate(&self) -> Result<(), AirStructureError> {
179        if self.preprocessed_trace().is_some() {
180            return Err(AirStructureError::PreprocessedTrace);
181        }
182        if self.aux_width() == 0 {
183            return Err(AirStructureError::ZeroAuxWidth);
184        }
185        for (i, col) in self.periodic_columns().iter().enumerate() {
186            if col.is_empty() || !col.len().is_power_of_two() {
187                return Err(AirStructureError::InvalidPeriodicColumn {
188                    index: i,
189                    length: col.len(),
190                });
191            }
192        }
193        Ok(())
194    }
195
196    /// Evaluate all AIR constraints using the provided builder.
197    fn eval<AB: LiftedAirBuilder<F = F>>(&self, builder: &mut AB);
198
199    /// Log₂ of the number of quotient chunks, inferred from symbolic constraint analysis.
200    ///
201    /// Evaluates the AIR on a [`SymbolicAirBuilder`](crate::symbolic::SymbolicAirBuilder) to
202    /// determine the maximum constraint degree M, then returns `log2_ceil(M - 1)` (padded so M
203    /// ≥ 2).
204    ///
205    /// Uses `SymbolicAirBuilder<F>` (i.e. `EF = F`) which is sufficient for degree
206    /// computation since extension-field operations have the same degree structure.
207    ///
208    /// # Why `M − 1` chunks?
209    ///
210    /// Let N be the trace height (so trace columns are polynomials of degree < N).
211    /// Symbolic evaluation assigns each constraint a *degree multiple* M, meaning the
212    /// resulting numerator polynomial C(X) has degree bounded by roughly M·(N − 1).
213    ///
214    /// In a STARK, the constraint numerator is divisible by the trace vanishing
215    /// polynomial `Z_H(X) = Xᴺ − 1`, so the quotient polynomial
216    /// `Q(X) = C(X) / Z_H(X)` has
217    ///
218    /// `deg(Q) ≤ deg(C) − N ≤ M·(N − 1) − N < (M − 1)·N`.
219    ///
220    /// We commit to Q(X) by splitting it into D chunks of degree < N. The bound above
221    /// shows that D = M − 1 chunks suffice; we then round D up to a power of two and
222    /// return `log2(D)`.
223    ///
224    /// We clamp M ≥ 2 so that D ≥ 1. If M = 1 then `deg(C) < N`, and divisibility by
225    /// `Z_H` would force C(X) to be the zero polynomial (i.e. the constraint carries no
226    /// information about the trace).
227    fn log_quotient_degree(&self) -> usize
228    where
229        Self: Sized,
230    {
231        let mut builder = SymbolicAirBuilder::<F>::new(self.air_layout());
232        self.eval(&mut builder);
233
234        let base_degree_multiple =
235            |constraint: &SymbolicExpression<F>| constraint.degree_multiple();
236        let ext_degree_multiple =
237            |constraint: &SymbolicExpressionExt<F, F>| constraint.degree_multiple();
238
239        let base_degree =
240            builder.base_constraints().iter().map(base_degree_multiple).max().unwrap_or(0);
241        let ext_degree = builder
242            .extension_constraints()
243            .iter()
244            .map(ext_degree_multiple)
245            .max()
246            .unwrap_or(0);
247        let constraint_degree = base_degree.max(ext_degree).max(2);
248
249        log2_ceil_usize(constraint_degree - 1)
250    }
251
252    /// Number of quotient chunks: `2^log_quotient_degree()`.
253    fn constraint_degree(&self) -> usize
254    where
255        Self: Sized,
256    {
257        1 << self.log_quotient_degree()
258    }
259
260    /// Check that a builder's dimensions match this AIR.
261    ///
262    /// Verifies every data-carrying accessor on [`LiftedAirBuilder`]: main trace,
263    /// preprocessed trace, aux trace, public values, randomness, aux values, and
264    /// periodic values.
265    ///
266    /// This guards the invariant that makes [`eval`](Self::eval) panic-free: if
267    /// the symbolic evaluation in [`log_quotient_degree`](Self::log_quotient_degree)
268    /// succeeds and this check passes, then `eval()` cannot panic from
269    /// out-of-bounds access on the builder's accessors.
270    fn is_valid_builder<AB: LiftedAirBuilder<F = F>>(
271        &self,
272        builder: &AB,
273    ) -> Result<(), AirStructureError> {
274        let check =
275            |part: TracePart, expected: usize, actual: usize| -> Result<(), AirStructureError> {
276                if actual != expected {
277                    return Err(AirStructureError::BuilderMismatch { part, expected, actual });
278                }
279                Ok(())
280            };
281
282        let main = builder.main();
283        // Check current and next slices of the main trace.
284        check(TracePart::Main, self.width(), main.current_slice().len())?;
285        check(TracePart::Main, self.width(), main.next_slice().len())?;
286
287        // Check current and next slices of the aux trace.
288        let perm = builder.permutation();
289        check(TracePart::Aux, self.aux_width(), perm.current_slice().len())?;
290        check(TracePart::Aux, self.aux_width(), perm.next_slice().len())?;
291
292        check(TracePart::PublicValues, self.num_public_values(), builder.public_values().len())?;
293        check(
294            TracePart::Randomness,
295            self.num_randomness(),
296            builder.permutation_randomness().len(),
297        )?;
298        check(TracePart::AuxValues, self.num_aux_values(), builder.permutation_values().len())?;
299        check(
300            TracePart::PeriodicValues,
301            self.periodic_columns().len(),
302            builder.periodic_values().len(),
303        )?;
304
305        Ok(())
306    }
307}

309/// Which part of the trace a builder mismatch refers to.
310#[derive(Copy, Clone, Debug)]
311pub enum TracePart {
312    Main,
313    Aux,
314    PublicValues,
315    Randomness,
316    AuxValues,
317    PeriodicValues,
318}

320/// Errors intrinsic to a single AIR definition, independent of any instance
321/// data. Returned by [`LiftedAir::validate`] and [`LiftedAir::is_valid_builder`].
322#[derive(Debug, Error)]
323pub enum AirStructureError {
324    #[error("periodic column {index}: length must be positive power of two, got {length}")]
325    InvalidPeriodicColumn { index: usize, length: usize },
326    #[error("preprocessed traces are not supported")]
327    PreprocessedTrace,
328    #[error("{part:?} dimension mismatch: expected {expected}, got {actual}")]
329    BuilderMismatch {
330        part: TracePart,
331        expected: usize,
332        actual: usize,
333    },
334    #[error("aux width must be positive")]
335    ZeroAuxWidth,
336}
