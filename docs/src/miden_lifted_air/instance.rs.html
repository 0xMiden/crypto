1//! Shared instance types for the lifted STARK prover and verifier.
2//!
3//! - [`AirWitness`]: Prover witness — trace + public values
4//! - [`AirInstance`]: Verifier instance — log trace height + public values
5
6use p3_field::Field;
7use p3_matrix::{Matrix, dense::RowMajorMatrix};
8
9use crate::{LiftedAir, VarLenPublicInputs, air::AirValidationError, util::log2_strict_u8};
10
11/// Prover witness: trace matrix, public values, and variable-length public inputs.
12///
13/// Validates on construction that the trace height is a power of two.
14///
15/// **Commitment:** callers **must** bind both `public_values` and
16/// `var_len_public_inputs` to the Fiat-Shamir challenger state before proving.
17pub struct AirWitness<'a, F> {
18    /// Main trace matrix.
19    pub trace: &'a RowMajorMatrix<F>,
20    /// Public values for this AIR.
21    pub public_values: &'a [F],
22    /// Variable-length public inputs (reducible inputs for bus identity checks).
23    pub var_len_public_inputs: VarLenPublicInputs<'a, F>,
24}
25
26impl<'a, F> AirWitness<'a, F> {
27    /// Create a new prover witness with validation.
28    ///
29    /// # Panics
30    ///
31    /// - If `trace.height()` is not a power of two
32    pub fn new(
33        trace: &'a RowMajorMatrix<F>,
34        public_values: &'a [F],
35        var_len_public_inputs: VarLenPublicInputs<'a, F>,
36    ) -> Self
37    where
38        F: Field,
39    {
40        assert!(
41            trace.height().is_power_of_two(),
42            "trace height must be power of two, got {}",
43            trace.height()
44        );
45        Self {
46            trace,
47            public_values,
48            var_len_public_inputs,
49        }
50    }
51
52    /// Convert to a verifier instance (drops the trace, keeps log height).
53    ///
54    /// Returns an error if the trace height is not a power of two.
55    pub fn to_instance(&self) -> Result<AirInstance<'a, F>, AirValidationError>
56    where
57        F: Clone + Send + Sync,
58    {
59        let height = self.trace.height();
60        if !height.is_power_of_two() {
61            return Err(AirValidationError::InvalidTraceHeight { height });
62        }
63        Ok(AirInstance {
64            log_trace_height: log2_strict_u8(height),
65            public_values: self.public_values,
66            var_len_public_inputs: self.var_len_public_inputs,
67        })
68    }
69}

71/// Verifier instance: log trace height, public values, and variable-length inputs.
72///
73/// Both the prover and verifier carry `var_len_public_inputs`. The verifier uses
74/// them in [`LiftedAir::reduced_aux_values`](crate::LiftedAir::reduced_aux_values)
75/// for the cross-AIR identity check.
76#[derive(Clone, Copy)]
77pub struct AirInstance<'a, F> {
78    /// Log₂ of the trace height.
79    pub log_trace_height: u8,
80    /// Public values for this AIR.
81    pub public_values: &'a [F],
82    /// Reducible inputs for the cross-AIR identity check. Empty slice if no buses.
83    pub var_len_public_inputs: VarLenPublicInputs<'a, F>,
84}
85
86/// Validate AIR and instance dimensions for a slice of `(air, instance)` pairs.
87///
88/// Checks that instances are non-empty, sorted by ascending log height, and that
89/// the maximum trace height is at least 2 (required for the 2-row transition window).
90/// Runs [`LiftedAir::validate`] + [`AirInstance::validate`] on each pair.
91///
92/// Returns the log of the maximum trace height.
93pub fn validate_instances<F, EF, A>(
94    instances: &[(&A, AirInstance<'_, F>)],
95) -> Result<u8, AirValidationError>
96where
97    F: Field,
98    A: LiftedAir<F, EF>,
99{
100    let mut log_prev_height: u8 = 0;
101    for (air, inst) in instances {
102        air.validate()?;
103        inst.validate(*air)?;
104        if inst.log_trace_height < log_prev_height {
105            return Err(AirValidationError::NotAscending);
106        }
107        log_prev_height = inst.log_trace_height;
108    }
109    // log_prev_height == 0 means either no instances or all traces have height 1,
110    // both invalid for a protocol with a 2-row transition window.
111    if log_prev_height == 0 {
112        return Err(AirValidationError::Empty);
113    }
114    Ok(log_prev_height)
115}

117impl<'a, F> AirInstance<'a, F> {
118    /// Validate that this instance matches an AIR's specification.
119    ///
120    /// Checks public values length, var-len public inputs count, and that the
121    /// trace height is at least the max periodic column length.
122    pub fn validate<EF>(&self, air: &impl LiftedAir<F, EF>) -> Result<(), AirValidationError>
123    where
124        F: Field,
125    {
126        let expected = air.num_public_values();
127        let actual = self.public_values.len();
128        if actual != expected {
129            return Err(AirValidationError::PublicValuesMismatch { expected, actual });
130        }
131        let expected = air.num_var_len_public_inputs();
132        let actual = self.var_len_public_inputs.len();
133        if actual != expected {
134            return Err(AirValidationError::VarLenPublicInputsMismatch { expected, actual });
135        }
136        let trace_height = 1 << self.log_trace_height as usize;
137        let max_period = air.periodic_columns().iter().map(|c| c.len()).max().unwrap_or(0);
138        if trace_height < max_period {
139            return Err(AirValidationError::TraceHeightBelowPeriod { trace_height, max_period });
140        }
141        Ok(())
142    }
143}
