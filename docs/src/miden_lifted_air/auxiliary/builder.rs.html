1//! The `AuxBuilder` trait for constructing auxiliary traces.
2//!
3//! This trait decouples auxiliary trace *building* from the AIR definition,
4//! allowing the prover to supply a separate builder per instance.
5
6use alloc::vec::Vec;
7
8use p3_field::{ExtensionField, Field};
9use p3_matrix::dense::RowMajorMatrix;
10
11/// Builder for constructing the auxiliary trace from a main trace and challenges.
12///
13/// Decoupled from [`LiftedAir`](crate::LiftedAir) so that prover-side trace
14/// construction is not part of the AIR trait. Each prover instance can supply
15/// its own `AuxBuilder`.
16pub trait AuxBuilder<F: Field, EF: ExtensionField<F>> {
17    /// Build the auxiliary trace and return aux values.
18    ///
19    /// # Arguments
20    /// - `main`: The main trace matrix
21    /// - `challenges`: Extension-field challenges for aux trace construction
22    ///
23    /// # Returns
24    /// `(aux_trace, aux_values)` where:
25    /// - `aux_trace`: The auxiliary trace matrix (EF columns)
26    /// - `aux_values`: Extension-field scalars committed to the Fiat-Shamir transcript. Their
27    ///   meaning is AIR-defined — typically the aux trace's last row, but the protocol does not
28    ///   require this. The AIR's [`eval`](crate::LiftedAir::eval) should constrain how they relate
29    ///   to the committed trace, and [`reduced_aux_values`](crate::LiftedAir::reduced_aux_values)
30    ///   uses them for cross-AIR bus identity checking.
31    fn build_aux_trace(
32        &self,
33        main: &RowMajorMatrix<F>,
34        challenges: &[EF],
35    ) -> (RowMajorMatrix<EF>, Vec<EF>);
36}
