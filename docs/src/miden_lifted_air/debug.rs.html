1//! Debug-only structural checks for AIRs. These panic on contract violation and
2//! are meant for tests / setup; the prover and verifier hot paths assume AIRs
3//! are well-formed.
4//!
5//! # When to use what
6//!
7//! - [`assert_multi_air_valid`]: verify a [`MultiAir`] satisfies the structural contract assumed by
8//!   the rest of the protocol — per AIR positive auxiliary width and power-of-two periodic columns;
9//!   across AIRs a shared `num_public_values`. This also cross-checks the overridable
10//!   [`MultiAir::num_air_inputs`] / [`LiftedAir::max_periodic_length`] against the raw AIR data, so
11//!   an override that lies about either is caught here.
12//! - [`check_builder_shape`]: verify a concrete builder's accessor dimensions match an AIR before
13//!   calling [`LiftedAir::eval`]. Used as a `cfg(debug_assertions)` belt-and-suspenders inside the
14//!   prover and verifier loops.
15//!
16//! Runtime checks on caller-supplied data live on the constructors
17//! [`Statement::new`](crate::Statement::new) /
18//! [`ProverStatement::new`](crate::ProverStatement::new).
19
20use p3_field::{ExtensionField, Field};

25/// Assert a [`MultiAir`] is structurally well-formed.
26///
27/// This checks the structural invariants that prover/verifier hot paths trust:
28/// - [`MultiAir::airs`] is non-empty.
29/// - All AIRs agree on [`BaseAir::num_public_values`].
30/// - [`MultiAir::num_air_inputs`] agrees with the per-AIR public value count.
31/// - Each AIR has positive auxiliary width.
32/// - Each AIR's [`BaseAir::preprocessed_width`] agrees with [`BaseAir::preprocessed_trace`]
33///   presence and width.
34/// - Each periodic column is non-empty and has power-of-two length.
35/// - [`LiftedAir::max_periodic_length`] agrees with the raw periodic columns.
36///
37/// Panics on any violation. An empty [`MultiAir`] is a malformed trusted AIR
38/// definition, not a typed [`Statement::new`](crate::Statement::new) error.
39pub fn assert_multi_air_valid<F, EF, MA>(multi_air: &MA)
40where
41    F: Field,
42    EF: ExtensionField<F>,
43    MA: MultiAir<F, EF>,
44{
45    let airs = multi_air.airs();
46    assert!(!airs.is_empty(), "MultiAir::airs() must be non-empty");
47
48    // Derive the shared count from the raw AIRs and confirm the overridable
49    // `num_air_inputs` agrees.
50    let num_air_inputs = airs[0].num_public_values();
51    assert!(
52        airs.iter().all(|air| air.num_public_values() == num_air_inputs),
53        "AIRs disagree on num_public_values",
54    );
55    assert!(
56        multi_air.num_air_inputs() == num_air_inputs,
57        "num_air_inputs() = {} disagrees with per-AIR num_public_values() = {num_air_inputs}",
58        multi_air.num_air_inputs(),
59    );
60
61    for (idx, air) in airs.iter().enumerate() {
62        check_one_air::<F, EF, _>(idx, air);
63    }
64}

66/// Assert one AIR satisfies the structural contract.
67fn check_one_air<F, EF, A>(idx: usize, air: &A)
68where
69    F: Field,
70    A: LiftedAir<F, EF>,
71{
72    assert!(air.aux_width() > 0, "AIR {idx}: aux_width must be positive");
73
74    let preprocessed_width = air.preprocessed_width();
75    match air.preprocessed_trace() {
76        Some(trace) => {
77            assert!(
78                preprocessed_width > 0,
79                "AIR {idx}: preprocessed_trace returned Some but preprocessed_width() is 0",
80            );
81            assert_eq!(
82                trace.width(),
83                preprocessed_width,
84                "AIR {idx}: preprocessed_trace width disagrees with preprocessed_width()",
85            );
86            assert!(
87                trace.height().is_power_of_two(),
88                "AIR {idx}: preprocessed_trace height must be a positive power of two, got {height}",
89                height = trace.height(),
90            );
91        },
92        None => {
93            assert_eq!(
94                preprocessed_width, 0,
95                "AIR {idx}: preprocessed_width() is {preprocessed_width} but preprocessed_trace returned None",
96            );
97        },
98    }
99
100    // Derive the max period from the raw columns (asserting positive-power-of-two)
101    // and confirm the overridable `max_periodic_length` agrees.
102    let mut max_period = 0;
103    for (i, col) in air.periodic_columns().iter().enumerate() {
104        assert!(
105            !col.is_empty() && col.len().is_power_of_two(),
106            "AIR {idx}: periodic column {i}: length must be a positive power of two, \
107             got {len}",
108            len = col.len(),
109        );
110        max_period = max_period.max(col.len());
111    }
112    assert!(
113        air.max_periodic_length() == max_period,
114        "AIR {idx}: max_periodic_length() = {} disagrees with periodic_columns() max = {max_period}",
115        air.max_periodic_length(),
116    );
117}

119/// Assert a concrete builder's accessor dimensions match `air` — preprocessed,
120/// main and aux trace, public values, randomness, aux values, and periodic values.
121///
122/// Guards the invariant that makes [`LiftedAir::eval`] panic-free: if symbolic
123/// evaluation in `constraint_degree` succeeds and this check passes, `eval()`
124/// cannot panic from out-of-bounds accessor access.
125pub fn check_builder_shape<F, EF, A, AB>(air: &A, builder: &AB)
126where
127    F: Field,
128    A: LiftedAir<F, EF>,
129    AB: LiftedAirBuilder<F = F>,
130{
131    let check = |part: &str, expected: usize, actual: usize| {
132        assert!(
133            actual == expected,
134            "{part} dimension mismatch: expected {expected}, got {actual}"
135        );
136    };
137
138    let preprocessed = builder.preprocessed();
139    check(
140        "preprocessed (current)",
141        air.preprocessed_width(),
142        preprocessed.current_slice().len(),
143    );
144    check("preprocessed (next)", air.preprocessed_width(), preprocessed.next_slice().len());
145
146    let main = builder.main();
147    check("main (current)", air.width(), main.current_slice().len());
148    check("main (next)", air.width(), main.next_slice().len());
149
150    let perm = builder.permutation();
151    check("aux (current)", air.aux_width(), perm.current_slice().len());
152    check("aux (next)", air.aux_width(), perm.next_slice().len());
153
154    check("public values", air.num_public_values(), builder.public_values().len());
155    check("randomness", air.num_randomness(), builder.permutation_randomness().len());
156    check("aux values", air.num_aux_values(), builder.permutation_values().len());
157    check("periodic values", air.periodic_columns().len(), builder.periodic_values().len());
158}
