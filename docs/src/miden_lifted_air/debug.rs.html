1//! Debug-only structural checks for AIRs. These panic on contract violation and
2//! are meant for tests / setup; the prover and verifier hot paths assume AIRs
3//! are well-formed.
4//!
5//! Runtime checks on caller-supplied data live instead on
6//! [`Statement::new`](crate::Statement::new) /
7//! [`ProverStatement::new`](crate::ProverStatement::new).
8
9use p3_field::{ExtensionField, Field};
10
11use crate::{BaseAir, LiftedAir, LiftedAirBuilder, MultiAir, WindowAccess};

13/// Assert a [`MultiAir`] is structurally well-formed.
14///
15/// This checks the structural invariants that prover/verifier hot paths trust:
16/// - [`MultiAir::airs`] is non-empty.
17/// - All AIRs agree on [`BaseAir::num_public_values`].
18/// - [`MultiAir::num_air_inputs`] agrees with the per-AIR public value count.
19/// - Each AIR has no preprocessed trace.
20/// - Each AIR has positive auxiliary width.
21/// - Each periodic column is non-empty and has power-of-two length.
22/// - [`LiftedAir::max_periodic_length`] agrees with the raw periodic columns.
23///
24/// Panics on any violation. An empty [`MultiAir`] is a malformed trusted AIR
25/// definition, not a typed [`Statement::new`](crate::Statement::new) error.
26pub fn assert_multi_air_valid<F, EF, MA>(multi_air: &MA)
27where
28    F: Field,
29    EF: ExtensionField<F>,
30    MA: MultiAir<F, EF>,
31{
32    let airs = multi_air.airs();
33    assert!(!airs.is_empty(), "MultiAir::airs() must be non-empty");
34
35    // Derive the shared count from the raw AIRs and confirm the overridable
36    // `num_air_inputs` agrees.
37    let num_air_inputs = airs[0].num_public_values();
38    assert!(
39        airs.iter().all(|air| air.num_public_values() == num_air_inputs),
40        "AIRs disagree on num_public_values",
41    );
42    assert!(
43        multi_air.num_air_inputs() == num_air_inputs,
44        "num_air_inputs() = {} disagrees with per-AIR num_public_values() = {num_air_inputs}",
45        multi_air.num_air_inputs(),
46    );
47
48    for (idx, air) in airs.iter().enumerate() {
49        check_one_air::<F, EF, _>(idx, air);
50    }
51}
52
53/// Assert one AIR satisfies the structural contract.
54fn check_one_air<F, EF, A>(idx: usize, air: &A)
55where
56    F: Field,
57    A: LiftedAir<F, EF>,
58{
59    assert!(
60        air.preprocessed_trace().is_none(),
61        "AIR {idx}: preprocessed traces are not supported"
62    );
63    assert!(air.aux_width() > 0, "AIR {idx}: aux_width must be positive");
64
65    // Derive the max period from the raw columns (asserting positive-power-of-two)
66    // and confirm the overridable `max_periodic_length` agrees.
67    let mut max_period = 0;
68    for (i, col) in air.periodic_columns().iter().enumerate() {
69        assert!(
70            !col.is_empty() && col.len().is_power_of_two(),
71            "AIR {idx}: periodic column {i}: length must be a positive power of two, \
72             got {len}",
73            len = col.len(),
74        );
75        max_period = max_period.max(col.len());
76    }
77    assert!(
78        air.max_periodic_length() == max_period,
79        "AIR {idx}: max_periodic_length() = {} disagrees with periodic_columns() max = {max_period}",
80        air.max_periodic_length(),
81    );
82}

84/// Assert a concrete builder's accessor dimensions match `air` — main and aux
85/// trace, public values, randomness, aux values, and periodic values.
86///
87/// Guards the invariant that makes [`LiftedAir::eval`] panic-free: if symbolic
88/// evaluation in `constraint_degree` succeeds and this check passes, `eval()`
89/// cannot panic from out-of-bounds accessor access.
90pub fn check_builder_shape<F, EF, A, AB>(air: &A, builder: &AB)
91where
92    F: Field,
93    A: LiftedAir<F, EF>,
94    AB: LiftedAirBuilder<F = F>,
95{
96    let check = |part: &str, expected: usize, actual: usize| {
97        assert!(
98            actual == expected,
99            "{part} dimension mismatch: expected {expected}, got {actual}"
100        );
101    };
102
103    let main = builder.main();
104    check("main (current)", air.width(), main.current_slice().len());
105    check("main (next)", air.width(), main.next_slice().len());
106
107    let perm = builder.permutation();
108    check("aux (current)", air.aux_width(), perm.current_slice().len());
109    check("aux (next)", air.aux_width(), perm.next_slice().len());
110
111    check("public values", air.num_public_values(), builder.public_values().len());
112    check("randomness", air.num_randomness(), builder.permutation_randomness().len());
113    check("aux values", air.num_aux_values(), builder.permutation_values().len());
114    check("periodic values", air.periodic_columns().len(), builder.periodic_values().len());
115}
