1//! This module contains the definition of [`History`], a simple container for some number of
2//! historical versions of a given merkle tree.
3//!
4//! This history consists of a series of _deltas_ from the current state of the tree, moving
5//! backward in history away from that current state. These deltas are then used to form an overlay
6//! that represents the changes to be made on top of the current tree to put it _back_ in that
7//! historical state.
8//!
9//! It provides functionality for adding new states to the history, as well as for querying the
10//! history at a given point in time.
11//!
12//! # Complexity
13//!
14//! Versions in this structure are _complete_. This means that the data stored for any given
15//! historical version is sufficient to apply atop the current tree to return it to the state
16//! corresponding to that historical version.
17//!
18//! # Performance
19//!
20//! This structure operates entirely in memory, and is hence reasonably quick to query. As of the
21//! current time, no detailed benchmarking has taken place for the history.
22

49/// A collection of changes to arbitrary non-leaf nodes in a merkle tree.
50///
51/// All changes to nodes between versions `v` and `v + 1` must be explicitly "undone" in the
52/// `NodeChanges` representing version `v`. This includes nodes that were defaulted in version `v`
53/// that were given an explicit value in version `v + 1`, where the `NodeChanges` must explicitly
54/// set those nodes back to the default.
55///
56/// Failure to do so will result in incorrect values when those nodes are queried at a point in the
57/// history corresponding to version `v`.
58pub type NodeChanges = Map<NodeIndex, Word>;
59
60/// The type of the keys that need to be changed by the delta to return the target tree to the state
61/// represented by the delta.
62pub type ChangedKeys = Map<Word, Word>;

67/// A History contains a sequence of versions atop a given tree.
68///
69/// The versions are _cumulative_, meaning that querying the history must account for changes from
70/// the current tree that take place in versions that are not the queried version or the current
71/// tree.
72#[derive(Clone, Debug)]
73pub struct History {
74    /// The maximum number of historical versions to be stored.
75    max_count: usize,
76
77    /// The deltas that make up the history for this tree.
78    ///
79    /// It will never contain more than `max_count` deltas, and is ordered with the oldest data at
80    /// the lowest index.
81    ///
82    /// # Implementation Note
83    ///
84    /// We use a [`VecDeque`] instead of a [`Vec`] so that we can both have efficient access to any
85    /// delta, while also making removals from the end efficient. As [`Self::truncate`] only ever
86    /// removes some oldest `n` entries, this is the exact behavior we want.
87    deltas: VecDeque<Delta>,
88}
89
90impl History {
91    /// Constructs a new history container, containing at most `max_count` historical versions for
92    /// a tree.
93    #[must_use]
94    pub fn empty(max_count: usize) -> Self {
95        let deltas = VecDeque::new();
96        Self { max_count, deltas }
97    }
98
99    /// Gets the maximum number of versions that this history can store.
100    #[must_use]
101    pub fn max_versions(&self) -> usize {
102        self.max_count
103    }
104
105    /// Gets the current number of versions in the history.
106    #[must_use]
107    pub fn num_versions(&self) -> usize {
108        self.deltas.len()
109    }
110
111    /// Returns all the roots that the history knows about.
112    ///
113    /// The iteration order of the roots is guaranteed to move backward in time, with earlier items
114    /// being roots from versions closer to the present.
115    ///
116    /// # Complexity
117    ///
118    /// Calling this method provides an iterator whose consumption requires a traversal of all the
119    /// versions. The method's complexity is thus `O(n)` in the number of versions.
120    pub fn roots(&self) -> impl Iterator<Item = RootValue> {
121        self.deltas.iter().rev().map(|d| d.root)
122    }
123
124    /// Returns the root value that corresponds to the provided `version`.
125    pub fn root_for_version(&self, version: VersionId) -> Result<RootValue> {
126        let ix = self.find_latest_corresponding_version(version)?;
127
128        // The direct index is safe here because `find_latest_...` will have returned an error if
129        // there is no such version, and is hence guaranteed to have returned a valid index.
130        Ok(self.deltas[ix].root)
131    }
132
133    /// Adds a version to the history with the provided `root` and represented by the changes from
134    /// the current tree given in `nodes` and `changed_keys`, with `entry_count` as the total
135    /// number of entries for the tree that corresponds to this version.
136    ///
137    /// If adding this version would result in exceeding `self.max_count` historical versions, then
138    /// the oldest of the versions is automatically removed.
139    ///
140    /// # Gotchas
141    ///
142    /// When constructing the `nodes` and `leaves`, keep in mind that those collections must contain
143    /// entries for the **default value of a node or leaf** at any position where the tree was
144    /// sparse in the state represented by `root`. If this is not done, incorrect values may be
145    /// returned.
146    ///
147    /// This is necessary because the changes are the _reverse_ from what one might expect. Namely,
148    /// the changes in a given version `v` must "_revert_" the changes made in the transition from
149    /// version `v` to version `v + 1`.
150    ///
151    /// # Errors
152    ///
153    /// - [`HistoryError::NonMonotonicVersions`] if the provided version is not greater than the
154    ///   previously added version.
155    pub fn add_version(
156        &mut self,
157        root: RootValue,
158        version_id: VersionId,
159        nodes: NodeChanges,
160        changed_keys: ChangedKeys,
161        entry_count: usize,
162    ) -> Result<()> {
163        // We need to fail early if the provided new version is not monotonic with respect to the
164        // latest version in the history.
165        if let Some(v) = self.deltas.back()
166            && v.version_id >= version_id
167        {
168            return Err(HistoryError::NonMonotonicVersions(version_id, v.version_id));
169        }
170
171        // We then check if we would exceed our version count limit, and remove the oldest if so.
172        if self.num_versions() >= self.max_versions() {
173            self.deltas.pop_front();
174        }
175
176        // We then need to update all the older deltas with the necessary additional changes
177        // represented in this newly-added version.
178        for delta in &mut self.deltas {
179            // The root and the version remain the same, but we may need to change the nodes and
180            // keys.
181            for (ix, value) in &nodes {
182                delta.nodes.entry(*ix).or_insert(*value);
183            }
184            for (key, value) in &changed_keys {
185                // If the delta has removed something, we never want to re-add it over the top.
186                delta.changed_keys.entry(*key).or_insert(*value);
187            }
188        }
189
190        self.deltas
191            .push_back(Delta::new(root, version_id, nodes, changed_keys, entry_count));
192
193        Ok(())
194    }
195
196    /// Adds a version to the history, represented by the changes from the current tree given
197    /// `mutations`, with `entry_count` corresponding to the number of entries in the full tree
198    /// corresponding to this version.
199    ///
200    /// If adding this version would result in exceeding `self.max_count` historical versions, then
201    /// the oldest of the versions is automatically removed.
202    ///
203    /// # Gotchas
204    ///
205    /// When constructing the `mutations`, keep in mind that the set must contain entries for the
206    /// **default value of a node or leaf** at any position where the tree was sparse in the state
207    /// represented by `root`. If this is not done, incorrect values may be returned.
208    ///
209    /// This is necessary because the changes are the _reverse_ from what one might expect. Namely,
210    /// the changes in a given version `v` must "_revert_" the changes made in the transition from
211    /// version `v` to version `v + 1`.
212    ///
213    /// # Errors
214    ///
215    /// - [`HistoryError::NonMonotonicVersions`] if the provided version is not greater than the
216    ///   previously added version.
217    pub fn add_version_from_mutation_set(
218        &mut self,
219        version_id: VersionId,
220        mutations: MutationSet,
221        entry_count: usize,
222    ) -> Result<()> {
223        let mut changed_keys = ChangedKeys::default();
224        mutations.new_pairs.into_iter().for_each(|(k, v)| {
225            changed_keys.insert(k, v);
226        });
227
228        // The node changes are more complex, as we have to explicitly handle reversions to empty
229        // specially.
230        let node_changes: NodeChanges = mutations
231            .node_mutations
232            .into_iter()
233            .map(|(ix, m)| match m {
234                NodeMutation::Removal => (ix, *EmptySubtreeRoots::entry(SMT_DEPTH, ix.depth())),
235                NodeMutation::Addition(n) => (ix, n.hash()),
236            })
237            .collect();
238
239        // Now we can simply delegate to the standard function.
240        self.add_version(mutations.new_root, version_id, node_changes, changed_keys, entry_count)
241    }
242
243    /// Returns the index in the sequence of deltas of the version that corresponds to the provided
244    /// `version_id`.
245    ///
246    /// To "correspond" means that it either has the provided `version_id`, or is the newest version
247    /// with a `version_id` less than the provided id. In either case, it is the correct version to
248    /// be used to query the tree state in the provided `version_id`.
249    ///
250    /// # Complexity
251    ///
252    /// Finding the latest corresponding version in the history requires a linear traversal of the
253    /// history entries, and hence has complexity `O(n)` in the number of versions.
254    ///
255    /// # Errors
256    ///
257    /// - [`HistoryError::HistoryEmpty`] if the history is empty and hence there is no version to
258    ///   find.
259    /// - [`HistoryError::VersionTooOld`] if the history does not contain the data to provide a
260    ///   coherent overlay for the provided `version_id` due to `version_id` being older than the
261    ///   oldest version stored.
262    fn find_latest_corresponding_version(&self, version_id: VersionId) -> Result<usize> {
263        // If the version is older than the oldest, we error.
264        if let Some(oldest_version) = self.deltas.front() {
265            if oldest_version.version_id > version_id {
266                return Err(HistoryError::VersionTooOld);
267            }
268        } else {
269            return Err(HistoryError::HistoryEmpty);
270        }
271
272        // As we want the NEWEST delta that satisfies the version, we look for the position at which
273        // the delta cannot be used and move back by one.
274        let ix = self
275            .deltas
276            .iter()
277            .position(|d| d.version_id > version_id)
278            .unwrap_or_else(|| self.num_versions())
279            .checked_sub(1)
280            .expect(
281                "Subtraction should not overflow as we have ruled out the no-version \
282                case, and in the other cases the left operand will be >= 1",
283            );
284
285        Ok(ix)
286    }
287
288    /// Returns a view of the history that allows querying as a single unified overlay on the
289    /// current state of the merkle tree as if the overlay was reverting the tree to the state
290    /// corresponding to the specified `version_id`.
291    ///
292    /// Note that the history may not contain a version that directly corresponds to `version_id`.
293    /// In such a case, the view will instead use the newest version coherent with the provided
294    /// `version_id`, as this is the correct version for the provided id. Note that this will be
295    /// incorrect if the versions stored in the history do not represent contiguous changes from the
296    /// current tree.
297    ///
298    /// # Complexity
299    ///
300    /// The computational complexity of this method is linear in the number of versions stored in
301    /// the history.
302    ///
303    /// # Errors
304    ///
305    /// - [`HistoryError::VersionTooOld`] if the history does not contain the data to provide a
306    ///   coherent overlay for the provided `version_id` due to `version_id` being older than the
307    ///   oldest version stored.
308    pub fn get_view_at(&self, version_id: VersionId) -> Result<HistoryView<'_>> {
309        HistoryView::new_of(version_id, self)
310    }
311
312    /// Removes all versions in the history that are older than the version denoted by the provided
313    /// `version_id`.
314    ///
315    /// If `version_id` is not a version known by the history, it will keep the newest version that
316    /// is capable of serving as that version in queries.
317    ///
318    /// # Complexity
319    ///
320    /// The computational complexity of this method is linear in the number of versions stored in
321    /// the history prior to any removals.
322    pub fn truncate(&mut self, version_id: VersionId) -> usize {
323        // We start by getting the index to truncate to, though it is not an error to remove
324        // something too old.
325        let truncate_ix = self.find_latest_corresponding_version(version_id).unwrap_or(0);
326
327        for _ in 0..truncate_ix {
328            self.deltas.pop_front();
329        }
330
331        truncate_ix
332    }
333
334    /// Removes all versions from the history.
335    pub fn clear(&mut self) {
336        self.deltas.clear();
337    }
338}

354/// A read-only view of the history overlay on the tree at a specified place in the history.
355#[derive(Copy, Clone, Debug)]
356pub struct HistoryView<'history> {
357    /// The delta corresponding to this overlay.
358    delta: &'history Delta,
359}
360
361impl<'history> HistoryView<'history> {
362    /// Constructs a new history view that acts as a single overlay of the state represented by the
363    /// history at the provided `version`.
364    ///
365    /// # Complexity
366    ///
367    /// The computational complexity of this method is linear in the number of versions stored in
368    /// the history.
369    ///
370    /// # Errors
371    ///
372    /// - [`HistoryError::VersionTooOld`] if the history does not contain the data to provide a
373    ///   coherent overlay for the provided `version`.
374    fn new_of(version: VersionId, history: &'history History) -> Result<Self> {
375        let version_ix = history.find_latest_corresponding_version(version)?;
376        let delta = &history.deltas[version_ix];
377        Ok(Self { delta })
378    }
379
380    /// Gets the value of the node in the history at the provided `index`, or returns `None` if the
381    /// version does not overlay the current tree at that node.
382    #[must_use]
383    pub fn node_value(&self, index: &NodeIndex) -> Option<&Word> {
384        self.delta.nodes.get(index)
385    }
386
387    /// Queries the value of a specific `key` in a leaf in the overlay, returning the value for that
388    /// `key` if it has been changed, and [`None`] otherwise.
389    #[must_use]
390    pub fn value(self, key: &Word) -> Option<Word> {
391        self.delta.changed_keys.get(key).cloned()
392    }
393
394    /// Returns `true` if the key is removed by this delta, and `false` otherwise.
395    #[must_use]
396    pub fn is_key_removed(self, key: &Word) -> bool {
397        self.delta.changed_keys.get(key).map(Word::is_empty).unwrap_or(false)
398    }
399
400    /// Returns an iterator which yields the entries that are added by this view.
401    pub fn changed_keys(self) -> impl Iterator<Item = (Word, Word)> + 'history {
402        self.delta.changed_keys.iter().map(|(k, v)| (*k, *v))
403    }
404
405    /// Returns the total number of entries in the tree at this historical version.
406    #[must_use]
407    pub fn entry_count(self) -> usize {
408        self.delta.entry_count
409    }
410}

415/// A delta for a state `n` represents the changes (to both nodes and leaves) that need to be
416/// applied on top of the state `n + 1` to yield the correct tree for state `n`.
417///
418/// # Cumulative Deltas and Temporal Ordering
419///
420/// In order to best represent the history of a merkle tree, these deltas are constructed to take
421/// advantage of two main properties:
422///
423/// - They are _cumulative_, which reduces their practical memory usage. This does, however, mean
424///   that querying the state of older blocks is more expensive than querying newer ones.
425/// - Deltas are applied in **temporally reversed order** from what one might expect. Most
426///   conventional applications of deltas bring something from the past into the future through
427///   application. In our case, the application of one or more deltas moves the tree into a **past
428///   state**.
429///
430/// # Construction
431///
432/// While the [`Delta`] type is visible in the interface of the history, it is only intended to be
433/// constructed by the history. Users should not be allowed to construct it directly.
434#[derive(Clone, Debug, PartialEq)]
435struct Delta {
436    /// The root of the tree in the `version` corresponding to the application of the reversions in
437    /// this delta to the previous tree state.
438    root: RootValue,
439
440    /// The version of the tree represented by the delta.
441    version_id: VersionId,
442
443    /// Any changes to the non-leaf nodes in the tree for this delta.
444    nodes: NodeChanges,
445
446    /// The keys that need to be changed by the delta to return the target tree to the state
447    /// represented by the delta. This includes pairs that either add or mutate a value under a
448    /// key, and pairs where the value is the `EMPTY_WORD` and hence represent removals.
449    changed_keys: ChangedKeys,
450
451    /// The total number of entries that existed in the tree at the version represented by this
452    /// delta, stored eagerly to avoid recomputation.
453    entry_count: usize,
454}
455
456impl Delta {
457    /// Creates a new delta with the provided `root`, representing the provided changes to the
458    /// `nodes` in the merkle tree, using `changed_keys` to represent the changes to entries in
459    /// the tree, and storing `entry_count` as the total number of entries at this version.
460    #[must_use]
461    fn new(
462        root: RootValue,
463        version_id: VersionId,
464        nodes: NodeChanges,
465        changed_keys: ChangedKeys,
466        entry_count: usize,
467    ) -> Self {
468        Self {
469            root,
470            version_id,
471            nodes,
472            changed_keys,
473            entry_count,
474        }
475    }
