32/// A read-only, point-in-time snapshot of an [`InMemoryBackend`].
33///
34/// This type intentionally implements only [`BackendReader`], not [`Backend`]. It is returned by
35/// [`InMemoryBackend::reader`] to hand out a detached copy of the backend state without exposing
36/// any mutation capabilities.
37#[derive(Clone, Debug, Eq, PartialEq)]
38pub struct InMemoryBackendSnapshot(InMemoryBackend);
39
40impl BackendReader for InMemoryBackendSnapshot {
41    fn open(&self, lineage: LineageId, key: Word) -> Result<SmtProof> {
42        self.0.open(lineage, key)
43    }
44
45    fn get_leaf(&self, lineage: LineageId, leaf_index: LeafIndex<SMT_DEPTH>) -> Result<SmtLeaf> {
46        self.0.get_leaf(lineage, leaf_index)
47    }
48
49    fn get(&self, lineage: LineageId, key: Word) -> Result<Option<Word>> {
50        self.0.get(lineage, key)
51    }
52
53    fn version(&self, lineage: LineageId) -> Result<VersionId> {
54        self.0.version(lineage)
55    }
56
57    fn lineages(&self) -> Result<impl Iterator<Item = LineageId>> {
58        self.0.lineages()
59    }
60
61    fn trees(&self) -> Result<impl Iterator<Item = TreeWithRoot>> {
62        self.0.trees()
63    }
64
65    fn entry_count(&self, lineage: LineageId) -> Result<usize> {
66        self.0.entry_count(lineage)
67    }
68
69    fn entries(&self, lineage: LineageId) -> Result<impl Iterator<Item = Result<TreeEntry>>> {
70        self.0.entries(lineage)
71    }
72}

77/// The in-memory backend itself.
78#[derive(Clone, Debug, Eq, PartialEq)]
79pub struct InMemoryBackend {
80    /// The storage for the full trees that are stored in this backend, always guaranteed to be the
81    /// latest tree in the lineage.
82    trees: Map<LineageId, TreeData>,
83}
84
85/// Prepared mutations for [`InMemoryBackend`].
86///
87/// This is the in-memory backend's concrete [`Backend::PreparedMutations`] type. It stores the
88/// forward SMT mutation sets that were computed during the first phase of a forest update. Applying
89/// it mutates the in-memory trees directly without recomputing the update batches.
90///
91/// The fields are private because callers should treat prepared mutation data as opaque and pass it
92/// back through
93/// [`LargeSmtForest::apply_mutations`](crate::merkle::smt::LargeSmtForest::apply_mutations).
94#[derive(Clone, Debug, Eq, PartialEq)]
95pub struct InMemoryPreparedMutations {
96    entries: Vec<InMemoryPreparedLineageMutation>,
97}
98
99#[derive(Clone, Debug, Eq, PartialEq)]
100struct InMemoryPreparedLineageMutation {
101    lineage: LineageId,
102    old_version: Option<VersionId>,
103    version: VersionId,
104    forward: MutationSet,
105    kind: LineageMutationKind,
106}
107
108impl InMemoryBackend {
109    /// Constructs a new instance of the in-memory backend.
110    pub fn new() -> Self {
111        let trees = Map::default();
112        Self { trees }
113    }
114
115    /// Converts this backend into a read-only snapshot.
116    pub fn into_snapshot(self) -> InMemoryBackendSnapshot {
117        InMemoryBackendSnapshot(self)
118    }
119
120    fn mutation_from_tree(
121        lineage: LineageId,
122        old_version: Option<VersionId>,
123        new_version: VersionId,
124        kind: LineageMutationKind,
125        forward: MutationSet,
126    ) -> (LineageMutation, InMemoryPreparedLineageMutation) {
127        let old_root = forward.old_root();
128        let new_root = forward.root();
129
130        let mutation =
131            LineageMutation::new(lineage, old_version, new_version, old_root, new_root, kind);
132        let prepared = InMemoryPreparedLineageMutation {
133            lineage,
134            old_version,
135            version: new_version,
136            forward,
137            kind,
138        };
139
140        (mutation, prepared)
141    }
142}

147impl BackendReader for InMemoryBackend {
148    /// Returns an opening for the specified `key` in the SMT with the specified `lineage`.
149    ///
150    /// # Errors
151    ///
152    /// - [`BackendError::UnknownLineage`] if the provided `lineage` is one not known by this
153    ///   backend.
154    fn open(&self, lineage: LineageId, key: Word) -> Result<SmtProof> {
155        let tree = self.trees.get(&lineage).ok_or(BackendError::UnknownLineage(lineage))?;
156        Ok(tree.tree.open(&key))
157    }
158
159    /// Returns the leaf stored at `leaf_index` in the SMT with the specified `lineage`.
160    ///
161    /// If no leaf is explicitly stored at the given index, an empty leaf for that index is
162    /// returned.
163    ///
164    /// # Errors
165    ///
166    /// - [`BackendError::UnknownLineage`] if the provided `lineage` is one not known by this
167    ///   backend.
168    fn get_leaf(&self, lineage: LineageId, leaf_index: LeafIndex<SMT_DEPTH>) -> Result<SmtLeaf> {
169        let tree = self.trees.get(&lineage).ok_or(BackendError::UnknownLineage(lineage))?;
170        Ok(tree
171            .tree
172            .get_leaf_by_index(leaf_index)
173            .unwrap_or_else(|| SmtLeaf::new_empty(leaf_index)))
174    }
175
176    /// Returns the value associated with the provided `key` in the SMT with the specified
177    /// `lineage`, or [`None`] if no such value exists.
178    ///
179    /// # Errors
180    ///
181    /// - [`BackendError::UnknownLineage`] if the provided `lineage` is one not known by this
182    ///   backend.
183    fn get(&self, lineage: LineageId, key: Word) -> Result<Option<Word>> {
184        let tree = self.trees.get(&lineage).ok_or(BackendError::UnknownLineage(lineage))?;
185        let value = tree.tree.get_value(&key);
186        let value = if value == EMPTY_WORD { None } else { Some(value) };
187
188        Ok(value)
189    }
190
191    /// Returns the version of the tree with the specified `lineage`.
192    ///
193    /// # Errors
194    ///
195    /// - [`BackendError::UnknownLineage`] If the provided `lineage` is one not known by this
196    ///   backend.
197    fn version(&self, lineage: LineageId) -> Result<VersionId> {
198        let tree = self.trees.get(&lineage).ok_or(BackendError::UnknownLineage(lineage))?;
199        Ok(tree.version)
200    }
201
202    /// Returns an iterator over all the lineages that the backend knows about.
203    fn lineages(&self) -> Result<impl Iterator<Item = LineageId>> {
204        Ok(self.trees.keys().cloned())
205    }
206
207    /// Returns an iterator over all the trees that the backend knows about.
208    ///
209    /// The iteration order is unspecified.
210    fn trees(&self) -> Result<impl Iterator<Item = TreeWithRoot>> {
211        Ok(self.trees.iter().map(|(l, t)| TreeWithRoot::new(*l, t.version, t.tree.root())))
212    }
213
214    /// Returns the total number of (key-value) entries in the specified `tree`.
215    ///
216    /// # Errors
217    ///
218    /// - [`BackendError::UnknownLineage`] if the provided `lineage` is one not known by this
219    ///   backend.
220    fn entry_count(&self, lineage: LineageId) -> Result<usize> {
221        let tree = self.trees.get(&lineage).ok_or(BackendError::UnknownLineage(lineage))?;
222        Ok(tree.tree.num_entries())
223    }
224
225    /// Returns an iterator that yields the populated (key-value) entries for the specified
226    /// `lineage`.
227    ///
228    /// It yields entries in an arbitrary order, and never yields entries where the value is the
229    /// empty word.
230    ///
231    /// # Errors
232    ///
233    /// - [`BackendError::UnknownLineage`] if the provided `lineage` is one not known by this
234    ///   backend.
235    fn entries(&self, lineage: LineageId) -> Result<impl Iterator<Item = Result<TreeEntry>>> {
236        let tree = self.trees.get(&lineage).ok_or(BackendError::UnknownLineage(lineage))?;
237        Ok(tree.tree.entries().map(|(k, v)| Ok(TreeEntry { key: *k, value: *v })))
238    }
239}

244impl Backend for InMemoryBackend {
245    type Reader = InMemoryBackendSnapshot;
246    type PreparedMutations = InMemoryPreparedMutations;
247
248    fn reader(&self) -> Result<Self::Reader> {
249        Ok(self.clone().into_snapshot())
250    }
251
252    /// Computes the mutations required to add the provided `lineage` to the forest.
253    ///
254    /// # Errors
255    ///
256    /// - [`BackendError::DuplicateLineage`] if the provided `lineage` is the same as an
257    ///   already-known lineage. No data is changed in this case.
258    /// - [`BackendError::Merkle`] if the provided `updates` cannot be applied to the empty tree.
259    fn compute_add_lineage_mutations(
260        &self,
261        lineage: LineageId,
262        version: VersionId,
263        updates: SmtUpdateBatch,
264    ) -> Result<(Vec<LineageMutation>, Self::PreparedMutations)> {
265        if self.trees.contains_key(&lineage) {
266            return Err(BackendError::DuplicateLineage(lineage));
267        }
268
269        let tree = Smt::new();
270        let forward = tree.compute_mutations(updates.into_iter().map(Into::into))?;
271        let (mutation, prepared) = Self::mutation_from_tree(
272            lineage,
273            None,
274            version,
275            LineageMutationKind::AddLineage,
276            forward,
277        );
278
279        Ok((vec![mutation], InMemoryPreparedMutations { entries: vec![prepared] }))
280    }
281
282    /// Computes the mutations required to perform the provided `updates` on the tree with the
283    /// specified `lineage`, returning the mutation set.
284    ///
285    /// At most one new root is added to the backend for the entire batch.
286    ///
287    /// # Errors
288    ///
289    /// - [`BackendError::Merkle`] if an error occurs with the merkle tree semantics.
290    /// - [`BackendError::UnknownLineage`] if the provided `lineage` is not known by this backend.
291    fn compute_update_tree_mutations(
292        &self,
293        lineage: LineageId,
294        new_version: VersionId,
295        updates: SmtUpdateBatch,
296    ) -> Result<(Vec<LineageMutation>, Self::PreparedMutations)> {
297        let tree_data = self.trees.get(&lineage).ok_or(BackendError::UnknownLineage(lineage))?;
298        let forward = tree_data.tree.compute_mutations(updates.into_iter().map(Into::into))?;
299        let (mutation, prepared) = Self::mutation_from_tree(
300            lineage,
301            Some(tree_data.version),
302            new_version,
303            LineageMutationKind::UpdateTree,
304            forward,
305        );
306
307        Ok((vec![mutation], InMemoryPreparedMutations { entries: vec![prepared] }))
308    }
309
310    /// Computes the mutations required to add multiple new `lineages` to the tree, creating
311    /// an empty tree for each and applying the provided modifications to it, with the result
312    /// being given the specified `version`. Returns the mutation set.
313    ///
314    /// If the provide batch of modifications is empty for any given lineage, then the **empty tree
315    /// will be added** as the first version in that lineage.
316    ///
317    /// # Errors
318    ///
319    /// - [`BackendError::DuplicateLineage`] if any of the provided lineages already exists in the
320    ///   backend.
321    /// - [`BackendError::Merkle`] if an error occurs with the merkle tree semantics.
322    fn compute_add_lineages_mutations(
323        &self,
324        version: VersionId,
325        lineages: SmtForestUpdateBatch,
326    ) -> Result<(Vec<LineageMutation>, Self::PreparedMutations)> {
327        let updates = lineages
328            .into_iter()
329            .map(|(lineage, ops)| {
330                if self.trees.contains_key(&lineage) {
331                    return Err(BackendError::DuplicateLineage(lineage));
332                }
333                Ok((lineage, ops))
334            })
335            .collect::<Result<Vec<_>>>()?;
336
337        let mut mutations = Vec::with_capacity(updates.len());
338        let mut prepared = Vec::with_capacity(updates.len());
339        for (lineage, ops) in updates {
340            let tree = Smt::new();
341            let forward = tree.compute_mutations(ops.into_iter().map(Into::into))?;
342            let (mutation, prepared_entry) = Self::mutation_from_tree(
343                lineage,
344                None,
345                version,
346                LineageMutationKind::AddLineage,
347                forward,
348            );
349            mutations.push(mutation);
350            prepared.push(prepared_entry);
351        }
352
353        Ok((mutations, InMemoryPreparedMutations { entries: prepared }))
354    }
355
356    /// Computes the mutations required to apply the provided `updates` on the entire forest,
357    /// returning the mutation sets.
358    ///
359    /// The order of application of these mutations is unspecified.
360    ///
361    /// # Errors
362    ///
363    /// - [`BackendError::Merkle`] if an error occurs with the merkle tree semantics.
364    /// - [`BackendError::UnknownLineage`] if the provided `lineage` is not known by this backend.
365    fn compute_update_forest_mutations(
366        &self,
367        new_version: VersionId,
368        updates: SmtForestUpdateBatch,
369    ) -> Result<(Vec<LineageMutation>, Self::PreparedMutations)> {
370        let updates = updates
371            .into_iter()
372            .map(|(lineage, ops)| {
373                if !self.trees.contains_key(&lineage) {
374                    return Err(BackendError::UnknownLineage(lineage));
375                }
376
377                Ok((lineage, ops))
378            })
379            .collect::<Result<Vec<_>>>()?;
380
381        let mut mutations = Vec::with_capacity(updates.len());
382        let mut prepared = Vec::with_capacity(updates.len());
383        for (lineage, ops) in updates {
384            let tree_data = self.trees.get(&lineage).expect("Tree known to be present was not");
385            let forward = tree_data.tree.compute_mutations(ops.into_iter().map(Into::into))?;
386            let (mutation, prepared_entry) = Self::mutation_from_tree(
387                lineage,
388                Some(tree_data.version),
389                new_version,
390                LineageMutationKind::UpdateTree,
391                forward,
392            );
393            mutations.push(mutation);
394            prepared.push(prepared_entry);
395        }
396
397        Ok((mutations, InMemoryPreparedMutations { entries: prepared }))
398    }
399
400    /// Apply a mutation set to the entire forest, returning the mutation sets that would reverse
401    /// the changes to each lineage in the forest.
402    ///
403    /// - [`BackendError::Merkle`] if an error occurs with the merkle tree semantics.
404    /// - [`BackendError::UnknownLineage`] if the provided `lineage` is not known by this backend.
405    fn apply_mutations(
406        &mut self,
407        mutations: Self::PreparedMutations,
408    ) -> Result<Vec<AppliedLineageMutation>> {
409        // We start by checking that all lineages referred to in the `mutations` are valid,
410        // failing early with an error if need be.
411        for mutation in &mutations.entries {
412            match mutation.kind {
413                LineageMutationKind::AddLineage => {
414                    if self.trees.contains_key(&mutation.lineage) {
415                        return Err(BackendError::DuplicateLineage(mutation.lineage));
416                    }
417                },
418                LineageMutationKind::UpdateTree => {
419                    let tree_data = self
420                        .trees
421                        .get(&mutation.lineage)
422                        .ok_or(BackendError::UnknownLineage(mutation.lineage))?;
423
424                    if Some(tree_data.version) != mutation.old_version {
425                        return Err(BackendError::BadVersion {
426                            provided: mutation.old_version.unwrap_or_default(),
427                            latest: tree_data.version,
428                        });
429                    }
430
431                    let old_root = mutation.forward.old_root();
432                    let latest_root = tree_data.tree.root();
433                    if latest_root != old_root {
434                        return Err(MerkleError::ConflictingRoots {
435                            expected_root: old_root,
436                            actual_root: latest_root,
437                        }
438                        .into());
439                    }
440                },
441            }
442        }
443
444        let mut applied = Vec::with_capacity(mutations.entries.len());
445
446        // Apply mutations to each lineage.
447        for mutation in mutations.entries {
448            let old_root = mutation.forward.old_root();
449            let new_root = mutation.forward.root();
450            match mutation.kind {
451                LineageMutationKind::AddLineage => {
452                    let mut tree = Smt::new();
453                    let reverse = MutationSet::default();
454
455                    if !mutation.forward.is_empty() {
456                        tree.apply_mutations(mutation.forward)
457                            .map_err(BackendError::internal_from)?;
458                    }
459
460                    applied.push(AppliedLineageMutation::new(
461                        mutation.lineage,
462                        mutation.old_version,
463                        mutation.version,
464                        old_root,
465                        new_root,
466                        0,
467                        reverse,
468                        mutation.kind,
469                    ));
470                    self.trees
471                        .insert(mutation.lineage, TreeData { version: mutation.version, tree });
472                },
473                LineageMutationKind::UpdateTree => {
474                    let tree_data = self
475                        .trees
476                        .get_mut(&mutation.lineage)
477                        .ok_or(BackendError::UnknownLineage(mutation.lineage))?;
478                    let old_entry_count = tree_data.tree.num_entries();
479
480                    let reverse = if mutation.forward.is_empty() {
481                        mutation.forward
482                    } else {
483                        let reverse = tree_data
484                            .tree
485                            .apply_mutations_with_reversion(mutation.forward)
486                            .map_err(BackendError::internal_from)?;
487                        tree_data.version = mutation.version;
488                        reverse
489                    };
490
491                    applied.push(AppliedLineageMutation::new(
492                        mutation.lineage,
493                        mutation.old_version,
494                        mutation.version,
495                        old_root,
496                        new_root,
497                        old_entry_count,
498                        reverse,
499                        mutation.kind,
500                    ));
501                },
502            }
503        }
504
505        Ok(applied)
506    }
507}

509// These are the implementations of helper methods used by the backend tests.
510#[cfg(test)]
511impl InMemoryBackend {
512    /// Adds the provided `lineage` to the forest.
513    ///
514    /// # Errors
515    ///
516    /// - [`BackendError::DuplicateLineage`] if the provided `lineage` is the same as an
517    ///   already-known lineage. No data is changed in this case.
518    /// - [`BackendError::Merkle`] if the provided `updates` cannot be applied to the empty tree.
519    pub(crate) fn add_lineage(
520        &mut self,
521        lineage: LineageId,
522        version: VersionId,
523        updates: SmtUpdateBatch,
524    ) -> Result<TreeWithRoot> {
525        let (_mutations, persistent_mutations) =
526            self.compute_add_lineage_mutations(lineage, version, updates)?;
527
528        let mut applied_mutations = self.apply_mutations(persistent_mutations)?;
529        let applied_mutation = applied_mutations
530            .pop()
531            .expect("should have applied exactly one lineage mutation");
532
533        // Finally we just return the necessary metadata.
534        Ok(TreeWithRoot::new(lineage, version, applied_mutation.new_root()))
535    }
536
537    /// Performs the provided `updates` on the tree with the specified `lineage`, returning the
538    /// mutation set that will revert the changes made to the tree.
539    ///
540    /// At most one new root is added to the backend for the entire batch.
541    ///
542    /// # Errors
543    ///
544    /// - [`BackendError::Merkle`] if the application of `updates` to the tree fails for any reason.
545    /// - [`BackendError::UnknownLineage`] If the provided `lineage` is one not known by this
546    ///   backend.
547    pub(crate) fn update_tree(
548        &mut self,
549        lineage: LineageId,
550        new_version: VersionId,
551        updates: SmtUpdateBatch,
552    ) -> Result<MutationSet> {
553        let (_mutations, persistent_mutations) =
554            self.compute_update_tree_mutations(lineage, new_version, updates)?;
555
556        let mut applied_mutations = self.apply_mutations(persistent_mutations)?;
557        let applied_mutation = applied_mutations
558            .pop()
559            .expect("should have applied exactly one lineage mutation");
560
561        // We then just return the reversion set for the operations in question.
562        Ok(applied_mutation.into_reverse())
563    }
564
565    /// Adds multiple new `lineages` to the tree, creating an empty tree for each and applying the
566    /// provided modifications to it, with the result being given the specified `version`.
567    ///
568    /// If the provide batch of modifications is empty for any given lineage, then the **empty tree
569    /// will be added** as the first version in that lineage.
570    ///
571    /// # Errors
572    ///
573    /// - [`BackendError::DuplicateLineage`] if any provided lineage conflicts with an already-known
574    ///   lineage. No data is changed in this case.
575    /// - [`BackendError::Merkle`] if any of the provided updates cannot be applied on top of the
576    ///   empty tree.
577    pub(crate) fn add_lineages(
578        &mut self,
579        version: VersionId,
580        lineages: SmtForestUpdateBatch,
581    ) -> Result<Vec<(LineageId, TreeWithRoot)>> {
582        let (_mutations, persistent_mutations) =
583            self.compute_add_lineages_mutations(version, lineages)?;
584
585        let applied_mutations = self.apply_mutations(persistent_mutations)?;
586
587        // Build the return value from the applied mutations.
588        let results = applied_mutations
589            .into_iter()
590            .map(|applied_mutation| (applied_mutation.lineage(), applied_mutation.result()))
591            .collect();
592
593        Ok(results)
594    }
595
596    /// Performs the provided `updates` on the entire forest, returning the mutation sets that would
597    /// reverse the changes to each tree in the forest.
598    ///
599    /// The order of application of these mutations is unspecified.
600    ///
601    /// # Errors
602    ///
603    /// - [`BackendError::Merkle`] if any set of operations on any lineage in the batch fail for any
604    ///   reason.
605    /// - [`BackendError::UnknownLineage`] if any lineage in the `updates` is not known by the
606    ///   backend.
607    ///
608    /// # Panics
609    ///
610    /// - If a tree that has been checked to be present is not present upon later access.
611    pub(crate) fn update_forest(
612        &mut self,
613        new_version: VersionId,
614        updates: SmtForestUpdateBatch,
615    ) -> Result<Vec<(LineageId, MutationSet)>> {
616        let (_mutations, persistent_mutations) =
617            self.compute_update_forest_mutations(new_version, updates)?;
618
619        let applied_mutations = self.apply_mutations(persistent_mutations)?;
620
621        // Build the return value from the applied mutations.
622        let reversion_sets = applied_mutations
623            .into_iter()
624            .map(|applied_mutation| (applied_mutation.lineage(), applied_mutation.into_reverse()))
625            .collect();
626
627        Ok(reversion_sets)
628    }
629}

643/// A container for the data associated with the latest tree in a given lineage within the backend.
644#[derive(Clone, Debug, Eq, PartialEq)]
645struct TreeData {
646    version: VersionId,
647    tree: Smt,
648}
