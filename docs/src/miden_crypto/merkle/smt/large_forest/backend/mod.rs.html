31/// The read-only interface for the SMT forest storage backend.
32///
33/// This trait provides the query operations necessary to read the full trees that make up the
34/// forest. It is a supertrait of [`Backend`], which extends it with write operations.
35///
36/// # Backend Data Storage
37///
38/// Having a generic [`BackendReader`] provides no guarantees to the user about how it stores data
39/// and what patterns are used for data access under the hood. It is, however, guaranteed to store
40/// _only_ the data necessary to describe the latest state of each tree in the forest.
41///
42/// # Error Handling
43///
44/// We separate errors in backend implementations into two semantic categories:
45///
46/// 1. **User-Derived Errors:** These are errors that arise downstream of data provided by the user.
47///    These errors must be signaled by returning an [`Err`] variant with an appropriate error.
48/// 2. **Internal Errors:** These are errors that are not derived from data provided by the user.
49///    Signaling such an error is up to the implementation, but can be done through both panicking
50///    and returning the [`BackendError::Internal`] variant as appropriate. These **may leave the
51///    backend in an inconsistent state** as they are designed to effect program termination or
52///    perform it directly.
53///
54/// The only reason that [`BackendError::Internal`] exists is to allow certain failures to result in
55/// termination at the level of the _forest_ instead of the _backend_ as this can sometimes lead to
56/// cleaner logic. If this is not appropriate, a panic is a better option.
57///
58/// # Expected Behavior
59///
60/// The following behavior is expected of all methods in implementations of this trait:
61///
62/// - For any failure derived from user input (see _User-Derived Errors_ above), the data and the
63///   backend must be **left in a consistent state** when the error is returned to the caller.
64/// - Failures derived from user input (see _User-Derived Errors_ above) must be signaled to the
65///   caller by returning a variant of [`BackendError`] that is **not [`BackendError::Internal`]**.
66///   Methods may place additional constraints on which errors are used to signal certain failures.
67///   Such failures should not lead to data corruption of any persistent data.
68pub trait BackendReader
69where
70    Self: Debug,
71{
72    /// Returns an opening for the specified `key` in the SMT with the specified `lineage`.
73    ///
74    /// It is the responsibility of the forest to ensure lineage existence before querying the
75    /// backend. The backend must return an error if the lineage does not exist.
76    fn open(&self, lineage: LineageId, key: Word) -> Result<SmtProof>;
77
78    /// Returns the leaf stored at the provided `leaf_index` in the SMT with the specified
79    /// `lineage`. If no leaf is explicitly stored at the given index, the backend must return
80    /// an empty leaf for that index.
81    ///
82    /// It is the responsibility of the forest to ensure lineage existence before querying the
83    /// backend. The backend must return an error if the lineage does not exist.
84    fn get_leaf(&self, lineage: LineageId, leaf_index: LeafIndex<SMT_DEPTH>) -> Result<SmtLeaf>;
85
86    /// Returns the value associated with the provided `key` in the SMT with the specified
87    /// `lineage`, or [`None`] if no such value exists.
88    ///
89    /// It is the responsibility of the forest to ensure lineage existence before querying the
90    /// backend. The backend must return an error if the lineage does not exist.
91    fn get(&self, lineage: LineageId, key: Word) -> Result<Option<Word>>;
92
93    /// Returns the version of the tree with the specified `lineage`.
94    ///
95    /// It is the responsibility of the forest to ensure lineage existence before querying the
96    /// backend. The backend must return an error if the lineage does not exist.
97    fn version(&self, lineage: LineageId) -> Result<VersionId>;
98
99    /// Returns an iterator over all the lineages that the backend knows about.
100    ///
101    /// The iteration order is unspecified.
102    fn lineages(&self) -> Result<impl Iterator<Item = LineageId>>;
103
104    /// Returns an iterator over all the trees (and their corresponding roots) that the backend
105    /// knows about.
106    ///
107    /// The iteration order is unspecified.
108    fn trees(&self) -> Result<impl Iterator<Item = TreeWithRoot>>;
109
110    /// Returns the total number of (key-value) entries in the specified `lineage`.
111    ///
112    /// It is the responsibility of the forest to ensure lineage existence before querying the
113    /// backend. The backend must return an error if the lineage does not exist.
114    ///
115    /// # Expected Behavior
116    ///
117    /// Implementations must guarantee the following behavior in addition to the global invariants:
118    ///
119    /// - This method must be **cheap** to call, not requiring network or disk I/O to service the
120    ///   result. This usually implies in-memory caching of the data.
121    /// - This method must not return errors other than if the lineage does not exist.
122    fn entry_count(&self, lineage: LineageId) -> Result<usize>;
123
124    /// Returns an iterator that yields the populated (key-value) entries for the specified
125    /// `lineage`.
126    ///
127    /// It is the responsibility of the forest to ensure lineage existence before querying the
128    /// backend. The backend must return an error if the lineage does not exist.
129    ///
130    /// The iterator may yield entries in any arbitrary order, but must not yield entries for which
131    /// the value is the empty word.
132    ///
133    /// # Expected Behavior
134    ///
135    /// Implementations must guarantee the following behavior in addition to the global invariants:
136    ///
137    /// - If any kind of error occurs during iteration that should be signaled to the user, the
138    ///   iterator must return `Some(Err(...))`. The caller should stop iteration after receiving an
139    ///   error as the iterator state is no longer valid.
140    /// - `None` will be returned upon successful completion, or at any time after an error has been
141    ///   returned.
142    fn entries(&self, lineage: LineageId) -> Result<impl Iterator<Item = Result<TreeEntry>>>;
143}

148/// The full read-write interface for the SMT forest storage backend.
149///
150/// This trait extends [`BackendReader`] with mutation operations, allowing the forest to add new
151/// lineages and update existing ones.
152///
153/// # Implementation Contract
154///
155/// Method-level doc comments describe invariants that cannot be encoded in the type system.
156/// Implementations are responsible for upholding them.
157pub trait Backend: BackendReader {
158    /// The read-only view type returned by [`Self::reader`].
159    ///
160    /// The returned type implements [`BackendReader`] but not [`Backend`], providing a read-only
161    /// guarantee. Implementations may return either a point-in-time snapshot or a live view, but
162    /// the view must always reflect a consistent committed state (not partial writes). Holding the
163    /// reader must not block writes in any way.
164    type Reader: BackendReader;
165
166    /// Backend-specific data prepared during mutation computation and consumed during application.
167    ///
168    /// This type is intentionally opaque to forest users. Implementations should store enough
169    /// information here to apply the already-computed mutations without repeating the expensive
170    /// tree update computation.
171    ///
172    /// The prepared value must represent only prospective changes. Computing it must not change
173    /// the backend's committed state. It may contain ordinary SMT mutation sets, storage-level
174    /// updates, serialized values, or any other implementation-specific data needed to apply the
175    /// mutation efficiently later.
176    type PreparedMutations;
177
178    /// Returns a read-only view of this backend that observes its current state.
179    fn reader(&self) -> Result<Self::Reader>;
180
181    // TWO-PHASE MODIFIERS
182    // ============================================================================================
183
184    /// Computes the backend data required to mutate lineages, without applying it.
185    ///
186    /// # Expected Behavior
187    ///
188    /// Implementations must guarantee the following behavior in addition to the global invariants:
189    ///
190    /// - The backend's committed state must not change.
191    /// - Each unknown lineage in `updates` is treated as an addition from the empty tree.
192    /// - Each known lineage in `updates` is treated as an update to its latest tree.
193    /// - Each lineage in `updates` must produce at most one [`LineageMutation`].
194    /// - No-op lineage updates must not allocate new backend tree versions when applied.
195    /// - The prepared mutations must be applicable atomically by [`Self::apply_mutations`] where
196    ///   the backend supports atomic writes.
197    fn compute_mutations(
198        &self,
199        new_version: VersionId,
200        updates: SmtForestUpdateBatch,
201    ) -> Result<(Vec<LineageMutation>, Self::PreparedMutations)>;
202
203    /// Applies previously-computed backend mutations.
204    ///
205    /// This method consumes the opaque prepared data returned by one of the backend compute
206    /// methods. It commits the backend's latest-tree state and returns the applied lineage data
207    /// needed by [`crate::merkle::smt::LargeSmtForest`] to update forest-level lineage metadata and
208    /// history.
209    ///
210    /// # Expected Behavior
211    ///
212    /// Implementations must guarantee the following behavior in addition to the global invariants:
213    ///
214    /// - The prepared mutation data must still be applicable to the current backend state before
215    ///   any mutation is written. For updates, the current version and root must match the
216    ///   version/root captured during the compute phase. For additions, the lineage must still be
217    ///   absent.
218    /// - User-derived errors must leave the backend in a consistent committed state.
219    /// - If the prepared data contains multiple lineage updates, they should be committed
220    ///   atomically when the backend's storage engine supports atomic batched writes.
221    /// - The method must not recompute Merkle mutations from the original user updates; that work
222    ///   belongs to the compute methods.
223    /// - On success, the returned [`AppliedLineageMutation`] values must correspond to the applied
224    ///   prepared mutations in the same lineage set, including reverse mutations and old entry
225    ///   counts for update history.
226    fn apply_mutations(
227        &mut self,
228        mutations: Self::PreparedMutations,
229    ) -> Result<Vec<AppliedLineageMutation>>;
230}

235/// The error type for use within Backends.
236#[derive(Debug, Error)]
237pub enum BackendError {
238    /// Raised when an update was prepared against a version that is no longer current.
239    #[error("Version {provided} is not current backend version {latest}")]
240    BadVersion { provided: VersionId, latest: VersionId },
241
242    /// Raised when corrupted data is encountered in the backend.
243    ///
244    /// It exists as a separate error variant to allow the forest itself to handle it better if
245    /// possible, but should be considered a fatal error.
246    #[error("Backend data corruption encountered: {0}")]
247    CorruptedData(String),
248
249    /// Raised when there is a conflict between an existing lineage ID and one already in the
250    /// forest.
251    #[error("Duplicate lineage ID {0} provided")]
252    DuplicateLineage(LineageId),
253
254    /// Raised for arbitrary errors that are not derived from user-input. These should be considered
255    /// fatal by callers, but exist to forward the termination decision up to an appropriate level.
256    #[error(transparent)]
257    Internal(Box<dyn core::error::Error + Sync + Send>),
258
259    /// Raised when there is an error with the merkle tree semantics within the backend.
260    #[error(transparent)]
261    Merkle(#[from] MerkleError),
262
263    /// Raised for arbitrary other errors within the backend that are derived from user-input and
264    /// hence non-fatal.
265    #[error(transparent)]
266    Other(Box<dyn core::error::Error + Sync + Send>),
267
268    /// Raised when the backend is queried for a lineage it doesn't know about.
269    #[error("Lineage {0} is not known by the backend")]
270    UnknownLineage(LineageId),
271
272    /// Raised for other errors in the backend that are user-specified.
273    #[error("Unspecified error: {0}")]
274    Unspecified(String),
275}
276
277impl BackendError {
278    /// Constructs an internal error variant from the provided concrete error `e`.
279    fn internal_from<E: core::error::Error + Sync + Send + 'static>(e: E) -> Self {
280        Self::Internal(Box::new(e))
281    }
282
283    /// Constructs an internal error variant from the provided `message`.
284    #[cfg(feature = "persistent-forest")]
285    fn internal_from_message(message: impl Into<String>) -> Self {
286        Self::internal_from(Self::Unspecified(message.into()))
287    }
288}

290/// The result type for use with backends.
291pub type Result<T> = core::result::Result<T, BackendError>;
