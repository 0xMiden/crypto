18// MEMORY STORAGE
19// ================================================================================================
20
21/// In-memory storage for a Sparse Merkle Tree (SMT), implementing the `SmtStorage` trait.
22///
23/// This structure stores the SMT's leaf nodes and subtrees directly in memory.
24///
25/// It is primarily intended for scenarios where data persistence to disk is not a
26/// primary concern. Common use cases include:
27/// - Testing environments.
28/// - Managing SMT instances with a limited operational lifecycle.
29/// - Situations where a higher-level application architecture handles its own data persistence
30///   strategy.
31#[derive(Debug, Clone)]
32pub struct MemoryStorage {
33    pub leaves: Map<u64, SmtLeaf>,
34    pub subtrees: Map<NodeIndex, Subtree>,
35}
36
37impl MemoryStorage {
38    /// Creates a new, empty in-memory storage for a Sparse Merkle Tree.
39    ///
40    /// Initializes empty maps for leaves and subtrees.
41    pub fn new() -> Self {
42        Self { leaves: Map::new(), subtrees: Map::new() }
43    }
44
45    /// Converts this storage into a read-only snapshot.
46    pub fn into_snapshot(self) -> MemoryStorageSnapshot {
47        MemoryStorageSnapshot(self)
48    }
49}
50
51impl Default for MemoryStorage {
52    fn default() -> Self {
53        Self::new()
54    }
55}
56
57impl SmtStorageReader for MemoryStorage {
58    /// Gets the total number of non-empty leaves currently stored.
59    fn leaf_count(&self) -> StorageResult<usize> {
60        Ok(self.leaves.len())
61    }
62
63    /// Gets the total number of key-value entries currently stored.
64    fn entry_count(&self) -> StorageResult<usize> {
65        Ok(self.leaves.values().map(SmtLeaf::num_entries).sum())
66    }
67
68    /// Retrieves a single leaf node.
69    fn get_leaf(&self, index: u64) -> StorageResult<Option<SmtLeaf>> {
70        Ok(self.leaves.get(&index).cloned())
71    }
72
73    /// Retrieves multiple leaf nodes. Returns Ok(None) for indices not found.
74    fn get_leaves(&self, indices: &[u64]) -> StorageResult<Vec<Option<SmtLeaf>>> {
75        let leaves = indices.iter().map(|idx| self.leaves.get(idx).cloned()).collect();
76        Ok(leaves)
77    }
78
79    /// Returns true if the storage has any leaves.
80    fn has_leaves(&self) -> StorageResult<bool> {
81        Ok(!self.leaves.is_empty())
82    }
83
84    /// Retrieves a single Subtree (representing deep nodes) by its root NodeIndex.
85    /// Assumes index.depth() >= IN_MEMORY_DEPTH. Returns Ok(None) if not found.
86    fn get_subtree(&self, index: NodeIndex) -> StorageResult<Option<Subtree>> {
87        Ok(self.subtrees.get(&index).cloned())
88    }
89
90    /// Retrieves multiple Subtrees.
91    /// Assumes index.depth() >= IN_MEMORY_DEPTH for all indices. Returns Ok(None) for indices not
92    /// found.
93    fn get_subtrees(&self, indices: &[NodeIndex]) -> StorageResult<Vec<Option<Subtree>>> {
94        let subtrees: Vec<_> = indices.iter().map(|idx| self.subtrees.get(idx).cloned()).collect();
95        Ok(subtrees)
96    }
97
98    /// Retrieves a single inner node from a Subtree.
99    ///
100    /// This function is intended for accessing nodes within a Subtree, meaning
101    /// `index.depth()` must be greater than or equal to `IN_MEMORY_DEPTH`.
102    ///
103    /// # Errors
104    /// - `StorageError::Unsupported`: If `index.depth() < IN_MEMORY_DEPTH`.
105    ///
106    /// Returns `Ok(None)` if the subtree or the specific inner node within it is not found.
107    fn get_inner_node(&self, index: NodeIndex) -> StorageResult<Option<InnerNode>> {
108        if index.depth() < IN_MEMORY_DEPTH {
109            return Err(StorageError::Unsupported(
110                "Cannot get inner node from upper part of the tree".into(),
111            ));
112        }
113        let subtree_root_index = Subtree::find_subtree_root(index);
114        Ok(self
115            .subtrees
116            .get(&subtree_root_index)
117            .and_then(|subtree| subtree.get_inner_node(index)))
118    }
119
120    /// Returns an iterator over all (index, SmtLeaf) pairs in the storage.
121    ///
122    /// The iterator provides access to the current state of the leaves.
123    fn iter_leaves(
124        &self,
125    ) -> StorageResult<Box<dyn Iterator<Item = StorageResult<(u64, SmtLeaf)>> + '_>> {
126        Ok(Box::new(self.leaves.iter().map(|(&k, v)| Ok((k, v.clone())))))
127    }
128
129    /// Returns an iterator over all Subtrees in the storage.
130    ///
131    /// The iterator provides access to the current subtrees from storage.
132    fn iter_subtrees(
133        &self,
134    ) -> StorageResult<Box<dyn Iterator<Item = StorageResult<Subtree>> + '_>> {
135        Ok(Box::new(self.subtrees.values().cloned().map(Ok)))
136    }
137
138    /// Retrieves roots of all subtrees at `IN_MEMORY_DEPTH` depth.
139    ///
140    /// Derived from the subtrees already in memory: for each subtree whose root sits at
141    /// `IN_MEMORY_DEPTH`, the root node's hash is the entry that `initialize_from_storage`
142    /// needs to reconstruct the in-memory top of the tree.
143    fn get_top_subtree_roots(&self) -> StorageResult<Vec<(u64, Word)>> {
144        let in_mem_roots = self
145            .subtrees
146            .values()
147            .filter(|subtree| subtree.root_index().depth() == IN_MEMORY_DEPTH)
148            .filter_map(|subtree| {
149                subtree
150                    .get_inner_node(subtree.root_index())
151                    .map(|node| (subtree.root_index().position(), node.hash()))
152            })
153            .collect();
154        Ok(in_mem_roots)
155    }
156}

158impl SmtStorage for MemoryStorage {
159    type Reader = MemoryStorageSnapshot;
160
161    /// Returns a read-only snapshot of this in-memory storage by cloning it.
162    fn reader(&self) -> StorageResult<Self::Reader> {
163        Ok(self.clone().into_snapshot())
164    }
165
166    /// Inserts a key-value pair into the leaf at the given index.
167    ///
168    /// - If the leaf at `index` does not exist, a new `SmtLeaf::Single` is created.
169    /// - If the leaf exists, the key-value pair is inserted into it.
170    /// - Returns the previous value associated with the key, if any.
171    ///
172    /// # Panics
173    /// Panics in debug builds if `value` is `EMPTY_WORD`.
174    fn insert_value(&mut self, index: u64, key: Word, value: Word) -> StorageResult<Option<Word>> {
175        debug_assert_ne!(value, EMPTY_WORD);
176
177        match self.leaves.get_mut(&index) {
178            Some(leaf) => Ok(leaf.insert(key, value)?),
179            None => {
180                self.leaves.insert(index, SmtLeaf::Single((key, value)));
181                Ok(None)
182            },
183        }
184    }
185
186    /// Removes a key-value pair from the leaf at the given `index`.
187    ///
188    /// - If the leaf at `index` exists and the `key` is found within that leaf, the key-value pair
189    ///   is removed, and the old `Word` value is returned in `Ok(Some(Word))`.
190    /// - If the leaf at `index` exists but the `key` is not found within that leaf, `Ok(None)` is
191    ///   returned (as `leaf.get_value(&key)` would be `None`).
192    /// - If the leaf at `index` does not exist, `Ok(None)` is returned, as no value could be
193    ///   removed.
194    fn remove_value(&mut self, index: u64, key: Word) -> StorageResult<Option<Word>> {
195        let old_value = match self.leaves.entry(index) {
196            MapEntry::Occupied(mut entry) => {
197                let (old_value, is_empty) = entry.get_mut().remove(key);
198                if is_empty {
199                    entry.remove();
200                }
201                old_value
202            },
203            // Leaf at index does not exist, so no value could be removed.
204            MapEntry::Vacant(_) => None,
205        };
206        Ok(old_value)
207    }
208
209    /// Sets multiple leaf nodes in storage.
210    ///
211    /// If a leaf at a given index already exists, it is overwritten.
212    fn set_leaves(&mut self, leaves_map: Map<u64, SmtLeaf>) -> StorageResult<()> {
213        self.leaves.extend(leaves_map);
214        Ok(())
215    }
216
217    /// Removes a single leaf node.
218    fn remove_leaf(&mut self, index: u64) -> StorageResult<Option<SmtLeaf>> {
219        Ok(self.leaves.remove(&index))
220    }
221
222    /// Sets a single Subtree (representing deep nodes) by its root NodeIndex.
223    ///
224    /// If a subtree with the same root NodeIndex already exists, it is overwritten.
225    /// Assumes `subtree.root_index().depth() >= IN_MEMORY_DEPTH`.
226    fn set_subtree(&mut self, subtree: &Subtree) -> StorageResult<()> {
227        self.subtrees.insert(subtree.root_index(), subtree.clone());
228        Ok(())
229    }
230
231    /// Sets multiple Subtrees (representing deep nodes) by their root NodeIndex.
232    ///
233    /// If a subtree with a given root NodeIndex already exists, it is overwritten.
234    /// Assumes `subtree.root_index().depth() >= IN_MEMORY_DEPTH` for all subtrees in the vector.
235    fn set_subtrees(&mut self, subtrees_vec: Vec<Subtree>) -> StorageResult<()> {
236        self.subtrees
237            .extend(subtrees_vec.into_iter().map(|subtree| (subtree.root_index(), subtree)));
238        Ok(())
239    }
240
241    /// Removes a single Subtree (representing deep nodes) by its root NodeIndex.
242    fn remove_subtree(&mut self, index: NodeIndex) -> StorageResult<()> {
243        self.subtrees.remove(&index);
244        Ok(())
245    }
246
247    /// Sets a single inner node within a Subtree.
248    ///
249    /// - `index.depth()` must be greater than or equal to `IN_MEMORY_DEPTH`.
250    /// - If the target Subtree does not exist, it is created.
251    /// - The `node` is then inserted into the Subtree.
252    ///
253    /// Returns the `InnerNode` that was previously at this `index`, if any.
254    ///
255    /// # Errors
256    /// - `StorageError::Unsupported`: If `index.depth() < IN_MEMORY_DEPTH`.
257    fn set_inner_node(
258        &mut self,
259        index: NodeIndex,
260        node: InnerNode,
261    ) -> StorageResult<Option<InnerNode>> {
262        if index.depth() < IN_MEMORY_DEPTH {
263            return Err(StorageError::Unsupported(
264                "Cannot set inner node in upper part of the tree".into(),
265            ));
266        }
267        let subtree_root_index = Subtree::find_subtree_root(index);
268        let mut subtree = self
269            .subtrees
270            .remove(&subtree_root_index)
271            .unwrap_or_else(|| Subtree::new(subtree_root_index));
272        let old_node = subtree.insert_inner_node(index, node);
273        self.subtrees.insert(subtree_root_index, subtree);
274        Ok(old_node)
275    }
276
277    /// Removes a single inner node from a Subtree.
278    ///
279    /// - `index.depth()` must be greater than or equal to `IN_MEMORY_DEPTH`.
280    /// - If the Subtree becomes empty after removing the node, the Subtree itself is removed from
281    ///   storage.
282    ///
283    /// Returns the `InnerNode` that was removed, if any.
284    ///
285    /// # Errors
286    /// - `StorageError::Unsupported`: If `index.depth() < IN_MEMORY_DEPTH`.
287    fn remove_inner_node(&mut self, index: NodeIndex) -> StorageResult<Option<InnerNode>> {
288        if index.depth() < IN_MEMORY_DEPTH {
289            return Err(StorageError::Unsupported(
290                "Cannot remove inner node from upper part of the tree".into(),
291            ));
292        }
293        let subtree_root_index = Subtree::find_subtree_root(index);
294
295        let inner_node: Option<InnerNode> =
296            self.subtrees.remove(&subtree_root_index).and_then(|mut subtree| {
297                let old_node = subtree.remove_inner_node(index);
298                if !subtree.is_empty() {
299                    self.subtrees.insert(subtree_root_index, subtree);
300                }
301                old_node
302            });
303        Ok(inner_node)
304    }
305
306    /// Applies a set of updates atomically to the storage.
307    ///
308    /// This method handles updates to:
309    /// - Leaves: Inserts new or updated leaves, removes specified leaves.
310    /// - Subtrees: Inserts new or updated subtrees, removes specified subtrees.
311    fn apply(&mut self, updates: StorageUpdates) -> StorageResult<()> {
312        let StorageUpdateParts {
313            leaf_updates,
314            subtree_updates,
315            leaf_count_delta: _,
316            entry_count_delta: _,
317        } = updates.into_parts();
318
319        for (index, leaf_opt) in leaf_updates {
320            if let Some(leaf) = leaf_opt {
321                self.leaves.insert(index, leaf);
322            } else {
323                self.leaves.remove(&index);
324            }
325        }
326        for update in subtree_updates {
327            match update {
328                SubtreeUpdate::Store { index, subtree } => {
329                    self.subtrees.insert(index, subtree);
330                },
331                SubtreeUpdate::Delete { index } => {
332                    self.subtrees.remove(&index);
333                },
334            }
335        }
336        Ok(())
337    }
338}

343/// Read-only snapshot of SMT storage data.
344///
345/// This type intentionally implements [`SmtStorageReader`] only. It is used as the reader view for
346/// storage backends that need to hand out a detached point-in-time copy without also exposing
347/// mutation methods through [`SmtStorage`].
348#[derive(Debug, Clone)]
349pub struct MemoryStorageSnapshot(MemoryStorage);
350
351impl SmtStorageReader for MemoryStorageSnapshot {
352    fn leaf_count(&self) -> StorageResult<usize> {
353        self.0.leaf_count()
354    }
355
356    fn entry_count(&self) -> StorageResult<usize> {
357        self.0.entry_count()
358    }
359
360    fn get_leaf(&self, index: u64) -> StorageResult<Option<SmtLeaf>> {
361        self.0.get_leaf(index)
362    }
363
364    fn get_leaves(&self, indices: &[u64]) -> StorageResult<Vec<Option<SmtLeaf>>> {
365        self.0.get_leaves(indices)
366    }
367
368    fn has_leaves(&self) -> StorageResult<bool> {
369        self.0.has_leaves()
370    }
371
372    fn get_subtree(&self, index: NodeIndex) -> StorageResult<Option<Subtree>> {
373        self.0.get_subtree(index)
374    }
375
376    fn get_subtrees(&self, indices: &[NodeIndex]) -> StorageResult<Vec<Option<Subtree>>> {
377        self.0.get_subtrees(indices)
378    }
379
380    fn get_leaf_and_subtrees(
381        &self,
382        leaf_index: u64,
383        subtree_indices: &[NodeIndex],
384    ) -> StorageResult<(Option<SmtLeaf>, Vec<Option<Subtree>>)> {
385        self.0.get_leaf_and_subtrees(leaf_index, subtree_indices)
386    }
387
388    fn get_inner_node(&self, index: NodeIndex) -> StorageResult<Option<InnerNode>> {
389        self.0.get_inner_node(index)
390    }
391
392    fn iter_leaves(
393        &self,
394    ) -> StorageResult<Box<dyn Iterator<Item = StorageResult<(u64, SmtLeaf)>> + '_>> {
395        self.0.iter_leaves()
396    }
397
398    fn iter_subtrees(
399        &self,
400    ) -> StorageResult<Box<dyn Iterator<Item = StorageResult<Subtree>> + '_>> {
401        self.0.iter_subtrees()
402    }
403
404    fn get_top_subtree_roots(&self) -> StorageResult<Vec<(u64, Word)>> {
405        self.0.get_top_subtree_roots()
406    }
