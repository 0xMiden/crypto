58/// A RocksDB-backed persistent storage implementation for a Sparse Merkle Tree (SMT).
59///
60/// Implements the `SmtStorage` trait, providing durable storage for SMT components
61/// including leaves, subtrees (for deeper parts of the tree), and metadata like the SMT root
62/// and counts. It leverages RocksDB column families to organize data:
63/// - `LEAVES_CF` ("leaves"): Stores `SmtLeaf` data, keyed by their logical u64 index.
64/// - `SUBTREE_16_CF` ("st16"): Stores serialized `Subtree` data at depth 16, keyed by their root
65///   `NodeIndex`.
66/// - `SUBTREE_24_CF` ("st24"): Stores serialized `Subtree` data at depth 24, keyed by their root
67///   `NodeIndex`.
68/// - `SUBTREE_32_CF` ("st32"): Stores serialized `Subtree` data at depth 32, keyed by their root
69///   `NodeIndex`.
70/// - `SUBTREE_40_CF` ("st40"): Stores serialized `Subtree` data at depth 40, keyed by their root
71///   `NodeIndex`.
72/// - `SUBTREE_48_CF` ("st48"): Stores serialized `Subtree` data at depth 48, keyed by their root
73///   `NodeIndex`.
74/// - `SUBTREE_56_CF` ("st56"): Stores serialized `Subtree` data at depth 56, keyed by their root
75///   `NodeIndex`.
76/// - `METADATA_CF` ("metadata"): Stores overall SMT metadata such as the current root hash, total
77///   leaf count, and total entry count.
78#[derive(Debug, Clone)]
79pub struct RocksDbStorage {
80    db: Arc<DB>,
81    durability_mode: RocksDbDurabilityMode,
82}

85    /// Opens or creates a RocksDB database at the specified `path` and configures it for SMT
86    /// storage.
87    ///
88    /// This method sets up the necessary column families (`leaves`, `subtrees`, `metadata`)
89    /// and applies various RocksDB options for performance, such as caching, bloom filters,
90    /// and compaction strategies tailored for SMT workloads.
91    ///
92    /// The default profile uses:
93    /// - a 1 GiB block cache shared by this database's column families
94    /// - up to 512 open files
95    /// - 16 KiB block-based table blocks with cached index/filter blocks
96    /// - 128 MiB write buffers with up to 3 memtables per write-heavy column family
97    /// - LZ4 compression for active data and ZSTD for bottommost files
98    ///
99    /// # Errors
100    /// Returns `StorageError::Backend` if the database cannot be opened or configured,
101    /// for example, due to path issues, permissions, or RocksDB internal errors.
102    pub fn open(config: RocksDbConfig) -> StorageResult<Self> {
103        let tuning_options = &config.tuning_options;
104
105        // Base DB options
106        let mut db_opts = Options::default();
107        // Create DB if it doesn't exist
108        db_opts.create_if_missing(true);
109        // Auto-create missing column families
110        db_opts.create_missing_column_families(true);
111        // Tune compaction threads to match CPU cores
112        db_opts.increase_parallelism(rayon::current_num_threads() as i32);
113        // Limit the number of open file handles
114        db_opts.set_max_open_files(config.max_open_files);
115        // Parallelize flush/compaction up to CPU count
116        db_opts.set_max_background_jobs(rayon::current_num_threads() as i32);
117        // Maximum WAL size
118        db_opts.set_max_total_wal_size(tuning_options.max_total_wal_size);
119
120        // Cache and optional write-buffer manager are shared across this DB's column families.
121        let cache = Cache::new_lru_cache(config.cache_size);
122        let write_buffer_manager = config.write_buffer_manager(&cache);
123
124        // Common table options for bloom filtering and cache
125        let mut table_opts = BlockBasedOptions::default();
126        configure_block_table_options(
127            &mut table_opts,
128            &cache,
129            tuning_options,
130            tuning_options.bloom_filter_bits_per_key.leaves,
131        );
132
133        // Column family for leaves
134        let mut leaves_opts = Options::default();
135        leaves_opts.set_block_based_table_factory(&table_opts);
136        configure_smt_cf_options(&mut leaves_opts);
137        if let Some(wbm) = write_buffer_manager.as_ref() {
138            db_opts.set_write_buffer_manager(wbm);
139            leaves_opts.set_write_buffer_manager(wbm);
140        }
141
142        // Helper to build subtree CF options with the tuned block-table profile
143        #[expect(clippy::items_after_statements)]
144        fn subtree_cf(
145            cache: &Cache,
146            tuning_options: &RocksDbTuningOptions,
147            bloom_filter_bits: f64,
148            write_buffer_manager: Option<&WriteBufferManager>,
149        ) -> Options {
150            let mut table_opts = BlockBasedOptions::default();
151            configure_block_table_options(
152                &mut table_opts,
153                cache,
154                tuning_options,
155                bloom_filter_bits,
156            );
157
158            let mut opts = Options::default();
159            opts.set_block_based_table_factory(&table_opts);
160            configure_smt_cf_options(&mut opts);
161            if let Some(wbm) = write_buffer_manager {
162                opts.set_write_buffer_manager(wbm);
163            }
164            opts
165        }
166
167        // In-memory-depth cache column family (uses its own bloom filter setting)
168        let mut in_mem_depth_table_opts = BlockBasedOptions::default();
169        configure_block_table_options(
170            &mut in_mem_depth_table_opts,
171            &cache,
172            tuning_options,
173            tuning_options.bloom_filter_bits_per_key.in_mem_depth,
174        );
175
176        let mut in_mem_depth_opts = Options::default();
177        in_mem_depth_opts.set_compression_type(DBCompressionType::Lz4);
178        in_mem_depth_opts.set_bottommost_compression_type(DBCompressionType::Zstd);
179        // Enable the bottommost compression setting; selecting ZSTD alone is not enough.
180        in_mem_depth_opts
181            .set_bottommost_zstd_max_train_bytes(DEFAULT_BOTTOMMOST_ZSTD_MAX_TRAIN_BYTES, true);
182        in_mem_depth_opts.set_block_based_table_factory(&in_mem_depth_table_opts);
183        if let Some(wbm) = write_buffer_manager.as_ref() {
184            in_mem_depth_opts.set_write_buffer_manager(wbm);
185        }
186
187        // Metadata CF with no compression
188        let mut metadata_opts = Options::default();
189        metadata_opts.set_compression_type(DBCompressionType::None);
190        if let Some(wbm) = write_buffer_manager.as_ref() {
191            metadata_opts.set_write_buffer_manager(wbm);
192        }
193
194        let bloom = &tuning_options.bloom_filter_bits_per_key;
195
196        // Define column families with tailored options
197        let cfs = vec![
198            ColumnFamilyDescriptor::new(LEAVES_CF, leaves_opts),
199            ColumnFamilyDescriptor::new(
200                SUBTREE_16_CF,
201                subtree_cf(&cache, tuning_options, bloom.subtree_16, write_buffer_manager.as_ref()),
202            ),
203            ColumnFamilyDescriptor::new(
204                SUBTREE_24_CF,
205                subtree_cf(&cache, tuning_options, bloom.subtree_24, write_buffer_manager.as_ref()),
206            ),
207            ColumnFamilyDescriptor::new(
208                SUBTREE_32_CF,
209                subtree_cf(&cache, tuning_options, bloom.subtree_32, write_buffer_manager.as_ref()),
210            ),
211            ColumnFamilyDescriptor::new(
212                SUBTREE_40_CF,
213                subtree_cf(&cache, tuning_options, bloom.subtree_40, write_buffer_manager.as_ref()),
214            ),
215            ColumnFamilyDescriptor::new(
216                SUBTREE_48_CF,
217                subtree_cf(&cache, tuning_options, bloom.subtree_48, write_buffer_manager.as_ref()),
218            ),
219            ColumnFamilyDescriptor::new(
220                SUBTREE_56_CF,
221                subtree_cf(&cache, tuning_options, bloom.subtree_56, write_buffer_manager.as_ref()),
222            ),
223            ColumnFamilyDescriptor::new(METADATA_CF, metadata_opts),
224            ColumnFamilyDescriptor::new(IN_MEM_DEPTH_CF, in_mem_depth_opts),
225        ];
226
227        // Open the database with our tuned CFs
228        let db = DB::open_cf_descriptors(&db_opts, config.path, cfs)?;
229
230        Ok(Self {
231            db: Arc::new(db),
232            durability_mode: config.durability_mode,
233        })
234    }

236    /// Syncs the RocksDB database to disk.
237    ///
238    /// This ensures that all data is persisted to disk.
239    ///
240    /// # Errors
241    /// - Returns `StorageError::Backend` if the flush operation fails.
242    fn sync(&self) -> StorageResult<()> {
243        let mut fopts = FlushOptions::default();
244        fopts.set_wait(true);
245
246        for name in [
247            LEAVES_CF,
248            SUBTREE_16_CF,
249            SUBTREE_24_CF,
250            SUBTREE_32_CF,
251            SUBTREE_40_CF,
252            SUBTREE_48_CF,
253            SUBTREE_56_CF,
254            METADATA_CF,
255            IN_MEM_DEPTH_CF,
256        ] {
257            let cf = self.cf_handle(name)?;
258            self.db.flush_cf_opt(cf, &fopts)?;
259        }
260
261        self.db.flush_wal(true)?;
262        Ok(())
263    }

410    ///
411    /// This method groups requests by subtree depth into column family buckets,

439    /// Batch-retrieves multiple subtrees from RocksDB by their node indices.
440    ///
441    /// This method groups requests by subtree depth into column family buckets,
442    /// then performs parallel `multi_get` operations to efficiently retrieve
443    /// all subtrees. Results are deserialized and placed in the same order as
444    /// the input indices.
445    ///
446    /// # Parameters
447    /// - `indices`: A slice of subtree root indices to retrieve.
448    ///
449    /// # Returns
450    /// - A `Vec<Option<Subtree>>` where each index corresponds to the original input.
451    /// - `Ok(...)` if all fetches succeed.
452    /// - `Err(StorageError)` if any RocksDB access or deserialization fails.
453    fn get_subtrees(&self, indices: &[NodeIndex]) -> StorageResult<Vec<Option<Subtree>>> {
454        use p3_maybe_rayon::prelude::*;
455
456        let mut depth_buckets: [Vec<(usize, NodeIndex)>; 6] = Default::default();
457
458        for (original_index, &node_index) in indices.iter().enumerate() {
459            let depth = node_index.depth();
460            let bucket_index = match depth {
461                56 => 0,
462                48 => 1,
463                40 => 2,
464                32 => 3,
465                24 => 4,
466                16 => 5,
467                _ => {
468                    return Err(StorageError::Unsupported(format!(
469                        "unsupported subtree depth {depth}"
470                    )));
471                },
472            };
473            depth_buckets[bucket_index].push((original_index, node_index));
474        }
475        let mut results = vec![None; indices.len()];
476
477        // Process depth buckets in parallel
478        let bucket_results: StorageResult<Vec<_>> = depth_buckets
479            .into_par_iter()
480            .enumerate()
481            .filter(|(_, bucket)| !bucket.is_empty())
482            .map(|(bucket_index, bucket)| -> StorageResult<Vec<(usize, Option<Subtree>)>> {
483                let depth = LargeSmt::<RocksDbStorage>::SUBTREE_DEPTHS[bucket_index];
484                let cf = self.cf_handle(cf_for_depth(depth))?;
485                let keys: Vec<_> =
486                    bucket.iter().map(|(_, idx)| Self::subtree_db_key(*idx)).collect();
487
488                let db_results = self.db.multi_get_cf(keys.iter().map(|k| (cf, k.as_ref())));
489
490                // Process results for this bucket
491                bucket
492                    .into_iter()
493                    .zip(db_results)
494                    .map(|((original_index, node_index), db_result)| {
495                        let subtree = match db_result {
496                            Ok(Some(bytes)) => Some(Subtree::from_vec(node_index, &bytes)?),
497                            Ok(None) => None,
498                            Err(e) => return Err(e.into()),
499                        };
500                        Ok((original_index, subtree))
501                    })
502                    .collect()
503            })
504            .collect();
505
506        // Flatten results and place them in correct positions
507        for bucket_result in bucket_results? {
508            for (original_index, subtree) in bucket_result {
509                results[original_index] = subtree;
510            }
511        }
512
513        Ok(results)
514    }

557    /// Returns an iterator over all `Subtree` instances across all subtree column families.
558    ///
559    /// The iterator uses a RocksDB snapshot and iterates in lexicographical order of keys
560    /// (subtree root NodeIndex) across all depth column families (24, 32, 40, 48, 56).
561    /// Errors during iteration (e.g., deserialization issues) are returned as iterator items.
562    ///
563    /// # Errors
564    /// - `StorageError::Backend`: If any subtree column family is missing or a RocksDB error occurs
565    ///   during iterator creation.
566    fn iter_subtrees(
567        &self,
568    ) -> StorageResult<Box<dyn Iterator<Item = StorageResult<Subtree>> + '_>> {
569        // All subtree column family names in order
570        const SUBTREE_CFS: [&str; 6] = [
571            SUBTREE_16_CF,
572            SUBTREE_24_CF,
573            SUBTREE_32_CF,
574            SUBTREE_40_CF,
575            SUBTREE_48_CF,
576            SUBTREE_56_CF,
577        ];
578
579        let mut cf_handles = Vec::new();
580        for cf_name in SUBTREE_CFS {
581            cf_handles.push(self.cf_handle(cf_name)?);
582        }
583
584        Ok(Box::new(RocksDbSubtreeIterator::new(&self.db, cf_handles)))
585    }

587    /// Retrieves roots of all top level subtrees for efficient startup reconstruction.
588    ///
589    /// # Errors
590    /// - `StorageError::Backend`: If the in-memory-depth column family is missing or a RocksDB
591    ///   error occurs.
592    /// - `StorageError::Value`: If any hash bytes are corrupt.
593    fn get_top_subtree_roots(&self) -> StorageResult<Vec<(u64, Word)>> {
594        let cf = self.cf_handle(IN_MEM_DEPTH_CF)?;
595        let iter = self.db.iterator_cf(cf, IteratorMode::Start);
596        let mut hashes = Vec::new();
597
598        for item in iter {
599            let (key_bytes, value_bytes) = item?;
600
601            let index = index_from_key_bytes(&key_bytes)?;
602            let hash = Word::read_from_bytes_with_budget(&value_bytes, value_bytes.len())?;
603
604            hashes.push((index, hash));
605        }
606
607        Ok(hashes)
608    }

619    /// Inserts a key-value pair into the SMT leaf at the specified logical `index`.
620    ///
621    /// This operation involves:
622    /// 1. Retrieving the current leaf (if any) at `index`.
623    /// 2. Inserting the new key-value pair into the leaf.
624    /// 3. Updating the leaf and entry counts in the metadata column family.
625    /// 4. Writing all changes (leaf data, counts) to RocksDB in a single batch.
626    ///
627    /// Note: This only updates the leaf. Callers are responsible for recomputing and
628    /// persisting the corresponding inner nodes.
629    ///
630    /// # Errors
631    /// - `StorageError::Backend`: If column families are missing or a RocksDB error occurs.
632    /// - `StorageError::DeserializationError`: If existing leaf data is corrupt.
633    fn insert_value(&mut self, index: u64, key: Word, value: Word) -> StorageResult<Option<Word>> {
634        debug_assert_ne!(value, EMPTY_WORD);
635
636        let mut batch = WriteBatch::default();
637
638        // Fetch initial counts.
639        let mut current_leaf_count = self.leaf_count()?;
640        let mut current_entry_count = self.entry_count()?;
641
642        let leaves_cf = self.cf_handle(LEAVES_CF)?;
643        let db_key = Self::index_db_key(index);
644
645        let maybe_leaf = self.get_leaf(index)?;
646
647        let value_to_return: Option<Word> = match maybe_leaf {
648            Some(mut existing_leaf) => {
649                let old_value = existing_leaf.insert(key, value).expect("Failed to insert value");
650                // Determine if the overall SMT entry_count needs to change.
651                // entry_count increases if:
652                //   1. The key was not present in this leaf before (`old_value` is `None`).
653                //   2. The key was present but held `EMPTY_WORD` (`old_value` is
654                //      `Some(EMPTY_WORD)`).
655                if old_value.is_none_or(|old_v| old_v == EMPTY_WORD) {
656                    current_entry_count += 1;
657                }
658                // current_leaf_count does not change because the leaf itself already existed.
659                batch.put_cf(leaves_cf, db_key, existing_leaf.to_bytes());
660                old_value
661            },
662            None => {
663                // Leaf at `index` does not exist, so create a new one.
664                let new_leaf = SmtLeaf::Single((key, value));
665                // A new leaf is created.
666                current_leaf_count += 1;
667                // This new leaf contains one new SMT entry.
668                current_entry_count += 1;
669                batch.put_cf(leaves_cf, db_key, new_leaf.to_bytes());
670                // No previous value, as the leaf (and thus the key in it) was new.
671                None
672            },
673        };
674
675        // Add updated metadata counts to the batch.
676        let metadata_cf = self.cf_handle(METADATA_CF)?;
677        batch.put_cf(metadata_cf, LEAF_COUNT_KEY, current_leaf_count.to_be_bytes());
678        batch.put_cf(metadata_cf, ENTRY_COUNT_KEY, current_entry_count.to_be_bytes());
679
680        // Atomically write all changes (leaf data and metadata counts).
681        self.write_batch(batch)?;
682
683        Ok(value_to_return)
684    }

686    /// Removes a key-value pair from the SMT leaf at the specified logical `index`.
687    ///
688    /// This operation involves:
689    /// 1. Retrieving the leaf at `index`.
690    /// 2. Removing the `key` from the leaf. If the leaf becomes empty, it's deleted from RocksDB.
691    /// 3. Updating the leaf and entry counts in the metadata column family.
692    /// 4. Writing all changes (leaf data/deletion, counts) to RocksDB in a single batch.
693    ///
694    /// Returns `Ok(None)` if the leaf at `index` does not exist or the `key` is not found.
695    ///
696    /// Note: This only updates the leaf. Callers are responsible for recomputing and
697    /// persisting the corresponding inner nodes.
698    ///
699    /// # Errors
700    /// - `StorageError::Backend`: If column families are missing or a RocksDB error occurs.
701    /// - `StorageError::DeserializationError`: If existing leaf data is corrupt.
702    fn remove_value(&mut self, index: u64, key: Word) -> StorageResult<Option<Word>> {
703        let Some(mut leaf) = self.get_leaf(index)? else {
704            return Ok(None);
705        };
706
707        let mut batch = WriteBatch::default();
708        let cf = self.cf_handle(LEAVES_CF)?;
709        let metadata_cf = self.cf_handle(METADATA_CF)?;
710        let db_key = Self::index_db_key(index);
711        let mut entry_count = self.entry_count()?;
712        let mut leaf_count = self.leaf_count()?;
713
714        let (current_value, is_empty) = leaf.remove(key);
715        if let Some(current_value) = current_value
716            && current_value != EMPTY_WORD
717        {
718            entry_count -= 1;
719        }
720        if is_empty {
721            leaf_count -= 1;
722            batch.delete_cf(cf, db_key);
723        } else {
724            batch.put_cf(cf, db_key, leaf.to_bytes());
725        }
726        batch.put_cf(metadata_cf, LEAF_COUNT_KEY, leaf_count.to_be_bytes());
727        batch.put_cf(metadata_cf, ENTRY_COUNT_KEY, entry_count.to_be_bytes());
728        self.write_batch(batch)?;
729        Ok(current_value)
730    }

732    /// Sets or updates multiple SMT leaf nodes in the `LEAVES_CF` column family.
733    ///
734    /// This method performs a batch write to RocksDB. It also updates the global
735    /// leaf and entry counts in the `METADATA_CF` based on the provided `leaves` map,
736    /// overwriting any previous counts.
737    ///
738    /// Note: This method assumes the provided `leaves` map represents the entirety
739    /// of leaves to be stored or that counts are being explicitly reset.
740    /// Note: This only updates the leaves. Callers are responsible for recomputing and
741    /// persisting the corresponding inner nodes.
742    ///
743    /// # Errors
744    /// - `StorageError::Backend`: If column families are missing or a RocksDB error occurs.
745    fn set_leaves(&mut self, leaves: Map<u64, SmtLeaf>) -> StorageResult<()> {
746        let cf = self.cf_handle(LEAVES_CF)?;
747        let leaf_count: usize = leaves.len();
748        let entry_count: usize = leaves.values().map(|leaf| leaf.entries().len()).sum();
749        let mut batch = WriteBatch::default();
750        for (idx, leaf) in leaves {
751            let key = Self::index_db_key(idx);
752            let value = leaf.to_bytes();
753            batch.put_cf(cf, key, &value);
754        }
755        let metadata_cf = self.cf_handle(METADATA_CF)?;
756        batch.put_cf(metadata_cf, LEAF_COUNT_KEY, leaf_count.to_be_bytes());
757        batch.put_cf(metadata_cf, ENTRY_COUNT_KEY, entry_count.to_be_bytes());
758        self.write_batch(batch)?;
759        Ok(())
760    }

763    ///
764    /// Important: This method currently *does not* update the global leaf and entry counts
765    /// in the metadata. Callers are responsible for managing these counts separately
766    /// if using this method directly, or preferably use `apply` or `remove_value` which handle
767    /// counts.
768    ///
769    /// Note: This only removes the leaf. Callers are responsible for recomputing and
770    /// persisting the corresponding inner nodes.
771    ///
772    /// # Errors
773    /// - `StorageError::Backend`: If the leaves column family is missing or a RocksDB error occurs.
774    /// - `StorageError::DeserializationError`: If the retrieved (to be returned) leaf data is
775    ///   corrupt.
776    fn remove_leaf(&mut self, index: u64) -> StorageResult<Option<SmtLeaf>> {
777        let key = Self::index_db_key(index);

784            SmtLeaf::read_from_bytes_with_budget(&bytes, bytes.len())

786        }))

789    /// Stores a single subtree in RocksDB and optionally updates the in-memory-depth root cache.
790    ///
791    /// The subtree is serialized and written to its corresponding column family.
792    /// If it’s an in-memory-depth subtree, the root node’s hash is also stored in the
793    /// dedicated `IN_MEM_DEPTH_CF` cache to support top-level reconstruction.
794    ///
795    /// # Parameters
796    /// - `subtree`: A reference to the subtree to be stored.
797    ///
798    /// # Errors
799    /// - Returns `StorageError` if column family lookup, serialization, or the write operation
800    ///   fails.
801    fn set_subtree(&mut self, subtree: &Subtree) -> StorageResult<()> {
802        let subtrees_cf = self.subtree_cf(subtree.root_index());
803        let mut batch = WriteBatch::default();
804
805        let key = Self::subtree_db_key(subtree.root_index());
806        let value = subtree.to_vec();
807        batch.put_cf(subtrees_cf, key, value);
808
809        // Also update in-memory-depth hash cache if this is an in-memory-depth subtree
810        if subtree.root_index().depth() == IN_MEMORY_DEPTH {
811            let root_hash = subtree
812                .get_inner_node(subtree.root_index())
813                .ok_or_else(|| StorageError::Unsupported("Subtree root node not found".into()))?
814                .hash();
815
816            let in_mem_depth_cf = self.cf_handle(IN_MEM_DEPTH_CF)?;
817            let hash_key = Self::index_db_key(subtree.root_index().position());
818            batch.put_cf(in_mem_depth_cf, hash_key, root_hash.to_bytes());
819        }
820
821        self.write_batch(batch)?;
822        Ok(())
823    }

835    /// - `subtrees`: A vector of `Subtree` objects to be serialized and persisted.

845            let key = Self::subtree_db_key(subtree.root_index());

852                let hash_key = Self::index_db_key(subtree.root_index().position());
853                batch.put_cf(in_mem_depth_cf, hash_key, root_node.hash().to_bytes());
854            }
855        }
856
857        self.write_batch(batch)?;
858        Ok(())
859    }

886    /// This method is intended for `index.depth() >= IN_MEMORY_DEPTH`.
887    /// If the target Subtree does not exist, it is created. The `node` is then
888    /// inserted into the Subtree, and the modified Subtree is written back to storage.
889    ///
890    /// # Errors
891    /// - `StorageError::Backend`: If `index.depth() < IN_MEMORY_DEPTH`, or if RocksDB errors occur.
892    /// - `StorageError::Value`: If existing Subtree data is corrupt.
893    fn set_inner_node(
894        &mut self,
895        index: NodeIndex,
896        node: InnerNode,
897    ) -> StorageResult<Option<InnerNode>> {
898        if index.depth() < IN_MEMORY_DEPTH {
899            return Err(StorageError::Unsupported(
900                "Cannot set inner node in upper part of the tree".into(),
901            ));
902        }
903
904        let subtree_root_index = Subtree::find_subtree_root(index);
905        let mut subtree = self
906            .get_subtree(subtree_root_index)?
907            .unwrap_or_else(|| Subtree::new(subtree_root_index));
908        let old_node = subtree.insert_inner_node(index, node);
909        self.set_subtree(&subtree)?;
910        Ok(old_node)

913    /// Removes a single inner node (non-leaf node) from within a Subtree.
914    ///
915    /// This method is intended for `index.depth() >= IN_MEMORY_DEPTH`.
916    /// If the Subtree becomes empty after removing the node, the Subtree itself
917    /// is removed from storage.
918    ///
919    /// # Errors
920    /// - `StorageError::Backend`: If `index.depth() < IN_MEMORY_DEPTH`, or if RocksDB errors occur.
921    /// - `StorageError::Value`: If existing Subtree data is corrupt.
922    fn remove_inner_node(&mut self, index: NodeIndex) -> StorageResult<Option<InnerNode>> {
923        if index.depth() < IN_MEMORY_DEPTH {
924            return Err(StorageError::Unsupported(
925                "Cannot remove inner node from upper part of the tree".into(),
926            ));
927        }
928
929        let subtree_root_index = Subtree::find_subtree_root(index);
930        self.get_subtree(subtree_root_index)
931            .and_then(|maybe_subtree| match maybe_subtree {
932                Some(mut subtree) => {
933                    let old_node = subtree.remove_inner_node(index);
934                    let db_operation_result = if subtree.is_empty() {
935                        self.remove_subtree(subtree_root_index)
936                    } else {
937                        self.set_subtree(&subtree)
938                    };
939                    db_operation_result.map(|_| old_node)
940                },
941                None => Ok(None),
942            })
943    }

945    /// Applies a batch of `StorageUpdates` atomically to the RocksDB backend.
946    ///
947    /// This is the primary method for persisting changes to the SMT. It constructs a single
948    /// RocksDB `WriteBatch` containing all specified changes:
949    /// - Leaf updates/deletions in `LEAVES_CF`.
950    /// - Subtree updates/deletions in `SUBTREE_24_CF`, `SUBTREE_32_CF`, `SUBTREE_40_CF`,
951    ///   `SUBTREE_48_CF`, `SUBTREE_56_CF`.
952    /// - Updates to leaf and entry counts in `METADATA_CF` based on `leaf_count_delta` and
953    ///   `entry_count_delta`.
954    ///
955    /// All operations in the batch are applied atomically by RocksDB.
956    ///
957    /// # Errors
958    /// - `StorageError::Backend`: If any column family is missing or a RocksDB write error occurs.
959    fn apply(&mut self, updates: StorageUpdates) -> StorageResult<()> {
960        use p3_maybe_rayon::prelude::*;
961
962        let mut batch = WriteBatch::default();
963
964        let leaves_cf = self.cf_handle(LEAVES_CF)?;
965        let metadata_cf = self.cf_handle(METADATA_CF)?;
966        let in_mem_depth_cf = self.cf_handle(IN_MEM_DEPTH_CF)?;
967
968        let StorageUpdateParts {
969            leaf_updates,
970            subtree_updates,
971            leaf_count_delta,
972            entry_count_delta,
973        } = updates.into_parts();
974
975        // Process leaf updates
976        for (index, maybe_leaf) in leaf_updates {
977            let key = Self::index_db_key(index);
978            match maybe_leaf {
979                Some(leaf) => batch.put_cf(leaves_cf, key, leaf.to_bytes()),
980                None => batch.delete_cf(leaves_cf, key),
981            }
982        }
983
984        // Helper for in-memory-depth operations
985        let is_in_mem_depth = |index: NodeIndex| index.depth() == IN_MEMORY_DEPTH;
986
987        // Parallel preparation of subtree operations
988        let subtree_ops: StorageResult<Vec<_>> = subtree_updates
989            .into_par_iter()
990            .map(|update| -> StorageResult<_> {
991                let (index, maybe_bytes, in_mem_depth_op) = match update {
992                    SubtreeUpdate::Store { index, subtree } => {
993                        let bytes = subtree.to_vec();
994                        let in_mem_depth_op = is_in_mem_depth(index)
995                            .then(|| subtree.get_inner_node(index))
996                            .flatten()
997                            .map(|root_node| {
998                                let hash_key = Self::index_db_key(index.position());
999                                (hash_key, Some(root_node.hash().to_bytes()))
1000                            });
1001                        (index, Some(bytes), in_mem_depth_op)
1002                    },
1003                    SubtreeUpdate::Delete { index } => {
1004                        let in_mem_depth_op = is_in_mem_depth(index).then(|| {
1005                            let hash_key = Self::index_db_key(index.position());
1006                            (hash_key, None)
1007                        });
1008                        (index, None, in_mem_depth_op)
1009                    },
1010                };
1011
1012                let key = Self::subtree_db_key(index);
1013                let subtrees_cf = self.subtree_cf(index);
1014
1015                Ok((subtrees_cf, key, maybe_bytes, in_mem_depth_op))
1016            })
1017            .collect();
1018
1019        // Sequential batch building
1020        for (subtrees_cf, key, maybe_bytes, in_mem_depth_op) in subtree_ops? {
1021            match maybe_bytes {
1022                Some(bytes) => batch.put_cf(subtrees_cf, key, bytes),
1023                None => batch.delete_cf(subtrees_cf, key),
1024            }
1025
1026            if let Some((hash_key, maybe_hash_bytes)) = in_mem_depth_op {
1027                match maybe_hash_bytes {
1028                    Some(hash_bytes) => batch.put_cf(in_mem_depth_cf, hash_key, hash_bytes),
1029                    None => batch.delete_cf(in_mem_depth_cf, hash_key),
1030                }
1031            }
1032        }
1033
1034        if leaf_count_delta != 0 || entry_count_delta != 0 {
1035            let current_leaf_count = self.leaf_count()?;
1036            let current_entry_count = self.entry_count()?;
1037
1038            let new_leaf_count = current_leaf_count.saturating_add_signed(leaf_count_delta);
1039            let new_entry_count = current_entry_count.saturating_add_signed(entry_count_delta);
1040
1041            batch.put_cf(metadata_cf, LEAF_COUNT_KEY, new_leaf_count.to_be_bytes());
1042            batch.put_cf(metadata_cf, ENTRY_COUNT_KEY, new_entry_count.to_be_bytes());
1043        }
1044
1045        self.write_batch(batch)?;
1046
1047        Ok(())
1048    }
1049}

1090/// An iterator over subtrees from multiple RocksDB column families.
1091///
1092/// Iterates through all subtree column families (24, 32, 40, 48, 56) sequentially.
1093/// When one column family is exhausted, it moves to the next one.
1094struct RocksDbSubtreeIterator<'a> {
1095    db: &'a DB,
1096    cf_handles: Vec<&'a rocksdb::ColumnFamily>,
1097    current_cf_index: usize,
1098    current_iter: Option<DBIteratorWithThreadMode<'a, DB>>,
1099}
1100
1101impl<'a> RocksDbSubtreeIterator<'a> {
1102    fn new(db: &'a DB, cf_handles: Vec<&'a rocksdb::ColumnFamily>) -> Self {
1103        let mut iterator = Self {
1104            db,
1105            cf_handles,
1106            current_cf_index: 0,
1107            current_iter: None,
1108        };
1109        iterator.advance_to_next_cf();
1110        iterator
1111    }
1112
1113    fn advance_to_next_cf(&mut self) {
1114        if self.current_cf_index < self.cf_handles.len() {
1115            let cf = self.cf_handles[self.current_cf_index];
1116            let mut read_opts = ReadOptions::default();
1117            read_opts.set_total_order_seek(true);
1118            self.current_iter = Some(self.db.iterator_cf_opt(cf, read_opts, IteratorMode::Start));
1119        } else {
1120            self.current_iter = None;
1121        }
1122    }
1123
1124    fn next_from_iter(
1125        iter: &mut DBIteratorWithThreadMode<DB>,
1126        cf_index: usize,
1127    ) -> Option<StorageResult<Subtree>> {
1128        iter.next().map(|result| {
1129            result.map_err(StorageError::from).and_then(|(key_bytes, value_bytes)| {
1130                let depth = IN_MEMORY_DEPTH + (cf_index * 8) as u8;
1131
1132                let node_idx = subtree_root_from_key_bytes(&key_bytes, depth)?;
1133                let value_vec = value_bytes.into_vec();
1134                Ok(Subtree::from_vec(node_idx, &value_vec)?)
1135            })
1136        })
1137    }
1138}

1140impl Iterator for RocksDbSubtreeIterator<'_> {
1141    type Item = StorageResult<Subtree>;
1142
1143    fn next(&mut self) -> Option<Self::Item> {
1144        loop {
1145            let iter = self.current_iter.as_mut()?;
1146
1147            // Try to get the next valid subtree from current iterator
1148            if let Some(result) = Self::next_from_iter(iter, self.current_cf_index) {
1149                return Some(result);
1150            }
1151
1152            // Current CF exhausted, advance to next
1153            self.current_cf_index += 1;
1154            self.advance_to_next_cf();
1155
1156            // If no more CFs, we're done
1157            self.current_iter.as_ref()?;
1158        }
1159    }
1160}

1471/// Deserializes an index (u64) from a RocksDB key byte slice.
1472/// Expects `key_bytes` to be exactly 8 bytes long.
1473///
1474/// # Errors
1475/// - `StorageError::BadKeyLen`: If `key_bytes` is not 8 bytes long or conversion fails.
1476fn index_from_key_bytes(key_bytes: &[u8]) -> StorageResult<u64> {
1477    if key_bytes.len() != 8 {
1478        return Err(StorageError::BadKeyLen { expected: 8, found: key_bytes.len() });
1479    }
1480    let mut arr = [0u8; 8];
1481    arr.copy_from_slice(key_bytes);
1482    Ok(u64::from_be_bytes(arr))
1483}

1485fn read_count(what: &'static str, bytes: &[u8]) -> StorageResult<usize> {

1487        what,
1488        expected: 8,
1489        found: bytes.len(),

1492}

1511/// Reconstructs a `NodeIndex` from the variable-length subtree key stored in RocksDB.
1512///
1513/// * `key_bytes` is the big-endian tail of the 64-bit value:
1514///   - depth 56 → 7 bytes
1515///   - depth 48 → 6 bytes
1516///   - depth 40 → 5 bytes
1517///   - depth 32 → 4 bytes
1518///   - depth 24 → 3 bytes
1519///   - depth 16 → 2 bytes
1520///
1521/// # Errors
1522/// * `StorageError::Unsupported` -  `depth` is not one of 24/32/40/48/56.
1523/// * `StorageError::DeserializationError` - `key_bytes.len()` does not match the length required by
1524///   `depth`.
1525#[inline(always)]
1526fn subtree_root_from_key_bytes(key_bytes: &[u8], depth: u8) -> StorageResult<NodeIndex> {
1527    let expected = match depth {
1528        16 => 2,
1529        24 => 3,
1530        32 => 4,
1531        40 => 5,
1532        48 => 6,
1533        56 => 7,
1534        d => return Err(StorageError::Unsupported(format!("unsupported subtree depth {d}"))),
1535    };
1536
1537    if key_bytes.len() != expected {
1538        return Err(StorageError::BadSubtreeKeyLen { depth, expected, found: key_bytes.len() });
1539    }
1540    let mut buf = [0u8; 8];
1541    buf[8 - expected..].copy_from_slice(key_bytes);
1542    let value = u64::from_be_bytes(buf);
1543    Ok(NodeIndex::new_unchecked(depth, value))
1544}

1581/// Owns a RocksDB snapshot together with the database it borrows from.
1582///
1583/// `rocksdb::Snapshot<'a>` borrows the database used to create it. This type stores an `Arc<DB>`
1584/// beside the snapshot and releases the snapshot before the `Arc` is dropped, so the borrowed

1591impl RocksDbSnapshotInner {

1595        // `db`. `RocksDbSnapshotInner` keeps that `Arc<DB>` alive and its `Drop` implementation
1596        // manually releases the snapshot before the `Arc<DB>` field is dropped.
1597        let snapshot = unsafe {
1598            core::mem::transmute::<rocksdb::Snapshot<'_>, rocksdb::Snapshot<'static>>(snapshot)

1601            snapshot: ManuallyDrop::new(snapshot),

1603        }

1610        // dropped exactly once here, before `db` is dropped by Rust's normal field cleanup.
1611        unsafe {
1612            ManuallyDrop::drop(&mut self.snapshot);

1614    }

1726                        "unsupported subtree depth {depth}"
1727                    )));
1728                },
1729            };
1730            depth_buckets[bucket_index].push((original_index, node_index));
1731        }
1732        let mut results = vec![None; indices.len()];
1733
1734        let bucket_results: StorageResult<Vec<_>> = depth_buckets
1735            .into_par_iter()
1736            .enumerate()
1737            .filter(|(_, bucket)| !bucket.is_empty())
1738            .map(|(bucket_index, bucket)| -> StorageResult<Vec<(usize, Option<Subtree>)>> {
1739                let depth = LargeSmt::<RocksDbStorage>::SUBTREE_DEPTHS[bucket_index];
1740                let cf = self.cf_handle(cf_for_depth(depth))?;
1741                let keys: Vec<_> =
1742                    bucket.iter().map(|(_, idx)| RocksDbStorage::subtree_db_key(*idx)).collect();
1743
1744                let db_results =
1745                    self.inner.snapshot.multi_get_cf(keys.iter().map(|k| (cf, k.as_ref())));
1746
1747                bucket
1748                    .into_iter()
1749                    .zip(db_results)
1750                    .map(|((original_index, node_index), db_result)| {
1751                        let subtree = match db_result {
1752                            Ok(Some(bytes)) => Some(Subtree::from_vec(node_index, &bytes)?),
1753                            Ok(None) => None,
1754                            Err(e) => return Err(e.into()),
1755                        };
1756                        Ok((original_index, subtree))
1757                    })
1758                    .collect()
1759            })
1760            .collect();
1761
1762        for bucket_result in bucket_results? {
1763            for (original_index, subtree) in bucket_result {
1764                results[original_index] = subtree;
1765            }
1766        }
1767
1768        Ok(results)
1769    }

1771    /// Retrieves a single inner node from within a snapshot subtree.
1772    fn get_inner_node(&self, index: NodeIndex) -> StorageResult<Option<InnerNode>> {
1773        if index.depth() < IN_MEMORY_DEPTH {
1774            return Err(StorageError::Unsupported(
1775                "Cannot get inner node from upper part of the tree".into(),
1776            ));
1777        }
1778        let subtree_root_index = Subtree::find_subtree_root(index);
1779        Ok(self
1780            .get_subtree(subtree_root_index)?
1781            .and_then(|subtree| subtree.get_inner_node(index)))
1782    }

1825/// An iterator over subtrees from multiple RocksDB column families in a single snapshot.
1826struct RocksDbSnapshotSubtreeIterator<'a> {
1827    snapshot: &'a rocksdb::Snapshot<'static>,
1828    cf_handles: Vec<&'a rocksdb::ColumnFamily>,
1829    current_cf_index: usize,
1830    current_iter: Option<DBIteratorWithThreadMode<'a, DB>>,
1831}
1832
1833impl<'a> RocksDbSnapshotSubtreeIterator<'a> {
1834    fn new(
1835        snapshot: &'a rocksdb::Snapshot<'static>,
1836        cf_handles: Vec<&'a rocksdb::ColumnFamily>,
1837    ) -> Self {
1838        let mut iterator = Self {
1839            snapshot,
1840            cf_handles,
1841            current_cf_index: 0,
1842            current_iter: None,
1843        };
1844        iterator.advance_to_next_cf();
1845        iterator
1846    }
1847
1848    fn advance_to_next_cf(&mut self) {
1849        if self.current_cf_index < self.cf_handles.len() {
1850            let cf = self.cf_handles[self.current_cf_index];
1851            let mut read_opts = ReadOptions::default();
1852            read_opts.set_total_order_seek(true);
1853            self.current_iter =
1854                Some(self.snapshot.iterator_cf_opt(cf, read_opts, IteratorMode::Start));
1855        } else {
1856            self.current_iter = None;
1857        }
1858    }
1859}
1860
1861impl Iterator for RocksDbSnapshotSubtreeIterator<'_> {
1862    type Item = StorageResult<Subtree>;
1863
1864    fn next(&mut self) -> Option<Self::Item> {
1865        loop {
1866            let iter = self.current_iter.as_mut()?;
1867
1868            if let Some(result) =
1869                RocksDbSubtreeIterator::next_from_iter(iter, self.current_cf_index)
1870            {
1871                return Some(result);
1872            }
1873
1874            self.current_cf_index += 1;
1875            self.advance_to_next_cf();
1876
1877            self.current_iter.as_ref()?;
1878        }
1879    }
1880}
