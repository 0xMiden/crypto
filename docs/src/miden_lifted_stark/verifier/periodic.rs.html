1//! Verifier-side periodic column handling.
2//!
3//! Periodic columns are stored as polynomial coefficients for efficient evaluation
4//! at the OOD point using Horner's method.
5
6extern crate alloc;
7
8use alloc::vec::Vec;
9
10use p3_dft::{NaiveDft, TwoAdicSubgroupDft};
11use p3_field::{ExtensionField, TwoAdicField};
12
13use crate::util::horner::horner_acc;
14
15/// Verifier-side periodic polynomials for OOD evaluation.
16///
17/// Stores polynomial coefficients computed from the AIR's periodic columns.
18/// Used to evaluate periodic values at the OOD point during verification.
19#[derive(Clone, Debug)]
20pub(super) struct PeriodicPolys<F> {
21    /// Polynomial coefficients for each column.
22    polys: Vec<Vec<F>>,
23}
24
25impl<F: TwoAdicField> PeriodicPolys<F> {
26    /// Construct from subgroup evaluations (canonical order).
27    ///
28    /// Converts subgroup evaluations to polynomial coefficients via inverse DFT.
29    ///
30    /// # Panics
31    /// Panics if any column length is zero or not a power of two.
32    /// This is a trusted path — the AIR is assumed structurally valid (see
33    /// [`assert_multi_air_valid`](miden_lifted_air::debug::assert_multi_air_valid) for the
34    /// debug-only check).
35    pub fn new(column_evals: &[Vec<F>]) -> Self {
36        let dft = NaiveDft;
37        let mut polys = Vec::with_capacity(column_evals.len());
38
39        for (i, column) in column_evals.iter().enumerate() {
40            let p = column.len();
41            assert!(
42                p > 0 && p.is_power_of_two(),
43                "periodic column {i}: length must be positive power of two, got {p}"
44            );
45            let coeffs = dft.idft(column.clone());
46            polys.push(coeffs);
47        }
48
49        Self { polys }
50    }
51
52    /// Evaluate all periodic polynomials at the OOD point.
53    ///
54    /// For a column with period `p`, evaluates at `z^(trace_height / p)`.
55    /// Uses Horner's method for efficient polynomial evaluation.
56    ///
57    /// # Arguments
58    /// - `trace_height`: Height of the trace
59    /// - `z`: The OOD evaluation point
60    ///
61    /// The evaluation point is `z^{n/p}` rather than `z` directly. A periodic
62    /// column with period p is a polynomial P(X) of degree < p defined on the subgroup of
63    /// order p. At trace row i, the value is P(ωₙⁱ) = P(ωₚ^{i mod p}).
64    /// Since ωₙ^{n/p} = ωₚ, the map X → X^{n/p} collapses the trace domain H
65    /// (order n) onto the periodic subgroup (order p). So evaluating P at z^{n/p} gives
66    /// the same result as evaluating the periodic extension of P at z on the full trace
67    /// domain.
68    pub fn eval_at<EF>(&self, trace_height: usize, z: EF) -> Vec<EF>
69    where
70        EF: ExtensionField<F>,
71    {
72        let mut result = Vec::with_capacity(self.polys.len());
73
74        for coeffs in &self.polys {
75            let period = coeffs.len();
76            let y = z.exp_u64((trace_height / period) as u64);
77            // Coefficients are stored in ascending degree (idft output): [c₀, c₁, ..., cₙ₋₁].
78            // Horner needs descending order (highest degree first), hence `.rev()`.
79            result.push(horner_acc(EF::ZERO, y, coeffs.iter().rev().copied()));
80        }
81
82        result
83    }
84}
