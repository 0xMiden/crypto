1//! Lifted STARK verifier.
2//!
3//! This module provides:
4//! - [`verify_single`]: Verify a single AIR instance.
5//! - [`verify_multi`]: Verify multiple AIR instances with traces of different heights.
6//!
7//! These functions take a challenger (consumed by value) and proof data, construct
8//! the verifier transcript internally, and return a [`StarkDigest`] on success.
9//! The caller must check that the digest matches the prover's digest.
10//!
11//! # Fiat-Shamir / transcript binding
12//!
13//! The caller must produce the same challenger state as the prover — see the
14//! prover module-level docs for the full binding contract and recommended
15//! pattern.
16//!
17//! Log trace heights are carried on the [`StarkProof`] and observed into the
18//! challenger by [`verify_multi`]. Callers must not pre-observe them.
19//!
20//! # Statement-bound trace heights
21//!
22//! The verifier accepts whatever trace heights the proof carries; it never
23//! compares them against a caller-supplied expectation. If your statement
24//! fixes the trace size (e.g. a proof for a 2^16-row execution), parse it
25//! with
26//! [`StarkTranscript::from_proof`](crate::proof::StarkTranscript::from_proof)
27//! and check `transcript.instance_shapes.log_trace_heights()` yourself.
28//!
29//! # Transcript boundaries (strict consumption)
30//!
31//! [`verify_multi`] finalizes the transcript internally: it rejects proofs with
32//! trailing data (via [`TranscriptError::TrailingData`]) and returns a binding
33//! digest that must match the prover's digest.
34//!
35//! If you want to bundle extra data alongside the proof, you must manage
36//! boundaries yourself (e.g. parse and validate that data first, then pass the
37//! remaining transcript to [`verify_multi`]).

65/// Errors that can occur during verification.
66#[derive(Debug, Error)]
67pub enum VerifierError {
68    #[error("instance validation failed: {0}")]
69    Instance(#[from] InstanceValidationError),
70    #[error("PCS verification failed: {0}")]
71    Pcs(#[from] PcsError),
72    #[error("transcript error: {0}")]
73    Transcript(#[from] TranscriptError),
74    #[error("invalid aux shape")]
75    InvalidAuxShape,
76    #[error("constraint mismatch: quotient * vanishing != folded constraints")]
77    ConstraintMismatch,
78    #[error(
79        "constraint degree exceeds blowup: \
80         log_quotient_degree {log_quotient_degree} > log_blowup {log_blowup}"
81    )]
82    ConstraintDegreeTooHigh { log_quotient_degree: u8, log_blowup: u8 },
83    #[error("global reduced aux identity check failed")]
84    InvalidReducedAux,
85    #[error("aux value reduction failed: {0}")]
86    Reduction(ReductionError),
87}

89/// Verify a single AIR. Convenience wrapper around [`verify_multi`].
90///
91/// The caller's challenger must already be bound to the full statement
92/// — see the prover module-level docs.
93pub fn verify_single<F, EF, A, SC>(
94    config: &SC,
95    air: &A,
96    public_values: &[F],
97    var_len_public_inputs: VarLenPublicInputs<'_, F>,
98    proof: &StarkProof<F, EF, SC>,
99    challenger: SC::Challenger,
100) -> Result<StarkDigest<F, EF, SC>, VerifierError>
101where
102    F: TwoAdicField,
103    EF: ExtensionField<F>,
104    SC: StarkConfig<F, EF>,
105    A: LiftedAir<F, EF>,
106{
107    let instance = AirInstance { public_values, var_len_public_inputs };
108    verify_multi(config, &[(air, instance)], proof, challenger)
109}

111/// Verify multiple AIRs with traces of different heights.
112///
113/// The verifier uses [`InstanceShapes::air_order`](crate::InstanceShapes::air_order) from the proof
114/// to match the caller's instances to the proof's ordering. The caller's challenger
115/// must already be bound to the full statement (protocol parameters, AIR
116/// configurations, AIR ordering, and public inputs — both fixed and
117/// variable-length) — see the prover module-level docs.
118///
119/// The verifier mirrors the prover's protocol:
120///
121/// 1. Validate instance shapes and observe log trace heights into the challenger
122/// 2. Receive commitments and sample challenges in the same order as the prover
123/// 3. For each AIR, evaluate constraints at the lifted OOD point yⱼ = z^{rⱼ}
124/// 4. Accumulate folded constraints with β: acc = acc·β + foldedⱼ
125/// 5. Check quotient identity: `acc == Q(z) * Z_{H_max}(z)`
126///
127/// Lifting: for a trace of height nⱼ lifted by factor rⱼ, the committed
128/// codeword encodes `p_lift(X) = p(X^{rⱼ})`; opening at `[z, z · h_max]`
129/// yields the local/next row pair for the original trace domain.
130///
131/// **Statement-bound heights:** this function does not compare the proof's
132/// declared heights against any caller expectation. If your statement fixes
133/// trace dimensions, parse via
134/// [`StarkTranscript::from_proof`](crate::proof::StarkTranscript::from_proof)
135/// and check `instance_shapes.log_trace_heights()` before calling this. See
136/// the module-level docs for the full contract.
137pub fn verify_multi<F, EF, A, SC>(
138    config: &SC,
139    instances: &[(&A, AirInstance<'_, F>)],
140    proof: &StarkProof<F, EF, SC>,
141    mut challenger: SC::Challenger,
142) -> Result<StarkDigest<F, EF, SC>, VerifierError>
143where
144    F: TwoAdicField,
145    EF: ExtensionField<F>,
146    SC: StarkConfig<F, EF>,
147    A: LiftedAir<F, EF>,
148{
149    let instance_shapes = &proof.instance_shapes;
150    let air_order = instance_shapes.air_order();
151
152    // Validate air_order and reorder caller instances to the proof's AIR ordering.
153    validate_air_order(air_order, instances.len())?;
154    let instances = instance_shapes.reorder(instances.to_vec())?;
155
156    let log_blowup = config.pcs().log_blowup();
157
158    let log_max_trace_height = validate_inputs(&instances, instance_shapes, log_blowup)?;
159    let log_trace_heights = instance_shapes.log_trace_heights();
160
161    instance_shapes.observe_heights::<F, _>(&mut challenger);
162
163    let mut channel = VerifierTranscript::from_data(challenger, &proof.transcript);
164
165    // Clear the challenger's absorb buffer after observing instance shapes by
166    // squeezing a throwaway extension element. Must mirror the prover exactly.
167    let _instance_challenge: EF = channel.sample_algebra_element::<EF>();
168
169    // Infer constraint degree from symbolic AIR analysis (max across all AIRs).
170    // NOTE: `log_quotient_degree()` runs symbolic eval and may panic if the AIR is
171    // invalid. Callers must ensure `validate_inputs` (above) passes first.
172    let log_constraint_degree =
173        instances.iter().map(|(air, _)| air.log_quotient_degree()).max().unwrap_or(1) as u8;
174
175    if log_constraint_degree > log_blowup {
176        return Err(VerifierError::ConstraintDegreeTooHigh {
177            log_quotient_degree: log_constraint_degree,
178            log_blowup,
179        });
180    }
181
182    let constraint_degree = 1 << log_constraint_degree as usize;
183
184    let max_trace_height = 1 << log_max_trace_height as usize;
185    let log_lde_height = log_max_trace_height + log_blowup;
186
187    // Max LDE coset (for the largest trace, no lifting)
188    let max_lde_coset = LiftedCoset::unlifted(log_max_trace_height, log_blowup);
189
190    // 1. Receive main trace commitment
191    let main_commit = channel.receive_commitment()?.clone();
192
193    // 2. Sample randomness for aux traces
194    let max_num_randomness =
195        instances.iter().map(|(air, _)| air.num_randomness()).max().unwrap_or(0);
196
197    let randomness: Vec<EF> = (0..max_num_randomness)
198        .map(|_| channel.sample_algebra_element::<EF>())
199        .collect();
200
201    // 3. Receive aux trace commitment
202    let aux_commit = channel.receive_commitment()?.clone();
203
204    // Receive aux values from the transcript (one EF element per aux value, per instance).
205    // When no AIR has aux columns, each entry is empty so nothing is received.
206    let all_aux_values: Vec<Vec<EF>> = instances
207        .iter()
208        .map(|(air, _)| {
209            let count = air.num_aux_values();
210            (0..count)
211                .map(|_| channel.receive_algebra_element::<EF>())
212                .collect::<Result<Vec<_>, _>>()
213        })
214        .collect::<Result<Vec<_>, _>>()?;
215
216    // 4. Sample constraint folding alpha and accumulation beta
217    let alpha: EF = channel.sample_algebra_element::<EF>();
218    let beta: EF = channel.sample_algebra_element::<EF>();
219
220    // 5. Receive quotient commitment
221    let quotient_commit = channel.receive_commitment()?.clone();
222
223    // 6. Sample OOD point (outside max trace domain H and max LDE coset gK)
224    let z: EF = max_lde_coset.sample_ood_point(&mut channel);
225    let h = F::two_adic_generator(log_max_trace_height.into());
226    let z_next = z * h;
227
228    // 7. Widths per commitment group (unpadded data widths).
229    let main_widths: Vec<usize> = instances.iter().map(|(air, _)| air.width()).collect();
230    let aux_widths: Vec<usize> =
231        instances.iter().map(|(air, _)| air.aux_width() * EF::DIMENSION).collect();
232    let quotient_widths: Vec<usize> = vec![constraint_degree * EF::DIMENSION];
233
234    // Build commitments with original (unpadded) widths.
235    // The PCS aligned wrapper handles alignment and truncation internally.
236    let commitments = vec![
237        (main_commit, main_widths),
238        (aux_commit, aux_widths),
239        (quotient_commit, quotient_widths),
240    ];
241
242    // 8. Verify PCS openings (returns per-matrix RowMajorMatrix<EF>, truncated to original widths)
243    let opened = verify_aligned::<F, EF, SC::Lmcs, _, 2>(
244        config.pcs(),
245        config.lmcs(),
246        &commitments,
247        log_lde_height,
248        [z, z_next],
249        &mut channel,
250    )?;
251
252    // 9. Group indices for accessing opened matrices: [main, aux, quotient].
253    let (main_g, aux_g, quot_g) = (0, 1, 2);
254
255    // 10. Per-AIR constraint evaluation and beta accumulation.
256    //
257    // opened[g] has one matrix per AIR (for main/aux) or one matrix total (quotient).
258    // Each matrix has N=2 rows: row 0 = local (z), row 1 = next (z·h).
259    //
260    // Instances are in the proof's AIR ordering (ascending height), so j
261    // indexes both AIR and trace position directly.
262    debug_assert_eq!(opened[main_g].len(), instances.len());
263    debug_assert_eq!(opened[aux_g].len(), instances.len());
264    let mut accumulated = EF::ZERO;
265    let mut reduced_aux = ReducedAuxValues::<EF>::identity();
266
267    for (j, (air, inst)) in instances.iter().enumerate() {
268        let coset_j = LiftedCoset::new(log_trace_heights[j], log_blowup, log_max_trace_height);
269
270        // opened[main_g][j] is a 2-row RowMajorMatrix (local, next) already truncated.
271        let main_window = RowWindow::from_view(&opened[main_g][j].as_view());
272
273        // Extract aux trace opened values (reconstitute EF from base field components).
274        let aux_mat = &opened[aux_g][j];
275        let aux_local = row_to_packed_ext::<F, EF>(&aux_mat.row_slice(0).expect("aux row 0"))?;
276        let aux_next = row_to_packed_ext::<F, EF>(&aux_mat.row_slice(1).expect("aux row 1"))?;
277        let aux_window = RowWindow::from_two_rows(&aux_local, &aux_next);
278
279        // Selectors at the lifted OOD point yⱼ = z^{rⱼ} (encapsulated in LiftedCoset).
280        let selectors = coset_j.selectors_at::<F, _>(z);
281
282        // Periodic values: for a column with period p, eval_at computes z^{n/p}.
283        // Using (max_trace_height, z) gives z^{max_n / p}, which equals
284        // y_j^{n_j / p} = (z^{max_n/n_j})^{n_j/p} = z^{max_n/p}. This avoids
285        // computing y_j = z^{r_j} explicitly.
286        let periodic_polys = PeriodicPolys::new(&air.periodic_columns());
287        let periodic_values = periodic_polys.eval_at::<EF>(max_trace_height, z);
288
289        let aux_values_j = &all_aux_values[j];
290        let num_rand = air.num_randomness();
291        let mut folder = ConstraintFolder {
292            main: main_window,
293            aux: aux_window,
294            randomness: &randomness[..num_rand],
295            public_values: inst.public_values,
296            periodic_values: &periodic_values,
297            permutation_values: aux_values_j,
298            selectors,
299            alpha,
300            accumulator: EF::ZERO,
301            _phantom: PhantomData,
302        };
303
304        air.is_valid_builder(&folder).map_err(InstanceValidationError::from)?;
305        air.eval(&mut folder);
306
307        // Accumulate: acc = acc * beta + folded_j
308        accumulated = accumulated * beta + folder.accumulator;
309
310        // Compute reduced aux contribution and accumulate.
311        let contribution = air
312            .reduced_aux_values(
313                aux_values_j,
314                &randomness[..num_rand],
315                inst.public_values,
316                inst.var_len_public_inputs,
317            )
318            .map_err(VerifierError::Reduction)?;
319        reduced_aux.combine_in_place(&contribution);
320    }
321
322    // 11. Reconstruct Q(z) and check quotient identity Q(z) * Z_{H_max}(z)
323    // Quotient group has a single matrix; row 0 is the evaluation at z.
324    let quot_row = opened[quot_g][0].row_slice(0).expect("quotient row 0");
325    let quotient_chunks = row_to_packed_ext::<F, EF>(&quot_row)?;
326    let quotient_z = reconstruct_quotient::<F, EF>(z, &max_lde_coset, &quotient_chunks);
327
328    let vanishing = max_lde_coset.vanishing_at::<F, _>(z);
329    if accumulated != quotient_z * vanishing {
330        return Err(VerifierError::ConstraintMismatch);
331    }
332
333    // 12. Check global reduced aux identity (all bus contributions combine to identity)
334    if !reduced_aux.is_identity() {
335        return Err(VerifierError::InvalidReducedAux);
336    }
337
338    // 13. Finalize transcript: check emptiness and return digest
339    Ok(channel.finalize()?)
340}
