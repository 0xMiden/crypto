1//! Lifted STARK verifier.
2//!
3//! This module provides:
4//! - [`VerifierInstance::verify`](crate::VerifierInstance::verify): Verify a [`Statement`].
5//!
6//! Takes a challenger (consumed by value) and proof data, constructs the
7//! verifier transcript internally, and returns a [`StarkDigest`] on success.
8//! The caller must check that the digest matches the prover's digest.
9//!
10//! # Fiat-Shamir / transcript binding
11//!
12//! The caller must produce the same challenger state as the prover — see the
13//! prover module-level docs for the full binding contract and recommended
14//! pattern.
15//!
16//! The proof's per-AIR log trace heights are carried on [`StarkProofData`] (in
17//! instance order); [`VerifierInstance::verify`](crate::VerifierInstance::verify)
18//! observes the derived instance count and those heights at the protocol layer.
19//! Callers must not pre-observe them.
20//!
21//! # Statement-bound trace heights
22//!
23//! The verifier accepts whatever trace heights the proof carries; it never
24//! compares them against a caller-supplied expectation. If your statement
25//! fixes the trace size (e.g. a proof for a 2^16-row execution), parse it
26//! with [`StarkProof::from_data`](crate::proof::StarkProof::from_data) using the same
27//! [`VerifierInstance`], and check `proof.log_trace_heights()` yourself.
28//!
29//! # Transcript boundaries (strict consumption)
30//!
31//! [`VerifierInstance::verify`](crate::VerifierInstance::verify) finalizes the
32//! transcript internally: it rejects proofs with trailing data (via
33//! [`TranscriptError::TrailingData`]) and returns a binding digest that must
34//! match the prover's digest.
35//!
36//! If you want to bundle extra data alongside the proof, you must manage
37//! boundaries yourself (e.g. parse and validate that data first, then pass the
38//! remaining transcript to
39//! [`VerifierInstance::verify`](crate::VerifierInstance::verify)).

75/// Verifier-side bundle: a [`StarkConfig`], a borrowed [`Statement`], and the
76/// optional preprocessed commitment (a trusted setup input, not read from the
77/// proof).
78///
79/// Construction validates preprocessed presence parity, so holding a
80/// `VerifierInstance` guarantees the commitment is present exactly when the
81/// AIRs declare preprocessed columns.
82pub struct VerifierInstance<'a, F, EF, MA, SC>
83where
84    F: TwoAdicField,
85    EF: ExtensionField<F>,
86    MA: MultiAir<F, EF>,
87    SC: StarkConfig<F, EF>,
88{
89    config: &'a SC,
90    statement: &'a Statement<F, EF, MA>,
91    preprocessed_commitment: Option<<SC::Lmcs as Lmcs>::Commitment>,
92}
93
94impl<'a, F, EF, MA, SC> VerifierInstance<'a, F, EF, MA, SC>
95where
96    F: TwoAdicField,
97    EF: ExtensionField<F>,
98    MA: MultiAir<F, EF>,
99    SC: StarkConfig<F, EF>,
100{
101    /// Bundle a config + statement with an optional preprocessed commitment.
102    ///
103    /// The commitment must be `Some` exactly when some AIR declares preprocessed
104    /// columns; otherwise this errors with
105    /// [`PreprocessedValidationError::PresenceMismatch`].
106    pub fn new(
107        config: &'a SC,
108        statement: &'a Statement<F, EF, MA>,
109        preprocessed_commitment: Option<<SC::Lmcs as Lmcs>::Commitment>,
110    ) -> Result<Self, PreprocessedValidationError> {
111        let expected = statement.airs().iter().any(|a| a.preprocessed_width() > 0);
112        let actual = preprocessed_commitment.is_some();
113        if expected != actual {
114            return Err(PreprocessedValidationError::PresenceMismatch { expected, actual });
115        }
116        Ok(Self {
117            config,
118            statement,
119            preprocessed_commitment,
120        })
121    }
122
123    /// Verify a proof against this instance.
124    pub fn verify(
125        &self,
126        proof: &StarkProofData<F, EF, SC>,
127        challenger: SC::Challenger,
128    ) -> Result<StarkDigest<F, EF, SC>, VerifierError> {
129        verify(self, proof, challenger)
130    }
131
132    /// Borrow the STARK configuration.
133    pub fn config(&self) -> &SC {
134        self.config
135    }
136
137    /// Borrow the wrapped air-crate statement.
138    pub fn statement(&self) -> &Statement<F, EF, MA> {
139        self.statement
140    }
141
142    /// Borrow the preprocessed commitment, if any.
143    pub fn preprocessed_commitment(&self) -> Option<&<SC::Lmcs as Lmcs>::Commitment> {
144        self.preprocessed_commitment.as_ref()
145    }
146}

148/// Errors that can occur during verification.
149///
150/// Returned exclusively for runtime instance / proof-shape failures or
151/// cryptographic verification failures. AIR structural correctness is
152/// trusted — call [`crate::debug::assert_prover_setup`] (or
153/// [`miden_lifted_air::debug::assert_multi_air_valid`]) from tests.
154#[derive(Debug, Error)]
155pub enum VerifierError {
156    #[error(transparent)]
157    Instance(#[from] InstanceError),
158    #[error(transparent)]
159    Shape(#[from] ShapeError),
160    #[error(transparent)]
161    Domain(#[from] DomainError),
162    #[error(transparent)]
163    Pcs(#[from] PcsError),
164    #[error(transparent)]
165    Transcript(#[from] TranscriptError),
166    #[error("external assertion evaluation failed: {0}")]
167    Reduction(ReductionError),
168    #[error("constraint mismatch: quotient * vanishing != folded constraints")]
169    ConstraintMismatch,
170    #[error("external assertion {assertion} is non-zero")]
171    ExternalAssertionFailed {
172        /// Index into the assertions vector returned by
173        /// [`Statement::eval_external`].
174        assertion: usize,
175    },
176}

178/// Verify a [`Statement`].
179///
180/// The verifier reads per-AIR log trace heights from the proof (in caller
181/// order, matching [`Statement::airs`]) and reconstructs the proof's AIR
182/// ordering deterministically from those heights. The caller's challenger
183/// must already be bound to protocol parameters and AIR configurations —
184/// see the prover module-level docs. The statement's inputs are absorbed via
185/// [`Statement::observe`], then the instance count and proof's log trace heights
186/// are observed in instance order.
187///
188/// The verifier mirrors the prover's protocol:
189///
190/// 1. Validate runtime statement/proof shape data, absorb statement-owned inputs, and observe the
191///    instance count plus log trace heights in instance order
192/// 2. Receive commitments and sample challenges in the same order as the prover
193/// 3. For each AIR (in proof order), evaluate constraints at the lifted OOD point yⱼ = z^{rⱼ}
194/// 4. Accumulate folded constraints with β: acc = acc·β + foldedⱼ
195/// 5. Check quotient identity: `acc == Q(z) * Z_{H_max}(z)`
196/// 6. Evaluate [`Statement::eval_external`] with aux values reordered back to instance order
197///
198/// Lifting: for a trace of height nⱼ lifted by factor rⱼ, the committed
199/// codeword encodes `p_lift(X) = p(X^{rⱼ})`; opening at `[z, z · h_max]`
200/// yields the local/next row pair for the original trace domain.
201///
202/// **Statement-bound heights:** this function does not compare the proof's
203/// declared heights against any caller expectation. If your statement fixes
204/// trace dimensions, parse via
205/// [`StarkProof::from_data`](crate::proof::StarkProof::from_data) using this instance and check
206/// `proof.log_trace_heights()` before calling this. See the module-level docs for the full
207/// contract.
208///
209/// # Trust contract
210///
211/// `verify` validates the runtime statement plus everything carried on the
212/// proof; the AIR list is **trusted** (run
213/// [`miden_lifted_air::debug::assert_multi_air_valid`] from tests).
214///
215/// ## Validated
216/// - Same statement checks as [`prove`](crate::ProverInstance::prove) minus trace shape (no traces
217///   here)
218/// - Same `log_quotient_degree <= log_blowup` compat check
219/// - Proof shape via the internal trace-order reconstruction from log heights
220/// - Proof byte parsing (transcript channel)
221/// - PCS / FRI / DEEP / LMCS / transcript / constraint identity
222/// - External assertions from [`Statement::eval_external`]
223/// - Preprocessed openings against the trusted commitment (via the PCS)
224///
225/// ## Trusted (NOT validated)
226/// - AIR structural shape (same list as in [`prove`](crate::ProverInstance::prove))
227pub(crate) fn verify<F, EF, MA, SC>(
228    instance: &VerifierInstance<'_, F, EF, MA, SC>,
229    proof: &StarkProofData<F, EF, SC>,
230    mut challenger: SC::Challenger,
231) -> Result<StarkDigest<F, EF, SC>, VerifierError>
232where
233    F: TwoAdicField,
234    EF: ExtensionField<F>,
235    SC: StarkConfig<F, EF>,
236    MA: MultiAir<F, EF>,
237{
238    // --- Trust boundary (see doc-block above). -------------------------------
239    let config = instance.config();
240    let statement: &Statement<F, EF, MA> = instance.statement();
241    let preprocessed_commitment = instance.preprocessed_commitment();
242    //
243    // `TraceOrder::from_log_heights` validates the (untrusted) proof heights
244    // against the AIRs: it bounds `log_h` within the host's `usize` width
245    // (later code dereferences `1usize << log_h` and would otherwise overflow on
246    // a malicious proof) and checks per-AIR periodic-height feasibility.
247    // Statement::new already enforced the input-side contracts before construction.
248    let trace_order = TraceOrder::from_log_heights::<F, EF, _>(
249        statement.airs(),
250        proof.log_trace_heights.clone(),
251    )?;
252
253    // Preprocessed trace↔AIR mappings (instance order), reconstructed from the
254    // heights — the same ones the prover used.
255    let preprocessed_trace_to_air =
256        trace_order.preprocessed_air_for_trace_index::<F, EF, _>(statement.airs());
257    let air_to_preprocessed_trace =
258        trace_order.preprocessed_trace_index_for_air::<F, EF, _>(statement.airs());
259
260    let air_refs: Vec<&MA::Air> = statement.airs().iter().collect();
261    let proof_ordered_airs = trace_order.to_proof_order(&air_refs);
262    let air_inputs = statement.air_inputs();
263
264    let log_blowup = config.pcs().log_blowup();
265    let log_max_trace_height = trace_order.max_log_height();
266    let max_lde_domain = LiftedDomain::<F>::try_canonical(log_max_trace_height, log_blowup)?;
267    let instance_domains: Vec<_> = trace_order
268        .log_heights_proof()
269        .iter()
270        .map(|&log_h| max_lde_domain.try_sub_domain(log_h))
271        .collect::<Result<_, _>>()?;
272
273    // Observe the preprocessed commitment first (when present); mirrors the
274    // prover. It is a trusted statement input, not read from the proof.
275    if let Some(commitment) = preprocessed_commitment {
276        challenger.observe(commitment.clone());
277    }
278
279    // `Statement::observe` absorbs statement-owned inputs. The protocol then
280    // binds the instance count and each log trace height in instance order.
281    statement.observe(&mut challenger, trace_order.log_heights());
282    trace_order.observe_shape::<F, _>(&mut challenger);
283
284    let mut channel = VerifierTranscript::from_data(challenger, &proof.transcript);
285
286    // Infer constraint degree from symbolic AIR analysis (max across all AIRs).
287    // NOTE: `log_quotient_degree` runs symbolic eval and may panic if the AIR is
288    // invalid. The AIR is trusted (see `miden_lifted_air::debug::assert_multi_air_valid`
289    // for the debug-only structural check).
290    let max_log_quotient_degree = proof_ordered_airs
291        .iter()
292        .map(|&air| log_quotient_degree::<F, EF, _>(air))
293        .max()
294        .expect("TraceOrder construction rejects empty AIR sets");
295    if max_log_quotient_degree > log_blowup {
296        return Err(DomainError::ConstraintDegreeTooHigh {
297            log_quotient: max_log_quotient_degree,
298            log_blowup,
299        }
300        .into());
301    }
302
303    // Pair the max LDE domain with the constraint degree for the constraint layer.
304    let max_eval_domain = max_lde_domain.evaluation_domain(max_log_quotient_degree);
305
306    let quotient_degree = 1 << max_log_quotient_degree as usize;
307
308    let max_trace_height = max_lde_domain.trace_height();
309
310    // 1. Receive main trace commitment
311    let main_commit = channel.receive_commitment()?.clone();
312
313    // 2. Sample randomness for aux traces
314    let max_num_randomness =
315        proof_ordered_airs.iter().map(|air| air.num_randomness()).max().unwrap_or(0);
316
317    let randomness: Vec<EF> = (0..max_num_randomness)
318        .map(|_| channel.sample_algebra_element::<EF>())
319        .collect();
320
321    // 3. Receive aux trace commitment
322    let aux_commit = channel.receive_commitment()?.clone();
323
324    // Receive aux values from the transcript (one EF element per aux value, per instance).
325    // When no AIR has aux columns, each entry is empty so nothing is received.
326    let all_aux_values: Vec<Vec<EF>> = proof_ordered_airs
327        .iter()
328        .map(|air| {
329            let count = air.num_aux_values();
330            (0..count)
331                .map(|_| channel.receive_algebra_element::<EF>())
332                .collect::<Result<Vec<_>, _>>()
333        })
334        .collect::<Result<Vec<_>, _>>()?;
335
336    // 4. Sample constraint folding alpha and accumulation beta
337    let alpha: EF = channel.sample_algebra_element::<EF>();
338    let beta: EF = channel.sample_algebra_element::<EF>();
339
340    // 5. Receive quotient commitment
341    let quotient_commit = channel.receive_commitment()?.clone();
342
343    // 6. Sample OOD point (outside max trace domain H and max LDE coset gK)
344    let z: EF = max_lde_domain.sample_ood_point(&mut channel);
345    let h = max_lde_domain.trace_subgroup().generator();
346    let z_next = z * h;
347
348    // 7. Widths per commitment group (unpadded data widths).
349    let main_widths: Vec<usize> = proof_ordered_airs.iter().map(|air| air.width()).collect();
350    let aux_widths: Vec<usize> =
351        proof_ordered_airs.iter().map(|air| air.aux_width() * EF::DIMENSION).collect();
352    let quotient_widths: Vec<usize> = vec![quotient_degree * EF::DIMENSION];
353
354    // Build commitments with original (unpadded) widths.
355    // The PCS aligned wrapper handles alignment and truncation internally.
356    // The preprocessed group (when present) is first, mirroring the prover; its
357    // widths are in committed preprocessed trace order, while main/aux are in proof order.
358    //
359    // Group indices, in batch order `[preprocessed?, main, aux, quotient]`: the
360    // preprocessed group occupies index 0 only when present, shifting the rest
361    // up by one. The prover builds its `trees` vector in this same order.
362    let s = preprocessed_commitment.is_some() as usize;
363    let (preproc_g, main_g, aux_g, quot_g) =
364        (preprocessed_commitment.is_some().then_some(0), s, s + 1, s + 2);
365    let full_log_height = log_max_trace_height + log_blowup;
366    let mut commitments = Vec::with_capacity(4);
367    if let Some(commitment) = preprocessed_commitment {
368        let preprocessed_widths: Vec<usize> = preprocessed_trace_to_air
369            .iter()
370            .map(|&air_idx| statement.airs()[air_idx as usize].preprocessed_width())
371            .collect();
372        // The preprocessed tree is committed at its own setup-fixed depth, determined by the
373        // tallest preprocessed trace. The PCS virtually lifts it to the max when shorter.
374        let preprocessed_log_height = preprocessed_trace_to_air
375            .iter()
376            .map(|&air_idx| trace_order.log_heights()[air_idx as usize])
377            .max()
378            .expect("preprocessed group is non-empty when a commitment is present")
379            + log_blowup;
380        commitments.push(CommitmentGroup {
381            root: commitment.clone(),
382            widths: preprocessed_widths,
383            log_height: preprocessed_log_height,
384        });
385    }
386    commitments.push(CommitmentGroup {
387        root: main_commit,
388        widths: main_widths,
389        log_height: full_log_height,
390    });
391    commitments.push(CommitmentGroup {
392        root: aux_commit,
393        widths: aux_widths,
394        log_height: full_log_height,
395    });
396    commitments.push(CommitmentGroup {
397        root: quotient_commit,
398        widths: quotient_widths,
399        log_height: full_log_height,
400    });
401
402    // 8. Verify PCS openings (returns per-matrix RowMajorMatrix<EF>, truncated to original widths)
403    let opened = verify_aligned::<F, EF, SC::Lmcs, _, 2>(
404        config.pcs(),
405        config.lmcs(),
406        &commitments,
407        &max_lde_domain,
408        [z, z_next],
409        &mut channel,
410    )?;
411
412    // 9. Per-AIR constraint evaluation and beta accumulation.
413    //
414    // opened[g] has one matrix per AIR (for main/aux) or one matrix total (quotient).
415    // Each matrix has N=2 rows: row 0 = local (z), row 1 = next (z·h).
416    //
417    // AIRs are in the proof's ordering (ascending height), so j indexes both
418    // AIR and trace position directly.
419    debug_assert_eq!(opened[main_g].len(), proof_ordered_airs.len());
420    debug_assert_eq!(opened[aux_g].len(), proof_ordered_airs.len());
421    let mut accumulated = EF::ZERO;
422
423    for (j, air) in proof_ordered_airs.iter().enumerate() {
424        let domain_j = instance_domains[j];
425
426        // opened[main_g][j] is a 2-row RowMajorMatrix (local, next) already truncated.
427        let main_window = RowWindow::from_view(&opened[main_g][j].as_view());
428
429        // Extract aux trace opened values (reconstitute EF from base field components).
430        // Row widths were validated against `air.aux_width() * EF::DIMENSION`
431        // by `verify_aligned` upstream; reaching here with a mismatch would
432        // indicate a framework bug.
433        let aux_mat = &opened[aux_g][j];
434        let aux_local = row_to_packed_ext::<F, EF>(&aux_mat.row_slice(0).expect("aux row 0"))
435            .expect("aux row width should match: PCS verify_aligned validates this upstream");
436        let aux_next = row_to_packed_ext::<F, EF>(&aux_mat.row_slice(1).expect("aux row 1"))
437            .expect("aux row width should match: PCS verify_aligned validates this upstream");
438        let aux_window = RowWindow::from_two_rows(&aux_local, &aux_next);
439
440        // Selectors at the lifted OOD point yⱼ = z^{rⱼ} (encapsulated in LiftedDomain).
441        let selectors = domain_j.selectors_at(z);
442
443        // Periodic values: for a column with period p, eval_at computes z^{n/p}.
444        // Using (max_trace_height, z) gives z^{max_n / p}, which equals
445        // y_j^{n_j / p} = (z^{max_n/n_j})^{n_j/p} = z^{max_n/p}. This avoids
446        // computing y_j = z^{r_j} explicitly.
447        let periodic_polys = PeriodicPolys::new(&air.periodic_columns());
448        let periodic_values = periodic_polys.eval_at::<EF>(max_trace_height, z);
449
450        let aux_values_j = &all_aux_values[j];
451        let num_rand = air.num_randomness();
452
453        // Extract the opened preprocessed window when this AIR declares
454        // preprocessed columns. The preprocessed trace index comes from the inverse
455        // `air_to_preprocessed_trace` mapping; the opened matrix is a 2-row
456        // `RowMajorMatrix` already truncated to the declared width by `verify_aligned`, so this
457        // is a zero-copy view (mirrors the main window above).
458        let instance_idx = trace_order.instance_indices()[j] as usize;
459        let preprocessed_window = match air_to_preprocessed_trace[instance_idx] {
460            Some(preprocessed_trace_idx) => RowWindow::from_view(
461                &opened[preproc_g.expect("preproc group present")][preprocessed_trace_idx]
462                    .as_view(),
463            ),
464            None => RowWindow::from_two_rows(&[], &[]),
465        };
466
467        let mut folder = ConstraintFolder {
468            main: main_window,
469            preprocessed: preprocessed_window,
470            aux: aux_window,
471            randomness: &randomness[..num_rand],
472            public_values: air_inputs,
473            periodic_values: &periodic_values,
474            permutation_values: aux_values_j,
475            selectors,
476            alpha,
477            accumulator: EF::ZERO,
478            _phantom: PhantomData,
479        };
480
481        #[cfg(debug_assertions)]
482        miden_lifted_air::debug::check_builder_shape(*air, &folder);
483        air.eval(&mut folder);
484
485        // Accumulate: acc = acc * beta + folded_j
486        accumulated = accumulated * beta + folder.accumulator;
487    }
488
489    // 11. Evaluate the proof's external assertions. Aux values came off the
490    // wire in proof order; reorder them back to instance order before handing
491    // them to `eval_external`, which is defined in instance-order terms.
492    let aux_instance = trace_order.to_instance_order(&all_aux_values);
493    let aux_views: Vec<&[EF]> = aux_instance.iter().map(Vec::as_slice).collect();
494    let assertions = statement
495        .eval_external(&randomness, &aux_views, trace_order.log_heights())
496        .map_err(VerifierError::Reduction)?;
497    for (k, assertion) in assertions.iter().enumerate() {
498        if *assertion != EF::ZERO {
499            return Err(VerifierError::ExternalAssertionFailed { assertion: k });
500        }
501    }
502
503    // 12. Reconstruct Q(z) and check quotient identity Q(z) * Z_{H_max}(z)
504    // Quotient group has a single matrix; row 0 is the evaluation at z.
505    let quot_row = opened[quot_g][0].row_slice(0).expect("quotient row 0");
506    let quotient_chunks = row_to_packed_ext::<F, EF>(&quot_row)
507        .expect("quotient row width should match: PCS verify_aligned validates this upstream");
508    let quotient_z = max_eval_domain.reconstruct_quotient::<EF>(z, &quotient_chunks);
509
510    // `max_lde_domain` is the tallest (lift_ratio = 0), so lifted == unlifted here.
511    let vanishing = max_lde_domain.trace_subgroup().vanishing_at(z);
512    if accumulated != quotient_z * vanishing {
513        return Err(VerifierError::ConstraintMismatch);
514    }
515
516    // 13. Finalize transcript: check emptiness and return digest
517    Ok(channel.finalize()?)
518}
