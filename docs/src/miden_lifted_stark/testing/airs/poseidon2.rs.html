1//! Wraps Plonky3's [`Poseidon2Air`] as a [`LiftedAir`].
2//!
3//! Uses the standard Goldilocks configuration: WIDTH=12, SBOX_DEGREE=7, SBOX_REGISTERS=1,
4//! HALF_FULL_ROUNDS=4, PARTIAL_ROUNDS=22.
5
6use alloc::vec::Vec;
7
8use miden_lifted_air::{Air, BaseAir, LiftedAir, LiftedAirBuilder};
9use p3_field::Field;
10use p3_goldilocks::{GenericPoseidon2LinearLayersGoldilocks, Goldilocks};
11use p3_matrix::dense::RowMajorMatrix;
12use p3_poseidon2_air::{Poseidon2Air, RoundConstants, num_cols};
13
14/// Goldilocks Poseidon2 configuration constants.
15pub const WIDTH: usize = 12;
16pub const SBOX_DEGREE: u64 = 7;
17pub const SBOX_REGISTERS: usize = 1;
18pub const HALF_FULL_ROUNDS: usize = 4;
19pub const PARTIAL_ROUNDS: usize = 22;
20
21/// Number of trace columns for the Goldilocks Poseidon2 AIR.
22pub const NUM_POSEIDON2_COLS: usize =
23    num_cols::<WIDTH, SBOX_DEGREE, SBOX_REGISTERS, HALF_FULL_ROUNDS, PARTIAL_ROUNDS>();
24
25type GoldilocksRoundConstants = RoundConstants<Goldilocks, WIDTH, HALF_FULL_ROUNDS, PARTIAL_ROUNDS>;
26
27type InnerAir = Poseidon2Air<
28    Goldilocks,
29    GenericPoseidon2LinearLayersGoldilocks,
30    WIDTH,
31    SBOX_DEGREE,
32    SBOX_REGISTERS,
33    HALF_FULL_ROUNDS,
34    PARTIAL_ROUNDS,
35>;
36
37/// [`Poseidon2Air`] adapted for the lifted STARK prover.
38///
39/// Poseidon2 is a main-trace-only AIR with no preprocessed, periodic, or auxiliary columns.
40/// Each row represents one full Poseidon2 permutation (1 row per hash).
41pub struct LiftedPoseidon2Air {
42    inner: InnerAir,
43}
44
45impl LiftedPoseidon2Air {
46    pub fn new(constants: GoldilocksRoundConstants) -> Self {
47        Self { inner: InnerAir::new(constants) }
48    }
49}
50
51impl<F> BaseAir<F> for LiftedPoseidon2Air {
52    fn width(&self) -> usize {
53        NUM_POSEIDON2_COLS
54    }
55}
56
57impl<EF: Field> LiftedAir<Goldilocks, EF> for LiftedPoseidon2Air {
58    fn num_randomness(&self) -> usize {
59        1
60    }
61
62    fn aux_width(&self) -> usize {
63        1
64    }
65
66    fn num_aux_values(&self) -> usize {
67        0
68    }
69
70    fn num_var_len_public_inputs(&self) -> usize {
71        0
72    }
73
74    fn eval<AB: LiftedAirBuilder<F = Goldilocks>>(&self, builder: &mut AB) {
75        Air::eval(&self.inner, builder);
76    }
77}
78
79/// Generate a Poseidon2 trace for the given inputs.
80///
81/// Each input is a WIDTH=12 element Goldilocks array. The number of inputs must
82/// be a power of two. The trace has `inputs.len()` rows and
83/// [`NUM_POSEIDON2_COLS`] columns.
84pub fn generate_poseidon2_trace(
85    inputs: Vec<[Goldilocks; WIDTH]>,
86    constants: &GoldilocksRoundConstants,
87) -> RowMajorMatrix<Goldilocks> {
88    p3_poseidon2_air::generate_trace_rows::<
89        Goldilocks,
90        GenericPoseidon2LinearLayersGoldilocks,
91        WIDTH,
92        SBOX_DEGREE,
93        SBOX_REGISTERS,
94        HALF_FULL_ROUNDS,
95        PARTIAL_ROUNDS,
96    >(inputs, constants, 0)
97}
