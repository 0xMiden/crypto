30/// A dummy AIR with a single degree-9 constraint and auxiliary columns.
31///
32/// The constraint is `local[0] * local[1] * ... * local[8] == 0`, which has
33/// degree 9 and produces `log_quotient_degree = 3` (8 quotient chunks).
34pub struct DummyMidenAir {
35    width: usize,
36    num_aux_cols: usize,
37}
38
39impl DummyMidenAir {
40    pub fn new(width: usize, num_aux_cols: usize) -> Self {
41        assert!(width >= 9, "DummyMidenAir needs at least 9 columns for the degree-9 constraint");
42        Self { width, num_aux_cols }
43    }
44}
45
46/// Shared constraint logic: `local[0] * local[1] * ... * local[8] == 0`.
47fn eval_miden_constraints<AB: AirBuilder>(builder: &mut AB) {
48    let main = builder.main();
49    let local = main.current_slice();
50    let product = (0..9).fold(AB::Expr::ONE, |acc, j| acc * local[j].into());
51    builder.assert_zero(product);
52}

90/// Generate a dummy trace with the given width and log₂ height.
91///
92/// Column 0 is zero everywhere (satisfying the product constraint).
93/// Columns 1..width are filled with deterministic pseudo-random values.
94pub fn generate_dummy_trace<F, R>(width: usize, log_height: u8, rng: &mut R) -> RowMajorMatrix<F>
95where
96    F: Field,
97    R: rand::Rng,
98    rand::distr::StandardUniform: rand::distr::Distribution<F>,
99{
100    use rand::RngExt;
101
102    let height = 1 << log_height as usize;
103    let mut values = F::zero_vec(height * width);
104
105    for row in 0..height {
106        // Column 0 stays zero (already initialized).
107        for col in 1..width {
108            values[row * width + col] = rng.random();
109        }
110    }
111
112    RowMajorMatrix::new(values, width)
113}
