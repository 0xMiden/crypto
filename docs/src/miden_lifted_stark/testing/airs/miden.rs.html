28// ---------------------------------------------------------------------------

32/// A dummy AIR with a single degree-9 constraint and auxiliary columns.
33///
34/// The constraint is `local[0] * local[1] * ... * local[8] == 0`, which has
35/// degree 9 and produces `log_quotient_degree = 3` (8 quotient chunks).
36pub struct DummyMidenAir {
37    width: usize,
38    num_aux_cols: usize,
39}
40
41impl DummyMidenAir {
42    pub fn new(width: usize, num_aux_cols: usize) -> Self {
43        assert!(width >= 9, "DummyMidenAir needs at least 9 columns for the degree-9 constraint");
44        Self { width, num_aux_cols }
45    }
46}
47
48/// Shared constraint logic: `local[0] * local[1] * ... * local[8] == 0`.
49fn eval_miden_constraints<AB: AirBuilder>(builder: &mut AB) {
50    let main = builder.main();
51    let local = main.current_slice();
52    let product = (0..9).fold(AB::Expr::ONE, |acc, j| acc * local[j].into());
53    builder.assert_zero(product);
54}

66impl<F: Field, EF: Field> LiftedAir<F, EF> for DummyMidenAir {
67    fn num_randomness(&self) -> usize {
68        2
69    }
70
71    fn aux_width(&self) -> usize {
72        self.num_aux_cols
73    }
74
75    fn num_aux_values(&self) -> usize {
76        self.num_aux_cols
77    }
78
79    fn build_aux_trace(
80        &self,
81        main: &RowMajorMatrix<F>,
82        _air_inputs: &[F],
83        _aux_inputs: &[F],
84        _challenges: &[EF],
85    ) -> (RowMajorMatrix<EF>, Vec<EF>) {
86        // Constraints touch only the main trace: emit an all-zero aux trace.
87        let aux =
88            RowMajorMatrix::new(EF::zero_vec(main.height() * self.num_aux_cols), self.num_aux_cols);
89        (aux, EF::zero_vec(self.num_aux_cols))
90    }
91
92    fn eval<AB: LiftedAirBuilder<F = F>>(&self, builder: &mut AB) {
93        eval_miden_constraints(builder);
94    }
95}

101/// Generate a dummy trace with the given width and log₂ height.
102///
103/// Column 0 is zero everywhere (satisfying the product constraint).
104/// Columns 1..width are filled with deterministic pseudo-random values.
105pub fn generate_dummy_trace<F, R>(width: usize, log_height: u8, rng: &mut R) -> RowMajorMatrix<F>
106where
107    F: Field,
108    R: rand::Rng,
109    rand::distr::StandardUniform: rand::distr::Distribution<F>,
110{
111    use rand::RngExt;
112
113    let height = 1 << log_height as usize;
114    let mut values = F::zero_vec(height * width);
115
116    for row in 0..height {
117        // Column 0 stays zero (already initialized).
118        for col in 1..width {
119            values[row * width + col] = rng.random();
120        }
121    }
122
123    RowMajorMatrix::new(values, width)
124}
