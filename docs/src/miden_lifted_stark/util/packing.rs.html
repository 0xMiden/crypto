1//! Packing / column-transpose helpers for SIMD operations on packed values
2//! and packed extension-field elements.
3
4use alloc::vec::Vec;
5
6use p3_field::{ExtensionField, Field};
7
8/// Reconstitute EF elements from opened base field polynomial evaluations.
9///
10/// When an EF polynomial is committed, it becomes DIM base field polynomials.
11/// Opening at EF point z gives DIM EF values (F-polys evaluated at EF point).
12/// Reconstruct each EF element: `vᵢ = Σⱼ basisⱼ·row[i·DIM + j]`.
13///
14/// Returns `None` if `row.len()` is not a multiple of `EF::DIMENSION`.
15pub(crate) fn row_to_packed_ext<F, EF>(row: &[EF]) -> Option<Vec<EF>>
16where
17    F: Field,
18    EF: ExtensionField<F>,
19{
20    if !row.len().is_multiple_of(EF::DIMENSION) {
21        return None;
22    }
23    Some(
24        row.chunks_exact(EF::DIMENSION)
25            .map(|chunk| EF::from_ext_basis_coefficients(chunk).unwrap())
26            .collect(),
27    )
28}
