1//! Packing / column-transpose helpers for SIMD operations on packed values
2//! and packed extension-field elements.
3
4use alloc::vec::Vec;
5use core::array;
6
7use p3_field::{ExtensionField, Field, PackedFieldExtension, PackedValue};
8
9/// Extension trait for [`PackedFieldExtension`] adding `pack_ext_columns` and
10/// `to_ext_slice` methods for column-wise SIMD operations on extension field elements.
11pub trait PackedFieldExtensionExt<
12    BaseField: Field,
13    ExtField: ExtensionField<BaseField, ExtensionPacking = Self>,
14>: PackedFieldExtension<BaseField, ExtField>
15{
16    /// Pack N columns from WIDTH rows into N packed extension field elements.
17    ///
18    /// Input: `rows[lane][col]` - WIDTH rows, each with N extension field elements.
19    /// Output: `result[col]` - N packed values, where each packs WIDTH lanes.
20    fn pack_ext_columns<const N: usize>(rows: &[[ExtField; N]]) -> [Self; N] {
21        let width = BaseField::Packing::WIDTH;
22        debug_assert_eq!(rows.len(), width);
23        array::from_fn(|col| {
24            let col_elems: Vec<ExtField> = (0..width).map(|lane| rows[lane][col]).collect();
25            Self::from_ext_slice(&col_elems)
26        })
27    }
28
29    /// Extract all lanes to an output slice.
30    fn to_ext_slice(&self, out: &mut [ExtField]) {
31        let width = BaseField::Packing::WIDTH;
32        for (lane, slot) in out.iter_mut().enumerate().take(width) {
33            *slot = self.extract(lane);
34        }
35    }
36}
37
38impl<
39    BaseField: Field,
40    ExtField: ExtensionField<BaseField, ExtensionPacking = P>,
41    P: PackedFieldExtension<BaseField, ExtField>,
42> PackedFieldExtensionExt<BaseField, ExtField> for P
43{
44}

46/// Reconstitute EF elements from opened base field polynomial evaluations.
47///
48/// When an EF polynomial is committed, it becomes DIM base field polynomials.
49/// Opening at EF point z gives DIM EF values (F-polys evaluated at EF point).
50/// Reconstruct each EF element: `vᵢ = Σⱼ basisⱼ·row[i·DIM + j]`.
51///
52/// Returns `None` if `row.len()` is not a multiple of `EF::DIMENSION`.
53pub fn row_to_packed_ext<F, EF>(row: &[EF]) -> Option<Vec<EF>>
54where
55    F: Field,
56    EF: ExtensionField<F>,
57{
58    if !row.len().is_multiple_of(EF::DIMENSION) {
59        return None;
60    }
61    Some(
62        row.chunks_exact(EF::DIMENSION)
63            .map(|chunk| {
64                chunk
65                    .iter()
66                    .enumerate()
67                    .map(|(j, &c)| EF::ith_basis_element(j).unwrap() * c)
68                    .sum()
69            })
70            .collect(),
71    )
72}
