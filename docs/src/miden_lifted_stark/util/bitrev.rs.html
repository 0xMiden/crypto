23/// A matrix that supports bit-reversed row reordering.
24///
25/// Local copy of `p3_matrix::bitrev::BitReversibleMatrix` extended with impls for
26/// [`FlatMatrixView`].
27pub trait BitReversibleMatrix<T: Send + Sync + Clone>: Matrix<T> {
28    /// The type returned when this matrix is viewed in bit-reversed order.
29    type BitRev: BitReversibleMatrix<T>;
30
31    /// Return a version of the matrix with its row order reversed by bit index.
32    fn bit_reverse_rows(self) -> Self::BitRev;
33}

39impl<T, S> BitReversibleMatrix<T> for DenseMatrix<T, S>
40where
41    T: Clone + Send + Sync,
42    S: DenseStorage<T>,
43{
44    type BitRev = BitReversedMatrixView<Self>;
45
46    fn bit_reverse_rows(self) -> Self::BitRev {
47        BitReversalPerm::new_view(self)
48    }
49}

54    S: DenseStorage<T>,
55{
56    type BitRev = DenseMatrix<T, S>;

59        self.inner
60    }

93/// Materialize a matrix into domain-ordered `BitReversedMatrixView<RowMajorMatrix<T>>`.
94///
95/// Temporary adapter for types that implement the upstream
96/// [`p3_matrix::bitrev::BitReversibleMatrix`] but not this crate's local copy.
97/// The returned type implements both traits and can be passed directly to
98/// [`Lmcs::build_tree`](crate::lmcs::Lmcs::build_tree) /
99/// [`Lmcs::build_aligned_tree`](crate::lmcs::Lmcs::build_aligned_tree).
100///
101/// Remove alongside [`BitReversibleMatrix`] when upstream impls cover all DFT output types.
102pub fn materialize_bitrev<T: Clone + Send + Sync>(
103    evals: impl p3_matrix::bitrev::BitReversibleMatrix<T>,
104) -> BitReversedMatrixView<RowMajorMatrix<T>> {
105    BitReversalPerm::new_view(evals.bit_reverse_rows().to_row_major_matrix())
106}

108/// Coset points `gK` in bit-reversed order.
109///
110/// Note: the coset shift `g` is fixed to `F::GENERATOR` by convention in this PCS.
111///
112/// Bit-reversal gives two properties essential for lifting:
113/// - **Adjacent negation**: `gK[2i+1] = -gK[2i]`, so both square to the same value
114/// - **Squaring gives prefix**: `(gK[2i])² = (gK)²[i]` — the even-indexed elements, when squared,
115///   form the half-size sub-coset. Generalizes to r-th powers.
116///
117/// Together these enable iterative weight folding in barycentric evaluation.
118///
119/// # Panics
120/// Panics if the two-adic coset construction fails (e.g., `log_n` exceeds the field's
121/// two-adicity), since this unwraps `TwoAdicMultiplicativeCoset::new`.
122pub fn bit_reversed_coset_points<F: TwoAdicField>(log_n: u8) -> Vec<F> {
123    let coset =
124        p3_field::coset::TwoAdicMultiplicativeCoset::new(F::GENERATOR, log_n as usize).unwrap();
125    let mut pts: Vec<F> = coset.iter().collect();
126    reverse_slice_index_bits(&mut pts);
127    pts
128}
