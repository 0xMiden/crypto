1//! Horner-style polynomial evaluation helpers.
2
3use core::ops::{Add, Mul};
4
5/// Horner fold with an explicit accumulator.
6///
7/// Computes `acc·xⁿ + v₀·xⁿ⁻¹ + v₁·xⁿ⁻² + ... + vₙ₋₁·x⁰` where n = len(vals).
8/// Equivalently: `((acc·x + v₀)·x + v₁)·x + ... + vₙ₋₁`.
9/// The first element gets the highest power of `x`.
10///
11/// For polynomial evaluation `p(x) = Σᵢ cᵢ·xⁱ`, pass coefficients in
12/// descending degree order `[cₙ, ..., c₁, c₀]`.
13#[inline]
14pub fn horner_acc<Acc, Val, X, I>(acc: Acc, x: X, vals: I) -> Acc
15where
16    I: IntoIterator<Item = Val>,
17    Acc: Mul<X, Output = Acc> + Add<Val, Output = Acc>,
18    X: Clone,
19{
20    vals.into_iter().fold(acc, |acc, val| acc * x.clone() + val)
21}
22
23/// Horner fold starting from zero.
24///
25/// See [`horner_acc`] for the evaluation convention.
26#[inline]
27pub fn horner<Acc, Val, X, I>(x: X, vals: I) -> Acc
28where
29    I: IntoIterator<Item = Val>,
30    Acc: Default + Mul<X, Output = Acc> + Add<Val, Output = Acc>,
31    X: Clone,
32{
33    horner_acc(Acc::default(), x, vals)
34}
