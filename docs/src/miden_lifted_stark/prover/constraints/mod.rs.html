29/// Row-blocks (`i_start = r * packing_width`) processed per rayon task.
30const ROW_BLOCKS_PER_PARALLEL_TASK: usize = 32;
31
32/// Type alias for packed base field from F.
33type PackedVal<F> = <F as Field>::Packing;
34
35/// Type alias for packed extension field from EF.
36type PackedExt<F, EF> = <EF as ExtensionField<F>>::ExtensionPacking;
37
38/// Evaluate an AIR's constraints on its native quotient coset and write the
39/// per-AIR quotient evaluations (constraint numerator divided by `Z_{H_j}`)
40/// into `output`.
41///
42/// `coset` is the AIR's native quotient evaluation coset `gJ_j` of size `n_j * D_j`,
43/// where `n_j` is the AIR's trace height and `D_j = 2^log_quotient_degree` is its
44/// per-AIR constraint-degree bound. For each point on `gJ_j` we evaluate every
45/// constraint, fold with powers of `alpha`, multiply by the precomputed `1 / Z_H`
46/// value, and write the result:
47///
48/// `output[i] = folded_constraints(x_i) / Z_{H_j}(x_i)`.
49///
50/// `inv_z_h` is a length-`D_j` slice: `Z_{H_j}(x)` takes only `D_j` distinct
51/// values over `gJ_j` by periodicity, so batch-inverting them once suffices
52/// (use [`crate::domain::EvaluationDomain::inv_vanishing_evals`]). Fusing the
53/// divide into the write loop saves a second pass over the `n_j · D_j`-point
54/// output buffer.
55///
56/// `output` must be a fresh zero-initialized buffer of length `n_j * D_j`; each
57/// point is written once. Upsampling to the batch-wide target and beta-accumulation
58/// into the shared quotient accumulator happen in the caller.
59///
60/// Trace views must be [`BitReversedMatrixView`] over dense row-major storage (as
61/// returned by [`crate::prover::commit::Committed::evals_on_quotient_domain`]), in
62/// natural order on `gJ_j`.
63///
64/// Uses SIMD-packed parallel iteration via rayon for optimal performance:
65/// - Processes `WIDTH` points simultaneously using packed field types
66/// - Main trace stays in base field, only aux trace uses extension field
67/// - Constraints are collected then finalized in batches via decomposed alpha powers
68///
69/// Why we fold with `alpha`: the prover does not want to carry K separate constraint
70/// polynomials through the rest of the protocol. A random linear combination
71///
72/// `C_fold(x) = Σₖ α^{K−1−k}·Cₖ(x)`
73///
74/// collapses them into one numerator polynomial while preserving soundness (a non-zero
75/// constraint survives with high probability).
76///
77/// Why we evaluate on the native coset: the quotient `Q_j = C_j / Z_{H_j}` has degree
78/// `< n_j * D_j` by construction, so `n_j * D_j` evaluation points suffice to determine
79/// it. The committed LDE coset (size `n_j * B`, with `B >= D_j`) contains `gJ_j` as a
80/// subset, so the truncated view the caller passes in is zero-copy.
81#[allow(clippy::too_many_arguments)]
82pub fn evaluate_constraints_into<F, EF, A>(
83    output: &mut [EF],
84    air: &A,
85    main_on_gj: &BitReversedMatrixView<RowMajorMatrixView<'_, F>>,
86    aux_on_gj: &BitReversedMatrixView<RowMajorMatrixView<'_, F>>,
87    eval_domain: &EvaluationDomain<F>,
88    alpha: EF,
89    randomness: &[EF],
90    public_values: &[F],
91    periodic_lde: &PeriodicLde<F>,
92    layout: &ConstraintLayout,
93    permutation_values: &[EF],
94    inv_z_h: &[F],
95) where
96    F: TwoAdicField,
97    EF: ExtensionField<F>,
98    PackedExt<F, EF>: Algebra<EF> + Algebra<PackedVal<F>> + BasedVectorSpace<PackedVal<F>>,
99    A: LiftedAir<F, EF>,
100{
101    type P<F> = PackedVal<F>;
102    type PE<F, EF> = PackedExt<F, EF>;
103
104    let quotient_degree = eval_domain.quotient_degree();
105    let gj_height = eval_domain.size();
106    assert_eq!(output.len(), gj_height);
107    let width = P::<F>::WIDTH;
108
109    assert_eq!(gj_height % width, 0, "quotient height must be divisible by packing width");
110    assert_eq!(inv_z_h.len(), quotient_degree, "inv_z_h length must equal D_j");
111    // Bitmask for `i % inv_z_h.len()`; len is `2^log_blowup` by construction.
112    let inv_z_h_mask: usize = inv_z_h.len() - 1;
113
114    // Precompute selectors over the quotient evaluation coset.
115    let sels = eval_domain.selectors();
116
117    // ─── Decompose alpha powers by constraint layout ───
118    let aux_ef_width = air.aux_width();
119    let constraint_count = layout.total_constraints();
120    let base_count = layout.base_indices.len();
121    let ext_count = layout.ext_indices.len();
122    let (base_alpha_powers, ext_alpha_powers) = layout.decompose_alpha(alpha);
123
124    // Main trace width
125    let main_width = main_on_gj.width();
126
127    // Pack randomness for aux trace
128    let packed_randomness: Vec<PE<F, EF>> = randomness.iter().copied().map(Into::into).collect();
129
130    // Pack permutation values
131    let packed_perm_values: Vec<PE<F, EF>> =
132        permutation_values.iter().copied().map(Into::into).collect();
133
134    let main_vals = main_on_gj.inner.values;
135    let aux_vals = aux_on_gj.inner.values;
136    let aux_scalar_width = aux_on_gj.width();
137    let main_trace_view = RowMajorMatrixView::new(main_vals, main_width);
138    let aux_trace_view = RowMajorMatrixView::new(aux_vals, aux_scalar_width);
139
140    let points_per_task = width * ROW_BLOCKS_PER_PARALLEL_TASK;
141
142    let eval_big_slice = |main_buf: &mut Vec<P<F>>,
143                          aux_base_buf: &mut Vec<P<F>>,
144                          aux_pe_buf: &mut Vec<PE<F, EF>>,
145                          g: usize,
146                          big_slice: &mut [EF]| {
147        for (sub_r, chunk) in big_slice.chunks_exact_mut(width).enumerate() {
148            let r = g * ROW_BLOCKS_PER_PARALLEL_TASK + sub_r;
149            let i_start = r * width;
150
151            // Extract packed selectors from precomputed vectors
152            let selectors = sels.packed_at::<P<F>>(i_start);
153
154            // Get main trace as packed row pair (stays in base field)
155            collect_vertically_packed_row_pair_bitrev_into::<F, P<F>>(
156                &main_trace_view,
157                i_start,
158                quotient_degree,
159                main_buf,
160            );
161            let main_mat = RowMajorMatrixView::new(main_buf.as_slice(), main_width);
162
163            // Get aux trace as packed row pair and convert to packed extension field
164            collect_vertically_packed_row_pair_bitrev_into::<F, P<F>>(
165                &aux_trace_view,
166                i_start,
167                quotient_degree,
168                aux_base_buf,
169            );
170
171            // Convert from packed base field to packed extension field
172            // Each EF element is formed from DIMENSION consecutive base field elements
173            aux_pe_buf.clear();
174            aux_pe_buf.reserve(aux_ef_width * 2);
175            for i in 0..aux_ef_width * 2 {
176                aux_pe_buf.push(PE::<F, EF>::from_basis_coefficients_fn(|j| {
177                    aux_base_buf[i * EF::DIMENSION + j]
178                }));
179            }
180            let aux_mat = RowMajorMatrixView::new(aux_pe_buf.as_slice(), aux_ef_width);
181
182            // Get packed periodic values
183            let periodic_values: Vec<P<F>> = periodic_lde.packed_values_at(i_start).collect();
184
185            // Build packed folder and evaluate constraints
186            let mut folder: ProverConstraintFolder<'_, F, EF, P<F>, PE<F, EF>> =
187                ProverConstraintFolder {
188                    main: RowWindow::from_view(&main_mat),
189                    aux: RowWindow::from_view(&aux_mat),
190                    packed_randomness: &packed_randomness,
191                    public_values,
192                    periodic_values: &periodic_values,
193                    permutation_values: &packed_perm_values,
194                    selectors,
195                    base_alpha_powers: &base_alpha_powers,
196                    ext_alpha_powers: &ext_alpha_powers,
197                    constraint_index: 0,
198                    constraint_count,
199                    base_constraints: Vec::with_capacity(base_count),
200                    ext_constraints: Vec::with_capacity(ext_count),
201                    _phantom: PhantomData,
202                };
203
204            #[cfg(debug_assertions)]
205            air.is_valid_builder(&folder).expect("builder dimensions must match AIR");
206            air.eval(&mut folder);
207            let folded = folder.finalize_constraints();
208
209            // Unpack the folded result, multiply by 1/Z_H (modular indexing since Z_H
210            // takes only D_j distinct values on gJ_j), and write into the output chunk.
211            for (k, (slot, val)) in
212                chunk.iter_mut().zip(PE::<F, EF>::to_ext_iter([folded])).enumerate()
213            {
214                *slot = val * inv_z_h[(i_start + k) & inv_z_h_mask];
215            }
216        }
217    };
218
219    #[cfg(feature = "parallel")]
220    output.par_chunks_mut(points_per_task).enumerate().for_each_init(
221        || (Vec::<P<F>>::new(), Vec::<P<F>>::new(), Vec::<PE<F, EF>>::new()),
222        |(main_buf, aux_base_buf, aux_pe_buf), (g, big_slice)| {
223            eval_big_slice(main_buf, aux_base_buf, aux_pe_buf, g, big_slice);
224        },
225    );
226
227    #[cfg(not(feature = "parallel"))]
228    {
229        let mut main_buf = Vec::<P<F>>::new();
230        let mut aux_base_buf = Vec::<P<F>>::new();
231        let mut aux_pe_buf = Vec::<PE<F, EF>>::new();
232        output.par_chunks_mut(points_per_task).enumerate().for_each(|(g, big_slice)| {
233            eval_big_slice(&mut main_buf, &mut aux_base_buf, &mut aux_pe_buf, g, big_slice);
234        });
235    }
236}
