1//! Prover-side periodic column handling.
2//!
3//! Periodic columns are stored as LDE values in a row-major matrix for efficient
4//! constraint evaluation on the LDE domain. The key optimization is that a periodic
5//! column with period `p` only needs `p * blowup` LDE values (not `trace_height * blowup`),
6//! which are accessed via modular indexing.
7//!
8//! Uses NaiveDft since periodic column periods are typically small.
9
10use miden_lifted_air::log2_strict_u8;
11use p3_dft::{NaiveDft, TwoAdicSubgroupDft};
12use p3_field::{PackedValue, TwoAdicField};
13use p3_matrix::{Matrix, dense::RowMajorMatrix};
14
15use crate::domain::{Coset, EvaluationDomain};

17/// Prover-side periodic LDE values for constraint evaluation.
18///
19/// Stores precomputed LDE values as a row-major matrix in natural order. The key insight
20/// is that by repeating each column's values to the maximum period, we can use batch DFT
21/// methods and store only `max_period * blowup` rows instead of `trace_height * blowup`.
22///
23/// A periodic column of period `p` repeats every `p` rows on the trace domain, so its LDE
24/// repeats every `p * blowup` rows on the quotient/LDE domains. We therefore only need to
25/// store `p * blowup` rows for that column. To share one buffer across many periodic columns,
26/// we repeat each column up to `max_period` and LDE-extend once; columns with smaller periods
27/// are accessed via modular indexing.
28#[derive(Clone, Debug)]
29pub(super) struct PeriodicLde<F: TwoAdicField> {
30    /// LDE values in natural order (height = max_period * blowup).
31    /// `None` when there are no periodic columns.
32    ldes: Option<RowMajorMatrix<F>>,
33}
34
35impl<F: TwoAdicField> PeriodicLde<F> {
36    /// Build periodic LDEs from a periodic column matrix.
37    ///
38    /// Takes the output of [`crate::air::LiftedAir::periodic_columns_matrix`], where
39    /// columns with smaller periods have been repeated cyclically to the maximum period.
40    /// Uses NaiveDft since periodic column periods are typically small.
41    ///
42    /// # Arguments
43    /// - `domain`: The evaluation domain (trace + LDE + constraint degree)
44    /// - `repeated_matrix`: Periodic columns extended to a common height (max period), or `None` if
45    ///   there are no periodic columns
46    ///
47    /// # Panics
48    /// Panics if the matrix height exceeds the trace height or is not a power of two.
49    pub fn build(domain: &EvaluationDomain<F>, repeated_matrix: Option<RowMajorMatrix<F>>) -> Self {
50        let Some(repeated_matrix) = repeated_matrix else {
51            return Self { ldes: None };
52        };
53
54        let max_period = repeated_matrix.height();
55        let log_max_period = log2_strict_u8(max_period);
56        assert!(
57            domain.log_trace_height() >= log_max_period,
58            "periodic column period ({max_period}) exceeds trace height ({})",
59            1 << domain.log_trace_height() as usize,
60        );
61        let log_blowup = domain.log_quotient_degree() as usize;
62
63        // Compute the coset shift for the max-period subgroup.
64        //
65        // Periodic polynomials are naturally defined on a subgroup of order `max_period`.
66        // The quotient evaluation coset shares the LDE coset's shift; map it down
67        // from trace height to `max_period` via a power-of-two ratio.
68        let log_ratio = domain.log_trace_height() - log_max_period;
69        let period_shift: F = domain.shift().exp_power_of_2(log_ratio as usize);
70
71        // Compute LDE using NaiveDft (periods are small)
72        let ldes = NaiveDft
73            .coset_lde_batch(repeated_matrix, log_blowup, period_shift)
74            .to_row_major_matrix();
75
76        Self { ldes: Some(ldes) }
77    }
78
79    /// Get packed values for consecutive natural indices [i, i+1, ..., i+WIDTH-1].
80    ///
81    /// Returns an empty iterator when there are no periodic columns.
82    #[inline]
83    pub fn packed_values_at<P: PackedValue<Value = F>>(
84        &self,
85        i: usize,
86    ) -> impl Iterator<Item = P> + '_ {
87        self.ldes.iter().flat_map(move |ldes| {
88            let height = ldes.height();
89            (0..ldes.width()).map(move |col| {
90                P::from_fn(|k| {
91                    let row = (i + k) % height;
92                    // SAFETY: `row < height` is guaranteed by the modulo operation,
93                    // and `col < width` is guaranteed by the iterator bounds (0..ldes.width()).
94                    unsafe { ldes.get_unchecked(row, col) }
95                })
96            })
97        })
98    }
99}

116    /// Verify that periodic LDE values match the full LDE computation.
117    fn assert_periodic_lde_matches_full(
118        columns: &[Vec<gl::Felt>],
119        log_trace_height: u8,
120        log_blowup: u8,
121    ) {
122        let trace_height = 1 << log_trace_height as usize;
123        let lde_height = trace_height << log_blowup as usize;
124
125        // Create an evaluation domain at max height (no lifting), with constraint
126        // degree = log_blowup (the max-degree case for this test).
127        let lifted: LiftedDomain<gl::Felt> = canonical_domain(log_trace_height, log_blowup);
128        let domain = lifted.evaluation_domain(log_blowup);
129
130        // Build the repeated matrix (same logic as periodic_columns_matrix)
131        let max_period = columns.iter().map(Vec::len).max().unwrap();
132        let num_cols = columns.len();
133        let mut values = Vec::with_capacity(max_period * num_cols);
134        for row in 0..max_period {
135            for col in columns {
136                values.push(col[row % col.len()]);
137            }
138        }
139        let repeated_matrix = RowMajorMatrix::new(values, num_cols);
140
141        let periodic_lde = PeriodicLde::build(&domain, Some(repeated_matrix));
142
143        // Compute expected LDE for each column via full expansion (natural order).
144        // The PeriodicLde uses `domain.lde_shift()` internally — match it here.
145        let expected_shift = domain.shift();
146        let expected: Vec<Vec<gl::Felt>> = columns
147            .iter()
148            .map(|col| {
149                let full: Vec<gl::Felt> = (0..trace_height).map(|i| col[i % col.len()]).collect();
150                let matrix = RowMajorMatrix::new(full, 1);
151                NaiveDft
152                    .coset_lde_batch(matrix, log_blowup.into(), expected_shift)
153                    .to_row_major_matrix()
154                    .values
155            })
156            .collect();
157
158        // Verify all LDE rows match (natural indices)
159        let ldes = periodic_lde.ldes.as_ref().expect("expected Some for non-empty columns");
160        let height = ldes.height();
161        for i in 0..lde_height {
162            let row = i % height;
163            let actual: Vec<gl::Felt> = ldes.row_slice(row).unwrap().to_vec();
164            for (col_idx, (&actual_val, expected_col)) in actual.iter().zip(&expected).enumerate() {
165                assert_eq!(actual_val, expected_col[i], "col {col_idx} mismatch at row {i}");
166            }
167        }
168
169        // Verify packed_values_at returns correct packed values
170        type P = gl::PackedFelt;
171        let pack_width = P::WIDTH;
172        for start in (0..lde_height).step_by(pack_width) {
173            let packed: Vec<P> = periodic_lde.packed_values_at(start).collect();
174            assert_eq!(packed.len(), columns.len());
175
176            // Verify each lane matches scalar access
177            for k in 0..pack_width {
178                let idx = start + k;
179                let row = idx % height;
180                let scalar: Vec<gl::Felt> = ldes.row_slice(row).unwrap().to_vec();
181                for (col_idx, (&packed_val, &scalar_val)) in packed.iter().zip(&scalar).enumerate()
182                {
183                    assert_eq!(
184                        packed_val.as_slice()[k],
185                        scalar_val,
186                        "packed mismatch col {col_idx} row {idx} lane {k}"
187                    );
188                }
189            }
190        }
191    }

193    #[test]
194    fn test_periodic_lde_matches_full_lde() {
195        // Period 2, blowup 2
196        assert_periodic_lde_matches_full(&[vec![gl::Felt::ZERO, gl::Felt::ONE]], 3, 1);
197
198        // Period 4, blowup 2
199        let col4: Vec<gl::Felt> = [1, 2, 3, 4].into_iter().map(gl::Felt::from_u64).collect();
200        assert_periodic_lde_matches_full(&[col4], 3, 1);
201
202        // Period 2, blowup 8 (higher blowup)
203        let col2: Vec<gl::Felt> = [5, 7].into_iter().map(gl::Felt::from_u64).collect();
204        assert_periodic_lde_matches_full(&[col2], 4, 3);
205
206        // Multiple columns with different periods
207        let col_p2: Vec<gl::Felt> = [1, 2].into_iter().map(gl::Felt::from_u64).collect();
208        let col_p4: Vec<gl::Felt> = [10, 20, 30, 40].into_iter().map(gl::Felt::from_u64).collect();
209        assert_periodic_lde_matches_full(&[col_p2, col_p4], 3, 2);
210    }
