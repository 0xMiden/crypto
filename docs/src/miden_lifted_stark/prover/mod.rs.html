1//! Lifted STARK prover.
2//!
3//! This module provides:
4//! - [`prove_single`]: Prove a single AIR instance.
5//! - [`prove_multi`]: Prove multiple AIR instances with traces of different heights.
6//!
7//! These functions write the proof into a [`miden_stark_transcript::ProverChannel`]
8//! (commitments, grinding witnesses, and openings).
9//!
10//! # Fiat-Shamir / transcript binding (initial challenger state)
11//!
12//! This crate intentionally does **not** prescribe the *initial* transcript state.
13//! Different applications may transport some statement data out-of-band (e.g. public
14//! inputs shipped alongside a proof) and do not want it duplicated inside the proof.
15//!
16//! The flip side is that the caller must ensure the Fiat-Shamir challenger inside
17//! `channel` is initialized and bound to the statement in whatever way is appropriate
18//! for their use case.
19//!
20//! The protocol implementation assumes that *all inputs that may vary* (including
21//! `public_values` and `var_len_public_inputs`) have been observed by the challenger.
22//! This is required so callers can avoid including public inputs in the proof when
23//! they are available out-of-band.
24//!
25//! In particular, the caller **MUST** bind both `public_values` and
26//! `var_len_public_inputs` to the challenger state. Otherwise, Fiat-Shamir
27//! challenges sampled during proving/verification are independent of the public
28//! inputs.
29//!
30//! Because the `air` is a concrete Rust type, you often do not need to explicitly
31//! observe it into the challenger *if your application has a single fixed AIR version
32//! compiled in*. If you support multiple AIRs/versions/protocol configurations, you
33//! should also bind some `air_id` / version tag / domain separator to prevent
34//! cross-protocol replay.
35//!
36//! ## Recommended pattern: pre-seed the challenger (no proof bloat)
37//!
38//! The transcript channel traits do not currently expose an “observe-only” operation
39//! (observe into the challenger without recording into the proof). If you want to
40//! bind public inputs without bloating the proof, the most ergonomic pattern is to
41//! pre-seed the challenger and then construct the transcript.
42//!
43//!
44//! ```ignore
45//! use p3_goldilocks::{Goldilocks, Poseidon2Goldilocks};
46//! use p3_challenger::{CanObserve, DuplexChallenger};
47//! use p3_dft::Radix2DitParallel;
48//! use p3_field::extension::BinomialExtensionField;
49//! use crate::{StarkConfig, prove_multi};
50//! use crate::{AirWitness, AirInstance};
51//! use crate::lmcs::LmcsConfig;
52//! use miden_stateful_hasher::StatefulSponge;
53//! use miden_stark_transcript::{ProverTranscript, VerifierTranscript};
54//! use p3_symmetric::TruncatedPermutation;
55//! use rand::rngs::SmallRng;
56//! use rand::SeedableRng;
57//!
58//! // Concrete instantiation matching the repository's Goldilocks+Poseidon2 defaults.
59//! const WIDTH: usize = 12;
60//! const RATE: usize = 8;
61//! const DIGEST: usize = 4;
62//!
63//! type F = Goldilocks;
64//! type EF = BinomialExtensionField<F, 2>;
65//! type P = <F as p3_field::Field>::Packing;
66//! type Perm = Poseidon2Goldilocks<WIDTH>;
67//! type Challenger = DuplexChallenger<F, Perm, WIDTH, RATE>;
68//! type Sponge = StatefulSponge<Perm, WIDTH, RATE, DIGEST>;
69//! type Compress = TruncatedPermutation<Perm, 2, DIGEST, WIDTH>;
70//! type Lmcs = LmcsConfig<P, P, Sponge, Compress, WIDTH, DIGEST>;
71//! type Dft = Radix2DitParallel<F>;
72//!
73//! // --- Build config ---
74//! let mut rng = SmallRng::seed_from_u64(0);
75//! let perm = Perm::new_from_rng_128(&mut rng);
76//! let sponge = Sponge::new(perm.clone());
77//! let compress = Compress::new(perm.clone());
78//! let config = StarkConfig { pcs: /* ... */, lmcs: Lmcs::new(sponge, compress), dft: Dft::default() };
79//!
80//! // --- Statement (out-of-band) ---
81//! let public_values: Vec<F> = /* ... */;
82//!
83//! // --- Prover: bind statement into Fiat-Shamir ---
84//! let mut ch = Challenger::new(perm.clone());
85//! // Domain separator: one Goldilocks element per ASCII byte.
86//! ch.observe_slice(&b"LSTARK0".map(|b| F::from_u8(b)));
87//! ch.observe_slice(&public_values);
88//! // If your app supports multiple AIRs/versions, also observe an application-level air_id here.
89//! // --- Prover ---
90//! let witness = AirWitness::new(&trace, &public_values, &[]);
91//! let output = prove_multi(&config, &[(&air, witness, &aux_builder)], ch)?;
92//!
93//! // --- Verifier: same binding, then verify ---
94//! let mut ch = Challenger::new(perm);
95//! ch.observe_slice(&b"LSTARK0".map(|b| F::from_u8(b)));
96//! ch.observe_slice(&public_values);
97//!
98//! let instance = AirInstance { public_values: &public_values, var_len_public_inputs: &[] };
99//! let verifier_digest = crate::verify_multi(&config, &[(&air, instance)], &output.proof, ch)?;
100//! assert_eq!(output.digest, verifier_digest);
101//! ```
102//!
103//! ## Alternative: write statement data into the transcript
104//!
105//! If you do want the proof to be self-contained, you can `send_field_slice` the
106//! statement data into the transcript before calling [`prove_single`] / [`prove_multi`].
107//! In that case, the verifier must *first* read and validate those values from the
108//! transcript, and only then call [`crate::verifier::verify_multi`].
109//!
110//! More generally, you can choose to obtain some parameters from the channel. If you
111//! do, you are responsible for validating them before use.
112//!
113//! # Multi-trace ordering
114//!
115//! [`prove_multi`] requires `instances` in ascending trace height order (smallest
116//! first). Internally, traces are committed and quotient numerators are accumulated
117//! in that order using cyclic extension, before a single vanishing division on the
118//! largest quotient domain.

147/// Errors that can occur during proving.
148#[derive(Debug, Error)]
149pub enum ProverError {
150    #[error("instance validation failed: {0}")]
151    Instance(#[from] InstanceValidationError),
152    #[error(
153        "constraint degree exceeds blowup: \
154         log_quotient_degree {log_quotient_degree} > log_blowup {log_blowup}"
155    )]
156    ConstraintDegreeTooHigh { log_quotient_degree: u8, log_blowup: u8 },
157}

159/// Prove a single AIR.
160///
161/// Transcript warning: the protocol assumes the challenger inside `channel` has
162/// already observed all variable statement inputs (in particular `public_values`
163/// and `var_len_public_inputs`). This lets callers keep public inputs out of the
164/// proof when they are available out-of-band. See the module-level docs for
165/// recommended patterns.
166///
167/// This is a convenience wrapper around [`prove_multi`] for the single-AIR case.
168///
169/// # Returns
170/// `Ok(StarkOutput { digest, proof })` on success, or a `ProverError` if validation fails.
171pub fn prove_single<F, EF, A, B, SC>(
172    config: &SC,
173    air: &A,
174    trace: &RowMajorMatrix<F>,
175    public_values: &[F],
176    var_len_public_inputs: VarLenPublicInputs<'_, F>,
177    aux_builder: &B,
178    challenger: SC::Challenger,
179) -> Result<StarkOutput<F, EF, SC>, ProverError>
180where
181    F: TwoAdicField,
182    EF: ExtensionField<F>,
183    SC: StarkConfig<F, EF>,
184    A: LiftedAir<F, EF>,
185    B: AuxBuilder<F, EF>,
186{
187    let witness = AirWitness::new(trace, public_values, var_len_public_inputs);
188    prove_multi(config, &[(air, witness, aux_builder)], challenger)
189}

191/// Prove multiple AIRs with traces of different heights.
192///
193/// Transcript warning: the protocol assumes the challenger inside `channel` has
194/// already observed all variable statement inputs (in particular each instance's
195/// `public_values`). This lets callers keep public inputs out of the proof when they
196/// are available out-of-band.
197///
198/// Instances must be provided in ascending height order (smallest first).
199/// Each trace may have a different height that is a power of 2. The quotient
200/// numerators are accumulated using cyclic extension:
201///
202/// 1. Compute numerator N₀ on the smallest quotient domain.
203/// 2. For each subsequent trace `j`:
204///    - cyclically extend the accumulator to the new (larger) quotient domain,
205///    - fold with the random challenge β: acc = acc·β + Nⱼ.
206/// 3. Divide by `Z_H` once on the largest quotient domain.
207///
208/// The ordering is important: cyclic extension only grows the accumulator (it does not
209/// shrink), and both prover and verifier must assign the same powers of `beta` to each
210/// instance's contribution.
211///
212/// # Arguments
213/// - `config`: STARK configuration (PCS params, LMCS, DFT)
214/// - `instances`: Pairs of (AIR, witness, aux_builder)
215/// - `challenger`: Fiat-Shamir challenger (heights are observed before use)
216///
217/// # Returns
218/// `Ok(StarkOutput { digest, proof })` on success, or a `ProverError` if validation fails.
219#[instrument(name = "prove", skip_all)]
220pub fn prove_multi<F, EF, A, B, SC>(
221    config: &SC,
222    instances: &[(&A, AirWitness<'_, F>, &B)],
223    mut challenger: SC::Challenger,
224) -> Result<StarkOutput<F, EF, SC>, ProverError>
225where
226    F: TwoAdicField,
227    EF: ExtensionField<F>,
228    SC: StarkConfig<F, EF>,
229    A: LiftedAir<F, EF>,
230    B: AuxBuilder<F, EF>,
231{
232    let verifier_instances: Vec<_> =
233        instances.iter().map(|(air, w, _)| (*air, w.to_instance())).collect();
234    let trace_heights: Vec<usize> = instances.iter().map(|(_, w, _)| w.trace.height()).collect();
235    let instance_shapes = InstanceShapes::from_trace_heights(trace_heights)?;
236
237    let log_blowup = config.pcs().log_blowup();
238
239    // Validate AIR structure, instance dimensions, heights, and trace widths.
240    let log_max_trace_height = validate_inputs(&verifier_instances, &instance_shapes, log_blowup)?;
241    for (air, w, _) in instances {
242        if w.trace.width() != air.width() {
243            return Err(InstanceValidationError::WidthMismatch {
244                expected: air.width(),
245                actual: w.trace.width(),
246            }
247            .into());
248        }
249    }
250
251    // Observe shape metadata before creating the transcript.
252    instance_shapes.observe::<F, _>(&mut challenger);
253
254    let mut channel = ProverTranscript::new(challenger);
255
256    // Clear the challenger's absorb buffer after observing instance shapes by
257    // squeezing a throwaway extension element. This guarantees later sampled
258    // challenges depend on all prior inputs regardless of sponge state.
259    let _instance_challenge: EF = channel.sample_algebra_element::<EF>();
260
261    // Infer constraint degree from symbolic AIR analysis (max across all AIRs)
262    let log_constraint_degree =
263        instances.iter().map(|(air, ..)| air.log_quotient_degree()).max().unwrap_or(1) as u8;
264
265    if log_constraint_degree > log_blowup {
266        return Err(ProverError::ConstraintDegreeTooHigh {
267            log_quotient_degree: log_constraint_degree,
268            log_blowup,
269        });
270    }
271
272    let log_lde_height = log_max_trace_height + log_blowup;
273
274    // Max LDE coset (for the largest trace, no lifting)
275    let max_lde_coset = LiftedCoset::unlifted(log_max_trace_height, log_blowup);
276    let max_quotient_coset = max_lde_coset.quotient_domain(log_constraint_degree);
277    let max_quotient_height = max_quotient_coset.lde_height();
278
279    // 1. Commit all main traces (trace order — ascending height).
280    //
281    // Clone with blowup × capacity so the DFT resize doesn't reallocate.
282    let blowup = 1 << log_blowup as usize;
283    let main_traces: Vec<_> = instances
284        .iter()
285        .map(|(_, w, _)| {
286            let src = &w.trace.values;
287            let mut values = Vec::with_capacity(src.len() * blowup);
288            values.extend_from_slice(src);
289            RowMajorMatrix::new(values, w.trace.width())
290        })
291        .collect();
292    let main_committed =
293        info_span!("commit to main traces").in_scope(|| commit_traces(config, main_traces));
294    channel.send_commitment(main_committed.root());
295
296    // 2. Sample randomness and build aux traces for all AIRs
297    let max_num_randomness =
298        instances.iter().map(|(air, ..)| air.num_randomness()).max().unwrap_or(0);
299
300    let randomness: Vec<EF> = (0..max_num_randomness)
301        .map(|_| channel.sample_algebra_element::<EF>())
302        .collect();
303
304    // Build aux traces via AuxBuilder
305    let (aux_traces_ef, all_aux_values): (Vec<RowMajorMatrix<EF>>, Vec<Vec<EF>>) =
306        info_span!("build aux traces").in_scope(|| {
307            let mut traces = Vec::with_capacity(instances.len());
308            let mut values = Vec::with_capacity(instances.len());
309            for (air, w, aux_builder) in instances {
310                let num_rand = air.num_randomness();
311                let (aux, aux_vals) = aux_builder.build_aux_trace(w.trace, &randomness[..num_rand]);
312
313                assert_eq!(aux.width(), air.aux_width(), "aux trace width mismatch");
314                assert_eq!(
315                    aux_vals.len(),
316                    air.num_aux_values(),
317                    "aux values length mismatch: build_aux_trace returned {} values, \
318                     but num_aux_values() is {}",
319                    aux_vals.len(),
320                    air.num_aux_values()
321                );
322                assert_eq!(aux.height(), w.trace.height());
323                traces.push(aux);
324                values.push(aux_vals);
325            }
326            (traces, values)
327        });
328
329    // Flatten EF -> F and commit aux traces
330    let aux_traces: Vec<RowMajorMatrix<F>> = aux_traces_ef
331        .into_iter()
332        .map(|aux| {
333            let base_width = aux.width() * EF::DIMENSION;
334            let base_values = <EF as BasedVectorSpace<F>>::flatten_to_base(aux.values);
335            RowMajorMatrix::new(base_values, base_width)
336        })
337        .collect();
338
339    let aux_committed =
340        info_span!("commit to aux traces").in_scope(|| commit_traces(config, aux_traces));
341    channel.send_commitment(aux_committed.root());
342
343    // Observe aux values into the transcript (binds to Fiat-Shamir state).
344    // When no AIR has aux columns, each entry is empty so nothing is sent.
345    for vals in &all_aux_values {
346        for &val in vals {
347            channel.send_algebra_element(val);
348        }
349    }
350
351    // 4. Sample constraint folding alpha and accumulation beta
352    let alpha: EF = channel.sample_algebra_element::<EF>();
353    let beta: EF = channel.sample_algebra_element::<EF>();
354
355    // 5. Evaluate constraints and accumulate with beta folding.
356    //
357    // Single accumulator, processed in trace order (ascending height):
358    //   1. Cyclically extend accumulator to the next quotient height
359    //   2. Multiply every element by beta (Horner)
360    //   3. Add constraint evaluations in-place: acc[i] += eval(i)
361    //
362    // Pre-allocate with LDE capacity so commit_quotient's resize doesn't reallocate.
363    let constraint_degree = 1 << log_constraint_degree as usize;
364    let mut accumulator: Vec<EF> = Vec::with_capacity(max_quotient_height * blowup);
365
366    // Pre-compute constraint layouts for each AIR (base/ext index mapping)
367    let layouts: Vec<_> = instances
368        .iter()
369        .map(|(air, ..)| get_constraint_layout::<F, EF, A>(*air))
370        .collect();
371
372    info_span!("evaluate constraints").in_scope(|| {
373        for (i, (air, w, _)) in instances.iter().enumerate() {
374            let trace_height = w.trace.height();
375            let log_trace_height = log2_strict_u8(trace_height);
376
377            // Create LiftedCoset for this trace (may be lifted relative to max)
378            let this_lde_coset =
379                LiftedCoset::new(log_trace_height, log_blowup, log_max_trace_height);
380            let this_quotient_coset = this_lde_coset.quotient_domain(log_constraint_degree);
381            let this_quotient_height = this_quotient_coset.lde_height();
382
383            // Truncate the committed LDE to the quotient evaluation domain gJ (size N·D).
384            // Since B ≥ D, the committed LDE on gK (size N·B) contains gJ as a prefix in
385            // bit-reversed storage, so this is a zero-copy view.
386            let main_on_gj = main_committed.evals_on_quotient_domain(i, constraint_degree);
387            let aux_on_gj = aux_committed.evals_on_quotient_domain(i, constraint_degree);
388
389            // Build periodic LDE for this trace via coset method
390            let periodic_lde =
391                PeriodicLde::build(&this_quotient_coset, air.periodic_columns_matrix());
392
393            // Cyclically extend accumulator to this quotient height and scale by beta.
394            // On the first iteration the accumulator is empty, so this is a no-op
395            // and evaluate_constraints_into writes into a zero-filled buffer.
396            tracing::debug_span!(
397                "cyclic_extend",
398                acc_len = accumulator.len(),
399                target = this_quotient_height
400            )
401            .in_scope(|| {
402                quotient::cyclic_extend_and_scale(&mut accumulator, this_quotient_height, beta);
403            });
404
405            let aux_values_i = &all_aux_values[i];
406
407            // Add constraint evaluations in-place: accumulator[i] += eval(i)
408            info_span!("eval_instance", instance = i, height = this_quotient_height).in_scope(
409                || {
410                    evaluate_constraints_into::<F, EF, A>(
411                        &mut accumulator,
412                        *air,
413                        &main_on_gj,
414                        &aux_on_gj,
415                        &this_quotient_coset,
416                        alpha,
417                        &randomness[..air.num_randomness()],
418                        w.public_values,
419                        &periodic_lde,
420                        &layouts[i],
421                        aux_values_i,
422                    );
423                },
424            );
425        }
426    });
427
428    // Verify we have the expected size (max quotient domain)
429    assert_eq!(accumulator.len(), max_quotient_height);
430
431    // 6. Divide by vanishing polynomial once on full gJ (in-place)
432    tracing::debug_span!("divide_by_vanishing", height = max_quotient_height).in_scope(|| {
433        quotient::divide_by_vanishing_in_place::<F, EF>(&mut accumulator, &max_quotient_coset);
434    });
435
436    // 7. Commit quotient
437    let quotient_committed = info_span!("commit to quotient poly chunks")
438        .in_scope(|| quotient::commit_quotient(config, accumulator, &max_lde_coset));
439    channel.send_commitment(quotient_committed.root());
440
441    // 8. Sample OOD point (outside H and gK)
442    let z: EF = max_lde_coset.sample_ood_point(&mut channel);
443    let h = F::two_adic_generator(log_max_trace_height.into());
444    let z_next = z * h;
445
446    // 9. Open via PCS
447    let trees = vec![main_committed.tree(), aux_committed.tree(), quotient_committed.tree()];
448
449    info_span!("open").in_scope(|| {
450        open_with_channel::<F, EF, SC::Lmcs, RowMajorMatrix<F>, _, 2>(
451            config.pcs(),
452            config.lmcs(),
453            log_lde_height,
454            [z, z_next],
455            &trees,
456            &mut channel,
457        )
458    });
459
460    let (digest, transcript) = channel.finalize();
461    let proof = StarkProof { instance_shapes, transcript };
462    Ok(StarkOutput { digest, proof })
463}
