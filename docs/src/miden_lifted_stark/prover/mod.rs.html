1//! Lifted STARK prover.
2//!
3//! This module provides:
4//! - [`ProverInstance::prove`](crate::ProverInstance::prove): Prove one or more AIR instances with
5//!   traces of (possibly) different heights.
6//!
7//! [`ProverInstance::prove`](crate::ProverInstance::prove) writes the proof into a
8//! [`miden_stark_transcript::ProverChannel`] (commitments, grinding witnesses,
9//! and openings).
10//!
11//! # Fiat-Shamir / transcript binding (initial challenger state)
12//!
13//! This crate does **not** prescribe the *initial* transcript state. The caller
14//! must bind protocol and AIR configuration data before calling
15//! [`ProverInstance::prove`](crate::ProverInstance::prove). Both prover and verifier must produce
16//! identical challenger states. Concretely, the caller **MUST** observe:
17//!
18//! 1. **Protocol parameters** — e.g. the STARK configuration, blowup factor, and any
19//!    application-level domain separator.
20//!
21//! 2. **AIR configurations** — The framework does not commit to the [`MultiAir::airs`] list. The
22//!    caller MUST bind every AIR configuration into the challenger before calling
23//!    [`ProverInstance::prove`](crate::ProverInstance::prove) /
24//!    [`VerifierInstance::verify`](crate::VerifierInstance::verify). The AIR ordering on the wire
25//!    is derived deterministically from the trace heights (stable sort on `(log_trace_height,
26//!    instance_index)`), so callers do not need to commit to it separately as long as they commit
27//!    to the AIR list and trace heights match.
28//!
29//! The statement's `air_inputs` and `aux_inputs` are absorbed automatically by
30//! [`Statement::observe`](crate::air::Statement::observe), followed by protocol-level
31//! absorption of the instance count and each AIR's log trace height in instance
32//! order. Callers do not bind these themselves.
33//!
34//! ## Recommended pattern
35//!
36//! Pre-seed the challenger so statement data stays out of the proof:
37//!
38//! ```ignore
39//! // --- Bind protocol parameters + AIR configurations into Fiat-Shamir ---
40//! let mut ch = Challenger::new(perm.clone());
41//! ch.observe_slice(&b"MY_APP_V1".map(|b| F::from_u8(b)));  // domain separator
42//! ch.observe(F::from_u8(config.pcs().log_blowup()));        // protocol parameters
43//! // ... bind AIR configurations + air ordering (see below) ...
44//!
45//! // --- Prove ---
46//! let prover_instance = ProverInstance::new(&config, &prover_statement, None)?;
47//! let output = prover_instance.prove(ch)?;
48//!
49//! // --- Verify (identical binding + the same statement) ---
50//! let mut ch = Challenger::new(perm);
51//! ch.observe_slice(&b"MY_APP_V1".map(|b| F::from_u8(b)));
52//! ch.observe(F::from_u8(config.pcs().log_blowup()));
53//! let verifier_instance = VerifierInstance::new(&config, prover_instance.statement(), None)?;
54//! let verifier_digest = verifier_instance.verify(&output.proof, ch)?;
55//! assert_eq!(output.digest, verifier_digest);
56//! ```
57//!
58//! ## Multi-AIR binding example
59//!
60//! ```text
61//! // Commit to AIRs in instance order — the proof's wire-format ordering is
62//! // derived from the heights inside the framework, so binding the instance
63//! // order is enough.
64//! for air in statement.airs() {
65//!     challenger.observe(air.commitment());
66//! }
67//! ```

105/// Prover-side bundle: a [`StarkConfig`], a borrowed [`ProverStatement`], and
106/// the optional borrowed [`Preprocessed`] data.
107///
108/// Construction validates preprocessed presence parity (and, when present, the
109/// bundle's shape against the AIRs and STARK config), so holding a
110/// `ProverInstance` is a guarantee its preprocessed shape is consistent;
111/// proving never re-checks it.
112pub struct ProverInstance<'a, F, EF, MA, SC>
113where
114    F: TwoAdicField,
115    EF: ExtensionField<F>,
116    MA: MultiAir<F, EF>,
117    SC: StarkConfig<F, EF>,
118{
119    config: &'a SC,
120    prover_statement: &'a ProverStatement<F, EF, MA>,
121    preprocessed: Option<&'a Preprocessed<F, SC::Lmcs>>,
122}
123
124impl<'a, F, EF, MA, SC> ProverInstance<'a, F, EF, MA, SC>
125where
126    F: TwoAdicField,
127    EF: ExtensionField<F>,
128    MA: MultiAir<F, EF>,
129    SC: StarkConfig<F, EF>,
130{
131    /// Bundle a config + prover statement with an optional preprocessed bundle.
132    ///
133    /// `preprocessed` must be `Some` exactly when some AIR declares preprocessed
134    /// columns; otherwise this errors with
135    /// [`PreprocessedValidationError::PresenceMismatch`]. When both hold, the
136    /// bundle's raw and committed shapes are validated against the AIRs and
137    /// config.
138    pub fn new(
139        config: &'a SC,
140        prover_statement: &'a ProverStatement<F, EF, MA>,
141        preprocessed: Option<&'a Preprocessed<F, SC::Lmcs>>,
142    ) -> Result<Self, PreprocessedValidationError> {
143        let expected =
144            prover_statement.statement().airs().iter().any(|a| a.preprocessed_width() > 0);
145        let actual = preprocessed.is_some();
146        if expected != actual {
147            return Err(PreprocessedValidationError::PresenceMismatch { expected, actual });
148        }
149        if let Some(p) = preprocessed {
150            validate_preprocessed(config, prover_statement, p)?;
151        }
152        Ok(Self { config, prover_statement, preprocessed })
153    }
154
155    /// Prove this instance.
156    pub fn prove(&self, challenger: SC::Challenger) -> Result<StarkOutput<F, EF, SC>, ProverError> {
157        prove(self, challenger)
158    }
159
160    /// Borrow the STARK configuration.
161    pub fn config(&self) -> &SC {
162        self.config
163    }
164
165    /// Borrow the wrapped air-crate prover statement.
166    pub fn prover_statement(&self) -> &ProverStatement<F, EF, MA> {
167        self.prover_statement
168    }
169
170    /// Borrow the verifier-side statement (the AIRs + public inputs).
171    pub fn statement(&self) -> &Statement<F, EF, MA> {
172        self.prover_statement.statement()
173    }
174
175    /// Commitment to the preprocessed tree, for the verifier's
176    /// [`VerifierInstance::new`](crate::VerifierInstance::new); `None` when there
177    /// is none.
178    pub fn preprocessed_commitment(&self) -> Option<<SC::Lmcs as Lmcs>::Commitment> {
179        self.preprocessed.map(Preprocessed::commitment)
180    }
181
182    /// Borrow the preprocessed bundle, if any.
183    pub(crate) fn preprocessed(&self) -> Option<&Preprocessed<F, SC::Lmcs>> {
184        self.preprocessed
185    }
186}

188/// Prove a [`ProverInstance`].
189///
190/// The caller's challenger must already be bound to protocol parameters and
191/// AIR configurations — see the module-level docs. The statement's `air_inputs`
192/// and `aux_inputs` are absorbed internally via
193/// [`Statement::observe`](crate::air::Statement::observe); both prover and verifier
194/// must carry the same statement.
195///
196/// # Trust contract
197///
198/// `prove` validates ONLY untrusted runtime inputs and returns typed
199/// [`ProverError`] on failure. AIR structural correctness is the
200/// implementer's contract — call [`crate::debug::assert_prover_setup`]
201/// from your test harness to enforce it in debug builds.
202///
203/// ## Validated
204/// - per AIR: `air.num_public_values() == statement.air_inputs().len()`
205/// - `statement.aux_inputs().len() <= multi_air.max_aux_inputs()`
206/// - `prover_statement.traces().len() == statement.airs().len()` and `<= u8::MAX + 1`
207/// - per AIR: `trace.width() == air.width()`
208/// - per AIR: `trace.height().is_power_of_two()`
209/// - per AIR: `trace.height() >= max periodic column length`
210/// - per AIR: `log_quotient_degree(air) <= config.pcs().log_blowup()`
211/// - LDE domain fits the field's two-adicity (via `LiftedDomain::try_canonical`)
212/// - Preprocessed bundle presence, shape, and config-dependent LDE dimensions via `ProverInstance`
213///   construction
214///
215/// ## Trusted (NOT validated)
216/// - AIR structural shape (positive `aux_width`, power-of-two periodic columns, window size 2)
217/// - [`crate::air::ProverStatement::build_aux_traces`] output dimensions — a malformed output is
218///   caught by the LDE/commit (panic) or by verification, since the verifier re-derives these
219///   shapes.
220///
221/// # Arguments
222/// - `instance`: the config, the validated prover statement (AIRs, shared `air_inputs`, and per-AIR
223///   traces, all in instance order), and the optional preprocessed bundle
224/// - `challenger`: Fiat-Shamir challenger pre-bound to protocol parameters and AIR configurations
225///
226/// # Returns
227/// `Ok(StarkOutput { digest, proof })`, or a [`ProverError`] if validation fails.
228#[instrument(name = "prove", skip_all)]
229pub(crate) fn prove<F, EF, MA, SC>(
230    instance: &ProverInstance<'_, F, EF, MA, SC>,
231    mut challenger: SC::Challenger,
232) -> Result<StarkOutput<F, EF, SC>, ProverError>
233where
234    F: TwoAdicField,
235    EF: ExtensionField<F>,
236    SC: StarkConfig<F, EF>,
237    MA: MultiAir<F, EF>,
238{
239    // --- Trust boundary (see doc-block above). -------------------------------
240    let config = instance.config();
241    let prover_statement = instance.prover_statement();
242    let preprocessed = instance.preprocessed();
243    let statement = prover_statement.statement();
244    let airs = statement.airs();
245    let air_inputs = statement.air_inputs();
246    let traces = prover_statement.traces();
247    let trace_heights: Vec<usize> = traces.iter().map(Matrix::height).collect();
248    let trace_order = TraceOrder::from_trace_heights::<F, EF, _>(airs, &trace_heights)
249        .expect("ProverStatement::new should reject malformed heights");
250
251    // Map each AIR to its preprocessed trace index. Only used when a preprocessed
252    // tree is present.
253    let air_to_preprocessed_trace = if preprocessed.is_some() {
254        trace_order.preprocessed_trace_index_for_air::<F, EF, _>(airs)
255    } else {
256        Vec::new()
257    };
258
259    // Borrow each AIR and trace, then reorder both into ascending-height (proof)
260    // order. AIRs are passed as `&MA::Air` (the existing constraint code expects
261    // a reference); traces likewise via `&RowMajorMatrix<F>`.
262    let air_refs: Vec<&MA::Air> = airs.iter().collect();
263    let trace_refs: Vec<&RowMajorMatrix<F>> = traces.iter().collect();
264    let proof_ordered_airs = trace_order.to_proof_order(&air_refs);
265    let proof_ordered_traces = trace_order.to_proof_order(&trace_refs);
266    let proof_ordered: Vec<_> = proof_ordered_airs
267        .iter()
268        .copied()
269        .zip(proof_ordered_traces.iter().copied())
270        .collect();
271
272    let log_blowup = config.pcs().log_blowup();
273    let log_max_trace_height = trace_order.max_log_height();
274    let max_lde_domain = LiftedDomain::<F>::try_canonical(log_max_trace_height, log_blowup)?;
275    let instance_domains: Vec<_> = trace_order
276        .log_heights_proof()
277        .iter()
278        .map(|&log_h| max_lde_domain.try_sub_domain(log_h))
279        .collect::<Result<_, _>>()?;
280
281    // Observe the preprocessed commitment first (when present); it is part of
282    // the statement and binds Fiat-Shamir before any other instance data.
283    if let Some(preprocessed) = preprocessed {
284        challenger.observe(preprocessed.commitment());
285    }
286
287    // `Statement::observe` absorbs statement-owned inputs. The protocol then
288    // binds the instance count and each log trace height in instance order.
289    statement.observe(&mut challenger, trace_order.log_heights());
290    trace_order.observe_shape::<F, _>(&mut challenger);
291
292    let mut channel = ProverTranscript::new(challenger);
293
294    // Infer per-AIR quotient degrees from symbolic analysis (per-AIR optimization).
295    let log_quotient_degrees: Vec<u8> = proof_ordered
296        .iter()
297        .map(|&(air, _)| log_quotient_degree::<F, EF, _>(air))
298        .collect();
299    let log_quotient_degree = log_quotient_degrees
300        .iter()
301        .copied()
302        .max()
303        .expect("TraceOrder construction rejects empty AIR sets");
304    if log_quotient_degree > log_blowup {
305        return Err(DomainError::ConstraintDegreeTooHigh {
306            log_quotient: log_quotient_degree,
307            log_blowup,
308        }
309        .into());
310    }
311
312    // Pair the max LDE domain with the quotient degree. The `EvaluationDomain`
313    // now flows through the constraint and quotient layers. Per-instance variants
314    // are pre-built so the constraint loop just indexes into a `Vec`.
315    let max_eval_domain = max_lde_domain.evaluation_domain(log_quotient_degree);
316    let instance_eval_domains: Vec<_> = instance_domains
317        .iter()
318        .map(|d| d.evaluation_domain(log_quotient_degree))
319        .collect();
320
321    // Quotient evaluation coset: a sub-coset of the LDE coset where Q is evaluated
322    // before being decomposed into D chunks and committed on the LDE coset itself.
323    let max_quotient_height = max_eval_domain.size();
324
325    // 1. Commit all main traces (trace order — ascending height).
326    //
327    // Clone with blowup × capacity so the DFT resize doesn't reallocate.
328    let blowup = 1 << log_blowup as usize;
329    let main_traces: Vec<_> = proof_ordered
330        .iter()
331        .map(|&(_, trace)| {
332            let src = &trace.values;
333            let mut values = Vec::with_capacity(src.len() * blowup);
334            values.extend_from_slice(src);
335            RowMajorMatrix::new(values, trace.width())
336        })
337        .collect();
338    let main_committed = info_span!("commit to main traces")
339        .in_scope(|| commit_traces(config, &instance_domains, main_traces));
340    channel.send_commitment(main_committed.root());
341
342    // 2. Sample randomness, build aux traces, and commit them
343    let max_num_randomness =
344        proof_ordered.iter().map(|&(air, _)| air.num_randomness()).max().unwrap_or(0);
345
346    let randomness: Vec<EF> = (0..max_num_randomness)
347        .map(|_| channel.sample_algebra_element::<EF>())
348        .collect();
349
350    // Build aux traces in instance order. The output shapes are trusted (see
351    // trust contract above); a malformed output is caught downstream by the
352    // LDE/commit or by verification.
353    let (mut aux_traces_ef, mut all_aux_values) = info_span!("build aux traces").in_scope(|| {
354        let mut aux_traces = Vec::with_capacity(airs.len());
355        let mut aux_values = Vec::with_capacity(airs.len());
356        for (air, main) in airs.iter().zip(traces.iter()) {
357            let num_randomness = air.num_randomness();
358            debug_assert!(
359                randomness.len() >= num_randomness,
360                "AIR requested more aux randomness than the shared challenge pool contains",
361            );
362            let (trace, values) = air.build_aux_trace(
363                main,
364                air_inputs,
365                statement.aux_inputs(),
366                &randomness[..num_randomness],
367            );
368            debug_assert_eq!(trace.height(), main.height(), "aux trace height mismatch");
369            debug_assert_eq!(trace.width(), air.aux_width(), "aux trace width mismatch");
370            debug_assert_eq!(values.len(), air.num_aux_values(), "aux values length mismatch");
371            aux_traces.push(trace);
372            aux_values.push(values);
373        }
374        (aux_traces, aux_values)
375    });
376
377    // Mirror the verifier's external assertion evaluation while aux values are
378    // still in instance order. This is cheap and catches malformed statements
379    // early; it could become a debug assertion if proving needs to skip this
380    // verifier-side sanity check.
381    let aux_views: Vec<&[EF]> = all_aux_values.iter().map(Vec::as_slice).collect();
382    let assertions = statement
383        .eval_external(&randomness, &aux_views, trace_order.log_heights())
384        .map_err(ProverError::Reduction)?;
385    for (k, assertion) in assertions.iter().enumerate() {
386        if *assertion != EF::ZERO {
387            return Err(ProverError::ExternalAssertionFailed { assertion: k });
388        }
389    }
390
391    // External assertions are defined in instance-order terms; now reorder aux
392    // traces and aux values to proof order for commitment and the prover loop.
393    trace_order.reorder_to_proof_in_place(&mut aux_traces_ef);
394    trace_order.reorder_to_proof_in_place(&mut all_aux_values);
395
396    // Flatten EF -> F and commit aux traces
397    let aux_traces: Vec<RowMajorMatrix<F>> = aux_traces_ef
398        .into_iter()
399        .map(|aux| {
400            let base_width = aux.width() * EF::DIMENSION;
401            let base_values = <EF as BasedVectorSpace<F>>::flatten_to_base(aux.values);
402            RowMajorMatrix::new(base_values, base_width)
403        })
404        .collect();
405
406    let aux_committed = info_span!("commit to aux traces")
407        .in_scope(|| commit_traces(config, &instance_domains, aux_traces));
408    channel.send_commitment(aux_committed.root());
409
410    // Observe aux values into the transcript (binds to Fiat-Shamir state).
411    // When no AIR has aux columns, each entry is empty so nothing is sent.
412    for vals in &all_aux_values {
413        for &val in vals {
414            channel.send_algebra_element(val);
415        }
416    }
417
418    // 3. Sample constraint folding alpha and accumulation beta
419    let alpha: EF = channel.sample_algebra_element::<EF>();
420    let beta: EF = channel.sample_algebra_element::<EF>();
421
422    // 4. Evaluate constraints and accumulate quotient evaluations with beta folding.
423    //
424    // Per AIR (ascending height):
425    //   1. Evaluate Q_j = (alpha-folded constraints) / Z_{H_j} on the native quotient domain
426    //      (divide fused into the eval write).
427    //   2. If D_j < D_max, upsample Q_j to the per-trace target domain.
428    //   3. Cyclically extend the accumulator and Horner-fold: acc <- acc * beta + Q_j.
429    //
430    // Pre-allocate with LDE capacity so commit_quotient's resize doesn't reallocate.
431    let mut accumulator: Vec<EF> = Vec::with_capacity(max_quotient_height * blowup);
432
433    // Pre-compute per-AIR constraint layouts.
434    let layouts: Vec<_> = proof_ordered
435        .iter()
436        .map(|&(air, _)| get_constraint_layout::<F, EF, _>(air))
437        .collect();
438
439    info_span!("evaluate constraints").in_scope(|| {
440        for (i, &(air, _)) in proof_ordered.iter().enumerate() {
441            let this_log_quotient_degree = log_quotient_degrees[i];
442            let this_quotient_degree = 1usize << this_log_quotient_degree;
443
444            // Per-AIR native quotient evaluation domain `gJ_j` (size n_j · D_j,
445            // before upsampling to n_j · D_max).
446            let this_quotient_eval_domain =
447                instance_domains[i].evaluation_domain(this_log_quotient_degree);
448            // Target after upsample to D_max (size n_j · D_max).
449            let this_target_quotient_height = instance_eval_domains[i].size();
450
451            // Truncate the committed LDE to the AIR's native quotient evaluation domain gJ_j.
452            // Since B >= D_j, the committed LDE on gK (size N*B) contains gJ_j as a prefix in
453            // bit-reversed storage, so this is a zero-copy view.
454            let main_on_gj = main_committed.evals_on_quotient_domain(i, &this_quotient_eval_domain);
455            let aux_on_gj = aux_committed.evals_on_quotient_domain(i, &this_quotient_eval_domain);
456
457            // Preprocessed view: fetched only when this AIR declares preprocessed
458            // columns. Resolve the committed trace via the inverse mapping
459            // `air_to_preprocessed_trace[instance_idx]`; it shares the max-LDE coset with
460            // the main trace, so `evals_on_quotient_domain` truncates it to this
461            // AIR's quotient domain the same way.
462            let preproc_on_gj = preprocessed.and_then(|p| {
463                let instance_idx = trace_order.instance_indices()[i] as usize;
464                air_to_preprocessed_trace[instance_idx].map(|preprocessed_trace_idx| {
465                    p.committed().evals_on_quotient_domain(
466                        preprocessed_trace_idx,
467                        &this_quotient_eval_domain,
468                    )
469                })
470            });
471
472            let periodic_lde =
473                PeriodicLde::build(&this_quotient_eval_domain, air.periodic_columns_matrix());
474
475            let mut quotient_evals = EF::zero_vec(this_quotient_eval_domain.size());
476            let aux_values_i = &all_aux_values[i];
477            let inv_z_h = this_quotient_eval_domain.inv_vanishing_evals();
478
479            tracing::debug_span!(
480                "eval_instance",
481                instance = i,
482                native_height = this_quotient_eval_domain.size(),
483                target_height = this_target_quotient_height,
484                native_degree = this_quotient_degree,
485                target_degree = 1 << log_quotient_degree as usize,
486            )
487            .in_scope(|| {
488                evaluate_constraints_into::<F, EF, _>(
489                    &mut quotient_evals,
490                    air,
491                    &main_on_gj,
492                    preproc_on_gj.as_ref(),
493                    &aux_on_gj,
494                    &this_quotient_eval_domain,
495                    alpha,
496                    &randomness[..air.num_randomness()],
497                    air_inputs,
498                    &periodic_lde,
499                    &layouts[i],
500                    aux_values_i,
501                    &inv_z_h,
502                );
503            });
504
505            if this_log_quotient_degree < log_quotient_degree {
506                let added_bits = (log_quotient_degree - this_log_quotient_degree) as usize;
507                quotient_evals = tracing::debug_span!(
508                    "upsample_quotient",
509                    instance = i,
510                    from = this_quotient_eval_domain.size(),
511                    to = this_target_quotient_height,
512                )
513                .in_scope(|| {
514                    quotient::upsample_evals::<F, EF, _>(config.dft(), quotient_evals, added_bits)
515                });
516            }
517
518            debug_assert_eq!(quotient_evals.len(), this_target_quotient_height);
519
520            // Cyclically extend the running accumulator to the per-AIR target height and
521            // Horner-fold this AIR's contribution in: acc <- acc * beta + Q_j.
522            tracing::debug_span!(
523                "cyclic_extend_and_accumulate",
524                acc_len = accumulator.len(),
525                target = this_target_quotient_height
526            )
527            .in_scope(|| {
528                quotient::cyclic_extend_and_accumulate(&mut accumulator, quotient_evals, beta);
529            });
530        }
531    });
532
533    debug_assert_eq!(accumulator.len(), max_quotient_height);
534
535    // 5. Commit quotient.
536    let quotient_committed = info_span!("commit to quotient poly chunks")
537        .in_scope(|| quotient::commit_quotient(config, accumulator, &max_eval_domain));
538    channel.send_commitment(quotient_committed.root());
539
540    // 6. Sample OOD point (outside H and gK)
541    let z: EF = max_lde_domain.sample_ood_point(&mut channel);
542    let h = max_lde_domain.trace_subgroup().generator();
543    let z_next = z * h;
544
545    // 7. Open via PCS. The prover and verifier use the same group order:
546    // `[preprocessed?, main, aux, quotient]`.
547    let mut trees = Vec::with_capacity(4);
548    if let Some(p) = preprocessed {
549        trees.push(p.committed().tree());
550    }
551    trees.push(main_committed.tree());
552    trees.push(aux_committed.tree());
553    trees.push(quotient_committed.tree());
554
555    info_span!("open").in_scope(|| {
556        open_with_channel::<F, EF, SC::Lmcs, RowMajorMatrix<F>, _, 2>(
557            config.pcs(),
558            config.lmcs(),
559            &max_lde_domain,
560            [z, z_next],
561            &trees,
562            &mut channel,
563        )
564    });
565
566    let (digest, transcript) = channel.finalize();
567    let proof = StarkProofData {
568        log_trace_heights: trace_order.log_heights().to_vec(),
569        transcript,
570    };
571    Ok(StarkOutput { digest, proof })
572}

574/// Errors from proving — runtime validation failures of caller-supplied data.
575/// The AIR's structural contract is trusted (see the crate-level trust model).

580    #[error(transparent)]

586        /// Index into the assertions vector returned by
587        /// [`Statement::eval_external`](miden_lifted_air::Statement::eval_external).
588        assertion: usize,
589    },
