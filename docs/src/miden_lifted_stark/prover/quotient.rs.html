1//! Quotient polynomial helpers: accumulation, vanishing division, decomposition.
2//!
3//! The prover orchestrates the quotient pipeline (loop over instances, accumulate,
4//! divide, commit). This module provides the building blocks:
5//!
6//! - [`cyclic_extend_and_scale`]: Horner-style beta scaling + cyclic extension
7//! - [`divide_by_vanishing_in_place`]: Divide by Z_H on the quotient evaluation domain
8//! - [`commit_quotient`]: Decompose Q(gJ) into chunks and commit on gK
9

28// ============================================================================
29// Accumulation
30// ============================================================================
31
32/// Cyclically extend the accumulator to `target_len` and scale every element by `β`.
33///
34/// On the first call (empty accumulator) this simply zero-fills to `target_len`.
35/// On subsequent calls it scales the existing buffer by `β` (Horner folding)
36/// then doubles via `extend_from_within` until it reaches `target_len`.
37///
38/// Both `accumulator.len()` and `target_len` must be powers of two, and
39/// `target_len ≥ accumulator.len()`.
40///
41/// Cyclic extension is valid because H_small is a subgroup of H_big, so
42/// evaluations repeat cyclically. The β scaling implements Horner folding for
43/// multi-trace accumulation: `acc = acc·β + Nⱼ`.
44pub fn cyclic_extend_and_scale<EF: Field>(accumulator: &mut Vec<EF>, target_len: usize, beta: EF) {
45    if accumulator.is_empty() {
46        accumulator.resize(target_len, EF::ZERO);
47    } else {
48        // Horner: scale the smaller buffer by beta before upsampling
49        accumulator.par_iter_mut().for_each(|v| *v *= beta);
50        // Cyclic extension by repeated doubling (all sizes are powers of 2)
51        while accumulator.len() < target_len {
52            accumulator.extend_from_within(..);
53        }
54    }
55}

57// ============================================================================
58// Vanishing division
59// ============================================================================
60
61/// Divide quotient numerator by vanishing polynomial in-place (natural order).
62///
63/// Replaces each `numerator[i]` with `numerator[i] / Z_H(xᵢ)` where
64/// `Z_H(X) = Xᴺ − 1` and `N` is the trace height.
65///
66/// This uses a periodicity trick: on the quotient evaluation coset `gJ` of size `N·D`,
67/// the values `Z_H(x)` take only `D` distinct values, so we can batch-invert those `D`
68/// values once and reuse them by modular indexing.
69///
70/// Note that here `coset.log_blowup()` is `log2(D)` because `coset` is the *quotient*
71/// domain (blowup = constraint degree), not the PCS/FRI blowup `B`.
72pub fn divide_by_vanishing_in_place<F, EF>(numerator: &mut [EF], coset: &LiftedCoset)
73where
74    F: TwoAdicField,
75    EF: ExtensionField<F>,
76{
77    // D = constraint degree. On the quotient coset, log_blowup() = log₂(D).
78    let log_blowup = coset.log_blowup();
79    let num_distinct = 1 << log_blowup;
80
81    // The D distinct values of Z_H on gJ:
82    // Z_H(g·ω_Jⁱ) = sᴺ·ω_Dⁱ − 1 where
83    // - s is the coset shift
84    // - ω_D is a D-th root of unity.
85    let shift: F = coset.lde_shift();
86    let s_pow_n = shift.exp_power_of_2(coset.log_trace_height as usize);
87    let z_h_evals: Vec<F> = F::two_adic_generator(log_blowup)
88        .powers()
89        .take(num_distinct)
90        .map(|x| s_pow_n * x - F::ONE)
91        .collect();
92
93    let inv_van = batch_multiplicative_inverse(&z_h_evals);
94
95    // Parallel division using modular indexing for periodicity.
96    // Z_H has only num_distinct unique values on gJ; power-of-2 size
97    // lets us use bitmask: i & (num_distinct - 1) == i % num_distinct.
98    numerator.par_iter_mut().enumerate().for_each(|(i, n)| {
99        *n *= inv_van[i & (num_distinct - 1)];
100    });
101}

103// ============================================================================
104// Quotient decomposition + commitment
105// ============================================================================
106
107/// Commit the quotient polynomial by splitting across the `D` quotient cosets.
108///
109/// The quotient is naturally evaluated on the quotient evaluation coset `gJ` of size
110/// `N·D` (N = trace height, D = constraint degree). We view `J` as `D` disjoint
111/// `H`-cosets: `J = ⋃_{t=0..D−1} ω_Jᵗ·H`. Reshaping `Q(gJ)` into an `N×D`
112/// matrix makes column `t` the evaluations of a degree-`< N` polynomial qₜ on the
113/// coset `g·ω_Jᵗ·H`.
114///
115/// We commit to all qₜ by LDE-extending them to the PCS domain `gK` (size `N·B`) and
116/// hashing the resulting matrix. Naïvely this would require `D` separate coset-iDFT /
117/// coset-DFT pairs (one per chunk). The "fused scaling" trick below collapses all of
118/// them into a single plain iDFT, a diagonal scaling pass, and one plain DFT:
119///
120/// - a plain iDFT on each column yields coefficients multiplied by `(g·ω_Jᵗ)ᵏ` (the inverse coset
121///   shift is absorbed into the coefficients),
122/// - multiplying by `(ω_J⁻ᵏ)ᵗ` removes the per-chunk shift ω_Jᵗ while keeping the common factor gᵏ
123///   baked in,
124/// - a plain (unshifted) forward DFT then evaluates directly on the shifted coset `gK`, because gᵏ
125///   already accounts for the coset offset.
126///
127/// `q_evals` is consumed and flattened to the base field for commitment.
128///
129/// # Panics
130///
131/// - If `q_evals.len()` is not divisible by N
132/// - If blowup B < constraint degree D
133pub fn commit_quotient<F, EF, SC>(
134    config: &SC,
135    q_evals: Vec<EF>,
136    coset: &LiftedCoset,
137) -> Committed<F, RowMajorMatrix<F>, SC::Lmcs>
138where
139    F: TwoAdicField,
140    EF: ExtensionField<F>,
141    SC: StarkConfig<F, EF>,
142{
143    let n = coset.trace_height();
144    let d = q_evals.len() / n;
145    let log_d = log2_strict_usize(d);
146    let log_blowup = config.pcs().log_blowup();
147    let b = 1usize << log_blowup;
148
149    debug_assert_eq!(q_evals.len() % n, 0, "q_evals length must be divisible by N");
150    debug_assert!(b >= d, "blowup B must be >= constraint degree D");
151
152    // ═══════════════════════════════════════════════════════════════════════
153    // Step 0: Reshape to N × D matrix
154    // ═══════════════════════════════════════════════════════════════════════
155    // q_evals[r·D + t] = Q(g·ω_Jᵗ·ω_Hʳ), so column t gives
156    // qₜ evaluated on the coset g·ω_Jᵗ·H.
157    let m = RowMajorMatrix::new(q_evals, d);
158
159    // ═══════════════════════════════════════════════════════════════════════
160    // Step 1: Batched iDFT over H
161    // ═══════════════════════════════════════════════════════════════════════
162    // iDFT treats each column as evaluations on H (not the actual coset
163    // g·ω_Jᵗ·H), producing shifted coefficients:
164    //   c_hat[t, k] = a[t, k]·(g·ω_Jᵗ)ᵏ
165    // where a[t, k] are the true coefficients of qₜ.
166    let mut coeffs = info_span!("quotient iDFT", dims = %format!("{n}x{d}"))
167        .in_scope(|| config.dft().idft_algebra_batch(m));
168
169    // ═══════════════════════════════════════════════════════════════════════
170    // Step 2: Fused coefficient scaling
171    // ═══════════════════════════════════════════════════════════════════════
172    // Multiply c_hat[t, k] by (ω_Jᵗ)⁻ᵏ → a[t, k]·gᵏ.
173    // This removes the per-coset shift ω_Jᵗ while keeping gᵏ baked in.
174    info_span!("quotient scaling", n).in_scope(|| {
175        let omega_j_inv = F::two_adic_generator(coset.log_trace_height as usize + log_d).inverse();
176
177        // Precompute ω_J⁻ᵏ for k = 0..N with sequential multiplications
178        let row_bases: Vec<F> = omega_j_inv.powers().take(n).collect();
179
180        // Row k, column t: multiply by (ω_J⁻ᵏ)ᵗ
181        coeffs.par_rows_mut().zip(row_bases.par_iter()).for_each(|(row, &row_base)| {
182            for (val, scale) in row.iter_mut().zip(row_base.powers()) {
183                *val *= scale;
184            }
185        });
186    });
187
188    // ═══════════════════════════════════════════════════════════════════════
189    // Step 3: Flatten EF → F, zero-pad to N·B rows
190    // ═══════════════════════════════════════════════════════════════════════
191    // We flatten before the DFT (rather than using dft_algebra_batch) because
192    // we need base field for commitment anyway — this skips the reconstitute.
193    //
194    // Zero-padding from N to N·B rows is needed because `dft_batch` expects
195    // the full target-size buffer. The extra rows are zero because each qₜ has
196    // degree < N. We pad here (after iDFT + scaling) so those two steps work
197    // on the smaller N-row buffer.
198    //
199    // PERF: the full N·B-size DFT processes N·(B−1) zero rows through every
200    // butterfly stage, costing O(N·B·log(N·B)) instead of O(N·B·log N). For
201    // B = 4, N = 2^20 that is ≈ 9% overhead on this step (small relative to
202    // total proving time since the quotient matrix has only D·DIM columns).
203    //
204    // The existing `lde_batch`/`coset_lde_batch` APIs cannot help: they take
205    // *evaluations*, not coefficients. Using them would add a redundant DFT(N)
206    // → iDFT(N) round-trip.
207    //
208    // What is conceptually missing from `TwoAdicSubgroupDft` is an
209    // `added_bits` parameter on `dft_batch` / `coset_dft_batch` that evaluates
210    // degree-< N coefficients on a larger domain of size N·2^added_bits. The
211    // default would be zero-pad + the existing same-size DFT, but an optimized
212    // implementation (like `Radix2DftParallel`) could run B separate N-size
213    // DFTs — one per coset of H inside K — matching what its `coset_lde_batch`
214    // already does internally after the iDFT phase.
215    let base_width = d * EF::DIMENSION;
216    let mut base_coeffs = <EF as BasedVectorSpace<F>>::flatten_to_base(coeffs.values);
217    base_coeffs.resize(n * b * base_width, F::ZERO);
218    let coeffs_padded = RowMajorMatrix::new(base_coeffs, base_width);
219
220    // ═══════════════════════════════════════════════════════════════════════
221    // Step 4: Plain DFT (not coset DFT) on base field
222    // ═══════════════════════════════════════════════════════════════════════
223    // Because gᵏ is baked into the coefficients, the plain DFT evaluates
224    // on gK directly: entry (i, t) gives qₜ(g·ω_Kⁱ).
225    let quotient_matrix = info_span!("quotient DFT", dims = %format!("{}x{base_width}", n * b))
226        .in_scope(|| {
227            let lde = config.dft().dft_batch(coeffs_padded);
228
229            // ═══════════════════════════════════════════════════════════════
230            // Step 5: Wrap for commitment
231            // ═══════════════════════════════════════════════════════════════
232            materialize_bitrev(lde)
233        });
234
235    let tree = config.lmcs().build_aligned_tree(vec![quotient_matrix]);
236
237    Committed::new(tree, log_blowup)
238}
