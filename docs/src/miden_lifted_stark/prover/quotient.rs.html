1//! Quotient polynomial helpers used by the prover's per-AIR pipeline.
2//!
3//! - [`upsample_evals`]: Low-degree extend coset evaluations onto a larger two-adic coset (the
4//!   constraint-degree axis: `D_j -> D_max`, same polynomial, denser evaluations).
5//! - [`cyclic_extend_and_accumulate`]: Lift the running accumulator along the trace-height axis
6//!   (`n_j -> N`) by cyclic repetition, and Horner-fold a new AIR's contribution in via beta.
7//! - [`commit_quotient`]: Decompose Q(gJ) into chunks and commit on gK.
8

32/// Low-degree extend coset evaluations onto a larger two-adic coset.
33///
34/// Treats `evals` as evaluations of a polynomial `p` on a coset `g*H` of size
35/// `evals.len()`, and returns evaluations of the same `p` on the coset `g*K`
36/// of size `evals.len() << added_bits` (same shift `g`, larger two-adic
37/// subgroup).
38///
39/// # Precondition
40///
41/// `deg(p) < evals.len()`. If the input evaluations are of a polynomial whose
42/// actual degree is `>= evals.len()`, this function silently returns evaluations
43/// of a different polynomial (the unique degree-`< evals.len()` interpolant of
44/// the input). The caller is responsible for ensuring the degree bound.
45pub(crate) fn upsample_evals<F, EF, DFT>(dft: &DFT, evals: Vec<EF>, added_bits: usize) -> Vec<EF>
46where
47    F: TwoAdicField,
48    EF: ExtensionField<F>,
49    DFT: TwoAdicSubgroupDft<F>,
50{
51    if added_bits == 0 {
52        return evals;
53    }
54
55    dft.lde_algebra_batch(RowMajorMatrix::new_col(evals), added_bits).values
56}

58/// Fold a new AIR's quotient contribution into the cross-AIR accumulator via one
59/// Horner step, after lifting the prior accumulator onto the larger coset:
60///
61/// ```text
62/// acc <- lift_r(acc) * beta + contribution,    r = contribution.len() / acc.len()
63/// ```
64///
65/// On return, `accumulator.len() == contribution.len()`. `contribution.len()` and
66/// (when non-empty) `accumulator.len()` must be powers of two, with
67/// `contribution.len() >= accumulator.len()`.
68///
69/// # Why
70///
71/// - **Scale before extend.** Horner-multiplying the smaller buffer is strictly less work than the
72///   lifted one, and the lifted values are determined entirely by the smaller buffer via cyclic
73///   repetition.
74/// - **Cyclic repetition = polynomial lift on two-adic cosets.** In natural-order coset
75///   evaluations, `extended[i] = original[i mod n_old]` realises the composition `P(X) -> P(X^r)`:
76///   iterating `gJ` in natural order and raising to the `r`-th power cycles through `(gJ)^r` with
77///   period `|(gJ)^r|`.
78pub(super) fn cyclic_extend_and_accumulate<EF: Field>(
79    accumulator: &mut Vec<EF>,
80    contribution: Vec<EF>,
81    beta: EF,
82) {
83    debug_assert!(contribution.len().is_power_of_two());
84    debug_assert!(accumulator.is_empty() || accumulator.len().is_power_of_two());
85    debug_assert!(contribution.len() >= accumulator.len());
86
87    if accumulator.is_empty() {
88        accumulator.extend(contribution);
89        return;
90    }
91
92    if accumulator.len() == contribution.len() {
93        // No lift needed; fuse the Horner mul and add into a single packed pass by
94        // computing `contribution + beta * accumulator` in place on the (owned)
95        // contribution buffer and swapping it in.
96        let mut contribution = contribution;
97        par_add_scaled_slice_in_place(&mut contribution, accumulator, beta);
98        *accumulator = contribution;
99        return;
100    }
101
102    par_scale_slice_in_place(accumulator, beta);
103    while accumulator.len() < contribution.len() {
104        accumulator.extend_from_within(..);
105    }
106    // TODO: use parallel packed addition
107    accumulator
108        .par_iter_mut()
109        .zip(contribution.into_par_iter())
110        .for_each(|(a, c)| *a += c);
111}

113// ============================================================================
114// Quotient decomposition + commitment
115// ============================================================================
116
117/// Commit the quotient polynomial by splitting across the `D` quotient cosets.
118///
119/// The quotient is naturally evaluated on the quotient evaluation coset `gJ` of size
120/// `N·D` (N = trace height, D = constraint degree). We view `J` as `D` disjoint
121/// `H`-cosets: `J = ⋃_{t=0..D−1} ω_Jᵗ·H`. Reshaping `Q(gJ)` into an `N×D`
122/// matrix makes column `t` the evaluations of a degree-`< N` polynomial qₜ on the
123/// coset `g·ω_Jᵗ·H`.
124///
125/// We commit to all qₜ by LDE-extending them to the PCS domain `gK` (size `N·B`) and
126/// hashing the resulting matrix. Naïvely this would require `D` separate coset-iDFT /
127/// coset-DFT pairs (one per chunk). The "fused scaling" trick below collapses all of
128/// them into a single plain iDFT, a diagonal scaling pass, and one plain DFT:
129///
130/// - a plain iDFT on each column yields coefficients multiplied by `(g·ω_Jᵗ)ᵏ` (the inverse coset
131///   shift is absorbed into the coefficients),
132/// - multiplying by `(ω_J⁻ᵏ)ᵗ` removes the per-chunk shift ω_Jᵗ while keeping the common factor gᵏ
133///   baked in,
134/// - a plain (unshifted) forward DFT then evaluates directly on the shifted coset `gK`, because gᵏ
135///   already accounts for the coset offset.
136///
137/// `q_evals` is consumed and flattened to the base field for commitment.
138///
139/// # Panics
140///
141/// - If `q_evals.len()` is not divisible by N
142/// - If blowup B < constraint degree D
143pub fn commit_quotient<F, EF, SC>(
144    config: &SC,
145    q_evals: Vec<EF>,
146    domain: &EvaluationDomain<F>,
147) -> Committed<F, RowMajorMatrix<F>, SC::Lmcs>
148where
149    F: TwoAdicField,
150    EF: ExtensionField<F>,
151    SC: StarkConfig<F, EF>,
152{
153    let n = domain.trace_height();
154    let d = domain.quotient_degree();
155    let lde_height = domain.lifted().lde_height();
156
157    debug_assert_eq!(q_evals.len(), n * d, "q_evals length must equal N · D");
158    // D ≤ B (i.e. lde_height ≥ N · D) is enforced by `EvaluationDomain::new`.
159
160    // ═══════════════════════════════════════════════════════════════════════
161    // Step 0: Reshape to N × D and flatten EF → F
162    // ═══════════════════════════════════════════════════════════════════════
163    // q_evals[r·D + t] = Q(g·ω_Jᵗ·ω_Hʳ), so column t gives qₜ evaluated on the
164    // coset g·ω_Jᵗ·H. The commitment is base-valued and the iDFT/DFT are
165    // F-linear, so flattening to the base field commutes with both.
166    let base_width = d * EF::DIMENSION;
167    let coeffs =
168        RowMajorMatrix::new(<EF as BasedVectorSpace<F>>::flatten_to_base(q_evals), base_width);
169
170    // ═══════════════════════════════════════════════════════════════════════
171    // Steps 1–4: fused iDFT over H → coefficient scaling → coset LDE onto gK
172    // ═══════════════════════════════════════════════════════════════════════
173    // `coset_lde_batch_with_transform` runs the iDFT, applies `transform` to the
174    // coefficient buffer, zero-pads by `added_bits`, then evaluates on the
175    // coset. The iDFT treats each column as evaluations on H, producing shifted
176    // coefficients c_hat[k, t] = a[k, t]·(g·ω_Jᵗ)ᵏ. `transform` multiplies by
177    // (ω_Jᵗ)⁻ᵏ, removing the per-coset shift ω_Jᵗ while keeping gᵏ baked in, so a
178    // shift-1 LDE then evaluates directly on the shifted coset gK.
179    let added_bits = log2_strict_usize(lde_height) - log2_strict_usize(n);
180    let omega_j_inv = domain.subgroup().generator_inverse();
181    let row_bases: Vec<F> = omega_j_inv.powers().take(n).collect();
182    let log_n = log2_strict_usize(n);
183
184    let lde =
185        info_span!("quotient LDE", dims = %format!("{lde_height}x{base_width}")).in_scope(|| {
186            config.dft().coset_lde_batch_with_transform(
187                coeffs,
188                added_bits,
189                F::ONE,
190                |buf, layout| {
191                    // Row r holds coefficient index k — bit-reversed when the buffer is in
192                    // bit-reversed layout. Multiply column t (each spanning `EF::DIMENSION`
193                    // flattened base columns) by (ω_J⁻ᵏ)ᵗ.
194                    let width = buf.width;
195                    buf.values.par_chunks_mut(width).enumerate().for_each(|(r, row)| {
196                        let k = match layout {
197                            Layout::Natural => r,
198                            Layout::BitReversed => reverse_bits_len(r, log_n),
199                        };
200                        let row_base = row_bases[k];
201                        let mut scale = F::ONE;
202                        for chunk in row.chunks_exact_mut(EF::DIMENSION) {
203                            for val in chunk {
204                                *val *= scale;
205                            }
206                            scale *= row_base;
207                        }
208                    });
209                },
210            )
211        });
212
213    let tree = config.lmcs().build_aligned_tree(vec![lde]);
214
215    // The quotient is committed on the same LDE coset as the trace commits.
216    Committed::new(tree)
217}

219#[cfg(test)]
220mod tests {
221    use alloc::vec::Vec;
222
223    use p3_dft::{NaiveDft, TwoAdicSubgroupDft};
224    use p3_field::{Field, PrimeCharacteristicRing};
225    use p3_matrix::dense::RowMajorMatrix;
226
227    use super::upsample_evals;
228    use crate::testing::configs::goldilocks_poseidon2::{Felt, QuadFelt};
229
230    fn coeffs(height: usize) -> Vec<QuadFelt> {
231        (0..height).map(|i| QuadFelt::from_u64((i as u64) + 1)).collect()
232    }
233
234    /// Checks that `upsample_evals` on `dft` produces the same result as a direct
235    /// coset DFT of zero-padded coefficients.
236    fn assert_upsample_matches_direct<D: TwoAdicSubgroupDft<Felt>>(dft: &D, shift: Felt) {
237        let small_height = 8;
238        let added_bits = 2;
239        let large_height = small_height << added_bits;
240
241        let small_coeffs = RowMajorMatrix::new(coeffs(small_height), 1);
242        let small_evals = NaiveDft.coset_dft_algebra_batch(small_coeffs, shift).values;
243
244        let mut large_coeffs = coeffs(small_height);
245        large_coeffs.resize(large_height, QuadFelt::ZERO);
246        let direct_large = NaiveDft
247            .coset_dft_algebra_batch(RowMajorMatrix::new(large_coeffs, 1), shift)
248            .values;
249
250        let upsampled = upsample_evals::<Felt, QuadFelt, _>(dft, small_evals, added_bits);
251        assert_eq!(upsampled, direct_large);
252    }
253
254    #[test]
255    fn upsample_evals_matches_direct_coset_dft() {
256        assert_upsample_matches_direct(&NaiveDft, Felt::GENERATOR.exp_power_of_2(2));
257    }
258
259    /// Same check with the production DFT backend.
260    #[test]
261    fn upsample_evals_with_radix2_dit_parallel_matches_naive() {
262        use p3_dft::Radix2DitParallel;
263        let dft = Radix2DitParallel::<Felt>::default();
264        assert_upsample_matches_direct(&dft, Felt::GENERATOR.exp_power_of_2(2));
265    }
266
267    #[test]
268    fn upsample_evals_with_zero_added_bits_returns_input_unchanged() {
269        let evals = coeffs(8);
270        let out = upsample_evals::<Felt, QuadFelt, _>(&NaiveDft, evals.clone(), 0);
271        assert_eq!(out, evals);
272    }
