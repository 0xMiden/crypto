26// Committed
27// ============================================================================
28
29/// Committed polynomial evaluations.
30///
31/// Thin wrapper around an LMCS tree. Per-matrix sizing for quotient evaluation
32/// flows through the [`EvaluationDomain`] passed to
33/// [`evals_on_quotient_domain`](Self::evals_on_quotient_domain) — the wrapper
34/// itself doesn't carry domain metadata, so accessing matrix `m`'s evaluation
35/// view doesn't trigger a per-call sub-domain construction (which would compute
36/// a multiplicative inverse).
37///
38/// # Type Parameters
39///
40/// - `F`: Scalar field element type
41/// - `M`: Matrix type (e.g., `RowMajorMatrix<F>`)
42/// - `L`: LMCS configuration type
43pub struct Committed<F, M, L>
44where
45    F: TwoAdicField,
46    L: Lmcs<F = F>,
47    M: Matrix<F>,
48{
49    /// The underlying LMCS tree.
50    tree: L::Tree<M>,
51}
52
53impl<F, M, L> Committed<F, M, L>
54where
55    F: TwoAdicField,
56    L: Lmcs<F = F>,
57    M: Matrix<F>,
58{
59    /// Create a new `Committed` wrapper around an LMCS tree.
60    #[inline]
61    pub fn new(tree: L::Tree<M>) -> Self {
62        Self { tree }
63    }
64
65    /// Get the commitment root.
66    #[inline]
67    pub fn root(&self) -> L::Commitment {
68        self.tree.root()
69    }
70
71    /// Get a reference to the underlying tree.
72    #[inline]
73    pub fn tree(&self) -> &L::Tree<M> {
74        &self.tree
75    }
76}
77
78impl<F, L> Committed<F, RowMajorMatrix<F>, L>
79where
80    F: TwoAdicField,
81    L: Lmcs<F = F>,
82{
83    /// Return a zero-copy view of matrix `m` on the quotient evaluation domain.
84    ///
85    /// This returns evaluations over the quotient coset `gJ ⊆ gK` for matrix `m`,
86    /// sized to the per-matrix trace height times `eval_domain.quotient_degree()`.
87    ///
88    /// The tree commits to LDE evaluations on `gK` (size `N·B`). The `RowMajorMatrix`
89    /// stores bit-reversed evaluations; `gJ` appears as the first `N·D` rows, so this is
90    /// a zero-copy prefix view followed by `bit_reverse_rows()` to expose natural order.
91    ///
92    /// # Panics
93    ///
94    /// Panics if `m >= num_matrices()`.
95    pub fn evals_on_quotient_domain(
96        &self,
97        m: usize,
98        eval_domain: &EvaluationDomain<F>,
99    ) -> BitReversedMatrixView<RowMajorMatrixView<'_, F>> {
100        debug_assert_eq!(
101            eval_domain.lifted().lde_height(),
102            self.tree.leaves()[m].height(),
103            "eval_domain LDE height must match matrix m's tree height",
104        );
105        self.tree.leaves()[m].split_rows(eval_domain.size()).0.bit_reverse_rows()
106    }
107}

109// ============================================================================
110// commit_traces
111// ============================================================================

113/// Commit multiple trace matrices with lifting: LDE → LMCS tree.
114///
115/// Traces must be sorted by height in ascending order. Each trace is lifted to
116/// the max LDE domain using the appropriate nested coset shift.
117///
118/// The DFT output is wrapped in `BitReversedMatrixView` (zero-cost view) and
119/// passed directly to the LMCS — no materialization needed.
120///
121/// Returns a [`Committed`] wrapper providing:
122/// - Commitment root via [`Committed::root()`]
123/// - Underlying LMCS tree via [`Committed::tree()`]
124/// - Quotient domain views via [`Committed::evals_on_quotient_domain()`]
125///
126/// # Arguments
127/// - `config`: STARK configuration containing PCS params, LMCS, and DFT
128/// - `domains`: One pre-validated [`LiftedDomain`] per trace, in the same order as `traces`. The
129///   last entry is the batch's max-trace domain (heights are sorted ascending).
130/// - `traces`: Trace matrices, in the same order as `domains`. Each must have height matching its
131///   paired `domains[i].trace_height()`.
132///
133/// # Panics
134/// - If `domains` and `traces` have different lengths
135/// - If any trace's height doesn't match its paired domain's `trace_height()`
136///
137/// Lifting note: for a trace of height `n` embedded into a max height `n_max`, let
138/// `r = n_max / n`. The commitment should behave as if it contains evaluations of the
139/// lifted polynomial `f_lift(X) = f(Xʳ)` on the max LDE coset. This is achieved by
140/// evaluating the original trace on a *nested* coset with shift gʳ: the map
141/// `(g·ω)ʳ = gʳ·ωʳ` sends the max domain down to the smaller one.
142pub(super) fn commit_traces<F, EF, SC>(
143    config: &SC,
144    domains: &[LiftedDomain<F>],
145    traces: Vec<RowMajorMatrix<F>>,
146) -> Committed<F, RowMajorMatrix<F>, SC::Lmcs>
147where
148    F: TwoAdicField,
149    EF: ExtensionField<F>,
150    SC: StarkConfig<F, EF>,
151{
152    assert_eq!(domains.len(), traces.len(), "domains and traces must have matching lengths");
153    assert!(!traces.is_empty(), "at least one trace required");
154
155    let log_blowup = config.pcs().log_blowup();
156
157    let ldes: Vec<_> = traces
158        .into_iter()
159        .zip(domains)
160        .enumerate()
161        .map(|(idx, (trace, domain))| {
162            let width = trace.width();
163            assert_eq!(
164                trace.height(),
165                domain.trace_height(),
166                "trace {idx} height does not match its domain",
167            );
168
169            let log_trace_height = domain.log_trace_height();
170            let coset_shift = domain.lde_shift();
171
172            info_span!("LDE", trace = idx, log_height = log_trace_height, width)
173                .in_scope(|| config.dft().coset_lde_batch(trace, log_blowup.into(), coset_shift))
174        })
175        .collect();
176
177    // Build aligned LMCS tree and wrap in Committed
178    let tree = config.lmcs().build_aligned_tree(ldes);
179    Committed::new(tree)
180}

186#[cfg(test)]
187mod tests {
188    use alloc::vec;
189
190    use p3_field::PrimeCharacteristicRing;
191    use p3_util::reverse_bits_len;
192
193    use super::*;
194    use crate::testing::configs::goldilocks_poseidon2::Felt;
195
196    #[test]
197    fn split_rows_truncates_correctly() {
198        // Create a 16x4 matrix (LDE height = 16, width = 4)
199        let data: Vec<Felt> = (0u64..64).map(Felt::from_u64).collect();
200        let matrix = RowMajorMatrix::new(data, 4);
201
202        // Truncate to 8 rows via split_rows
203        let truncated = matrix.split_rows(8).0;
204        assert_eq!(truncated.height(), 8);
205        assert_eq!(truncated.width(), 4);
206
207        // Verify first row is unchanged
208        let row: Vec<Felt> = truncated.row(0).unwrap().into_iter().collect();
209        assert_eq!(
210            row,
211            vec![Felt::from_u64(0), Felt::from_u64(1), Felt::from_u64(2), Felt::from_u64(3)]
212        );
213    }
214
215    #[test]
216    fn bit_reverse_rows_gives_natural_order() {
217        // Create an 8x2 matrix with values that let us verify bit-reversal
218        // Row i (bit-reversed) contains [2*i, 2*i+1]
219        let data: Vec<Felt> = (0u64..16).map(Felt::from_u64).collect();
220        let matrix = RowMajorMatrix::new(data, 2);
221
222        let natural = matrix.as_view().bit_reverse_rows();
223        assert_eq!(natural.height(), 8);
224        assert_eq!(natural.width(), 2);
225
226        // General verification: natural row i should have values from bit-reversed row bitrev(i)
227        for i in 0..8 {
228            let br_i = reverse_bits_len(i, 3);
229            let natural_row: Vec<Felt> = natural.row(i).unwrap().into_iter().collect();
230            let expected: Vec<Felt> =
231                vec![Felt::from_u64((br_i * 2) as u64), Felt::from_u64((br_i * 2 + 1) as u64)];
232            assert_eq!(natural_row, expected, "mismatch at natural row {i}");
233        }
234    }
235
236    #[test]
237    fn truncate_then_bit_reverse() {
238        // Create a 16x2 matrix
239        let data: Vec<Felt> = (0u64..32).map(Felt::from_u64).collect();
240        let matrix = RowMajorMatrix::new(data, 2);
241
242        // Truncate to 8 rows and convert to natural order
243        let truncated_natural = matrix.split_rows(8).0.bit_reverse_rows();
244        assert_eq!(truncated_natural.height(), 8);
245        assert_eq!(truncated_natural.width(), 2);
246
247        for i in 0..8 {
248            assert_eq!(
249                truncated_natural.row(i).unwrap().into_iter().count(),
250                2,
251                "row {i} should have 2 elements"
252            );
253        }
254    }
