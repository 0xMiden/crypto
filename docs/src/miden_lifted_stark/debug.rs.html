1//! Debug helpers for lifted AIRs.
2//!
3//! Two flavours of helpers live here:
4//!
5//! - **Structural assertion** ([`assert_prover_setup`]) — a panic-based check over
6//!   [`miden_lifted_air::debug::assert_multi_air_valid`]. Call it from tests / setup; the prover
7//!   and verifier hot paths trust the AIR's structural contract.
8//! - **Constraint checker** ([`check_constraints`]) — evaluates AIR constraints row-by-row on
9//!   concrete trace values and panics on the first nonzero constraint. It derives deterministic
10//!   debug challenges; it does not replay the prover transcript.

30/// Assert the AIR's structural contract via [`assert_multi_air_valid`].
31///
32/// Only the *trusted* structural contract is asserted here. The AIR ↔ PCS
33/// compatibility bound (`log_quotient_degree <= log_blowup`) is a validated
34/// runtime input — prover and verifier surface it as
35/// [`DomainError::ConstraintDegreeTooHigh`](crate::DomainError) — so it is not
36/// re-checked here.
37///
38/// The preprocessed bundle's shape (tree presence, per-trace width, per-AIR
39/// height) is validated at [`ProverInstance::new`](crate::ProverInstance::new)
40/// construction time, so it is not re-checked here.
41pub fn assert_prover_setup<F, EF, MA>(prover_statement: &ProverStatement<F, EF, MA>)
42where
43    F: Field,
44    EF: ExtensionField<F>,
45    MA: MultiAir<F, EF>,
46{
47    assert_multi_air_valid::<F, EF, MA>(prover_statement.statement().multi_air());
48}

54/// Evaluate AIR constraints against concrete trace values and panic on failure.
55///
56/// Constraints are checked row-by-row using the trace + aux trace built by
57/// [`ProverStatement`]. All AIRs see the same `air_inputs` from `statement`.
58///
59/// Derives auxiliary-trace challenges from the supplied challenger using only
60/// statement-owned data plus the instance count and log trace heights. This is
61/// a local constraint debugger: it intentionally skips protocol commitments
62/// (including any preprocessed setup commitment), so sampled challenges need
63/// not match a full proof transcript produced by
64/// [`ProverInstance::prove`](crate::ProverInstance::prove).
65///
66/// # Panics
67///
68/// - If trace dimensions don't match their AIR
69/// - If any constraint evaluates to nonzero on any row
70pub fn check_constraints<F, EF, MA, Ch>(
71    prover_statement: &ProverStatement<F, EF, MA>,
72    mut challenger: Ch,
73) where
74    F: Field,
75    EF: ExtensionField<F>,
76    MA: MultiAir<F, EF>,
77    Ch: CanObserve<F> + CanSample<F>,
78{
79    let statement = prover_statement.statement();
80    let airs = statement.airs();
81    let traces = prover_statement.traces();
82    let air_inputs = statement.air_inputs();
83    let aux_inputs = statement.aux_inputs();
84    assert!(!airs.is_empty(), "no instances provided");
85    assert_eq!(airs.len(), traces.len(), "airs and traces counts must match");
86
87    // Seed deterministic debug challenges from statement/height observations only.
88    // Do not observe setup/trace commitments or replay the prover transcript here.
89    let trace_heights: Vec<usize> = traces.iter().map(Matrix::height).collect();
90    let trace_order = TraceOrder::from_trace_heights::<F, EF, _>(airs, &trace_heights)
91        .expect("ProverStatement::new should reject malformed heights");
92    statement.observe(&mut challenger, trace_order.log_heights());
93    trace_order.observe_shape::<F, _>(&mut challenger);
94    let max_num_randomness = airs.iter().map(LiftedAir::num_randomness).max().unwrap_or(0);
95    let challenges: Vec<EF> = (0..max_num_randomness)
96        .map(|_| EF::from_basis_coefficients_fn(|_| challenger.sample()))
97        .collect();
98
99    let mut aux_traces = Vec::with_capacity(airs.len());
100    let mut aux_values_per_air = Vec::with_capacity(airs.len());
101    for (air, main) in airs.iter().zip(traces.iter()) {
102        let num_randomness = air.num_randomness();
103        let (aux_trace, aux_values) =
104            air.build_aux_trace(main, air_inputs, aux_inputs, &challenges[..num_randomness]);
105        aux_traces.push(aux_trace);
106        aux_values_per_air.push(aux_values);
107    }
108
109    // Mirror the verifier's external-assertion check: the cross-AIR
110    // interactions must hold for these aux values and public inputs. Each
111    // assertion is a concrete value, so a zero-check is exact.
112    let aux_views: Vec<&[EF]> = aux_values_per_air.iter().map(Vec::as_slice).collect();
113    let assertions = statement
114        .eval_external(&challenges, &aux_views, trace_order.log_heights())
115        .expect("eval_external failed during check_constraints");
116    for (k, assertion) in assertions.iter().enumerate() {
117        assert_eq!(*assertion, EF::ZERO, "external assertion {k} is non-zero");
118    }
119
120    for (i, ((air, main), (aux_trace, aux_values))) in airs
121        .iter()
122        .zip(traces.iter())
123        .zip(aux_traces.iter().zip(aux_values_per_air.iter()))
124        .enumerate()
125    {
126        // `check_builder_shape` validates row-window widths per row, but aux trace
127        // height is invisible to a single row window — check it here so a short aux
128        // trace fails cleanly rather than via an opaque row-slice panic.
129        assert_eq!(
130            aux_trace.height(),
131            main.height(),
132            "instance {i}: aux trace height mismatch: expected {}, got {}",
133            main.height(),
134            aux_trace.height()
135        );
136
137        check_single_trace(air, main, aux_trace, aux_values, air_inputs, &challenges, i);
138    }
139}

141/// Check constraints for one instance's traces row by row.
142#[allow(clippy::too_many_arguments)]
143fn check_single_trace<F, EF, A>(
144    air: &A,
145    main: &RowMajorMatrix<F>,
146    aux_trace: &RowMajorMatrix<EF>,
147    aux_values: &[EF],
148    public_values: &[F],
149    challenges: &[EF],
150    instance_index: usize,
151) where
152    F: Field,
153    EF: ExtensionField<F>,
154    A: LiftedAir<F, EF>,
155{
156    let height = main.height();
157
158    // Preprocessed matrix comes straight off the AIR (debug-only; this
159    // re-materialises `BaseAir::preprocessed_trace`). Its height must match the main
160    // trace; width is checked per row by `check_builder_shape`.
161    let preprocessed = air.preprocessed_trace();
162    if let Some(preproc) = &preprocessed {
163        assert_eq!(
164            preproc.height(),
165            height,
166            "instance {instance_index}: preprocessed trace height mismatch: expected {height}, got {}",
167            preproc.height()
168        );
169    }
170
171    let periodic_matrix = air.periodic_columns_matrix();
172    for row in 0..height {
173        let next_row = (row + 1) % height;
174
175        // Main trace rows.
176        let main_current = main.row_slice(row).unwrap();
177        let main_next = main.row_slice(next_row).unwrap();
178
179        // Aux trace rows.
180        let aux_current = aux_trace.row_slice(row).unwrap();
181        let aux_next = aux_trace.row_slice(next_row).unwrap();
182
183        // Periodic values for this row via modulo indexing into the periodic table.
184        let periodic_row = periodic_matrix.as_ref().map(|m| m.row_slice(row % m.height()).unwrap());
185        let periodic_values: &[F] = periodic_row.as_deref().unwrap_or(&[]);
186
187        // Preprocessed rows (empty window when the AIR declares none).
188        let preprocessed_current = preprocessed.as_ref().map(|m| m.row_slice(row).unwrap());
189        let preprocessed_next = preprocessed.as_ref().map(|m| m.row_slice(next_row).unwrap());
190
191        let mut builder = DebugConstraintBuilder {
192            main: RowWindow::from_two_rows(&main_current, &main_next),
193            preprocessed: RowWindow::from_two_rows(
194                preprocessed_current.as_deref().unwrap_or(&[]),
195                preprocessed_next.as_deref().unwrap_or(&[]),
196            ),
197            permutation: RowWindow::from_two_rows(&aux_current, &aux_next),
198            randomness: &challenges[..air.num_randomness()],
199            public_values,
200            periodic_values,
201            permutation_values: aux_values,
202            is_first_row: F::from_bool(row == 0),
203            is_last_row: F::from_bool(row == height - 1),
204            is_transition: F::from_bool(row != height - 1),
205            instance_index,
206            row_index: row,
207        };
208
209        #[cfg(debug_assertions)]
210        miden_lifted_air::debug::check_builder_shape(air, &builder);
211
212        air.eval(&mut builder);
213    }
214}

220/// Lightweight constraint builder that checks constraints against concrete trace values.
221///
222/// Evaluates constraints row-by-row and panics immediately on the first nonzero constraint.
223/// Uses base field `F` for the main trace and extension field `EF` for the auxiliary
224/// (permutation) trace, matching the actual field layout of lifted STARK traces.
225struct DebugConstraintBuilder<'a, F: Field, EF: ExtensionField<F>> {
226    main: RowWindow<'a, F>,
227    preprocessed: RowWindow<'a, F>,
228    permutation: RowWindow<'a, EF>,
229    randomness: &'a [EF],
230    public_values: &'a [F],
231    periodic_values: &'a [F],
232    permutation_values: &'a [EF],
233    is_first_row: F,
234    is_last_row: F,
235    is_transition: F,
236    instance_index: usize,
237    row_index: usize,
238}
239
240impl<'a, F, EF> AirBuilder for DebugConstraintBuilder<'a, F, EF>
241where
242    F: Field,
243    EF: ExtensionField<F>,
244{
245    type F = F;
246    type Expr = F;
247    type Var = F;
248    type PreprocessedWindow = RowWindow<'a, F>;
249    type MainWindow = RowWindow<'a, F>;
250    type PublicVar = F;
251    type PeriodicVar = F;
252
253    fn main(&self) -> Self::MainWindow {
254        self.main
255    }
256
257    fn preprocessed(&self) -> &Self::PreprocessedWindow {
258        &self.preprocessed
259    }
260
261    fn is_first_row(&self) -> Self::Expr {
262        self.is_first_row
263    }
264
265    fn is_last_row(&self) -> Self::Expr {
266        self.is_last_row
267    }
268
269    fn is_transition(&self) -> Self::Expr {
270        self.is_transition
271    }
272
273    fn assert_zero<I: Into<Self::Expr>>(&mut self, x: I) {
274        assert_eq!(
275            x.into(),
276            F::ZERO,
277            "constraint not satisfied at instance {}, row {}",
278            self.instance_index,
279            self.row_index
280        );
281    }
282
283    fn public_values(&self) -> &[Self::PublicVar] {
284        self.public_values
285    }
286
287    fn periodic_values(&self) -> &[Self::PeriodicVar] {
288        self.periodic_values
289    }
290}

310            self.row_index

313}
