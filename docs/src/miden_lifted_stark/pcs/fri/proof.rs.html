1//! FRI structured proof types — parsed view of the FRI sub-transcript.
2
3use alloc::vec::Vec;
4
5use miden_stark_transcript::{TranscriptError, VerifierChannel};
6use p3_field::{ExtensionField, TwoAdicField};
7
8use crate::{domain::LiftedDomain, pcs::fri::FriParams};
9
10/// Structured view of a single FRI folding round.
11pub struct FriRoundProof<F, EF, Commitment> {
12    /// Commitment to the folded evaluation matrix for this round.
13    pub commitment: Commitment,
14    /// Proof-of-work witness sampled before `beta`.
15    pub pow_witness: F,
16    /// Folding challenge `β` for this round.
17    pub beta: EF,
18}
19
20/// Structured view of the full FRI sub-proof.
21pub struct FriProof<F, EF, Commitment> {
22    /// Per-round commitments and challenges.
23    pub rounds: Vec<FriRoundProof<F, EF, Commitment>>,
24    /// Coefficients of the final low-degree polynomial in descending degree order
25    /// `[cₙ, ..., c₁, c₀]`, ready for direct Horner evaluation.
26    pub final_poly: Vec<EF>,
27}
28
29impl<F, EF, Commitment> FriProof<F, EF, Commitment>
30where
31    F: TwoAdicField,
32    EF: ExtensionField<F>,
33    Commitment: Clone,
34{
35    /// Parse a [`FriProof`] from a verifier channel.
36    ///
37    /// Reads commitments, verifies PoW witnesses, samples challenges, and
38    /// reads the final polynomial. Does not verify low-degree claims;
39    /// that validation happens in `FriOracle::test_low_degree`.
40    pub(in crate::pcs) fn read_from_channel<Ch>(
41        params: &FriParams,
42        domain: &LiftedDomain<F>,
43        channel: &mut Ch,
44    ) -> Result<Self, TranscriptError>
45    where
46        Ch: VerifierChannel<F = F, Commitment = Commitment>,
47    {
48        let num_rounds = params.num_rounds(domain);
49        let mut rounds = Vec::with_capacity(num_rounds);
50
51        for _ in 0..num_rounds {
52            let commitment = channel.receive_commitment()?.clone();
53
54            let pow_witness = channel.grind(params.folding_pow_bits)?;
55
56            let beta: EF = channel.sample_algebra_element();
57            rounds.push(FriRoundProof { commitment, pow_witness, beta });
58        }
59
60        let final_degree = params.final_poly_degree(domain);
61        let final_poly = channel.receive_algebra_slice(final_degree)?;
62
63        Ok(Self { rounds, final_poly })
64    }
65}
