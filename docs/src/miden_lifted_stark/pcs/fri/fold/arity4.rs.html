1//! Arity-4 FRI folding using inverse FFT.
2//!
3//! Given evaluations of a polynomial `f` on a coset `s·⟨ω⟩` where `ω = i` is a primitive
4//! 4th root of unity, we recover the folded value `g(s⁴)` for a challenge `β`
5//! (and when `deg f < 4`, this equals `f(β)`).
6//!
7//! ## Setup
8//!
9//! Let `f(X) = c₀ + c₁X + c₂X² + c₃X³` with evaluations on the coset `s·⟨ω⟩`:
10//!
11//! ```text
12//! y₀ = f(s),   y₁ = f(ωs),   y₂ = f(ω²s),   y₃ = f(ω³s)
13//! ```
14//!
15//! We store these in **bit-reversed order**: `[y₀, y₂, y₁, y₃]`.
16//!
17//! We decompose `f` by residue class modulo 4:
18//! `f(X) = Σⱼ X^j · fⱼ(X⁴)` for j ∈ {0,1,2,3}.
19//! The folded polynomial is `g(X) = Σⱼ β^j · fⱼ(X)`.
20//!
21//! ## Algorithm
22//!
23//! 1. **Inverse FFT**: Recover coefficients of `f(sX)` from evaluations on `⟨ω⟩`.
24//! 2. **Evaluate**: Compute `f(sX)` at `X = β/s`, yielding the folded value `g(s⁴)`.
25
26use core::array;
27
28use p3_field::{Algebra, TwoAdicField};
29
30/// Evaluate the folded value `g(s⁴)` from evaluations on a coset
31/// (equals `f(β)` when `deg f < 4`).
32///
33/// ## Inputs
34///
35/// - `evals`: slice of 4 evaluations `[f(s), f(ω²s), f(ωs), f(ω³s)]` in bit-reversed order,
36///   equivalently `[f(s), f(−s), f(is), f(−is)]` since `ω = i`.
37/// - `s_inv`: the inverse of the coset generator `s`.
38/// - `beta`: the FRI folding challenge `β`.
39///
40/// ## FRI Context
41///
42/// In arity-4 FRI, the polynomial `f` is evaluated on cosets of the form `s·⟨ω⟩`.
43/// The verifier needs to check that the folded value `g(s⁴)` matches the prover's claim.
44/// This function recovers `g(s⁴)` from the four coset evaluations via interpolation.
45#[inline(always)]
46pub fn fold_evals<F, PF, PEF>(evals: &[PEF], s_inv: PF, beta: PEF) -> PEF
47where
48    F: TwoAdicField,
49    PF: Algebra<F> + Algebra<PF>,
50    PEF: Algebra<PF>,
51{
52    debug_assert_eq!(evals.len(), 4, "evals must have 4 elements");
53    let evals = array::from_fn(|i| evals[i].clone());
54    // Recover coefficients [c₀, c₁, c₂, c₃] of 4·f(sX) via inverse FFT.
55    let [c0, c1, c2, c3] = ifft4::<F, PF, PEF>(evals);
56
57    // Folded value g(s⁴) = (1/4) · (c₀ + c₁·x + c₂·x² + c₃·x³)  where x = β/s.
58    let x = beta * s_inv;
59    let terms = [
60        c0,              // c₀
61        c1 * x.clone(),  // c₁ · x
62        c2 * x.square(), // c₂ · x²
63        c3 * x.cube(),   // c₃ · x³
64    ];
65
66    // Divide by 4
67    let four_inv: PF = F::ONE.halve().halve().into();
68    PEF::sum_array::<4>(&terms) * four_inv
69}
70
71/// Size-4 inverse FFT (unscaled), input in bit-reversed order.
72///
73/// Returns coefficients `[c₀, c₁, c₂, c₃]` of `4·f(sX) = c₀ + c₁X + c₂X² + c₃X³`.
74#[inline(always)]
75fn ifft4<F, PF, PEF>(evals: [PEF; 4]) -> [PEF; 4]
76where
77    F: TwoAdicField,
78    PF: Algebra<F> + Algebra<PF>,
79    PEF: Algebra<PF>,
80{
81    // ω = i, primitive 4th root of unity
82    let w: PF = F::two_adic_generator(2).into();
83
84    // Input (bit-reversed): [y₀, y₂, y₁, y₃]
85    let [y0, y2, y1, y3] = evals;
86
87    // Inverse DFT formula (without 1/N normalization):
88    //   4cⱼ = Σₖ yₖ · ω^(−jk)
89    //
90    // Expanded for each coefficient (i = imaginary unit):
91    //   4c₀ = y₀ + y₁ + y₂ + y₃
92    //   4c₁ = y₀ − i·y₁ − y₂ + i·y₃
93    //   4c₂ = y₀ − y₁ + y₂ − y₃
94    //   4c₃ = y₀ + i·y₁ − y₂ − i·y₃
95
96    // -------------------------------------------------------------------------
97    // Stage 0: length-2 butterflies on bit-reversed pairs
98    // -------------------------------------------------------------------------
99    let s02 = y0.clone() + y2.clone(); // y₀ + y₂  (used in c₀, c₂)
100    let d02 = y0 - y2; // y₀ − y₂  (used in c₁, c₃)
101    let s13 = y1.clone() + y3.clone(); // y₁ + y₃  (used in c₀, c₂)
102    let d31 = y3 - y1; // y₃ − y₁  (note: negated so we can multiply by ω instead of ω⁻¹)
103
104    // -------------------------------------------------------------------------
105    // Stage 1: combine via length-4 butterflies
106    //
107    // Rewriting the target formulas using stage 0 results:
108    //   4c₀ = (y₀ + y₂) + (y₁ + y₃)           = s02 + s13
109    //   4c₂ = (y₀ + y₂) − (y₁ + y₃)           = s02 − s13
110    //   4c₁ = (y₀ − y₂) + i(y₃ − y₁)          = d02 + i·d31
111    //   4c₃ = (y₀ − y₂) − i(y₃ − y₁)          = d02 − i·d31
112    // -------------------------------------------------------------------------
113    let d31_w = d31 * w; // i · (y₃ − y₁)
114
115    [
116        s02.clone() + s13.clone(),   // 4c₀
117        d02.clone() + d31_w.clone(), // 4c₁
118        s02 - s13,                   // 4c₂
119        d02 - d31_w,                 // 4c₃
120    ]
121}

127#[cfg(test)]
128mod tests {
129    use p3_dft::{NaiveDft, TwoAdicSubgroupDft};
130    use p3_field::PrimeCharacteristicRing;
131    use p3_matrix::dense::RowMajorMatrix;
132    use rand::{RngExt, SeedableRng, distr::StandardUniform, prelude::SmallRng};
133
134    use super::*;
135    use crate::testing::configs::goldilocks_poseidon2::{Felt, QuadFelt};
136
137    /// Test that ifft4 correctly recovers polynomial coefficients from DFT evaluations.
138    #[test]
139    fn test_ifft4() {
140        let mut rng = SmallRng::seed_from_u64(42);
141
142        // Random polynomial coefficients
143        let coeffs: [QuadFelt; 4] = array::from_fn(|_| rng.sample(StandardUniform));
144
145        // Compute DFT using NaiveDft (standard order)
146        let coeffs_matrix = RowMajorMatrix::new(coeffs.to_vec(), 1);
147        let evals_matrix = NaiveDft.dft_batch(coeffs_matrix);
148        let evals_std = evals_matrix.values;
149
150        // Convert to bit-reversed order for ifft4
151        let evals_br: [QuadFelt; 4] = [evals_std[0], evals_std[2], evals_std[1], evals_std[3]];
152
153        // Run ifft4 (returns 4 * coefficients)
154        let recovered_scaled = ifft4::<Felt, Felt, QuadFelt>(evals_br);
155
156        // Verify: recovered_scaled[i] == 4 * coeffs[i]
157        for (i, (recovered, &original)) in recovered_scaled.iter().zip(coeffs.iter()).enumerate() {
158            let expected = original.double().double(); // 4 * original
159            assert_eq!(*recovered, expected, "Coefficient mismatch at index {i}");
160        }
161    }
