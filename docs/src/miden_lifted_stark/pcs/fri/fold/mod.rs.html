1//! FRI folding via polynomial interpolation.
2//!
3//! FRI (Fast Reed-Solomon IOP of Proximity) folds evaluations on a coset into a
4//! lower-degree polynomial value parameterized by a challenge `β`. Each row fold
5//! returns the folded value `g(s^r)` for its coset `s·⟨ω_r⟩` (and when `deg f < r`,
6//! this equals `f(β)`). This module provides a struct-based abstraction for FRI
7//! folding at different arities.
8//!
9//! ## Arity
10//!
11//! The **arity** determines how many evaluations are folded together in each round:
12//! - **Arity 2**: Fold pairs `{f(s), f(-s)}` using even-odd decomposition
13//! - **Arity 4**: Fold quadruples `{f(s), f(-s), f(is), f(-is)}` using inverse FFT
14//! - **Arity 8**: Fold octuples using size-8 inverse FFT
15//!
16//! Higher arity reduces the number of FRI rounds but increases per-round work.
17
18mod arity2;
19mod arity4;
20mod arity8;

28/// FRI folding strategy.
29///
30/// This struct encapsulates different folding arities (2, 4, 8).
31#[derive(Clone, Copy, Debug)]
32pub struct FriFold {
33    pub(crate) log_arity: u8,
34}
35
36impl FriFold {
37    /// Create a new folder for a supported log-arity (currently only 1, 2, 3).
38    pub const fn new(log_arity: u8) -> Option<Self> {
39        if log_arity == 1 || log_arity == 2 || log_arity == 3 {
40            Some(Self { log_arity })
41        } else {
42            None
43        }
44    }
45
46    #[inline]
47    pub const fn arity(self) -> usize {
48        1 << self.log_arity as usize
49    }
50
51    #[inline]
52    pub const fn log_arity(self) -> u8 {
53        self.log_arity
54    }
55
56    /// Fold evaluations from a slice of extension field elements.
57    ///
58    /// The slice must have exactly `arity()` elements.
59    /// Used by the verifier in scalar mode.
60    ///
61    /// Folding is the core FRI step: it turns `arity` evaluations of `f` on a coset
62    /// `s·⟨ω⟩` into a single evaluation of a new polynomial `g` on the folded domain.
63    ///
64    /// Conceptually, write `f(X)` as `Σⱼ Xʲ·fⱼ(X^arity)`. The fold interpolates the
65    /// `fⱼ` values from the row (an iFFT on the coset) and then takes a random linear
66    /// combination with challenge `β` to obtain `g(s^arity)`. The resulting `g` has
67    /// degree reduced by a factor of `arity`. If `deg(f) < arity`, folding recovers
68    /// `f(β)` exactly.
69    #[inline]
70    pub fn fold_evals<F: TwoAdicField, EF: ExtensionField<F>>(
71        self,
72        evals: &[EF],
73        s_inv: F,
74        beta: EF,
75    ) -> EF {
76        match self.log_arity {
77            1 => arity2::fold_evals::<F, F, EF>(evals, s_inv, beta),
78            2 => arity4::fold_evals::<F, F, EF>(evals, s_inv, beta),
79            3 => arity8::fold_evals::<F, F, EF>(evals, s_inv, beta),
80            _ => unreachable!("unsupported arity"),
81        }
82    }
83
84    /// Packed (SIMD) version of `fold_evals`.
85    #[inline]
86    fn fold_evals_packed<F: TwoAdicField, EF: ExtensionField<F>>(
87        self,
88        evals: &[EF::ExtensionPacking],
89        s_inv: F::Packing,
90        beta: EF,
91    ) -> EF::ExtensionPacking {
92        let beta_packed: EF::ExtensionPacking = beta.into();
93        match self.log_arity {
94            1 => {
95                arity2::fold_evals::<F, F::Packing, EF::ExtensionPacking>(evals, s_inv, beta_packed)
96            },
97            2 => {
98                arity4::fold_evals::<F, F::Packing, EF::ExtensionPacking>(evals, s_inv, beta_packed)
99            },
100            3 => {
101                arity8::fold_evals::<F, F::Packing, EF::ExtensionPacking>(evals, s_inv, beta_packed)
102            },
103            _ => unreachable!("unsupported arity"),
104        }
105    }

107    /// Fold a matrix of coset evaluations using the challenge `beta`.
108    ///
109    /// Each row contains evaluations on a coset `s·⟨ω⟩`. Returns folded
110    /// evaluations, one per row, maintaining bit-reversed order.
111    ///
112    /// Automatically dispatches to scalar or packed implementation based on matrix size.
113    pub fn fold_matrix<F: TwoAdicField, EF: ExtensionField<F>>(
114        self,
115        input: RowMajorMatrixView<'_, EF>,
116        s_invs: &[F],
117        beta: EF,
118    ) -> Vec<EF> {
119        let width = F::Packing::WIDTH;
120        if input.height() < width || width == 1 {
121            // Scalar path
122            let arity = self.arity();
123            assert_eq!(input.width, arity);
124            input
125                .values
126                .par_chunks(arity)
127                .zip(s_invs.par_iter())
128                .map(|(evals, &s_inv)| self.fold_evals(evals, s_inv, beta))
129                .collect()
130        } else {
131            match self.log_arity {
132                1 => self.fold_matrix_packed_impl::<2, F, EF>(input, s_invs, beta),
133                2 => self.fold_matrix_packed_impl::<4, F, EF>(input, s_invs, beta),
134                3 => self.fold_matrix_packed_impl::<8, F, EF>(input, s_invs, beta),
135                _ => unreachable!("unsupported arity"),
136            }
137        }
138    }
139
140    fn fold_matrix_packed_impl<const ARITY: usize, F, EF>(
141        self,
142        input: RowMajorMatrixView<'_, EF>,
143        s_invs: &[F],
144        beta: EF,
145    ) -> Vec<EF>
146    where
147        F: TwoAdicField,
148        EF: ExtensionField<F>,
149    {
150        assert_eq!(input.width, ARITY);
151        assert_eq!(input.values.len() % ARITY, 0);
152        let evals: &[[EF; ARITY]] = unsafe {
153            // SAFETY: the slice length is checked above to be a multiple of `ARITY`, and
154            // `[EF; ARITY]` is laid out contiguously like `ARITY` adjacent `EF` values.
155            core::slice::from_raw_parts(input.values.as_ptr().cast(), input.values.len() / ARITY)
156        };
157        let width = F::Packing::WIDTH;
158        assert_eq!(evals.len() % width, 0);
159
160        let mut new_evals = EF::zero_vec(evals.len());
161
162        new_evals
163            .par_chunks_exact_mut(width)
164            .zip(evals.par_chunks_exact(width))
165            .zip(s_invs.par_chunks_exact(width))
166            .for_each(|((new_evals_chunk, evals_chunk), s_inv_chunk)| {
167                let evals_packed =
168                    <EF::ExtensionPacking as PackedFieldExtension<F, EF>>::pack_ext_columns::<ARITY>(
169                        evals_chunk,
170                    );
171                let s_invs_packed = F::Packing::from_slice(s_inv_chunk);
172                let new_evals_packed =
173                    self.fold_evals_packed::<F, EF>(&evals_packed, *s_invs_packed, beta);
174                <EF::ExtensionPacking as PackedFieldExtension<F, EF>>::to_ext_slice(
175                    &new_evals_packed,
176                    new_evals_chunk,
177                );
178            });
179        new_evals
180    }
181}

213    /// Test fold_evals against NaiveDft coset evaluations for a specific arity.
214    ///
215    /// Generates a random polynomial, computes evaluations on a coset using NaiveDft,
216    /// then verifies fold_evals correctly recovers f(β).
217    fn test_fold_evals_naive_dft(fold: FriFold) {
218        let mut rng = SmallRng::seed_from_u64(42);
219        let arity = fold.arity();
220
221        // Polynomial of degree arity-1
222        let coeffs: Vec<QuadFelt> = (0..arity).map(|_| rng.sample(StandardUniform)).collect();
223
224        // Coset generator
225        let s: Felt = rng.sample(StandardUniform);
226        let s_inv = s.inverse();
227
228        // Compute evaluations using NaiveDft on coset s·⟨ω⟩
229        let mut coeffs_padded = coeffs.clone();
230        coeffs_padded.resize(arity, QuadFelt::ZERO);
231        let coeffs_matrix = RowMajorMatrix::new(coeffs_padded, 1);
232        let evals_matrix = NaiveDft.coset_dft_batch(coeffs_matrix, QuadFelt::from(s));
233        let mut evals: Vec<QuadFelt> = evals_matrix.values;
234        reverse_slice_index_bits(&mut evals);
235
236        // Fold with random beta
237        let beta: QuadFelt = rng.sample(StandardUniform);
238        let result = fold.fold_evals(&evals, s_inv, beta);
239
240        // Expected: direct Horner evaluation at beta
241        let expected = coeffs.iter().copied().horner(beta);
242        assert_eq!(result, expected, "fold_evals mismatch for arity {arity}");
243    }

245    /// Test FRI folding correctness for a specific arity.
246    ///
247    /// Creates a random polynomial of degree `arity - 1`, evaluates it on a coset
248    /// of size `arity`, then verifies that `fold_evals` correctly recovers `f(β)`.
249    fn test_fold_correctness<Base, Ext>(fold: FriFold)
250    where
251        Base: TwoAdicField,
252        Ext: ExtensionField<Base>,
253        StandardUniform: Distribution<Ext> + Distribution<Base>,
254    {
255        let rng = &mut SmallRng::seed_from_u64(1);
256        let beta: Ext = rng.sample(StandardUniform);
257        let arity = fold.arity();
258        let log_arity = fold.log_arity();
259
260        // Random polynomial of degree arity - 1
261        let poly: Vec<Ext> = (0..arity).map(|_| rng.sample(StandardUniform)).collect();
262
263        // Roots of unity in bit-reversed order for this arity.
264        let roots: Vec<Base> = TwoAdicSubgroup::<Base>::new(log_arity).bit_reversed_points();
265
266        let s: Base = rng.sample(StandardUniform);
267        let s_inv = s.inverse();
268
269        // Evaluate polynomial at coset points: [f(s·root) for root in roots]
270        let evals: Vec<Ext> =
271            roots.iter().map(|&root| poly.iter().copied().horner(root * s)).collect();
272
273        // Expected: f(beta)
274        let expected = poly.iter().copied().horner(beta);
275
276        // Test fold_evals
277        let result = fold.fold_evals(&evals, s_inv, beta);
278        assert_eq!(result, expected);
279    }

281    /// Test that `fold_matrix` scalar and packed paths produce identical results.
282    ///
283    /// Creates a matrix large enough to trigger the packed path, then verifies
284    /// the result matches row-by-row scalar `fold_evals` computation.
285    fn test_fold_matrix_scalar_packed_equivalence(fold: FriFold) {
286        let rng = &mut SmallRng::seed_from_u64(42);
287        let arity = fold.arity();
288
289        // Create input matrix with height = multiple of packing width (triggers packed path)
290        let height = Pf::WIDTH * 4;
291        let values: Vec<QuadFelt> =
292            (0..height * arity).map(|_| rng.sample(StandardUniform)).collect();
293        let input = RowMajorMatrix::new(values.clone(), arity);
294
295        // Generate random coset generators and their inverses
296        let s_values: Vec<Felt> =
297            (0..height).map(|_| rng.sample::<Felt, _>(StandardUniform)).collect();
298        let s_invs: Vec<Felt> = s_values.iter().map(Field::inverse).collect();
299
300        let beta: QuadFelt = rng.sample(StandardUniform);
301
302        // Scalar path: compute fold_evals for each row
303        let scalar_result: Vec<QuadFelt> = values
304            .chunks(arity)
305            .zip(s_invs.iter())
306            .map(|(evals, &s_inv)| fold.fold_evals(evals, s_inv, beta))
307            .collect();
308
309        // Packed path: call fold_matrix (uses packed impl for large matrices)
310        let packed_result = fold.fold_matrix(input.as_view(), &s_invs, beta);
311
312        assert_eq!(scalar_result, packed_result, "Scalar vs packed mismatch for arity {arity}");
313    }

315    /// Test that folding preserves low-degree structure.
316    ///
317    /// After folding a degree-d polynomial, the result should have degree d/arity.
318    /// Verifies by checking that high coefficients are zero after IDFT.
319    fn test_folding_preserves_low_degree(fold: FriFold) {
320        let rng = &mut SmallRng::seed_from_u64(42);
321        let arity = fold.arity();
322        let log_arity: u8 = fold.log_arity();
323
324        let log_blowup: u8 = 2;
325        let log_poly_degree: u8 = 4; // degree 16 polynomial
326        let poly_degree = 1usize << log_poly_degree;
327        let log_lde_size: u8 = log_poly_degree + log_blowup;
328        let lde_size = 1usize << log_lde_size;
329
330        // Generate random low-degree polynomial
331        let coeffs: Vec<QuadFelt> = (0..poly_degree).map(|_| rng.sample(StandardUniform)).collect();
332
333        // Compute LDE in bit-reversed order
334        let mut full_coeffs = coeffs;
335        full_coeffs.resize(lde_size, QuadFelt::ZERO);
336        let dft = Radix2DFTSmallBatch::<QuadFelt>::default();
337        let mut evals = dft.dft_algebra(full_coeffs);
338        reverse_slice_index_bits(&mut evals);
339
340        // Compute s_invs
341        let log_num_cosets = log_lde_size - log_arity;
342        let num_cosets = 1usize << log_num_cosets;
343        let g_inv = TwoAdicSubgroup::<Felt>::new(log_lde_size).generator_inverse();
344        let mut s_invs: Vec<Felt> = g_inv.powers().take(num_cosets).collect();
345        reverse_slice_index_bits(&mut s_invs);
346
347        // Fold with random beta
348        let beta: QuadFelt = rng.sample(StandardUniform);
349        let matrix = RowMajorMatrix::new(evals, arity);
350        let folded = fold.fold_matrix(matrix.as_view(), &s_invs, beta);
351
352        // IDFT the result to get coefficients
353        let mut folded_for_idft = folded;
354        reverse_slice_index_bits(&mut folded_for_idft);
355        let folded_coeffs = dft.idft_algebra(folded_for_idft);
356
357        // Check that all coefficients beyond degree/arity are zero
358        let expected_degree = poly_degree / arity;
359        for (i, coeff) in folded_coeffs.iter().enumerate().skip(expected_degree) {
360            assert_eq!(
361                *coeff,
362                QuadFelt::ZERO,
363                "Arity {arity}: High coefficient c[{i}] should be zero but was {coeff:?}"
364            );
365        }
366    }
