1//! # FRI Protocol Implementation
2//!
3//! Fast Reed-Solomon Interactive Oracle Proof for low-degree testing.
4//! Proves that a committed polynomial has degree below a target bound.
5//!
6//! ## Domain Convention
7//!
8//! This FRI implementation treats inputs as evaluations over the unshifted two-adic
9//! subgroup. If the PCS evaluates over a coset `gK`, the shift is absorbed into
10//! the polynomial: `Q'(X) = Q(g·X)`. The low-degree test is run on `Q'` using
11//! subgroup points.
12//!
13//! ## Type vocabulary
14//!
15//! FRI takes its initial domain as a [`LiftedDomain<F>`](crate::domain::LiftedDomain) —
16//! the protocol-level LDE object carrying both the LDE subgroup size and the blowup
17//! ratio relative to the trace. The unshifted subgroup view is reached via
18//! `domain.lde_coset().subgroup()`. Each fold round shrinks the working subgroup by the
19//! folding arity, derived via [`TwoAdicSubgroup::shrink`](crate::domain::TwoAdicSubgroup::shrink)
20//! (or per-round generator squaring inside the round loop, which is equivalent and avoids
21//! re-querying `F::two_adic_generator`). Internal `arity`-th roots of unity used by the fold
22//! operations come from `TwoAdicSubgroup::<F>::new(log_arity).generator()`. Routing every
23//! two-adic root and every multiplicative coset shift through these encapsulation types keeps
24//! `F::two_adic_generator` and `F::GENERATOR` confined to their single canonical sites.
25

36/// FRI protocol parameters.
37///
38/// Controls the trade-off between proof size, prover time, and verifier time.
39///
40/// Higher arity reduces the number of FRI rounds (fewer Merkle tree commitments) but increases
41/// per-query proof size (each opening reveals `arity` siblings). `log_final_degree` reduces the
42/// number of rounds and therefore the number of Merkle commitments; if too large, the final
43/// polynomial's coefficients dominate the proof size.
44///
45/// The LDE blowup factor is **not** stored here — it is a structural property of the codeword
46/// being tested and is read from the [`LiftedDomain`](crate::domain::LiftedDomain) passed to
47/// [`num_rounds`](Self::num_rounds), [`final_poly_degree`](Self::final_poly_degree),
48/// [`FriPolys::new`](prover::FriPolys::new), and
49/// [`FriOracle::new`](verifier::FriOracle::new).
50#[derive(Clone, Copy, Debug)]
51pub(crate) struct FriParams {
52    /// The FRI folding strategy.
53    ///
54    /// Determines the folding arity (2, 4, or 8).
55    pub(crate) fold: FriFold,
56
57    /// Log₂ of the final polynomial degree.
58    ///
59    /// Folding stops when degree reaches `2^log_final_degree`.
60    /// Final polynomial coefficients are sent in descending degree order
61    /// `[cₙ, ..., c₁, c₀]` for direct Horner evaluation by the verifier.
62    pub(crate) log_final_degree: u8,
63
64    /// Grinding bits before each folding challenge.
65    pub(crate) folding_pow_bits: usize,
66}
67
68impl FriParams {
69    /// Compute the number of folding rounds for an LDE codeword evaluated on `domain`.
70    ///
71    /// Each round reduces the domain by `2^log_folding_factor`. We fold until the domain
72    /// size reaches `2^(log_final_degree + log_blowup)`, at which point the polynomial
73    /// degree is at most `2^log_final_degree`.
74    ///
75    /// Uses `div_ceil` to round up, ensuring we always reach the target degree even if
76    /// the domain size doesn't divide evenly by the folding factor. `PcsParams::new`
77    /// rejects parameter sets whose target final domain is too small to be reachable by
78    /// fixed-arity folding for all valid domains.
79    #[inline]
80    pub fn num_rounds<F: TwoAdicField>(&self, domain: &LiftedDomain<F>) -> usize {
81        // Maximum domain size needed to accommodate a degree-`2^log_final_degree` polynomial
82        // after folding `num_rounds` times.
83        let log_max_final_size = u16::from(self.log_final_degree) + u16::from(domain.log_blowup());
84        // Number of domain squarings required to reach a domain of size at most
85        // `log_max_final_size`. `saturating_sub` covers the degenerate "LDE already at or
86        // below target" case.
87        let num_steps = u16::from(domain.log_lde_height()).saturating_sub(log_max_final_size);
88        // Divide the number of steps by the folding factor to get the number of rounds.
89        // Round up so the final domain is ≤ `2^log_max_final_size` even when the
90        // folding factor doesn't divide `num_steps` evenly. The last round may
91        // overshoot, leaving the actual final degree strictly below the bound —
92        // see [`final_poly_degree`](Self::final_poly_degree).
93        num_steps.div_ceil(u16::from(self.fold.log_arity())) as usize
94    }
95
96    /// Compute the final polynomial degree after folding the codeword evaluated on `domain`.
97    ///
98    /// After `num_rounds` folding rounds, the LDE domain shrinks from
99    /// `2^domain.log_lde_height()` to `2^(log_lde_height − num_rounds × log_folding_factor)`.
100    /// The polynomial degree is then `domain_size / blowup`.
101    ///
102    /// Due to `div_ceil` in `num_rounds`, the actual final degree may be smaller than
103    /// `2^log_final_degree` when the folding doesn't divide evenly.
104    #[inline]
105    pub fn final_poly_degree<F: TwoAdicField>(&self, domain: &LiftedDomain<F>) -> usize {
106        let num_rounds = self.num_rounds(domain);
107        // log of the final domain size: starting LDE shrunk by num_rounds folds of
108        // factor 2^log_arity.
109        let log_final_domain_size = (domain.log_lde_height() as usize)
110            .saturating_sub(num_rounds * self.fold.log_arity() as usize);
111        let log_final_poly_degree =
112            log_final_domain_size.saturating_sub(domain.log_blowup() as usize);
113        // Poly degree = final domain size / blowup.
114        1 << log_final_poly_degree
115    }
116}
