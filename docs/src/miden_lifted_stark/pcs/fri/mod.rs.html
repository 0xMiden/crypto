1//! # FRI Protocol Implementation
2//!
3//! Fast Reed-Solomon Interactive Oracle Proof for low-degree testing.
4//! Proves that a committed polynomial has degree below a target bound.
5//!
6//! ## Domain Convention
7//!
8//! This FRI implementation treats inputs as evaluations over the unshifted two-adic subgroup.
9//! If the PCS evaluates over a coset `gK`, the shift is absorbed into the polynomial:
10//! `Q'(X) = Q(g·X)`. The low-degree test is run on `Q'` using subgroup points.
11
12pub mod fold;
13pub mod proof;
14pub mod prover;
15pub mod verifier;

17use fold::FriFold;

19/// FRI protocol parameters.
20///
21/// Controls the trade-off between proof size, prover time, and verifier time.
22///
23/// Higher `log_blowup` increases soundness per query (fewer queries needed) but introduces
24/// larger Merkle trees — increasing both proof size (longer authentication paths) and prover time
25/// (LDE over a larger domain). Higher arity reduces the number of FRI rounds (fewer Merkle
26/// tree commitments) but increases per-query proof size (each opening reveals `arity`
27/// siblings). `log_final_degree` reduces the number of rounds and therefore the number of
28/// Merkle commitments; if too large, the final polynomial's coefficients dominate the proof
29/// size.
30#[derive(Clone, Copy, Debug)]
31pub struct FriParams {
32    /// Log₂ of the blowup factor (LDE domain size / polynomial degree).
33    ///
34    /// Higher values increase soundness but also proof size and prover time.
35    /// Typical values: 2-4 (blowup factors of 4-16).
36    pub(crate) log_blowup: u8,
37
38    /// The FRI folding strategy.
39    ///
40    /// Determines the folding arity (2, 4, or 8).
41    pub(crate) fold: FriFold,
42
43    /// Log₂ of the final polynomial degree.
44    ///
45    /// Folding stops when degree reaches `2^log_final_degree`.
46    /// Final polynomial coefficients are sent in descending degree order
47    /// `[cₙ, ..., c₁, c₀]` for direct Horner evaluation by the verifier.
48    pub(crate) log_final_degree: u8,
49
50    /// Grinding bits before each folding challenge.
51    pub(crate) folding_pow_bits: usize,
52}
53
54impl FriParams {
55    /// Compute the number of folding rounds for a given initial evaluation domain size.
56    ///
57    /// Each round reduces the domain by `2^log_folding_factor`. We fold until the domain
58    /// size reaches `2^(log_final_degree + log_blowup)`, at which point the polynomial
59    /// degree is at most `2^log_final_degree`.
60    ///
61    /// Uses `div_ceil` to round up, ensuring we always reach the target degree even if
62    /// the domain size doesn't divide evenly by the folding factor.
63    #[inline]
64    pub fn num_rounds(&self, log_domain_size: u8) -> usize {
65        // Final domain size = final_degree × blowup = 2^(log_final_degree + log_blowup).
66        // Safety: PcsParams::new() validates this sum does not exceed MAX_LOG_DOMAIN_SIZE.
67        debug_assert!(
68            (self.log_final_degree as u16 + self.log_blowup as u16)
69                <= crate::pcs::params::MAX_LOG_DOMAIN_SIZE as u16,
70            "log_final_degree + log_blowup overflows; construct FriParams via PcsParams::new()",
71        );
72        let log_max_final_size = self.log_final_degree + self.log_blowup;
73        // Number of times we need to divide by 2^log_folding_factor
74        log_domain_size
75            .saturating_sub(log_max_final_size)
76            .div_ceil(self.fold.log_arity()) as usize
77    }
78
79    /// Compute the final polynomial degree after folding.
80    ///
81    /// After `num_rounds` folding rounds, the domain shrinks from `2^log_domain_size`
82    /// to `2^(log_domain_size - num_rounds × log_folding_factor)`. The polynomial
83    /// degree is then `domain_size / blowup`.
84    ///
85    /// Due to `div_ceil` in `num_rounds`, the actual final degree may be smaller than
86    /// `2^log_final_degree` when the folding doesn't divide evenly.
87    #[inline]
88    pub fn final_poly_degree(&self, log_domain_size: u8) -> usize {
89        let num_rounds = self.num_rounds(log_domain_size);
90        // log of final domain size after folding
91        let log_final_size = log_domain_size as usize - num_rounds * self.fold.log_arity() as usize;
92        // degree = domain_size / blowup = 2^(log_final_size - log_blowup)
93        1 << log_final_size.saturating_sub(self.log_blowup as usize)
94    }
95}
