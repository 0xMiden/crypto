24/// The DEEP quotient `Q(X)` evaluated over the LDE domain.
25///
26/// Combines all polynomial evaluation claims into a single low-degree polynomial.
27///
28/// Intuition: DEEP turns many separate opening claims into a single polynomial identity.
29///
30/// After reducing all opened columns with a random `α` into one polynomial `f_red(X)`,
31/// DEEP defines
32///
33/// ```text
34/// Q(X) = Σⱼ βʲ · (f_red(zⱼ) − f_red(X)) / (zⱼ − X)
35/// ```
36///
37/// If the claimed OOD values `f_red(zⱼ)` are correct, each numerator vanishes at
38/// `X = zⱼ` and cancels the denominator, so `Q` is a (low-degree) polynomial.
39/// If any claim is wrong, the corresponding term has a pole and `Q` is not even a
40/// polynomial; committing to `Q` via FRI then detects the inconsistency.
41pub struct DeepPoly<EF> {
42    /// The DEEP quotient polynomial evaluated over the domain.
43    /// `deep_evals[i]` is the evaluation at the i-th domain point (bit-reversed order).
44    pub(crate) deep_evals: Vec<EF>,
45}
46
47impl<EF> DeepPoly<EF> {
48    /// Construct `Q(X)` by evaluating trace trees at the opening points.
49    ///
50    /// This computes the LDE coset points from the trace tree height, evaluates the committed
51    /// matrices at `eval_points`, and then calls [`Self::from_evals`].
52    ///
53    /// Preconditions: `eval_points` must be distinct and lie outside the trace subgroup `H`
54    /// and LDE evaluation coset `gK`. The outer protocol is expected to enforce this.
55    pub fn from_trees<L, M, const N: usize, Ch>(
56        params: DeepParams,
57        trace_trees: &[&L::Tree<M>],
58        eval_points: [EF; N],
59        log_blowup: u8,
60        channel: &mut Ch,
61    ) -> Self
62    where
63        L: Lmcs,
64        L::F: TwoAdicField,
65        EF: ExtensionField<L::F>,
66        M: Matrix<L::F>,
67        Ch: ProverChannel<F = L::F, Commitment = L::Commitment>,
68    {
69        let lde_height = trace_trees.first().expect("at least one trace tree required").height();
70        assert!(
71            trace_trees.iter().all(|tree| tree.height() == lde_height),
72            "mixed trace tree heights are not supported"
73        );
74
75        let log_lde_height = log2_strict_u8(lde_height);
76        let coset_points = bit_reversed_coset_points::<L::F>(log_lde_height);
77
78        let matrices_groups: Vec<Vec<&M>> =
79            trace_trees.iter().map(|tree| tree.leaves().iter().collect()).collect();
80
81        let quotient = PointQuotients::new(FieldArray::from(eval_points), &coset_points);
82        let batched_evals = info_span!("evaluate at OOD points")
83            .in_scope(|| quotient.batch_eval_lifted(&matrices_groups, &coset_points, log_blowup));
84
85        let (deep_poly, _evals) =
86            Self::from_evals::<L, M, N, Ch>(params, trace_trees, batched_evals, &quotient, channel);
87        deep_poly
88    }

90    /// Construct `Q(X)` from committed matrices and batched evaluations at N opening points.
91    ///
92    /// # Arguments
93    /// - `trace_trees`: Trace trees used to derive alignment and matrix groups. All trees must
94    ///   share the same alignment; mixed alignments are not supported.
95    /// - `batched_evals`: One row per matrix, each row holding `FieldArray<EF, N>` per column.
96    ///   Widths match the unpadded matrices; alignment padding is applied lazily during channel
97    ///   writes and Horner reduction.
98    /// - `quotient`: Precomputed `1/(zⱼ − xᵢ)` for all opening points zⱼ and domain points xᵢ.
99    ///
100    /// Returns the constructed `DeepPoly` and the (unaligned) `batched_evals` for test inspection.
101    ///
102    /// Columns are reduced with a random `α` using Horner
103    /// (`f_red(X) = Σᵢ α^{W−1−i} · fᵢ(X)`). This lets the verifier stream the
104    /// reduction as it reads opened rows, and ensures a cheating prover cannot satisfy
105    /// some constraints while failing others with non-negligible probability.
106    ///
107    /// Implementation note: we fuse a sign change into the per-column coefficient stream
108    /// so reduction and quotient assembly can share a single traversal over the domain.
109    pub fn from_evals<L, M, const N: usize, Ch>(
110        params: DeepParams,
111        trace_trees: &[&L::Tree<M>],
112        batched_evals: RowList<FieldArray<EF, N>>,
113        quotient: &PointQuotients<L::F, EF, N>,
114        channel: &mut Ch,
115    ) -> (Self, RowList<FieldArray<EF, N>>)
116    where
117        L: Lmcs,
118        L::F: TwoAdicField,
119        EF: ExtensionField<L::F>,
120        M: Matrix<L::F>,
121        Ch: ProverChannel<F = L::F, Commitment = L::Commitment>,
122    {
123        // The alignment of the trees defines the number of virtual zero-values columns were
124        // inserted while hashing the rows of the matrices. The prover pads the opened rows of each
125        // matrix with zeros, so that the length of the row is a multiple of the alignment.
126        // The alignment is tied to the underlying cryptographic permutation's rate.
127        let alignment =
128            trace_trees.first().expect("at least one tree must be provided").alignment();
129        assert!(
130            trace_trees.iter().all(|tree| tree.alignment() == alignment),
131            "mixed trace tree alignments are not supported"
132        );
133
134        // Collect the LDE matrices from each committed tree, grouped by commitment.
135        // matrices_groups[group_idx][matrix_idx] is a reference to the LDE matrix
136        // whose rows are bit-reversed coset evaluations at height `lde_height`.
137        let matrices_groups: Vec<Vec<&M>> =
138            trace_trees.iter().map(|tree| tree.leaves().iter().collect()).collect();
139
140        // 1. Bind the prover's OOD evaluation claims into the Fiat-Shamir transcript. The DEEP
141        //    challenges (alpha, beta) are derived after this, so a cheating prover cannot adapt its
142        //    claims to the challenges. Each matrix row is zero-padded to the tree alignment,
143        //    matching the virtual zero columns the LMCS inserts when hashing rows. All matrices are
144        //    concatenated into a single flat slice per eval point.
145        for point_idx in 0..N {
146            let flat: Vec<EF> =
147                batched_evals.iter_aligned(alignment).map(|fa| fa[point_idx]).collect();
148            channel.send_algebra_slice(&flat);
149        }
150
151        // 2. Grind for proof-of-work witness
152        let _pow_witness = info_span!("DEEP grind", bits = params.deep_pow_bits)
153            .in_scope(|| channel.grind(params.deep_pow_bits));
154
155        // 3. Sample DEEP challenges
156        let challenge_columns: EF = channel.sample_algebra_element();
157        let challenge_points: EF = channel.sample_algebra_element();
158
159        // Pre-compute f_reduced(zⱼ) for all N points using Horner.
160        // Reduces across all matrices' aligned columns in flat order.
161        let f_reduced_at_points: FieldArray<EF, N> =
162            horner(challenge_columns, batched_evals.iter_aligned(alignment));
163
164        let w = <L::F as Field>::Packing::WIDTH;
165        let point_quotient = &quotient.point_quotient;
166        let n = point_quotient.len();
167
168        let group_sizes: Vec<usize> = matrices_groups.iter().map(Vec::len).collect();
169        let widths: Vec<usize> =
170            matrices_groups.iter().flat_map(|g| g.iter().map(|m| m.width())).collect();
171
172        // Align each matrix width so padding is explicit in the transcript.
173        let aligned_widths = aligned_widths(widths, alignment);
174
175        // Compute explicit coefficients for -f_reduced(X) = -Σᵢ α^{W−1−i}·fᵢ(X).
176        //
177        // The verifier computes f_reduced via `horner(alpha, columns)`, which assigns
178        // the highest power to the first column: column 0 gets α^{W−1}, column W-1
179        // gets α⁰. To match this with an explicit dot-product (needed for the LDE
180        // evaluation), we need coefficient[i] = −α^{W−1−i}.
181        //
182        // Construction:
183        //   shifted_powers(NEG_ONE) produces [−1, −α, −α², …, −α^{W−1}]
184        //   .rev() reverses to [−α^{W−1}, …, −α, −1]
185        //   Split into per-matrix chunks in commitment order.
186        //
187        // The negation is folded into the coefficients so the DEEP quotient loop can
188        // compute f_reduced(zⱼ) + neg_f_reduced(X) = f_reduced(zⱼ) − f_reduced(X)
189        // without a separate negation pass.
190        let total_width: usize = aligned_widths.iter().sum();
191        let mut neg_powers_iter = challenge_columns
192            .shifted_powers(EF::NEG_ONE)
193            .collect_n(total_width)
194            .into_iter()
195            .rev();
196        let neg_column_coeffs: Vec<Vec<EF>> = aligned_widths
197            .iter()
198            .map(|&width| neg_powers_iter.by_ref().take(width).collect())
199            .collect();
200
201        // Compute -f_reduced(X) = -Σᵢ α^{W−1−i}·fᵢ(X) over the LDE domain, then
202        // transform in-place into the DEEP quotient:
203        //   Q(X) = Σⱼ βʲ·(f_reduced(zⱼ) − f_reduced(X))·1/(zⱼ − X)
204        //
205        // The column reduction and quotient assembly are fused: the `neg_f_reduced`
206        // vector is consumed in-place to produce `deep_evals`, avoiding a separate
207        // full-domain allocation and improving cache locality.
208
209        let deep_evals = info_span!("DEEP reduce + assemble").in_scope(|| {
210            let mut neg_column_coeffs_iter = neg_column_coeffs.iter();
211            let mut neg_f_reduced = zip(matrices_groups.iter(), &group_sizes)
212                .map(|(matrices_group, &size)| {
213                    let group_coeffs: Vec<&Vec<EF>> =
214                        neg_column_coeffs_iter.by_ref().take(size).collect();
215                    accumulate_matrices(matrices_group, &group_coeffs)
216                })
217                .reduce(|mut acc, next| {
218                    debug_assert_eq!(acc.len(), next.len());
219                    acc.par_chunks_mut(w).zip(next.par_chunks(w)).for_each(
220                        |(acc_chunk, next_chunk)| {
221                            EF::add_slices(acc_chunk, next_chunk);
222                        },
223                    );
224                    acc
225                })
226                .unwrap_or_else(|| EF::zero_vec(n));
227
228            // Pre-compute βʲ for all N points
229            let point_coeffs: [EF; N] =
230                core::array::from_fn(|j| challenge_points.exp_u64(j as u64));
231
232            // Transform neg_f_reduced in-place into deep_evals.
233            // Q(x) = Σⱼ βʲ·qⱼ(x)·(f_reduced(zⱼ) + neg_f_reduced(x))
234            if w == 1 || n < w {
235                neg_f_reduced
236                    .par_iter_mut()
237                    .zip(point_quotient.par_iter())
238                    .for_each(|(neg, q)| {
239                        let mut result = q[0] * (f_reduced_at_points[0] + *neg);
240                        for j in 1..N {
241                            result += point_coeffs[j] * q[j] * (f_reduced_at_points[j] + *neg);
242                        }
243                        *neg = result;
244                    });
245            } else {
246                let f_reduced_packed: [EF::ExtensionPacking; N] =
247                    f_reduced_at_points.0.map(EF::ExtensionPacking::from);
248                let point_coeffs_packed: [EF::ExtensionPacking; N] =
249                    point_coeffs.map(EF::ExtensionPacking::from);
250
251                neg_f_reduced
252                    .par_chunks_exact_mut(w)
253                    .zip(point_quotient.par_chunks_exact(w))
254                    .for_each(|(neg_chunk, q_chunk)| {
255                        let neg_p = EF::ExtensionPacking::from_ext_slice(neg_chunk);
256
257                        // Transpose quotients: q_chunk[lane][point] -> q_packed[point] packs all
258                        // lanes
259                        let q_packed: [EF::ExtensionPacking; N] =
260                            EF::ExtensionPacking::pack_ext_columns(FieldArray::as_raw_slice(
261                                q_chunk,
262                            ));
263
264                        // First point (j=0) has coefficient β⁰ = 1, compute directly
265                        let mut result_p = q_packed[0] * (f_reduced_packed[0] + neg_p);
266
267                        // Remaining points (j>0) multiply by βʲ
268                        for j in 1..N {
269                            result_p += point_coeffs_packed[j]
270                                * q_packed[j]
271                                * (f_reduced_packed[j] + neg_p);
272                        }
273                        result_p.to_ext_slice(neg_chunk);
274                    });
275            }
276
277            neg_f_reduced // now contains deep_evals
278        });
279
280        (Self { deep_evals }, batched_evals)
281    }
282}

284/// Accumulate `f_reduced(X) = Σᵢ α^{W−1−i}·fᵢ(X)` across matrices of varying heights.
285///
286/// In bit-reversed order, the lifted polynomial `f(Xʳ)` repeats each evaluation r times
287/// (because adjacent bit-reversed indices differ only in their low bits, mapping to points
288/// that are r-th roots of the same value). This lets us upsample by repetition instead of
289/// recomputing the lifted polynomial: when crossing a height boundary, repeat entries to
290/// match the new height, then continue accumulating. Matrices must be sorted by ascending
291/// height.
292fn accumulate_matrices<F: Field, EF: ExtensionField<F>, M: Matrix<F>, C: AsRef<[EF]>>(
293    matrices: &[&M],
294    coeffs: &[C],
295) -> Vec<EF> {
296    debug_assert!(
297        matrices.windows(2).all(|w| w[0].height() <= w[1].height()),
298        "matrices must be sorted by ascending height"
299    );
300    let n = matrices.last().unwrap().height();
301
302    let mut acc = EF::zero_vec(n);
303    let mut scratch = EF::zero_vec(n);
304
305    let mut active_height = matrices.first().unwrap().height();
306
307    for (&matrix, coeffs) in zip(matrices, coeffs) {
308        let coeffs = coeffs.as_ref();
309        let height = matrix.height();
310        debug_assert!(height.is_power_of_two(), "matrix height must be a power of two");
311        debug_assert!(
312            matrix.width() <= coeffs.len(),
313            "matrix width {} exceeds coeffs length {}",
314            matrix.width(),
315            coeffs.len()
316        );
317
318        // Upsample: [a, b] → [a, a, b, b] when height doubles
319        if height > active_height {
320            let scaling_factor = height / active_height;
321            scratch[..height]
322                .par_chunks_mut(scaling_factor)
323                .zip(acc[..active_height].par_iter())
324                .for_each(|(chunk, &val)| chunk.fill(val));
325            acc[..height].swap_with_slice(&mut scratch[..height]);
326        }
327
328        // SIMD path using horizontal packing.
329        // Slice to matrix width to avoid packing alignment-padding coefficients.
330        let w = F::Packing::WIDTH;
331        let active_coeffs = &coeffs[..matrix.width()];
332        let packed_coeffs: Vec<EF::ExtensionPacking> = active_coeffs
333            .chunks(w)
334            .map(|chunk| {
335                if chunk.len() == w {
336                    EF::ExtensionPacking::from_ext_slice(chunk)
337                } else {
338                    // Pad with zeros for the last chunk
339                    let mut padded = EF::zero_vec(w);
340                    padded[..chunk.len()].copy_from_slice(chunk);
341                    EF::ExtensionPacking::from_ext_slice(&padded)
342                }
343            })
344            .collect();
345
346        matrix
347            .rowwise_packed_dot_product::<EF>(&packed_coeffs)
348            .zip(acc[..height].par_iter_mut())
349            .for_each(|(dot_result, acc_val)| {
350                *acc_val += dot_result;
351            });
352
353        active_height = height;
354    }
355
356    acc
357}

368    /// `reduce_with_powers` (Horner) must match explicit negative coeffs + dot product.
369    #[test]
370    fn neg_coeffs_match_neg_horner() {
371        let c: QuadFelt = QuadFelt::from_u64(2);
372        let alignment = 3;
373        let widths = [2usize, 3];
374        let aligned_widths = aligned_widths(widths.to_vec(), alignment);
375        let rows: Vec<Vec<Felt>> = vec![
376            vec![Felt::from_u64(1), Felt::from_u64(2)],
377            vec![Felt::from_u64(3), Felt::from_u64(4), Felt::from_u64(5)],
378        ];
379        let padded = RowList::from_rows_aligned(&rows, alignment);
380
381        let mut neg_powers_iter = c
382            .shifted_powers(QuadFelt::NEG_ONE)
383            .collect_n(aligned_widths.iter().sum())
384            .into_iter()
385            .rev();
386        let neg_coeffs: Vec<Vec<QuadFelt>> = aligned_widths
387            .iter()
388            .map(|&width| neg_powers_iter.by_ref().take(width).collect())
389            .collect();
390
391        // Explicit coefficient sum: Σᵢ coeffs[i] · rows[i]
392        let explicit: QuadFelt =
393            dot_product(neg_coeffs.iter().flatten().copied(), padded.iter_values());
394
395        // Horner using reduce_with_powers (same as used in verifier)
396        let horner: QuadFelt = horner(c, padded.iter_values());
397
398        assert_eq!(explicit, QuadFelt::NEG_ONE * horner);
399    }

401    /// Padding: negative coeffs match -Horner for various width/alignment combos.
402    #[test]
403    fn neg_coeffs_alignment() {
404        let c: QuadFelt = QuadFelt::from_u64(7);
405        let alignment = 4;
406        let widths = [3usize, 5, 2];
407        let aligned_widths = aligned_widths(widths.to_vec(), alignment);

409        let mut neg_powers_iter = c
410            .shifted_powers(QuadFelt::NEG_ONE)
411            .collect_n(aligned_widths.iter().sum())
412            .into_iter()
413            .rev();
414        let coeffs: Vec<Vec<QuadFelt>> = aligned_widths
415            .iter()
416            .map(|&width| neg_powers_iter.by_ref().take(width).collect())
417            .collect();
418
419        // Verify lengths match aligned widths
420        assert_eq!(coeffs[0].len(), aligned_widths[0]);
421        assert_eq!(coeffs[1].len(), aligned_widths[1]);
422        assert_eq!(coeffs[2].len(), aligned_widths[2]);
423
424        // Verify this matches Horner reduction with arbitrary test data
425        let rows: Vec<Vec<Felt>> = vec![
426            vec![Felt::from_u64(10), Felt::from_u64(20), Felt::from_u64(30)],
427            vec![
428                Felt::from_u64(1),
429                Felt::from_u64(2),
430                Felt::from_u64(3),
431                Felt::from_u64(4),
432                Felt::from_u64(5),
433            ],
434            vec![Felt::from_u64(100), Felt::from_u64(200)],
435        ];
436        let padded = RowList::from_rows_aligned(&rows, alignment);
437
438        let explicit: QuadFelt =
439            dot_product(coeffs.iter().flatten().copied(), padded.iter_values());
440        let horner: QuadFelt = horner(c, padded.iter_values());

443    }
