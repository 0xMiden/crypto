19/// Verifier's view of the DEEP quotient as a point-query oracle.
20///
21/// The prover claims OOD evaluations for all committed columns at a small set of points
22/// `zⱼ`. The verifier uses a random `α` to reduce (batch) all columns into a single
23/// polynomial `f_red`, and a random `β` to combine multiple opening points into one
24/// DEEP quotient polynomial:
25///
26/// ```text
27/// Q(X) = Σⱼ βʲ · (f_red(zⱼ) − f_red(X)) / (zⱼ − X)
28/// ```
29///
30/// This oracle stores the commitments and the reduced OOD claims `(zⱼ, f_red(zⱼ))`.
31/// At query time it:
32/// - verifies Merkle openings for all committed matrices at the query index,
33/// - reduces the opened row to `f_red(X)` using Horner with the same `α`,
34/// - reconstructs `Q(X)` and returns it to the FRI verifier.
35///
36/// Full-height groups live on the max domain. Shorter groups are interpreted by
37/// folding query indices to their committed depth.
38pub(in crate::pcs) struct DeepOracle<F: TwoAdicField, EF: ExtensionField<F>, L: Lmcs<F = F>> {
39    /// Committed groups (root + widths + tree depth), one per tree.
40    ///
41    /// Widths must match the committed rows (including any alignment padding if
42    /// `build_aligned_tree` was used). A group's `log_height` may be below the
43    /// query depth — a virtually-lifted, setup-fixed preprocessed tree.
44    commitments: Vec<CommitmentGroup<L::Commitment>>,
45
46    /// Max LDE coset; query indices are sampled at `domain.log_lde_height()` and
47    /// folded down to each group's committed depth when shorter.
48    domain: LiftedDomain<F>,
49
50    /// Reduced openings: pairs of `(zⱼ, f_reduced(zⱼ))` from the prover's claims.
51    reduced_openings: Vec<(EF, EF)>,
52
53    /// Challenge `α` for batching columns into `f_reduced`.
54    challenge_columns: EF,
55    /// Challenge `β` for batching opening points.
56    challenge_points: EF,
57
58    _marker: PhantomData<F>,
59}

61impl<F: TwoAdicField, EF: ExtensionField<F>, L: Lmcs<F = F>> DeepOracle<F, EF, L> {
62    /// Construct by reading evaluations, checking PoW, and sampling challenges.
63    ///
64    /// Commitment widths must match the committed rows (including any alignment padding).
65    /// Each group's `log_height` must be `≤ domain.log_lde_height()`; shorter groups
66    /// are virtually lifted at query time.
67    ///
68    /// Preconditions: `eval_points` must be distinct and lie outside the trace subgroup `H`
69    /// and LDE evaluation coset `gK`. The outer protocol is expected to enforce this.
70    ///
71    /// Returns the oracle and per-matrix evaluations: `evals[g][m]` is a
72    /// `RowMajorMatrix<EF>` with one row per evaluation point.
73    pub fn new<Ch>(
74        params: DeepParams,
75        eval_points: &[EF],
76        commitments: Vec<CommitmentGroup<L::Commitment>>,
77        domain: &LiftedDomain<F>,
78        channel: &mut Ch,
79    ) -> Result<(Self, OpenedValues<EF>), DeepError>
80    where
81        Ch: VerifierChannel<F = F, Commitment = L::Commitment>,
82    {
83        let group_widths: Vec<&[usize]> = commitments.iter().map(|g| g.widths.as_slice()).collect();
84        let evals = read_eval_matrices::<F, EF, Ch>(&group_widths, eval_points.len(), channel)?;
85
86        // 1. Check grinding witness
87        channel.grind(params.deep_pow_bits)?;
88
89        // 2. Sample DEEP challenges
90        let challenge_columns: EF = channel.sample_algebra_element();
91        let challenge_points: EF = channel.sample_algebra_element();
92
93        // Horner reduction: fold across all evals for each evaluation point
94        let reduced_openings: Vec<(EF, EF)> = eval_points
95            .iter()
96            .enumerate()
97            .map(|(p, &point)| {
98                let val = evals.iter().flat_map(|g| g.iter()).fold(EF::ZERO, |acc, mat| {
99                    // mat has num_eval_points rows (one per z), p < num_eval_points.
100                    horner_acc(
101                        acc,
102                        challenge_columns,
103                        mat.row(p).expect("eval point index in range"),
104                    )
105                });
106                (point, val)
107            })
108            .collect();
109
110        let oracle = Self {
111            commitments,
112            domain: *domain,
113            reduced_openings,
114            challenge_columns,
115            challenge_points,
116            _marker: PhantomData,
117        };
118
119        Ok((oracle, evals))
120    }
121
122    /// Open the oracle at given tree indices by reading proofs from a verifier channel.
123    ///
124    /// `tree_indices` are domain indices (sorted, deduplicated).
125    /// Returns a map from domain index to DEEP evaluation at that point.
126    ///
127    /// The reduction to `f_red` must match the prover's exactly.
128    ///
129    /// In particular, the prover streams columns in a fixed commitment-group order
130    /// (e.g. main, aux, quotient). The verifier must iterate groups in the same order so
131    /// that `horner_acc` assigns the same `α` powers to the same columns; otherwise the
132    /// reconstructed `Q(X)` will not match the FRI-committed codeword.
133    pub fn open_batch<Ch>(
134        &self,
135        lmcs: &L,
136        tree_indices: &TreeIndices,
137        channel: &mut Ch,
138    ) -> Result<BTreeMap<usize, EF>, DeepError>
139    where
140        Ch: VerifierChannel<F = F, Commitment = L::Commitment>,
141    {
142        let mut reduced_rows: BTreeMap<usize, EF> =
143            tree_indices.iter().map(|&idx| (idx, EF::ZERO)).collect();
144
145        for (group_idx, group) in self.commitments.iter().enumerate() {
146            // `open_lifted_batch` returns rows keyed by the original query indices, even when
147            // this group is committed at a shorter depth, so the reduction is uniform.
148            let opened_rows = lmcs
149                .open_lifted_batch(
150                    &group.root,
151                    &group.widths,
152                    tree_indices,
153                    group.log_height,
154                    channel,
155                )
156                .map_err(|source| DeepError::LmcsError { source, tree: group_idx })?;
157
158            // Reduce opened rows via Horner: f_reduced(X) = Σᵢ αᵂ⁻¹⁻ⁱ · fᵢ(X).
159            //
160            // `horner_acc` continues the running accumulation across commitment groups:
161            // group 0's columns get the highest powers, group 1's continue from where
162            // group 0 left off. The coefficient ordering must match the prover's exactly;
163            // otherwise the reconstructed DEEP quotient diverges from the FRI-committed
164            // codeword, causing verification failure.
165            for (tree_idx, acc) in reduced_rows.iter_mut() {
166                let rows_for_query = opened_rows
167                    .get(tree_idx)
168                    .ok_or(DeepError::InvalidOpening { tree: group_idx, tree_index: *tree_idx })?;
169                *acc = horner_acc(*acc, self.challenge_columns, rows_for_query.iter_values());
170            }
171        }
172
173        // Reconstruct Q(x) at each queried domain point x from the opened row data.
174        // If the prover's OOD claims were correct, these values lie on the
175        // low-degree polynomial committed via FRI.
176        let evals: BTreeMap<usize, EF> = reduced_rows
177            .into_iter()
178            .map(|(tree_idx, reduced_row)| {
179                // Recover domain point X = g·ω^{tree_idx} (tree index = domain index)
180                let row_point = self.domain.lde_coset().point_at(tree_idx as u64);
181
182                // DEEP quotient: Q(X) = Σⱼ βʲ · (f_reduced(zⱼ) - f_reduced(X)) / (zⱼ - X)
183                // Precondition: eval points lie outside the LDE domain.
184                let mut deep_eval = EF::ZERO;
185                for ((point, reduced_eval), coeff_point) in
186                    zip(&self.reduced_openings, self.challenge_points.powers())
187                {
188                    let denom_inv = (*point - row_point)
189                        .try_inverse()
190                        .ok_or(DeepError::EvalPointOnDomain { tree_index: tree_idx })?;
191                    deep_eval += coeff_point * (*reduced_eval - reduced_row) * denom_inv;
192                }
193                Ok((tree_idx, deep_eval))
194            })
195            .collect::<Result<_, DeepError>>()?;
196
197        Ok(evals)
198    }
199}

205/// Errors that can occur during DEEP oracle construction or verification.
206#[derive(Debug, Error)]
207pub enum DeepError {
208    #[error("LMCS verification failed for commitment group {tree}: {source}")]
209    LmcsError { source: LmcsError, tree: usize },
210    #[error("invalid opening for tree index {tree_index} in commitment group {tree}")]
211    InvalidOpening { tree: usize, tree_index: usize },
212    #[error("evaluation point coincides with domain point at tree index {tree_index}")]
213    EvalPointOnDomain { tree_index: usize },
214    #[error("transcript error: {0}")]
215    TranscriptError(#[from] TranscriptError),
216}
