15/// Open committed matrices at N evaluation points, writing to a prover channel.
16///
17/// # Preconditions
18/// - `eval_points` must lie outside both the trace-domain subgroup `H` and the LDE evaluation coset
19///   `gK` used by the PCS. If a point lies in either set, denominators `(zⱼ − X)` in the DEEP
20///   quotient become zero for some domain element, making the quotient undefined.
21/// - All trace trees must be built at the same LDE height `2^log_lde_height`. Multiple LDE heights
22///   are not supported yet and will panic.
23///
24/// `log_lde_height` is the log₂ of the LDE evaluation domain height (i.e. the height of
25/// the committed LDE matrices). When a trace degree is known, it is typically
26/// `log_trace_height + params.fri.log_blowup` (plus any extension used by the caller).
27/// In that common case, the trace subgroup `H` has size `2^(log_lde_height -
28/// params.fri.log_blowup)`, while the LDE coset `gK` has size `2^log_lde_height`.
29///
30/// Alignment is derived from the trace trees to pad DEEP evaluations consistently.
31/// Trace trees must be built with `build_aligned_tree` to match this padding.
32#[instrument(name = "PCS opening", skip_all)]
33pub fn open_with_channel<F, EF, L, M, Ch, const N: usize>(
34    params: &PcsParams,
35    lmcs: &L,
36    log_lde_height: u8,
37    eval_points: [EF; N],
38    trace_trees: &[&L::Tree<M>],
39    channel: &mut Ch,
40) where
41    F: TwoAdicField,
42    EF: ExtensionField<F>,
43    L: Lmcs<F = F>,
44    M: Matrix<F>,
45    Ch: ProverChannel<F = F, Commitment = L::Commitment>,
46{
47    const { assert!(N > 0, "at least one evaluation point required") };
48
49    // Determine LDE domain size from the supplied LDE height.
50    // For now, all trace trees must share this height; mixed LDE heights are not supported yet.
51    assert!(!trace_trees.is_empty(), "at least one trace tree required");
52    let expected_height = 1 << log_lde_height as usize;
53    assert!(
54        trace_trees.iter().all(|tree| tree.height() == expected_height),
55        "mixed LDE heights are not supported yet",
56    );
57    // ─────────────────────────────────────────────────────────────────────────
58    // Construct DEEP quotient (observes evals, grinds, samples alpha and beta)
59    // ─────────────────────────────────────────────────────────────────────────
60    let deep_poly = info_span!("DEEP quotient").in_scope(|| {
61        DeepPoly::from_trees::<L, M, N, Ch>(
62            params.deep,
63            trace_trees,
64            eval_points,
65            params.fri.log_blowup,
66            channel,
67        )
68    });
69
70    // ─────────────────────────────────────────────────────────────────────────
71    // FRI commit phase (observes commitments, grinds per-round, samples betas)
72    // ─────────────────────────────────────────────────────────────────────────
73    // The deep_poly contains evaluations on the LDE domain (size 2^log_lde_height).
74    // FRI will prove that this polynomial is low-degree.
75    let fri_polys = info_span!("FRI commit phase")
76        .in_scope(|| FriPolys::<F, EF, L>::new(&params.fri, lmcs, deep_poly.deep_evals, channel));
77
78    // ─────────────────────────────────────────────────────────────────────────
79    // Grind for query sampling
80    // ─────────────────────────────────────────────────────────────────────────
81    let _query_pow_witness = info_span!("query grind", bits = params.query_pow_bits())
82        .in_scope(|| channel.grind(params.query_pow_bits()));
83
84    // ─────────────────────────────────────────────────────────────────────────
85    // Sample query indices (domain indices)
86    // ─────────────────────────────────────────────────────────────────────────
87    // Sampled indices are domain indices: domain point = g·ω^{index}.
88    // The LMCS tree is indexed by domain order (no bit-reversal needed).
89    let sampled_indices_iter =
90        (0..params.num_queries()).map(|_| channel.sample_bits(log_lde_height as usize));
91    let tree_indices = TreeIndices::new(sampled_indices_iter, log_lde_height)
92        .expect("sampled indices are in range");
93
94    // ─────────────────────────────────────────────────────────────────────────
95    // Generate query proofs
96    // ─────────────────────────────────────────────────────────────────────────
97    info_span!("query phase").in_scope(|| {
98        // Open input trees at all query indices at once (one proof per tree)
99        info_span!("open input trees", n_trees = trace_trees.len()).in_scope(|| {
100            for tree in trace_trees {
101                tree.prove_batch(&tree_indices, channel);
102            }
103        });
104
105        // Open all FRI rounds at all query indices at once (one proof per round)
106        info_span!("open FRI trees").in_scope(|| {
107            fri_polys.prove_queries(&params.fri, tree_indices, channel);
108        });
109    });
110}
