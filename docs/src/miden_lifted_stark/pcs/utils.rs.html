17/// Horner fold with an explicit accumulator.
18///
19/// Computes `acc·xⁿ + v₀·xⁿ⁻¹ + v₁·xⁿ⁻² + ... + vₙ₋₁·x⁰` where n = len(vals).
20/// Equivalently: `((acc·x + v₀)·x + v₁)·x + ... + vₙ₋₁`.
21/// The first element gets the highest power of `x`.
22///
23/// For polynomial evaluation `p(x) = Σᵢ cᵢ·xⁱ`, pass coefficients in
24/// descending degree order `[cₙ, ..., c₁, c₀]`.
25#[inline]
26pub fn horner_acc<Acc, Val, X, I>(acc: Acc, x: X, vals: I) -> Acc
27where
28    I: IntoIterator<Item = Val>,
29    Acc: Mul<X, Output = Acc> + Add<Val, Output = Acc>,
30    X: Clone,
31{
32    vals.into_iter().fold(acc, |acc, val| acc * x.clone() + val)
33}
34
35/// Horner fold starting from zero.
36///
37/// See [`horner_acc`] for the evaluation convention.
38#[inline]
39pub fn horner<Acc, Val, X, I>(x: X, vals: I) -> Acc
40where
41    I: IntoIterator<Item = Val>,
42    Acc: Default + Mul<X, Output = Acc> + Add<Val, Output = Acc>,
43    X: Clone,
44{
45    horner_acc(Acc::default(), x, vals)
46}

48/// Extension trait adding `pack_ext_columns` and `to_ext_slice` methods.
49///
50/// These methods enable efficient SIMD operations on arrays of extension field elements
51/// by providing column-wise packing and unpacking utilities.
52pub trait PackedFieldExtensionExt<
53    BaseField: Field,
54    ExtField: ExtensionField<BaseField, ExtensionPacking = Self>,
55>: PackedFieldExtension<BaseField, ExtField>
56{
57    /// Pack N columns from WIDTH rows into N packed extension field elements.
58    ///
59    /// Input: `rows[lane][col]` - WIDTH rows, each with N extension field elements.
60    /// Output: `result[col]` - N packed values, where each packs WIDTH lanes.
61    fn pack_ext_columns<const N: usize>(rows: &[[ExtField; N]]) -> [Self; N] {
62        let width = BaseField::Packing::WIDTH;
63        debug_assert_eq!(rows.len(), width);
64        array::from_fn(|col| {
65            let col_elems: Vec<ExtField> = (0..width).map(|lane| rows[lane][col]).collect();
66            Self::from_ext_slice(&col_elems)
67        })
68    }
69
70    /// Extract all lanes to an output slice.
71    fn to_ext_slice(&self, out: &mut [ExtField]) {
72        let width = BaseField::Packing::WIDTH;
73        for (lane, slot) in out.iter_mut().enumerate().take(width) {
74            *slot = self.extract(lane);
75        }
76    }
77}

82    P: PackedFieldExtension<BaseField, ExtField>,

84{
85}

87/// Coset points `gK` in bit-reversed order.
88///
89/// Note: the coset shift `g` is fixed to `F::GENERATOR` by convention in this PCS.
90///
91/// Bit-reversal gives two properties essential for lifting:
92/// - **Adjacent negation**: `gK[2i+1] = -gK[2i]`, so both square to the same value
93/// - **Squaring gives prefix**: `(gK[2i])² = (gK)²[i]` — the even-indexed elements, when squared,
94///   form the half-size sub-coset. Generalizes to r-th powers.
95///
96/// Together these enable iterative weight folding in barycentric evaluation.
97///
98/// # Panics
99/// Panics if the two-adic coset construction fails (e.g., `log_n` exceeds the field's
100/// two-adicity), since this unwraps `TwoAdicMultiplicativeCoset::new`.
101pub fn bit_reversed_coset_points<F: TwoAdicField>(log_n: u8) -> Vec<F> {
102    let coset = TwoAdicMultiplicativeCoset::new(F::GENERATOR, log_n as usize).unwrap();
103    let mut pts: Vec<F> = coset.iter().collect();
104    reverse_slice_index_bits(&mut pts);
105    pts
106}
