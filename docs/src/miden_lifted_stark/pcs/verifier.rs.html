1//! PCS Verifier
2//!
3//! Verifies polynomial evaluation claims against commitments.
4//!
5//! Two entry points with the same signature, differing only in alignment handling:
6//!
7//! | Function          | Alignment |
8//! |-------------------|-----------|

35/// Verify polynomial evaluation claims against commitments.
36///
37/// Commitment widths must match the committed rows (including any alignment padding
38/// from `build_aligned_tree`). The PCS is alignment-agnostic; callers that use
39/// aligned trees must pass aligned widths and handle truncation themselves.
40/// See [`verify_aligned`] for automatic alignment handling.
41///
42/// Does **not** check that the channel is fully consumed after verification.
43/// Callers should use
44/// [`VerifierTranscript::finalize`](miden_stark_transcript::VerifierTranscript::finalize) to
45/// enforce transcript exhaustion.
46///
47/// # Preconditions
48/// - `eval_points` must lie outside both the trace-domain subgroup `H` and the LDE evaluation coset
49///   `gK`. Otherwise denominators `(zⱼ − X)` in the DEEP quotient become zero, making it undefined.
50/// - All commitments must be lifted to the same LDE height `2^log_lde_height`.
51///
52/// # Returns
53/// `opened[group][matrix]` as a `RowMajorMatrix<EF>` with `N` rows
54/// (one per evaluation point), using the same widths that were passed in.
55pub fn verify<F, EF, L, Ch, const N: usize>(
56    params: &PcsParams,
57    lmcs: &L,
58    commitments: &[(L::Commitment, Vec<usize>)],
59    log_lde_height: u8,
60    eval_points: [EF; N],
61    channel: &mut Ch,
62) -> Result<OpenedValues<EF>, PcsError>
63where
64    F: TwoAdicField,
65    EF: ExtensionField<F> + PartialEq + Clone,
66    L: Lmcs<F = F>,
67    Ch: VerifierChannel<F = F, Commitment = L::Commitment>,
68{
69    const { assert!(N > 0, "at least one evaluation point required") };
70
71    if commitments.is_empty() {
72        return Err(PcsError::NoCommitments);
73    }
74
75    // Construct verifier's DEEP oracle (observes evals, checks PoW, samples α/β)
76    let (deep_oracle, evals) = DeepOracle::<F, EF, L>::new(
77        params.deep,
78        &eval_points,
79        commitments.to_vec(),
80        log_lde_height,
81        channel,
82    )?;
83
84    // Create FRI oracle (observes commitments + final poly, checks per-round PoW)
85    let fri_oracle = FriOracle::new(&params.fri, log_lde_height, channel)?;
86
87    // Check query PoW witness and sample query indices
88    channel.grind(params.query_pow_bits())?;
89
90    // Sample query indices (domain indices). The LMCS tree is indexed by domain order.
91    let sampled_indices_iter =
92        (0..params.num_queries()).map(|_| channel.sample_bits(log_lde_height as usize));
93    let tree_indices = TreeIndices::new(sampled_indices_iter, log_lde_height)
94        .expect("sampled indices are in range");
95
96    // Verify DEEP openings for all queries at once
97    // tree_indices are bit-reversed positions; deep_evals is keyed by tree index
98    let deep_evals = deep_oracle.open_batch(lmcs, &tree_indices, channel)?;
99
100    // Test low-degree proximity for all queries at once
101    fri_oracle.test_low_degree(lmcs, &params.fri, deep_evals, tree_indices, channel)?;
102
103    Ok(evals)
104}

106/// Like [`verify`], but handles LMCS alignment automatically.
107///
108/// Commitment widths should be the original (unpadded) data widths. This function:
109/// 1. Aligns widths to `lmcs.alignment()`
110/// 2. Calls [`verify`] with aligned widths
111/// 3. Truncates returned evals back to original widths
112pub fn verify_aligned<F, EF, L, Ch, const N: usize>(
113    params: &PcsParams,
114    lmcs: &L,
115    commitments: &[(L::Commitment, Vec<usize>)],
116    log_lde_height: u8,
117    eval_points: [EF; N],
118    channel: &mut Ch,
119) -> Result<OpenedValues<EF>, PcsError>
120where
121    F: TwoAdicField,
122    EF: ExtensionField<F> + PartialEq + Clone,
123    L: Lmcs<F = F>,
124    Ch: VerifierChannel<F = F, Commitment = L::Commitment>,
125{
126    let alignment = lmcs.alignment();
127    let aligned_commitments: Vec<_> = commitments
128        .iter()
129        .map(|(c, widths)| (c.clone(), aligned_widths(widths.clone(), alignment)))
130        .collect();
131
132    let evals = verify(params, lmcs, &aligned_commitments, log_lde_height, eval_points, channel)?;
133
134    // Truncate each matrix back to original widths, removing alignment padding.
135    let truncated = evals
136        .into_iter()
137        .zip(commitments)
138        .map(|(group, (_, orig_widths))| {
139            group
140                .into_iter()
141                .zip(orig_widths)
142                .map(|(mat, &orig_w)| {
143                    HorizontallyTruncated::new(mat, orig_w)
144                        .expect("original width must not exceed aligned width")
145                        .to_row_major_matrix()
146                })
147                .collect()
148        })
149        .collect();
150
151    Ok(truncated)
152}

158/// Errors that can occur during PCS verification.
159#[derive(Debug, Error)]
160pub enum PcsError {
161    #[error("no commitments provided")]
162    NoCommitments,
163    #[error("DEEP error: {0}")]
164    DeepError(#[from] DeepError),
165    #[error("FRI error: {0}")]
166    FriError(#[from] FriError),
167    #[error("transcript error: {0}")]
168    TranscriptError(#[from] TranscriptError),
169}
