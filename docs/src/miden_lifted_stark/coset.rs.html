15// ============================================================================
16// LiftedCoset
17// ============================================================================
18
19/// Lifted coset for polynomial evaluation.
20///
21/// Represents a coset (gK)ʳ where:
22/// - K is the evaluation domain of size 2^log_lde_height
23/// - r = 2^log_lift_ratio is the lift factor (row repetition)
24/// - The shift is gʳ where g = F::GENERATOR
25///
26/// Key relationships:
27/// - log_blowup = log_lde_height - log_trace_height
28/// - log_lift_ratio = log_max_lde_height - log_lde_height
29/// - lde_shift = gʳ = F::GENERATOR.exp_power_of_2(log_lift_ratio)
30///
31/// # Invariants
32///
33/// - `log_lde_height = log_trace_height + log_blowup`
34/// - `log_lde_height <= log_max_lde_height`
35/// - All heights are powers of two (stored as log values)
36#[derive(Clone, Copy, Debug, PartialEq, Eq)]
37pub struct LiftedCoset {
38    /// Log₂ of the original trace height.
39    pub log_trace_height: u8,
40    /// Log₂ of this matrix's LDE height.
41    pub log_lde_height: u8,
42    /// Log₂ of the maximum LDE height in the commitment.
43    pub log_max_lde_height: u8,
44}
45
46impl LiftedCoset {
47    /// Create a new `LiftedCoset`.
48    ///
49    /// Both `log_lde_height` and `log_max_lde_height` are derived by adding
50    /// `log_blowup` to the respective trace heights.
51    ///
52    /// # Panics
53    ///
54    /// Panics if `log_trace_height > log_max_trace_height` or if
55    /// `log_max_trace_height + log_blowup > MAX_LOG_DOMAIN_SIZE`.
56    #[inline]
57    pub fn new(log_trace_height: u8, log_blowup: u8, log_max_trace_height: u8) -> Self {
58        assert!(
59            log_trace_height <= log_max_trace_height,
60            "trace height cannot exceed max trace height"
61        );
62        let log_lde_height = log_trace_height as u16 + log_blowup as u16;
63        let log_max_lde_height = log_max_trace_height as u16 + log_blowup as u16;
64        assert!(
65            log_max_lde_height <= MAX_LOG_DOMAIN_SIZE as u16,
66            "LDE height 2^{log_max_lde_height} exceeds maximum 2^{MAX_LOG_DOMAIN_SIZE}",
67        );
68        Self {
69            log_trace_height,
70            log_lde_height: log_lde_height as u8,
71            log_max_lde_height: log_max_lde_height as u8,
72        }
73    }
74
75    /// Create a `LiftedCoset` at max height (no lifting).
76    ///
77    /// Convenience for the common single-trace case where the LDE height
78    /// equals the max LDE height.
79    #[inline]
80    pub fn unlifted(log_trace_height: u8, log_blowup: u8) -> Self {
81        let log_lde_height = log_trace_height as u16 + log_blowup as u16;
82        assert!(
83            log_lde_height <= MAX_LOG_DOMAIN_SIZE as u16,
84            "LDE height 2^{log_lde_height} exceeds maximum 2^{MAX_LOG_DOMAIN_SIZE}",
85        );
86        Self {
87            log_trace_height,
88            log_lde_height: log_lde_height as u8,
89            log_max_lde_height: log_lde_height as u8,
90        }
91    }
92
93    // ============ Existing methods ============
94
95    /// Log₂ of the blowup factor for this matrix.
96    ///
97    /// Returns `log_lde_height - log_trace_height`.
98    #[inline]
99    pub fn log_blowup(&self) -> usize {
100        (self.log_lde_height - self.log_trace_height) as usize
101    }
102
103    /// Log₂ of the lift ratio for this matrix.
104    ///
105    /// The lift ratio is how many times this matrix's rows are virtually repeated
106    /// to match the max LDE height: `max_lde_height / lde_height`.
107    ///
108    /// Returns `log_max_lde_height - log_lde_height`.
109    #[inline]
110    pub fn log_lift_ratio(&self) -> usize {
111        (self.log_max_lde_height - self.log_lde_height) as usize
112    }
113
114    /// Whether this matrix is lifted (its LDE height is less than the max).
115    #[inline]
116    pub fn is_lifted(&self) -> bool {
117        self.log_lde_height < self.log_max_lde_height
118    }
119
120    /// Compute the coset shift for this matrix's LDE domain.
121    ///
122    /// For a matrix with lift ratio `r = 2^log_lift_ratio`, the coset shift is gʳ
123    /// where g is the field generator.
124    ///
125    /// Why gʳ: lifting embeds a smaller-domain polynomial into the max domain by
126    /// composition `p_lift(X) = p(Xʳ)`. Evaluating `p_lift` on the max coset `g·K_max`
127    /// corresponds to evaluating `p` on the nested coset `gʳ·K`, because
128    /// `(g·ω)ʳ = gʳ·ωʳ` and ωʳ ranges over K when ω ranges over `K_max`.
129    #[inline]
130    pub fn lde_shift<F: TwoAdicField>(&self) -> F {
131        F::GENERATOR.exp_power_of_2(self.log_lift_ratio())
132    }
133
134    /// The trace height (number of constraint rows).
135    #[inline]
136    pub fn trace_height(&self) -> usize {
137        1 << self.log_trace_height as usize
138    }
139
140    /// The LDE height for this matrix.
141    #[inline]
142    pub fn lde_height(&self) -> usize {
143        1 << self.log_lde_height as usize
144    }
145
146    /// The maximum LDE height across all matrices.
147    #[inline]
148    pub fn max_lde_height(&self) -> usize {
149        1 << self.log_max_lde_height as usize
150    }
151
152    /// The blowup factor for this matrix.
153    #[inline]
154    pub fn blowup(&self) -> usize {
155        1 << self.log_blowup()
156    }
157
158    // ============ Domain derivation ============
159
160    /// Derive the quotient domain coset from this LDE coset.
161    ///
162    /// For constraint evaluation, we need a coset of size `trace_height * constraint_degree`.
163    /// This transforms (gK)ʳ into (gJ)ʳ while preserving the lift ratio.
164    ///
165    /// # Panics
166    /// Panics if log_constraint_degree > log_blowup.
167    ///
168    /// The quotient domain is a strict subset of the committed LDE domain.
169    ///
170    /// If the constraint degree is `D`, the resulting quotient polynomial has degree
171    /// `< N * (D - 1)`, so `N * D` evaluation points suffice for commitment and for the
172    /// verifier's reconstruction. The PCS uses a larger blowup `B`, so the committed
173    /// LDE domain `gK` has `N * B` points, but constraint evaluation only needs the
174    /// sub-coset `gJ` of size `N * D` (with `D <= B`).
175    pub fn quotient_domain(&self, log_constraint_degree: u8) -> Self {
176        let log_blowup = self.log_lde_height - self.log_trace_height;
177        assert!(log_constraint_degree <= log_blowup, "constraint degree cannot exceed blowup");
178        let log_max_trace_height = self.log_max_lde_height - log_blowup;
179        Self {
180            log_trace_height: self.log_trace_height,
181            log_lde_height: self.log_trace_height + log_constraint_degree,
182            log_max_lde_height: log_max_trace_height + log_constraint_degree,
183        }
184    }
185
186    // ============ Selector computation ============
187
188    /// Compute selectors for evaluation over this coset in natural order.
189    ///
190    /// Returns is_first_row, is_last_row, is_transition for each point in the coset
191    /// (gK)ʳ. The trace domain H has size `2^log_trace_height`.
192    ///
193    /// Selectors use unnormalized Lagrange basis polynomials. The is_first_row selector
194    /// is `L₀(x) = Z_H(x) / (x − 1)`, which equals 0 on all of H except the first row.
195    /// When multiplied by a constraint C(x), it enforces C only at the first row:
196    /// `L₀(x)·C(x)` vanishes on H iff `C(1) = 0`. Similarly,
197    /// `is_last_row = Z_H(x) / (x − ω⁻¹)`.
198    ///
199    /// The is_transition selector is `(x − ω⁻¹)`, which is nonzero everywhere except the
200    /// last row, enforcing transition constraints on all consecutive row pairs.
201    /// These are "unnormalized" because we omit the constant factor 1/N that would make
202    /// them evaluate to exactly 1 at their target row. This is fine because both prover
203    /// and verifier evaluate the same unnormalized form: multiplying all boundary constraints
204    /// by a common nonzero constant does not affect whether the quotient is a polynomial.
205    pub fn selectors<F: TwoAdicField>(&self) -> Selectors<Vec<F>> {
206        let shift: F = self.lde_shift();
207        let coset_size = self.lde_height();
208        let log_blowup = self.log_blowup();
209
210        // Z_H(x) = xⁿ − 1 evaluated at coset points.
211        // Periodic with 2^log_blowup distinct values; expand to full coset size for zip.
212        let s_pow_n = shift.exp_power_of_2(self.log_trace_height as usize);
213        let z_h_periodic: Vec<F> = F::two_adic_generator(log_blowup)
214            .shifted_powers(s_pow_n)
215            .take(1 << log_blowup)
216            .map(|x| x - F::ONE)
217            .collect();
218        let period = z_h_periodic.len();
219
220        // Coset points in natural order: shift·ω_Jⁱ
221        let omega_j = F::two_adic_generator(self.log_lde_height as usize);
222        let xs: Vec<F> = omega_j.shifted_powers(shift).collect_n(coset_size);
223
224        let omega_h_inv = F::two_adic_generator(self.log_trace_height as usize).inverse();
225
226        // Unnormalized Lagrange selector: selᵢ = Z_H(xᵢ) / (xᵢ − basis_point)
227        // Uses modular indexing into z_h_periodic to avoid a full-size allocation.
228        let single_point_selector = |basis_point: F| -> Vec<F> {
229            let denoms: Vec<F> = xs.par_iter().map(|&x| x - basis_point).collect();
230            let invs = batch_multiplicative_inverse(&denoms);
231            (0..coset_size)
232                .into_par_iter()
233                .map(|i| z_h_periodic[i % period] * invs[i])
234                .collect()
235        };
236
237        Selectors {
238            is_first_row: single_point_selector(F::ONE),
239            is_last_row: single_point_selector(omega_h_inv),
240            is_transition: xs.into_par_iter().map(|x| x - omega_h_inv).collect(),
241        }
242    }
243
244    /// Lifted selectors at the OOD point (verifier).
245    ///
246    /// For a selector `s(x)` defined over the original trace domain of size `n_j`,
247    /// lifting evaluates `s(z_lift)` where `z_lift = z^r` and
248    /// `r = 2^log_lift_ratio = max_n / n_j`. This maps the OOD point `z`
249    /// (sampled in the max-trace domain) into the per-instance trace domain.
250    ///
251    /// # Formulas (unnormalized)
252    /// - `is_first_row = Z_H(z_lift) / (z_lift − 1)`
253    /// - `is_last_row  = Z_H(z_lift) / (z_lift − ω_{n_j}⁻¹)`
254    /// - `is_transition = z_lift − ω_{n_j}⁻¹`
255    ///
256    /// where `Z_H(z_lift) = z_lift^{n_j} − 1 = z^{max_n} − 1`.
257    pub fn selectors_at<F, EF>(&self, z: EF) -> Selectors<EF>
258    where
259        F: TwoAdicField,
260        EF: ExtensionField<F>,
261    {
262        let z_lift = z.exp_power_of_2(self.log_lift_ratio());
263        let vanishing = self.vanishing_at::<F, _>(z_lift);
264        let omega_inv = F::two_adic_generator(self.log_trace_height as usize).inverse();
265
266        Selectors {
267            is_first_row: vanishing / (z_lift - F::ONE),
268            is_last_row: vanishing / (z_lift - omega_inv),
269            is_transition: z_lift - omega_inv,
270        }
271    }
272
273    // ============ Vanishing polynomial ============
274
275    /// Vanishing polynomial at an out-of-domain point.
276    ///
277    /// Returns `Z_H(z) = zⁿ − 1` using `exp_power_of_2` (log-many squarings).
278    pub fn vanishing_at<F, EF>(&self, z: EF) -> EF
279    where
280        F: TwoAdicField,
281        EF: ExtensionField<F>,
282    {
283        z.exp_power_of_2(self.log_trace_height as usize) - EF::ONE
284    }
285
286    // ============ Domain membership ============
287
288    /// Check if a point is in the trace domain H.
289    ///
290    /// Returns true if `z^N == 1` where N is the trace height.
291    /// Points in H cause division by zero in vanishing polynomial inversion.
292    #[inline]
293    pub fn is_in_trace_domain<F, EF>(&self, z: EF) -> bool
294    where
295        F: TwoAdicField,
296        EF: ExtensionField<F>,
297    {
298        z.exp_power_of_2(self.log_trace_height as usize) == EF::ONE
299    }
300
301    /// Check if a point is in the LDE coset gK.
302    ///
303    /// Returns true if `(z/g)^|K| == 1` where g is the generator shift
304    /// and K is the LDE domain. Points in gK cause division by zero in DEEP quotients.
305    #[inline]
306    pub fn is_in_lde_coset<F, EF>(&self, z: EF) -> bool
307    where
308        F: TwoAdicField,
309        EF: ExtensionField<F>,
310    {
311        let shift: F = self.lde_shift();
312        let z_over_shift = z * shift.inverse();
313        z_over_shift.exp_power_of_2(self.log_lde_height as usize) == EF::ONE
314    }
315
316    // ============ OOD point sampling ============
317
318    /// Sample an OOD evaluation point from the channel that lies outside both the
319    /// trace-domain subgroup `H` and the LDE evaluation coset `gK`.
320    ///
321    /// Repeatedly draws `sample_algebra_element` candidates until one satisfies
322    /// both exclusion tests. This terminates with overwhelming probability because
323    /// `|H ∪ gK|` is negligible relative to the extension field size.
324    pub fn sample_ood_point<F, EF>(&self, channel: &mut impl Channel<F = F>) -> EF
325    where
326        F: TwoAdicField,
327        EF: ExtensionField<F>,
328    {
329        loop {
330            let candidate: EF = channel.sample_algebra_element();
331            if !self.is_in_trace_domain::<F, _>(candidate)
332                && !self.is_in_lde_coset::<F, _>(candidate)
333            {
334                break candidate;
335            }
336        }
337    }
338}

344#[cfg(test)]
345mod tests {
346    use p3_field::{Field, PrimeCharacteristicRing};
347
348    use super::*;
349    use crate::testing::configs::goldilocks_poseidon2::Felt;
350
351    #[test]
352    fn domain_info_basic() {
353        // Trace height 2^10, blowup 2^3, max trace 2^12
354        let info = LiftedCoset::new(10, 3, 12);
355
356        assert_eq!(info.log_trace_height, 10);
357        assert_eq!(info.log_lde_height, 13);
358        assert_eq!(info.log_max_lde_height, 15);
359
360        assert_eq!(info.log_blowup(), 3);
361        assert_eq!(info.log_lift_ratio(), 2);
362        assert!(info.is_lifted());
363
364        assert_eq!(info.trace_height(), 1024);
365        assert_eq!(info.lde_height(), 8192);
366        assert_eq!(info.max_lde_height(), 32768);
367        assert_eq!(info.blowup(), 8);
368    }
369
370    #[test]
371    fn domain_info_no_lift() {
372        // Matrix at max height (no lifting needed)
373        let info = LiftedCoset::unlifted(10, 3);
374
375        assert_eq!(info.log_lift_ratio(), 0);
376        assert!(!info.is_lifted());
377    }
378
379    #[test]
380    fn domain_info_lde_shift() {
381        // Trace height 2^10, blowup 2^3, max trace 2^12
382        let info = LiftedCoset::new(10, 3, 12);
383        let shift: Felt = info.lde_shift();
384
385        // shift = g^(2^2) = g^4
386        let expected = Felt::GENERATOR.exp_power_of_2(2);
387        assert_eq!(shift, expected);
388    }
389
390    #[test]
391    fn domain_info_no_lift_shift() {
392        // When not lifted, shift should be g^1 = g
393        let info = LiftedCoset::unlifted(10, 3);
394        let shift: Felt = info.lde_shift();
395
396        // shift = g^(2^0) = g
397        assert_eq!(shift, Felt::GENERATOR);
398    }
399
400    #[test]
401    fn quotient_domain_preserves_lift_ratio_and_updates_blowup() {
402        // Trace height 2^10, blowup 2^3 (B=8), max trace 2^12.
403        let lde = LiftedCoset::new(10, 3, 12);
404
405        // Constraint degree D = 4 (log D = 2), so quotient domain size is N*D.
406        let q = lde.quotient_domain(2);
407
408        // Trace height is unchanged; the evaluation domain becomes N*D.
409        assert_eq!(q.log_trace_height, 10);
410        assert_eq!(q.log_blowup(), 2);
411        assert_eq!(q.log_lde_height, 12);
412
413        // Max evaluation domain becomes N_max*D.
414        assert_eq!(q.log_max_lde_height, 14);
415
416        // Lift ratio is preserved.
417        assert_eq!(q.log_lift_ratio(), lde.log_lift_ratio());
418    }
