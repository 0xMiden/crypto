1//! Heap-indexed node address in a binary Merkle tree.
2
3/// A node address in a binary Merkle tree using heap indexing.
4///
5/// Depth 0 = root. At depth `d`, positions range over `0..2^d`.
6/// The heap index `(1 << depth) + position` yields a single `usize`
7/// whose natural ordering is top-down, left-to-right. Standard tree
8/// operations reduce to bit manipulation:
9///
10/// - `parent()` = `id >> 1`
11/// - `sibling()` = `id ^ 1`
12/// - `depth()` = `ilog2(id)`
13/// - `position()` = `id − 2^depth`
14#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
15pub struct NodeId(usize);
16
17impl NodeId {
18    /// Create a node ID from a (depth, position) pair.
19    ///
20    /// Callers must ensure `position < 2^depth` (except for the root at depth 0,
21    /// where position must be 0).
22    #[inline]
23    pub const fn new(depth: usize, position: usize) -> Self {
24        Self((1 << depth) + position)
25    }
26
27    /// The depth in the tree (0 = root).
28    #[inline]
29    pub const fn depth(&self) -> usize {
30        self.0.ilog2() as usize
31    }
32
33    /// The position within the depth level.
34    #[inline]
35    pub const fn position(&self) -> usize {
36        self.0 - (1 << self.depth())
37    }
38
39    /// The sibling node (same depth, position XOR 1).
40    #[inline]
41    pub const fn sibling(&self) -> Self {
42        Self(self.0 ^ 1)
43    }
44
45    /// The parent node (depth − 1, position >> 1).
46    #[inline]
47    pub const fn parent(&self) -> Self {
48        Self(self.0 >> 1)
49    }
50}
