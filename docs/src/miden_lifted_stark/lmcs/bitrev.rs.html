1//! Local copy of `BitReversibleMatrix` with additional impls for [`FlatMatrixView`].
2//!
3//! # Temporary stopgap
4//!
5//! The upstream `BitReversibleMatrix` trait in `p3-matrix` is only implemented for
6//! [`DenseMatrix`], not for [`FlatMatrixView`]. This module provides an identical
7//! trait with impls for all matrix types used by the LMCS and FRI.
8//!
9//! Once an upstream impl is available, this module can be removed and all uses
10//! replaced with `p3_matrix::bitrev::BitReversibleMatrix`.
11
12use p3_field::{ExtensionField, Field};
13use p3_matrix::{
14    Matrix,
15    bitrev::{BitReversalPerm, BitReversedMatrixView},
16    dense::{DenseMatrix, DenseStorage, RowMajorMatrix},
17    extension::FlatMatrixView,
18};
19
20/// Materialize a matrix into domain-ordered `BitReversedMatrixView<RowMajorMatrix<T>>`.
21///
22/// Temporary adapter for types that implement the upstream
23/// [`p3_matrix::bitrev::BitReversibleMatrix`] but not this crate's local copy.
24/// The returned type implements both traits and can be passed directly to
25/// [`Lmcs::build_tree`](crate::lmcs::Lmcs::build_tree) /
26/// [`Lmcs::build_aligned_tree`](crate::lmcs::Lmcs::build_aligned_tree).
27///
28/// Remove alongside this module when upstream impls cover all DFT output types.
29pub fn materialize_bitrev<T: Clone + Send + Sync>(
30    evals: impl p3_matrix::bitrev::BitReversibleMatrix<T>,
31) -> BitReversedMatrixView<RowMajorMatrix<T>> {
32    BitReversalPerm::new_view(evals.bit_reverse_rows().to_row_major_matrix())
33}
34
35/// A matrix that supports bit-reversed row reordering.
36///
37/// Local copy of `p3_matrix::bitrev::BitReversibleMatrix` extended with impls for
38/// [`FlatMatrixView`].
39pub trait BitReversibleMatrix<T: Send + Sync + Clone>: Matrix<T> {
40    /// The type returned when this matrix is viewed in bit-reversed order.
41    type BitRev: BitReversibleMatrix<T>;
42
43    /// Return a version of the matrix with its row order reversed by bit index.
44    fn bit_reverse_rows(self) -> Self::BitRev;
45}
46
47// ============================================================================
48// DenseMatrix impls (mirrors upstream)
49// ============================================================================
50
51impl<T, S> BitReversibleMatrix<T> for DenseMatrix<T, S>
52where
53    T: Clone + Send + Sync,
54    S: DenseStorage<T>,
55{
56    type BitRev = BitReversedMatrixView<Self>;
57
58    fn bit_reverse_rows(self) -> Self::BitRev {
59        BitReversalPerm::new_view(self)
60    }
61}
62
63impl<T, S> BitReversibleMatrix<T> for BitReversedMatrixView<DenseMatrix<T, S>>
64where
65    T: Clone + Send + Sync,
66    S: DenseStorage<T>,
67{
68    type BitRev = DenseMatrix<T, S>;
69
70    fn bit_reverse_rows(self) -> Self::BitRev {
71        self.inner
72    }
73}
