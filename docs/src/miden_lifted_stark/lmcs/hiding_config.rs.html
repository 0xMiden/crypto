24/// Configuration for hiding LMCS with random salt.
25///
26/// This type wraps a [`LmcsConfig`] and adds an RNG for generating salt
27/// during tree construction. The RNG is stored in a `RefCell` to allow
28/// salt generation without `&mut self` (required by `Mmcs::commit`).
29///
30/// `open_batch` delegates to the inner `LmcsConfig`, so hint layout and proof shape
31/// match the non-hiding implementation except for
32/// the presence of salt. The RNG is only used during tree construction.
33///
34/// # Type Parameters
35///
36/// - `PF`: Packed field element type for SIMD operations.
37/// - `PD`: Packed hash word element type.
38/// - `H`: Stateful hasher/sponge type.
39/// - `C`: 2-to-1 compression function type.
40/// - `R`: Random number generator type.
41/// - `WIDTH`: State width for the hasher.
42/// - `DIGEST`: Number of elements in a hash.
43/// - `SALT`: Number of salt elements per leaf (must be > 0).
44///
45/// # Security notes
46/// - `SALT` should be sized so `SALT * sizeof(PF::Value)` meets the target security parameter.
47/// - `R` should be an appropriately seeded CSPRNG; weaker RNGs can undermine hiding.
48/// - Cloning this config clones RNG state. Re-seed if you need independent salts per config.
49///
50/// # Example
51///
52/// ```ignore
53/// use p3_miden_lmcs::{HidingLmcsConfig, Lmcs, LmcsTree};
54/// use rand::rngs::StdRng;
55/// use rand::SeedableRng;
56///
57/// let rng = StdRng::seed_from_u64(42);
58/// let config =
59///     HidingLmcsConfig::<PF, PD, _, _, _, WIDTH, DIGEST, 4>::new(sponge, compress, rng);
60///
61/// let tree = config.build_aligned_tree(matrices);
62/// let root = tree.root();
63/// ```
64#[derive(Clone, Debug)]
65pub struct HidingLmcsConfig<
66    PF,
67    PD,
68    H,
69    C,
70    R,
71    const WIDTH: usize,
72    const DIGEST: usize,
73    const SALT: usize,
74> {
75    /// Inner non-hiding config with sponge and compression.
76    pub inner: LmcsConfig<PF, PD, H, C, WIDTH, DIGEST, SALT>,
77    /// RNG for salt generation. Uses `RefCell` for interior mutability.
78    rng: RefCell<R>,
79}
80
81impl<PF, PD, H, C, R, const WIDTH: usize, const DIGEST: usize, const SALT: usize>
82    HidingLmcsConfig<PF, PD, H, C, R, WIDTH, DIGEST, SALT>
83{
84    /// Create a new hiding LMCS configuration.
85    ///
86    /// # Compile-time Error
87    ///
88    /// Fails to compile if `SALT == 0`. Use [`LmcsConfig`] for non-hiding commitments.
89    #[inline]
90    pub fn new(sponge: H, compress: C, rng: R) -> Self {
91        const { assert!(SALT > 0) }
92        Self {
93            inner: LmcsConfig::new(sponge, compress),
94            rng: RefCell::new(rng),
95        }
96    }
97}

99impl<PF, PD, H, C, R, const WIDTH: usize, const DIGEST: usize, const SALT: usize> Lmcs
100    for HidingLmcsConfig<PF, PD, H, C, R, WIDTH, DIGEST, SALT>
101where
102    PF: PackedValue + Default,
103    PD: PackedValue + Default,
104    R: Rng + Clone,
105    StandardUniform: Distribution<PF::Value>,
106    H: StatefulHasher<PF::Value, [PD::Value; DIGEST], State = [PD::Value; WIDTH]>
107        + StatefulHasher<PF, [PD; DIGEST], State = [PD; WIDTH]>
108        + Alignable<PF::Value, PD::Value>
109        + Sync,
110    C: PseudoCompressionFunction<[PD::Value; DIGEST], 2>
111        + PseudoCompressionFunction<[PD; DIGEST], 2>
112        + Sync,
113{
114    type F = PF::Value;
115    type Commitment = Hash<PF::Value, PD::Value, DIGEST>;
116    type Tree<M: Matrix<PF::Value>> = LiftedMerkleTree<PF::Value, PD::Value, M, DIGEST, SALT>;
117    type BatchProof = BatchProof<PF::Value, Self::Commitment, SALT>;
118
119    /// Build a tree with per-leaf salt sampled from the RNG.
120    ///
121    /// Preconditions match `LmcsConfig::build_tree`; panics if `leaves` is empty.
122    fn build_tree<M: BitReversibleMatrix<Self::F>>(&self, leaves: Vec<M>) -> Self::Tree<M::BitRev> {
123        let tree_height = leaves.last().map(Matrix::height).unwrap_or(0);
124        let salt = RowMajorMatrix::rand(&mut *self.rng.borrow_mut(), tree_height, SALT);
125        LiftedMerkleTree::build_with_alignment::<M, PF, PD, H, C, WIDTH>(
126            &self.inner.sponge,
127            &self.inner.compress,
128            leaves,
129            Some(salt),
130            1,
131        )
132    }
133
134    /// Build a tree with per-leaf salt sampled from the RNG and hasher alignment padding.
135    ///
136    /// Preconditions match `LmcsConfig::build_tree`; panics if `leaves` is empty.
137    fn build_aligned_tree<M: BitReversibleMatrix<Self::F>>(
138        &self,
139        leaves: Vec<M>,
140    ) -> Self::Tree<M::BitRev> {
141        let tree_height = leaves.last().map(Matrix::height).unwrap_or(0);
142        let salt = RowMajorMatrix::rand(&mut *self.rng.borrow_mut(), tree_height, SALT);
143        LiftedMerkleTree::build_with_alignment::<M, PF, PD, H, C, WIDTH>(
144            &self.inner.sponge,
145            &self.inner.compress,
146            leaves,
147            Some(salt),
148            <H as Alignable<PF::Value, PD::Value>>::ALIGNMENT,
149        )
150    }
151
152    fn hash<'a, I>(&self, rows: I) -> Self::Commitment
153    where
154        I: IntoIterator<Item = &'a [Self::F]>,
155        Self::F: 'a,
156    {
157        self.inner.hash(rows)
158    }
159
160    fn compress(&self, left: Self::Commitment, right: Self::Commitment) -> Self::Commitment {
161        self.inner.compress(left, right)
162    }
163
164    fn open_batch<Ch>(
165        &self,
166        commitment: &Self::Commitment,
167        widths: &[usize],
168        indices: &TreeIndices,
169        channel: &mut Ch,
170    ) -> Result<OpenedRows<Self::F>, LmcsError>
171    where
172        Ch: VerifierChannel<F = Self::F, Commitment = Self::Commitment>,
173    {
174        self.inner.open_batch(commitment, widths, indices, channel)
175    }
176
177    fn read_batch_proof<Ch>(
178        &self,
179        widths: &[usize],
180        indices: &TreeIndices,
181        channel: &mut Ch,
182    ) -> Result<Self::BatchProof, LmcsError>
183    where
184        Ch: VerifierChannel<F = Self::F, Commitment = Self::Commitment>,
185    {
186        self.inner.read_batch_proof(widths, indices, channel)
187    }
188
189    fn alignment(&self) -> usize {
190        self.inner.alignment()
191    }
