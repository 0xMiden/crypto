1//! Merkle witness for batch proof verification.
2//!
3//! [`MerkleWitness`] reconstructs the minimal subset of a Merkle tree from
4//! opened leaves and provided sibling hashes. It supports [`root`](MerkleWitness::root)
5//! verification and [`path`](MerkleWitness::path) extraction.
6
7use alloc::vec::Vec;
8
9use crate::lmcs::node_id::NodeId;
10
11/// The minimal subset of a Merkle tree reconstructed from opened leaves and sibling hashes.
12///
13/// Contains every node on the authentication paths from the opened leaves to the
14/// root — the opened leaves themselves, their siblings, and all ancestors up to
15/// and including the root. Nodes not on any authentication path are absent.
16///
17/// Nodes are stored as `(NodeId, value)` pairs sorted by heap index, which gives
18/// natural top-down, left-to-right ordering.
19pub struct MerkleWitness<T> {
20    /// All nodes sorted by NodeId (heap index order).
21    nodes: Vec<(NodeId, T)>,
22    /// Tree depth (leaves are at this depth, root at 0).
23    tree_depth: usize,
24}
25
26impl<T: Clone> MerkleWitness<T> {
27    /// Look up a node by its ID.
28    fn get(&self, id: NodeId) -> Option<&T> {
29        self.nodes.binary_search_by_key(&id, |(k, _)| *k).ok().map(|i| &self.nodes[i].1)
30    }
31
32    /// Build a witness from sorted leaf hashes.
33    ///
34    /// Leaves must be sorted by position in ascending order with no duplicates.
35    ///
36    /// `fetch_sibling` is called with the [`NodeId`] of each missing sibling,
37    /// level-by-level, left-to-right, bottom-to-top, matching transcript order.
38    pub(super) fn build<E>(
39        leaves: impl IntoIterator<Item = (usize, T)>,
40        tree_depth: usize,
41        mut fetch_sibling: impl FnMut(NodeId) -> Result<T, E>,
42        compress: impl Fn(T, T) -> T,
43    ) -> Result<Self, E> {
44        let mut current: Vec<(NodeId, T)> = leaves
45            .into_iter()
46            .map(|(pos, val)| (NodeId::new(tree_depth, pos), val))
47            .collect();
48        debug_assert!(current.windows(2).all(|w| w[0].0 < w[1].0), "leaves must be sorted");
49
50        let mut nodes: Vec<(NodeId, T)> = Vec::new();
51        // Each level has at most ceil(n/2) parents; pre-allocate to avoid
52        // reallocation on the first level (subsequent levels reuse via swap).
53        let mut next: Vec<(NodeId, T)> = Vec::with_capacity(current.len().div_ceil(2));
54
55        for _ in 0..tree_depth {
56            let mut iter = current.drain(..).peekable();
57            while let Some((node, hash)) = iter.next() {
58                let sibling = node.sibling();
59
60                // Get sibling hash: from the set if present, otherwise fetched.
61                let sib_hash = iter
62                    .next_if(|(id, _)| *id == sibling)
63                    .map(|(_, h)| h)
64                    .map_or_else(|| fetch_sibling(sibling), Ok)?;
65
66                // Order children left-to-right, compress, and promote.
67                let (left, right) = if node < sibling {
68                    ((node, hash), (sibling, sib_hash))
69                } else {
70                    ((sibling, sib_hash), (node, hash))
71                };
72
73                let parent_hash = compress(left.1.clone(), right.1.clone());
74                next.push((node.parent(), parent_hash));
75
76                nodes.extend([left, right]);
77            }
78            drop(iter);
79            core::mem::swap(&mut current, &mut next);
80        }
81
82        // Root level (depth 0).
83        nodes.extend(current);
84
85        // Sort by heap index for binary search lookups.
86        nodes.sort_by_key(|(id, _)| *id);
87
88        Ok(Self { nodes, tree_depth })
89    }
90
91    /// The root hash, or `None` if the tree is empty.
92    pub fn root(&self) -> Option<&T> {
93        self.get(NodeId::new(0, 0))
94    }
95
96    /// Authentication path for a leaf index (sibling hashes, bottom-to-top).
97    pub fn path(&self, index: usize) -> Option<Vec<T>> {
98        if index >= (1usize << self.tree_depth) {
99            return None;
100        }
101
102        let mut path = Vec::with_capacity(self.tree_depth);
103        let mut id = NodeId::new(self.tree_depth, index);
104        for _ in 0..self.tree_depth {
105            path.push(self.get(id.sibling())?.clone());
106            id = id.parent();
107        }
108        Some(path)
109    }
110}

112#[cfg(test)]
113mod tests {
114    use alloc::vec;
115
116    use super::*;
117
118    #[test]
119    fn both_children_known() {
120        let tree = MerkleWitness::build(
121            [(0, 10u64), (1, 20)],
122            1,
123            |_| -> Result<u64, ()> { panic!("should not be called") },
124            |l, r| l + r,
125        )
126        .unwrap();
127        assert_eq!(tree.root(), Some(&30));
128    }
129
130    #[test]
131    fn fetches_missing_sibling() {
132        let tree = MerkleWitness::build(
133            [(0, 10u64)],
134            1,
135            |sib| {
136                assert_eq!(sib, NodeId::new(1, 1));
137                Ok::<_, ()>(20)
138            },
139            |l, r| l + r,
140        )
141        .unwrap();
142        assert_eq!(tree.root(), Some(&30));
143    }
144
145    #[test]
146    fn path_extraction() {
147        // tree_depth=2: 4 leaves, only positions 0 and 3 known.
148        let tree = MerkleWitness::build(
149            [(0, 1u64), (3, 4)],
150            2,
151            |sib| match sib {
152                s if s == NodeId::new(2, 1) => Ok::<_, ()>(2u64),
153                s if s == NodeId::new(2, 2) => Ok(3u64),
154                _ => panic!("unexpected sibling request: {sib:?}"),
155            },
156            |l, r| l + r,
157        )
158        .unwrap();
159
160        assert_eq!(tree.root(), Some(&10)); // (1+2) + (3+4) = 10
161
162        // path: sibling hashes from leaf to root
163        assert_eq!(tree.path(0).unwrap(), vec![2, 7]);
164        assert_eq!(tree.path(3).unwrap(), vec![3, 3]);
165        assert!(tree.path(4).is_none());
166    }
