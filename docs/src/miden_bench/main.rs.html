1//! Unified STARK prove/verify profiling binary.
2//!
3//! Proves and verifies a set of AIR instances, printing a configuration summary,
4//! proof size, and total time. Trace specifications are passed as positional
5//! arguments; all parameters have sensible defaults.
6//!
7//! By default the output is concise (config header + proof size + total time).
8//! Pass `-v` for the full hierarchical tracing tree, or set `RUST_LOG` for
9//! fine-grained control.
10//!
11//! ```bash
12//! # Quick run with defaults (blake3:15 keccak:18 poseidon2:19):
13//! cargo run -p miden-bench --features concurrent --release
14//!
15//! # Custom traces:
16//! cargo run -p miden-bench --features concurrent --release -- keccak:15 keccak:18 keccak:19
17//!
18//! # Full tracing tree:
19//! cargo run -p miden-bench --features concurrent --release -- -v keccak:15
20//!
21//! # Multi-iteration with warm-up (reports min/median/mean/max):
22//! cargo run -p miden-bench --features concurrent --release -- -n 5 keccak:15
23//!
24//! # Miden-shaped AIR (auto log_blowup=3):
25//! cargo run -p miden-bench --features concurrent --release -- miden:18:51 miden:19:20
26//!
27//! # Batch-STARK comparison:
28//! cargo run -p miden-bench --features concurrent --release -- -m batch keccak:15 keccak:18
29//! ```

84/// Captured output from a single prove/verify invocation.
85pub(crate) struct RunResult {
86    pub(crate) proof_size_bytes: usize,
87    /// Number of field elements in the proof (lifted only, 0 for batch).
88    pub(crate) field_elems: usize,
89    /// Number of commitments in the proof (lifted only, 0 for batch).
90    pub(crate) commitments: usize,
91}
92
93impl fmt::Display for RunResult {
94    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
95        write!(f, "proof size: {}", format_bytes(self.proof_size_bytes))?;
96        if self.field_elems > 0 || self.commitments > 0 {
97            write!(f, " ({} field elems, {} commitments)", self.field_elems, self.commitments,)?;
98        }
99        Ok(())
100    }
101}

107fn print_config(cli: &Cli, specs: &[TraceSpec], traces: &[RowMajorMatrix<Felt>], log_blowup: u8) {
108    let mode = match cli.mode {
109        Mode::Lifted => "lifted",
110        Mode::Batch => "batch",
111    };
112    eprintln!("{:<20} {mode}", "mode:");
113    eprintln!("{:<20} {}", "hash:", cli.hash);
114    eprintln!("{:<20} {}", "seed:", cli.seed);
115    for (i, (spec, trace)) in specs.iter().zip(traces).enumerate() {
116        let label = if i == 0 { "traces:" } else { "" };
117        eprintln!(
118            "{:<20} {}:{} (width={}, 2^{} = {} rows)",
119            label,
120            spec.air_type,
121            spec.log_height,
122            trace.width(),
123            spec.log_height,
124            trace.height(),
125        );
126    }
127    eprintln!("{:<20} {}", "log_blowup:", log_blowup);
128    eprintln!("{:<20} {}", "log_folding_arity:", cli.log_folding_arity);
129    eprintln!("{:<20} {}", "log_final_degree:", cli.log_final_degree);
130    eprintln!("{:<20} {}", "num_queries:", cli.num_queries);
131    eprintln!("{:<20} {}", "deep_pow_bits:", cli.deep_pow_bits);
132    eprintln!("{:<20} {}", "folding_pow_bits:", cli.folding_pow_bits);
133    eprintln!("{:<20} {}", "query_pow_bits:", cli.query_pow_bits);
134    eprintln!();
135}

137// ═══════════════════════════════════════════════════════════════════════════════
138// Trace generation (shared between modes)
139// ═══════════════════════════════════════════════════════════════════════════════
140
141fn generate_traces(
142    specs: &[TraceSpec],
143    rng: &mut SmallRng,
144    constants: Option<&GlRoundConstants>,
145) -> Vec<RowMajorMatrix<Felt>> {
146    specs
147        .iter()
148        .map(|spec| {
149            info_span!("generate trace", air = %spec.air_type, log_height = spec.log_height)
150                .in_scope(|| match spec.air_type {
151                    AirType::Keccak => {
152                        let n = (1usize << spec.log_height) / KECCAK_ROWS_PER_HASH;
153                        let inputs: Vec<[u64; 25]> = (0..n).map(|_| rng.random()).collect();
154                        generate_keccak_trace(inputs)
155                    },
156                    AirType::Poseidon2 => {
157                        let n = 1usize << spec.log_height;
158                        let inputs: Vec<[Felt; 12]> = (0..n).map(|_| rng.random()).collect();
159                        generate_poseidon2_trace(
160                            inputs,
161                            constants.expect("poseidon2 constants required"),
162                        )
163                    },
164                    AirType::Blake3 => {
165                        let n = 1usize << spec.log_height;
166                        let inputs: Vec<[u32; 24]> = (0..n).map(|_| rng.random()).collect();
167                        generate_blake3_trace(inputs)
168                    },
169                    AirType::Miden => generate_dummy_trace(spec.width, spec.log_height, rng),
170                })
171        })
172        .collect()
173}

176// Main
177// ═══════════════════════════════════════════════════════════════════════════════
178
179fn main() {
180    let cli = Cli::parse();
181
182    // Apply defaults.
183    let mut specs = if cli.traces.is_empty() {
184        vec![
185            TraceSpec {
186                air_type: AirType::Blake3,
187                log_height: 15,
188                width: 0,
189                num_aux_cols: 0,
190            },
191            TraceSpec {
192                air_type: AirType::Keccak,
193                log_height: 18,
194                width: 0,
195                num_aux_cols: 0,
196            },
197            TraceSpec {
198                air_type: AirType::Poseidon2,
199                log_height: 19,
200                width: 0,
201                num_aux_cols: 0,
202            },
203        ]
204    } else {
205        cli.traces.clone()
206    };
207
208    // Sort by ascending height for deterministic output.
209    specs.sort_by_key(|s| s.log_height);
210
211    let has_miden = specs.iter().any(|s| s.air_type == AirType::Miden);
212    let log_blowup = cli.log_blowup.unwrap_or(if has_miden { 3 } else { 1 });
213
214    // Set up tracing subscriber (quiet by default, -v for full tree).
215    init_tracing(cli.verbose);
216
217    // Generate Poseidon2 round constants (from RNG, before trace inputs).
218    let mut rng = SmallRng::seed_from_u64(cli.seed);
219    let poseidon2_constants: Option<GlRoundConstants> =
220        if specs.iter().any(|s| s.air_type == AirType::Poseidon2) {
221            Some(RoundConstants::from_rng(&mut rng))
222        } else {
223            None
224        };
225
226    // Generate traces.
227    let traces = generate_traces(&specs, &mut rng, poseidon2_constants.as_ref());
228
229    // Print configuration summary.
230    print_config(&cli, &specs, &traces, log_blowup);
231
232    // Build PCS params (shared across hash functions).
233    let pcs = PcsParams::new(
234        log_blowup,
235        cli.log_folding_arity,
236        cli.log_final_degree,
237        cli.folding_pow_bits,
238        cli.deep_pow_bits,
239        cli.num_queries,
240        cli.query_pow_bits,
241    )
242    .expect("invalid PCS params");
243
244    type Dft = p3_dft::Radix2DitParallel<Felt>;
245
246    // Run prove/verify.
247    let start = Instant::now();
248    let result = match cli.mode {
249        Mode::Lifted => match cli.hash {
250            HashFn::Poseidon2 => {
251                let config = GenericStarkConfig::new(
252                    pcs,
253                    gl::test_lmcs(),
254                    Dft::default(),
255                    gl::test_challenger(),
256                );
257                lifted::run_lifted(&config, &specs, traces, &poseidon2_constants, &cli)
258            },
259            HashFn::Keccak => {
260                let config = GenericStarkConfig::new(
261                    pcs,
262                    keccak::test_lmcs(),
263                    Dft::default(),
264                    keccak::test_challenger(),
265                );
266                lifted::run_lifted(&config, &specs, traces, &poseidon2_constants, &cli)
267            },
268            HashFn::Blake3 => {
269                let config = GenericStarkConfig::new(
270                    pcs,
271                    blake3::test_lmcs(),
272                    Dft::default(),
273                    blake3::test_challenger(),
274                );
275                lifted::run_lifted(&config, &specs, traces, &poseidon2_constants, &cli)
276            },
277            HashFn::Blake3_192 => {
278                let config = GenericStarkConfig::new(
279                    pcs,
280                    blake3_192::test_lmcs(),
281                    Dft::default(),
282                    blake3_192::test_challenger(),
283                );
284                lifted::run_lifted(&config, &specs, traces, &poseidon2_constants, &cli)
285            },
286        },
287        Mode::Batch => match cli.hash {
288            HashFn::Poseidon2 => {
289                batch::run_batch_poseidon2(&specs, &traces, &poseidon2_constants, log_blowup, &cli)
290            },
291            HashFn::Keccak => {
292                batch::run_batch_keccak(&specs, &traces, &poseidon2_constants, log_blowup, &cli)
293            },
294            HashFn::Blake3 => {
295                batch::run_batch_blake3(&specs, &traces, &poseidon2_constants, log_blowup, &cli)
296            },
297            HashFn::Blake3_192 => {
298                batch::run_batch_blake3_192(&specs, &traces, &poseidon2_constants, log_blowup, &cli)
299            },
300        },
301    };
302    let elapsed = start.elapsed();
303
304    println!("{result}");
305    println!("total time: {:.3} s", elapsed.as_secs_f64());
306}

312fn init_tracing(verbose: bool) {
313    let default_level = if verbose {
314        tracing_forest::util::LevelFilter::DEBUG
315    } else {
316        tracing_forest::util::LevelFilter::WARN
317    };
318
319    let env_filter = tracing_subscriber::EnvFilter::builder()
320        .with_default_directive(default_level.into())
321        .from_env_lossy();
322
323    Registry::default()
324        .with(tracing_forest::ForestLayer::default().with_filter(env_filter))
325        .init();
326}

328pub(crate) fn format_bytes(bytes: usize) -> String {
329    if bytes < 1024 {
330        format!("{bytes} B")
331    } else if bytes < 1024 * 1024 {
332        format!("{:.1} KiB", bytes as f64 / 1024.0)
333    } else {
334        format!("{:.2} MiB", bytes as f64 / (1024.0 * 1024.0))
335    }
