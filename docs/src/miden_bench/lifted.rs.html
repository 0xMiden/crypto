25// AIR enum
26// ═══════════════════════════════════════════════════════════════════════════════
27
28pub(crate) enum LiftedBenchAir {
29    Keccak(LiftedKeccakAir),
30    Poseidon2(Box<LiftedPoseidon2Air>),
31    Blake3(LiftedBlake3Air),
32    Miden(DummyMidenAir),
33}
34
35impl BaseAir<Felt> for LiftedBenchAir {
36    fn width(&self) -> usize {
37        match self {
38            Self::Keccak(a) => BaseAir::<Felt>::width(a),
39            Self::Poseidon2(a) => BaseAir::<Felt>::width(a.as_ref()),
40            Self::Blake3(a) => BaseAir::<Felt>::width(a),
41            Self::Miden(a) => BaseAir::<Felt>::width(a),
42        }
43    }
44}
45
46impl<EF: Field> LiftedAir<Felt, EF> for LiftedBenchAir {
47    fn num_randomness(&self) -> usize {
48        match self {
49            Self::Miden(a) => LiftedAir::<Felt, EF>::num_randomness(a),
50            _ => 1,
51        }
52    }
53
54    fn aux_width(&self) -> usize {
55        match self {
56            Self::Miden(a) => LiftedAir::<Felt, EF>::aux_width(a),
57            _ => 1,
58        }
59    }
60
61    fn num_aux_values(&self) -> usize {
62        match self {
63            Self::Miden(a) => LiftedAir::<Felt, EF>::num_aux_values(a),
64            _ => 0,
65        }
66    }
67
68    fn num_var_len_public_inputs(&self) -> usize {
69        0
70    }
71
72    fn eval<AB: LiftedAirBuilder<F = Felt>>(&self, builder: &mut AB) {
73        match self {
74            Self::Keccak(a) => LiftedAir::<Felt, EF>::eval(a, builder),
75            Self::Poseidon2(a) => LiftedAir::<Felt, EF>::eval(a.as_ref(), builder),
76            Self::Blake3(a) => LiftedAir::<Felt, EF>::eval(a, builder),
77            Self::Miden(a) => LiftedAir::<Felt, EF>::eval(a, builder),
78        }
79    }
80}

82// ═══════════════════════════════════════════════════════════════════════════════
83// Runner
84// ═══════════════════════════════════════════════════════════════════════════════
85
86pub(crate) fn run_lifted<SC>(
87    config: &SC,
88    specs: &[TraceSpec],
89    traces: &[RowMajorMatrix<Felt>],
90    constants: &Option<GlRoundConstants>,
91    cli: &Cli,
92) -> RunResult
93where
94    SC: StarkConfig<Felt, QuadFelt>,
95    miden_lifted_stark::StarkDigest<Felt, QuadFelt, SC>: PartialEq + fmt::Debug,
96{
97    let airs: Vec<LiftedBenchAir> = specs
98        .iter()
99        .map(|spec| match spec.air_type {
100            AirType::Keccak => LiftedBenchAir::Keccak(LiftedKeccakAir),
101            AirType::Poseidon2 => {
102                let c = constants.as_ref().expect("poseidon2 constants required");
103                LiftedBenchAir::Poseidon2(Box::new(LiftedPoseidon2Air::new(c.clone())))
104            },
105            AirType::Blake3 => LiftedBenchAir::Blake3(LiftedBlake3Air),
106            AirType::Miden => {
107                LiftedBenchAir::Miden(DummyMidenAir::new(spec.width, spec.num_aux_cols))
108            },
109        })
110        .collect();
111
112    let aux_builders: Vec<ZeroAuxBuilder> = specs
113        .iter()
114        .map(|spec| match spec.air_type {
115            AirType::Miden => ZeroAuxBuilder {
116                num_aux_cols: spec.num_aux_cols,
117                num_aux_values: spec.num_aux_cols,
118            },
119            _ => ZeroAuxBuilder::dummy(),
120        })
121        .collect();
122
123    let instances: Vec<_> = airs
124        .iter()
125        .zip(traces)
126        .zip(&aux_builders)
127        .map(|((air, trace), aux)| (air, AirWitness::new(trace, &[], &[]), aux))
128        .collect();
129
130    let output = info_span!("prove")
131        .in_scope(|| prove_multi(config, &instances, config.challenger()).expect("proving failed"));
132
133    let result = RunResult {
134        proof_size_bytes: output.proof.size_in_bytes(),
135        field_elems: output.proof.num_field_elements(),
136        commitments: output.proof.num_commitments(),
137    };
138
139    if !cli.no_verify {
140        info_span!("verify").in_scope(|| {
141            let verifier_instances: Vec<_> = airs
142                .iter()
143                .map(|air| {
144                    (
145                        air,
146                        AirInstance {
147                            public_values: &[],
148                            var_len_public_inputs: &[],
149                        },
150                    )
151                })
152                .collect();
153            let digest =
154                verify_multi(config, &verifier_instances, &output.proof, config.challenger())
155                    .expect("verification failed");
156            assert_eq!(output.digest, digest);
157        });
158    }
159
160    result
161}
