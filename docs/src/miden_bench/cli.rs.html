13/// Prove and verify a set of AIR instances with the lifted or batch STARK prover.
14///
15/// Prints a configuration summary, proof size, and total time.
16/// Pass -v for the full hierarchical tracing tree.
17///
18/// Trace spec format:  `AIR:LOG_HEIGHT[:WIDTH[:AUX_COLS]]`
19///
20/// Available AIR types (with short aliases):
21///
22///   keccak (k)     Keccak-f(1600) permutation, 24 rows/hash
23///   poseidon2 (p)  Poseidon2 permutation (Goldilocks), 1 row/hash
24///   blake3 (b)     Blake3 compression, 1 row/hash
25///   miden (m)      Dummy degree-9 constraint (Miden VM shape)
26///
27/// WIDTH and AUX_COLS only apply to `miden` (defaults: 51, 8).
28///
29/// Examples:
30///
31///   bench                                             # default: blake3:15 keccak:18 poseidon2:19
32///   bench keccak:15 keccak:18 keccak:19               # 3x Keccak at different heights
33///   bench -v keccak:15                                 # full tracing tree
34///   bench miden:18:51 miden:19:20                      # two Miden-shaped traces (auto blowup=3)
35///   bench -m batch keccak:15 keccak:18 keccak:19       # batch-STARK comparison
36///   bench -H keccak keccak:15                          # use Keccak hash for commitments
37///   bench -H blake3 keccak:15                          # use BLAKE3 (32B) hash for commitments
38///   bench -H blake3-192 keccak:15                      # use BLAKE3-192 (24B) hash for commitments
39///   bench --log-blowup 2 --num-queries 50 keccak:18    # override PCS parameters
40#[derive(Parser)]
41#[command(name = "bench", verbatim_doc_comment)]
42pub(crate) struct Cli {
43    /// Trace specs (`AIR:LOG_HEIGHT[:WIDTH[:AUX_COLS]]`).
44    ///
45    /// When omitted, defaults to: blake3:15 keccak:18 poseidon2:19
46    #[arg(value_name = "TRACE")]
47    pub(crate) traces: Vec<TraceSpec>,
48
49    /// Prover backend: `lifted` (LMCS-based) or `batch` (Plonky3 batch-STARK).
50    #[arg(long, short = 'm', value_enum, default_value_t = Mode::Lifted)]
51    pub(crate) mode: Mode,
52
53    /// Hash function for the commitment scheme.
54    ///
55    /// Only applies to lifted mode; batch mode always uses poseidon2.
56    #[arg(long, short = 'H', value_enum, default_value_t = HashFn::Poseidon2)]
57    pub(crate) hash: HashFn,
58
59    /// Print the full hierarchical tracing tree (default: summary only).
60    ///
61    /// RUST_LOG overrides this when set.
62    #[arg(long, short = 'v')]
63    pub(crate) verbose: bool,
64
65    /// RNG seed for reproducible trace generation.
66    #[arg(long, short = 's', default_value_t = 1)]
67    pub(crate) seed: u64,
68
69    /// Skip proof verification (prover-only profiling).
70    #[arg(long)]
71    pub(crate) no_verify: bool,
72
73    // ── PCS Parameters ──────────────────────────────────────────────────
74    /// Log₂ blowup factor for the LDE domain extension.
75    ///
76    /// Auto-detected when omitted: 1 for hash-only workloads, 3 when any
77    /// `miden` trace is present (degree-9 constraints need more blowup).
78    #[arg(long, help_heading = "PCS Parameters")]
79    pub(crate) log_blowup: Option<u8>,
80
81    /// Number of FRI query repetitions (higher = more soundness).
82    #[arg(long, default_value_t = DEFAULT_NUM_QUERIES, help_heading = "PCS Parameters")]
83    pub(crate) num_queries: usize,
84
85    /// Proof-of-work grinding bits for the DEEP challenge (lifted mode only).
86    #[arg(long, default_value_t = DEFAULT_POW_BITS, help_heading = "PCS Parameters")]
87    pub(crate) deep_pow_bits: usize,
88
89    /// Log₂ FRI folding arity (1, 2, or 3 for fold-by-2/4/8).
90    #[arg(long, default_value_t = 2, help_heading = "PCS Parameters")]
91    pub(crate) log_folding_arity: u8,
92
93    /// Log₂ final polynomial degree bound.
94    #[arg(long, default_value_t = 0, help_heading = "PCS Parameters")]
95    pub(crate) log_final_degree: u8,
96
97    /// Proof-of-work grinding bits per FRI folding round.
98    #[arg(long, default_value_t = 0, help_heading = "PCS Parameters")]
99    pub(crate) folding_pow_bits: usize,
100
101    /// Proof-of-work grinding bits before query index sampling.
102    #[arg(long, default_value_t = 0, help_heading = "PCS Parameters")]
103    pub(crate) query_pow_bits: usize,
104}

123        match self {
124            HashFn::Poseidon2 => write!(f, "poseidon2"),
125            HashFn::Keccak => write!(f, "keccak"),
126            HashFn::Blake3 => write!(f, "blake3"),

129    }

134#[derive(Clone)]
135pub(crate) struct TraceSpec {
136    pub(crate) air_type: AirType,
137    pub(crate) log_height: u8,
138    /// Main trace width (miden only).
139    pub(crate) width: usize,
140    /// Extension-field auxiliary columns (miden only).
141    pub(crate) num_aux_cols: usize,
142}
143
144#[derive(Clone, Copy, PartialEq, Eq)]
145pub(crate) enum AirType {
146    Keccak,
147    Poseidon2,
148    Blake3,
149    Miden,
150}
151
152impl fmt::Display for AirType {
153    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
154        match self {
155            Self::Keccak => write!(f, "keccak"),
156            Self::Poseidon2 => write!(f, "poseidon2"),
157            Self::Blake3 => write!(f, "blake3"),
158            Self::Miden => write!(f, "miden"),
159        }
160    }
161}
162
163impl FromStr for TraceSpec {
164    type Err = String;
165
166    fn from_str(s: &str) -> Result<Self, Self::Err> {
167        let parts: Vec<&str> = s.split(':').collect();
168        if parts.len() < 2 {
169            return Err(format!("expected <air>:<log_height>[:<width>[:<aux_cols>]], got '{s}'"));
170        }
171
172        let air_type = match parts[0] {
173            "keccak" | "k" => AirType::Keccak,
174            "poseidon2" | "p" => AirType::Poseidon2,
175            "blake3" | "b" => AirType::Blake3,
176            "miden" | "m" => AirType::Miden,
177            other => return Err(format!("unknown AIR type '{other}'")),
178        };
179
180        let log_height: u8 =
181            parts[1].parse().map_err(|_| format!("invalid log_height '{}'", parts[1]))?;
182
183        let width = if parts.len() > 2 {
184            parts[2].parse().map_err(|_| format!("invalid width '{}'", parts[2]))?
185        } else {
186            DEFAULT_MIDEN_WIDTH
187        };
188
189        let num_aux_cols = if parts.len() > 3 {
190            parts[3].parse().map_err(|_| format!("invalid aux_cols '{}'", parts[3]))?
191        } else {
192            DEFAULT_MIDEN_AUX_COLS
193        };
194
195        if air_type == AirType::Miden && width < 9 {
196            return Err("miden width must be at least 9".to_string());
197        }
198
199        Ok(Self {
200            air_type,
201            log_height,
202            width,
203            num_aux_cols,
204        })
205    }
